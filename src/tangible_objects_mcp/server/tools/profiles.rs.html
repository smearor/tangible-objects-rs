1//! Profile query tools - `list_profiles`.
2
3use rmcp::ErrorData;
4use rmcp::handler::server::wrapper::Parameters;
5use rmcp::model::CallToolResult;
6use rmcp::model::ContentBlock;
7use rmcp::tool;
8use rmcp::tool_router;
9use tangible_objects_model::Owner;
10
11use crate::server::engine_mcp::EngineMcp;
12use crate::server::helpers::json_pretty;
13use crate::tool_args::ListProfilesArgs;
14
15/// Loaded-profile query tools.
16#[tool_router(router = profiles_router, vis = "pub(crate)")]
17impl EngineMcp {
18    /// Loaded figure profiles - same `owner`/`static_only` filters as
19    /// the GraphQL `profiles` query.
20    #[tool(
21        description = "List the loaded figure profiles; 'owner' filters to one application, 'static_only' keeps only owner-less static profiles."
22    )]
23    async fn list_profiles(
24        &self,
25        Parameters(args): Parameters<ListProfilesArgs>,
26    ) -> Result<CallToolResult, ErrorData> {
27        let profiles = self
28            .handle
29            .profiles()
30            .into_iter()
31            .filter(|info| match &args.owner {
32                Some(owner) => info.owner.as_ref().map(Owner::as_str) == Some(owner.as_str()),
33                None => true,
34            })
35            .filter(|info| !(args.static_only.unwrap_or(false) && info.owner.is_some()))
36            .collect::<Vec<_>>();
37        Ok(CallToolResult::success(vec![ContentBlock::text(
38            json_pretty(&profiles)?,
39        )]))
40    }
41}
