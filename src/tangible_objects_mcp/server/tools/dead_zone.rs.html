22/// Dead-zone registration and query tools.
23#[tool_router(router = dead_zone_router, vis = "pub(crate)")]
24impl EngineMcp {
25    /// All active dead zones - blacklisted surface positions whose
26    /// contacts are dropped at ingest.
27    #[tool(
28        description = "List the active dead zones: blacklisted surface spots (name, center mm, radius mm) whose contacts are suppressed."
29    )]
30    async fn dead_zones(&self) -> Result<CallToolResult, ErrorData> {
31        let snapshot = self.handle.snapshot();
32        Ok(CallToolResult::success(vec![ContentBlock::text(
33            json_pretty(&snapshot.dead_zones)?,
34        )]))
35    }
36
37    /// Registers a dead zone: a circular blacklist spot (name, x/y mm,
38    /// radius mm - default 10) whose contacts are dropped at ingest.
39    /// Ghost contacts never reach tracking, matching or fields.
40    /// `lifetime: "persistent"` writes the zone to the runtime store
41    /// so it reloads on daemon start.
42    #[tool(
43        description = "Blacklist a circular surface spot (name, x/y mm, radius mm - default 10): contacts inside are dropped at ingest. lifetime `persistent` reloads it on daemon start."
44    )]
45    async fn register_dead_zone(
46        &self,
47        Parameters(args): Parameters<RegisterDeadZoneArgs>,
48    ) -> Result<CallToolResult, ErrorData> {
49        let zone = DeadZone::new(
50            args.name,
51            Vec2::from_mm(args.x, args.y),
52            args.radius.unwrap_or(DEFAULT_DEAD_ZONE_RADIUS),
53        );
54        let (command, rx) = ControlCommand::register(
55            Registration::DeadZone(zone),
56            args.owner,
57            parse_lifetime(&args.lifetime)?,
58        );
59        self.handle
60            .send_command(command)
61            .await
62            .map_err(|_| ErrorData::internal_error("device loop is not running", None))?;
63        match rx.await {
64            Ok(Ok(_)) => Ok(CallToolResult::success(vec![ContentBlock::text(
65                "dead zone registered".to_string(),
66            )])),
67            Ok(Err(reason)) => Ok(CallToolResult::error(vec![ContentBlock::text(format!(
68                "rejected: {reason}"
69            ))])),
70            Err(_) => Ok(CallToolResult::error(vec![ContentBlock::text(
71                "device loop closed the registration reply".to_string(),
72            )])),
73        }
74    }
75
76    /// Removes a dead zone by name.
77    #[tool(description = "Unregister a dead zone by name.")]
78    async fn unregister_dead_zone(
79        &self,
80        Parameters(args): Parameters<UnregisterDeadZoneArgs>,
81    ) -> Result<CallToolResult, ErrorData> {
82        self.handle
83            .send_command(ControlCommand::UnregisterDeadZone {
84                name: args.name.clone(),
85            })
86            .await
87            .map_err(|_| ErrorData::internal_error("device loop is not running", None))?;
88        Ok(CallToolResult::success(vec![ContentBlock::text(format!(
89            "unregistered dead zone {}",
90            args.name
91        ))]))
92    }
93}
