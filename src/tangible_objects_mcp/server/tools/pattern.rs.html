24/// Profile building shared by `register_pattern`/`update_profile` -
25/// validates the flat pin list into `Vec2<Millimeters>` pairs and
26/// fills the spawn/stage options.
27fn build_profile(
28    name: &str,
29    points: &[f64],
30    tolerance: PatternTolerance,
31    spawn_on_partial: Option<bool>,
32    spawn_stability: Option<u64>,
33    stage: &Option<StageName>,
34) -> Result<ProfileFile, ErrorData> {
35    if !points.len().is_multiple_of(2) || points.is_empty() {
36        return Err(ErrorData::invalid_params(
37            "points must be a non-empty even-length flat list",
38            None,
39        ));
40    }
41    let points: Vec<Vec2<Millimeters>> = points
42        .as_chunks::<2>()
43        .0
44        .iter()
45        .map(|pair| Vec2::new(pair[0], pair[1]))
46        .collect();
47    let mut profile = ProfileFile::from_points(name, &points, tolerance);
48    profile.spawn_on_partial = spawn_on_partial.unwrap_or(false);
49    profile.stage = stage.clone();
50    if let Some(stability) = spawn_stability {
51        profile.spawn_stability = tangible_objects_model::FrameCount::new(stability);
52    }
53    Ok(profile)
54}

56/// Pattern-registration tools.
57#[tool_router(router = pattern_router, vis = "pub(crate)")]
58impl EngineMcp {
59    /// Registers a figure profile at runtime.
60    #[tool(
61        description = "Register a figure profile by name, flat pin coordinates [x0,y0,...] mm and tolerance mm. Fingerprint collisions with loaded profiles are rejected by the daemon."
62    )]
63    async fn register_pattern(
64        &self,
65        Parameters(args): Parameters<RegisterPatternArgs>,
66    ) -> Result<CallToolResult, ErrorData> {
67        let profile = build_profile(
68            &args.name,
69            &args.points,
70            args.tolerance,
71            args.spawn_on_partial,
72            args.spawn_stability,
73            &args.stage,
74        )?;
75        let (command, rx) = ControlCommand::register(
76            Registration::Pattern(profile),
77            args.owner,
78            parse_lifetime(&args.lifetime)?,
79        );
80        self.handle
81            .send_command(command)
82            .await
83            .map_err(|_| ErrorData::internal_error("device loop is not running", None))?;
84        match rx.await {
85            Ok(Ok(_)) => Ok(CallToolResult::success(vec![ContentBlock::text(format!(
86                "registered: {}",
87                args.name
88            ))])),
89            Ok(Err(reason)) => Ok(CallToolResult::error(vec![ContentBlock::text(format!(
90                "rejected: {reason}"
91            ))])),
92            Err(_) => Ok(CallToolResult::error(vec![ContentBlock::text(
93                "device loop closed the registration reply".to_string(),
94            )])),
95        }
96    }
97
98    /// Replaces a registered pattern's geometry while keeping its ID.
99    #[tool(
100        description = "Replace a registered pattern's geometry (name, points, tolerance) while keeping its pattern ID. Applies to new matches."
101    )]
102    async fn update_profile(
103        &self,
104        Parameters(args): Parameters<UpdateProfileArgs>,
105    ) -> Result<CallToolResult, ErrorData> {
106        let profile = build_profile(
107            &args.name,
108            &args.points,
109            args.tolerance,
110            args.spawn_on_partial,
111            args.spawn_stability,
112            &args.stage,
113        )?;
114        self.handle
115            .send_command(ControlCommand::UpdateProfile {
116                id: args.pattern_id,
117                profile: Box::new(profile),
118            })
119            .await
120            .map_err(|_| ErrorData::internal_error("device loop is not running", None))?;
121        Ok(CallToolResult::success(vec![ContentBlock::text(format!(
122            "updated pattern {}",
123            args.pattern_id
124        ))]))
125    }
126
127    /// Removes a dynamically registered pattern.
128    #[tool(description = "Unregister a dynamically registered pattern by pattern ID.")]
129    async fn unregister_pattern(
130        &self,
131        Parameters(args): Parameters<UnregisterPatternArgs>,
132    ) -> Result<CallToolResult, ErrorData> {
133        self.handle
134            .send_command(ControlCommand::UnregisterPattern {
135                id: args.pattern_id,
136            })
137            .await
138            .map_err(|_| ErrorData::internal_error("device loop is not running", None))?;
139        Ok(CallToolResult::success(vec![ContentBlock::text(format!(
140            "unregistered pattern {}",
141            args.pattern_id
142        ))]))
143    }
