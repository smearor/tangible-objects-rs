1//! Shared helpers for tool and resource impls - serialization
2//! shorthands and the `tangible://` resource descriptor builder.
3
4use rmcp::ErrorData;
5use rmcp::model::Resource;
6use tangible_objects_model::Lifetime;
7
8/// Serializes a value to pretty JSON for tool/resource output.
9pub(crate) fn json_pretty<T: serde::Serialize + ?Sized>(value: &T) -> Result<String, ErrorData> {
10    serde_json::to_string_pretty(value)
11        .map_err(|err| ErrorData::internal_error(format!("serialization failed: {err}"), None))
12}
13
14/// Parses the optional `lifetime` tool argument (`session`/`persistent`).
15pub(crate) fn parse_lifetime(lifetime: &Option<String>) -> Result<Lifetime, ErrorData> {
16    lifetime
17        .as_deref()
18        .map(str::parse)
19        .transpose()
20        .map(|parsed| parsed.unwrap_or_default())
21        .map_err(|_| ErrorData::invalid_params("lifetime must be 'session' or 'persistent'", None))
22}
23
24/// Describes one `tangible://` resource in `resources/list`.
25pub(crate) fn resource(uri: &str, name: &str, description: &str) -> Resource {
26    Resource::new(uri, name)
27        .with_description(description)
28        .with_mime_type("application/json")
29}
