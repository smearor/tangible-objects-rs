1//! `UpdateProfileArgs` - arguments for the `update_profile` tool.
2
3use schemars::JsonSchema;
4use serde::Deserialize;
5use tangible_objects_model::PatternId;
6use tangible_objects_model::PatternTolerance;
7use tangible_objects_model::StageName;
8
9/// Arguments for the `update_profile` tool - replaces a registered
10/// pattern's geometry while keeping its ID.
11#[derive(Debug, Deserialize, JsonSchema)]
12pub struct UpdateProfileArgs {
13    /// Pattern ID from `list_profiles`.
14    pub pattern_id: PatternId,
15    /// Figure name.
16    pub name: String,
17    /// Flat pin coordinates `[x0, y0, x1, y1, ...]` in millimeters.
18    pub points: Vec<f64>,
19    /// Scalar matching tolerance in millimeters.
20    pub tolerance: PatternTolerance,
21    /// Whether a partial match may spawn this figure.
22    pub spawn_on_partial: Option<bool>,
23    /// Frames a partial spawn must hold before confirmation.
24    pub spawn_stability: Option<u64>,
25    /// Pipeline stage this pattern suppresses through
26    /// (EVENT_PIPELINE.md); `null` keeps the pattern's stage.
27    pub stage: Option<StageName>,
28}
