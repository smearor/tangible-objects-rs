1//! `SetCompletionArgs` - arguments for the `set_completion` tool.
2
3use schemars::JsonSchema;
4use serde::Deserialize;
5
6/// Arguments for the `set_completion` tool - every field optional, unset
7/// fields keep the current value.
8#[derive(Debug, Deserialize, JsonSchema)]
9pub struct SetCompletionArgs {
10    /// Completion mode: `match` (claim binds for tracking only) or
11    /// `exact` (completed partial counts as exact for suppression).
12    pub mode: Option<String>,
13    /// Claim radius as a multiple of the pattern's scalar tolerance.
14    pub radius_factor: Option<f64>,
15    /// Maximum missing pins a partial may complete (1 or 2).
16    pub max_missing: Option<u32>,
17    /// Whether claimed slots enter the tangible's slot binding.
18    pub persist: Option<bool>,
19    /// Incumbent re-match tolerance widening (`match_score <= factor`,
20    /// `1.0` disables).
21    pub affinity_factor: Option<f64>,
22    /// Per-slot position smoothing alpha (`0.0` disables).
23    pub smoothing_alpha: Option<f64>,
24}
