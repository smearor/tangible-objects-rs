1//! `UpdateSeatArgs` - arguments for the `update_seat` tool.
2
3use schemars::JsonSchema;
4use serde::Deserialize;
5use smearor_commons_layout::SeatPose;
6use tangible_objects_model::Owner;
7use tangible_objects_model::PatternId;
8use tangible_objects_model::SeatId;
9use tangible_objects_model::SeatName;
10
11/// Arguments for the `update_seat` tool - omitted fields keep their
12/// current value.
13#[derive(Debug, Deserialize, JsonSchema)]
14pub struct UpdateSeatArgs {
15    /// Seat to mutate - the globally unique ID `add_seat` returned.
16    pub seat: SeatId,
17    /// Owner mutating the seat - must own its session.
18    pub owner: Owner,
19    /// New seat name - omitted keeps the current one.
20    #[serde(default)]
21    pub name: Option<SeatName>,
22    /// Rebinds the seat to another marker profile - omitted keeps the
23    /// current marker unless `clear_marker` is set.
24    #[serde(default)]
25    pub marker: Option<PatternId>,
26    /// Clears the marker binding so the seat turns virtual - wins
27    /// over `marker` when both are passed.
28    #[serde(default)]
29    pub clear_marker: Option<bool>,
30    /// New pose - omitted keeps the current one. Honored for virtual
31    /// seats; a bound marker re-derives the pose on the next frame.
32    #[serde(default)]
33    pub pose: Option<SeatPose>,
34}
