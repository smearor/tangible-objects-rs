2//! [`GraphQLClient`] method; one file per topic. No GraphQL strings

20use tangible_objects_client::ClientError;
21use tangible_objects_client::GraphQLClient;
22
23use crate::args::Cli;
24use crate::args::Command;
25
26/// Dispatches the parsed command onto the client.
27pub async fn run(client: &GraphQLClient, cli: &Cli) -> Result<(), ClientError> {
28    match &cli.command {
29        Command::Devices => devices::devices(client, cli).await?,
30        Command::Status => status::status(client, cli).await?,
31        Command::Metrics(args) => metrics::metrics(client, args).await?,
32        Command::Message(args) => message::message(client, args).await?,
33        Command::Tangibles => tangibles::tangibles(client, cli).await?,
34        Command::Touches => touches::touches(client, cli).await?,
35        Command::Profiles(args) => profiles::profiles(client, cli, args).await?,
36        Command::FieldSets(args) => field_sets::field_sets(client, cli, args).await?,
37        Command::Stages(args) => stages::stages(client, args).await?,
38        Command::Filter(args) => filter::filter(client, args).await?,
39        Command::Completion(args) => completion::completion(client, cli, args).await?,
40        Command::Record(args) => record::record(client, cli, args).await?,
41        Command::Analyze(args) => analyze::analyze(client, args).await?,
42        Command::Watch(args) => watch::watch(client, cli, args).await?,
43    }
44    Ok(())
45}
