1//! Output helpers - `--json` emits the typed payload verbatim;
2//! otherwise rows render as a `tabled` table.
3
4use serde::Serialize;
5use tabled::builder::Builder;
6use tabled::settings::Style;
7
8/// Prints `value` pretty-JSON when `json` is set, else hands the same
9/// JSON to the `lines` formatter for a compact textual form.
10pub fn emit<T: Serialize>(json: bool, value: &T, lines: impl Fn(&serde_json::Value)) {
11    let value = serde_json::to_value(value).unwrap_or(serde_json::Value::Null);
12    if json {
13        println!(
14            "{}",
15            serde_json::to_string_pretty(&value).unwrap_or_default()
16        );
17    } else {
18        lines(&value);
19    }
20}
21
22/// Prints a `tabled` table of `headers` + `row(item)` mappings -
23/// `--json` bypasses rendering and prints the typed list.
24pub fn emit_table<T: Serialize>(
25    json: bool,
26    value: &[T],
27    headers: &[&str],
28    row: impl Fn(&T) -> Vec<String>,
29) {
30    if json {
31        println!(
32            "{}",
33            serde_json::to_string_pretty(value).unwrap_or_default()
34        );
35        return;
36    }
37    let mut builder = Builder::new();
38    builder.push_record(headers.iter().copied());
39    for item in value {
40        builder.push_record(row(item));
41    }
42    let mut table = builder.build();
43    table.with(Style::rounded());
44    println!("{table}");
45}

47/// Prints a single event line for `watch` (kind + the interesting bits).
48pub fn event_line(event: &tangible_objects_model::TangibleEvent) -> String {
49    let kind = serde_json::to_value(event.kind)
50        .ok()
51        .and_then(|v| v.as_str().map(str::to_owned))
52        .unwrap_or_else(|| format!("{:?}", event.kind));
53    let mut parts = vec![kind];
54    if let Some(t) = &event.tangible {
55        parts.push(format!(
56            "id={} {} ({:.1},{:.1})mm",
57            t.id.as_u32(),
58            t.pattern_name.as_str(),
59            t.center.x,
60            t.center.y
61        ));
62    }
63    if let Some(name) = &event.name {
64        parts.push(format!("name={name}"));
65    }
66    if let Some(detail) = &event.detail {
67        parts.push(detail.clone());
68    }
69    if let Some(skipped) = event.skipped {
70        parts.push(format!("skipped={skipped}"));
71    }
72    parts.join(" ")
73}
