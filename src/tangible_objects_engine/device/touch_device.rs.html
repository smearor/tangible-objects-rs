17/// An evdev multitouch input device.
18///
19/// Thin wrapper around [`evdev::Device`] that speaks the engine's
20/// [`RawEvent`] vocabulary instead of `evdev::InputEvent`.
21pub struct TouchDevice {
22    /// The opened evdev device - source of raw events and target of
23    /// `EVIOCGRAB`/`ungrab`/`EVIOCGABS` ioctls.
24    device: evdev::Device,
25    /// Queued events emitted ahead of the next `fetch_events` batch -
26    /// the startup kernel-state import.
27    pending: Vec<RawEvent>,
28}
29
30impl TouchDevice {
31    /// Opens an evdev device node (usually requires root or `input` group).
32    pub fn open(path: &Path) -> Result<Self, EngineError> {
33        let device = evdev::Device::open(path)?;
34        Ok(Self {
35            device,
36            pending: Vec::new(),
37        })
38    }
39
40    /// Queues the kernel's current MT slot state for emission on the
41    /// next `fetch_events` call - the startup import
42    /// (`DetectionConfig::import_slots`). Contacts held open before the
43    /// reader attached become engine-visible immediately; an ioctl
44    /// failure queues nothing (the stream is then the only source).
45    pub fn queue_state_import(&mut self) {
46        if let Ok(events) = self.resync_events() {
47            self.pending = events;
48        }
49    }
50
51    /// Lists all `/dev/input/event*` devices that can be opened.
52    pub fn enumerate() -> Vec<DeviceEntry> {
53        let mut entries = Vec::new();
54        for index in 0..32 {
55            let path = PathBuf::from(format!("/dev/input/event{index}"));
56            if let Ok(device) = evdev::Device::open(&path) {
57                let name = device.name().unwrap_or("(unknown)").to_string();
58                let is_multitouch = device
59                    .supported_absolute_axes()
60                    .map(|axes| axes.contains(evdev::AbsoluteAxisType::ABS_MT_SLOT))
61                    .unwrap_or(false);
62                entries.push(DeviceEntry {
63                    path,
64                    name,
65                    vendor: device.input_id().vendor(),
66                    product: device.input_id().product(),
67                    is_multitouch,
68                });
69            }
70        }
71        entries
72    }
73
74    /// Auto-detects the touchscreen: enumerates devices and matches their
75    /// `vendor:product` pair against known [`DeviceProfile`]s. Succeeds
76    /// only when exactly one known device is attached - zero matches ask
77    /// for `--device`, several matches list the candidates.
78    pub fn detect(
79        registry: &tangible_objects_model::DeviceRegistry,
80    ) -> Result<DeviceEntry, EngineError> {
81        let mut matches: Vec<DeviceEntry> = Self::enumerate()
82            .into_iter()
83            .filter(|entry| registry.find_by_ids(entry.vendor, entry.product).is_some())
84            .collect();
85        match matches.len() {
86            0 => Err(EngineError::InvalidInput(
87                "no device with a known profile detected - pass --device /dev/input/eventX"
88                    .to_string(),
89            )),
90            1 => Ok(matches.swap_remove(0)),
91            _ => {
92                let list = matches
93                    .iter()
94                    .map(|e| format!("{} ({e})", e.path.display()))
95                    .collect::<Vec<_>>()
96                    .join(", ");
97                Err(EngineError::InvalidInput(format!(
98                    "several known devices detected ({list}) - pass --device to disambiguate"
99                )))
100            }
101        }
102    }

104    /// Device name reported by the kernel, if any.
105    pub fn name(&self) -> Option<&str> {
106        self.device.name()
107    }
108
109    /// Raw file descriptor of the device node (for `poll()`-based loops).
110    pub fn as_raw_fd(&self) -> RawFd {
111        self.device.as_raw_fd()
112    }
113
114    /// Exclusive grab (`EVIOCGRAB`): all other processes stop receiving
115    /// events. Used by the filter proxy so tangible contacts stay invisible
116    /// to the rest of the system.
117    pub fn grab(&mut self) -> Result<(), EngineError> {
118        self.device.grab()?;
119        Ok(())
120    }
121
122    /// Read access to the wrapped evdev device - the filter proxy mirrors
123    /// its capabilities onto a uinput clone.
124    pub fn inner(&self) -> &evdev::Device {
125        &self.device
126    }
127
128    /// Reads the device calibration (`resolution` and min/max of the
129    /// `ABS_MT_POSITION_X/Y` axes, units/mm).
130    ///
131    /// Returns `None` when the device reports no usable resolution.
132    pub fn calibration(&self) -> Option<Calibration> {
133        let abs_state = self.device.get_abs_state().ok()?;
134        let x = &abs_state[ABS_MT_POSITION_X.as_u16() as usize];
135        let y = &abs_state[ABS_MT_POSITION_Y.as_u16() as usize];
136        let x_res = x.resolution as f64;
137        let y_res = y.resolution as f64;
138        if x_res > 0.0 && y_res > 0.0 {
139            Some(Calibration::with_bounds(
140                x_res,
141                y_res,
142                x.minimum as f64,
143                x.maximum as f64,
144                y.minimum as f64,
145                y.maximum as f64,
146            ))
147        } else {
148            None
149        }
150    }
151
152    /// Fetches all pending input events converted to [`RawEvent`]s.
153    ///
154    /// On `SYN_DROPPED` (the kernel's resync signal: the event queue
155    /// overflowed and terminations may have been lost) the kernel MT
156    /// slot state is queried via `EVIOCGMTSLOTS` and appended as an
157    /// authoritative resync batch: the marker followed by every open
158    /// slot's state and a `SYN_REPORT`. `TouchState` lifts every held
159    /// contact absent from that batch at frame end - stale slots die
160    /// where they were created, not by heuristic watchdog. When the
161    /// ioctl fails no marker is emitted (stale view beats a wipe).
162    pub fn fetch_events(&mut self) -> Result<Vec<RawEvent>, EngineError> {
163        let mut events = std::mem::take(&mut self.pending);
164        let mut dropped = false;
165        for event in self.device.fetch_events()? {
166            let raw = RawEvent::from(&event);
167            if raw.is_syn_dropped() {
168                dropped = true;
169                continue;
170            }
171            events.push(raw);
172        }
173        if dropped && let Ok(resync) = self.resync_events() {
174            events.extend(resync);
175        }
176        Ok(events)
177    }
178
179    /// Builds the authoritative resync batch from kernel MT slot state:
180    /// the `SYN_DROPPED` marker, `SLOT`/`TRACKING_ID`/`POSITION_X/Y`
181    /// for every kernel-open contact, then `SYN_REPORT`.
182    fn resync_events(&self) -> Result<Vec<RawEvent>, EngineError> {
183        let abs = self.device.get_abs_state()?;
184        let num_slots = abs[ABS_MT_SLOT.as_u16() as usize].maximum as usize + 1;
185        let tracking = self.mt_slots(ABS_MT_TRACKING_ID.as_u16(), num_slots)?;
186        let pos_x = self.mt_slots(ABS_MT_POSITION_X.as_u16(), num_slots)?;
187        let pos_y = self.mt_slots(ABS_MT_POSITION_Y.as_u16(), num_slots)?;
188
189        let mut events = vec![RawEvent::syn_dropped()];
190        for slot in 0..num_slots {
191            if tracking[slot] == -1 {
192                continue;
193            }
194            events.extend([
195                RawEvent::new(EventType::Abs, ABS_MT_SLOT, slot as i32),
196                RawEvent::new(EventType::Abs, ABS_MT_TRACKING_ID, tracking[slot]),
197                RawEvent::new(EventType::Abs, ABS_MT_POSITION_X, pos_x[slot]),
198                RawEvent::new(EventType::Abs, ABS_MT_POSITION_Y, pos_y[slot]),
199            ]);
200        }
201        events.push(RawEvent::syn_report());
202        Ok(events)
203    }
204
205    /// `EVIOCGMTSLOTS(code)` ioctl: current kernel values of one
206    /// `ABS_MT_*` axis across all slots - the resync source of truth.
207    fn mt_slots(&self, code: u16, num_slots: usize) -> Result<Vec<i32>, EngineError> {
208        let mut data = vec![0i32; num_slots + 1];
209        data[0] = i32::from(code);
210        // _IOC(_IOC_READ, 'E', 0x0a, 4 + 4 * num_slots)
211        let size = 4 + 4 * num_slots;
212        let request = 0x80000000_u64 | (u64::from(b'E') << 8) | 0x0A | ((size as u64) << 16);
213        let rc = unsafe { libc::ioctl(self.device.as_raw_fd(), request, data.as_mut_ptr()) };
214        if rc < 0 {
215            return Err(EngineError::Io(std::io::Error::last_os_error()));
216        }
217        data.remove(0);
218        Ok(data)
219    }
