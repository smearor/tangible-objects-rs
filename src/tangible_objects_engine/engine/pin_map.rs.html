1//! `PinMap` - a tracked tangible's persistent slot→reference-pin
2//! binding (ROTATION.md). Solved once at spawn, re-solved only when
3//! the genuine slot set changes; claimed pins follow their claim
4//! frame by frame.
5
6use std::collections::HashMap;
7
8use tangible_objects_model::CombinationBuf;
9use tangible_objects_model::PatternMatch;
10use tangible_objects_model::RefIndex;
11use tangible_objects_model::RefPoints;
12use tangible_objects_model::RotationRad;
13use tangible_objects_model::SlotSet;
14use tangible_objects_model::TouchPointSlot;
15use tangible_objects_model::TouchSet;

17/// Slot→pin identity of one tracked tangible, plus the pattern's
18/// reference geometry (stored at spawn - no pattern lookup needed on
19/// the incumbent path).
20pub(crate) struct PinMap {
21    /// The pattern's reference pin positions.
22    refs: RefPoints,
23    /// Genuinely matched slots → reference pin index. Rewritten only
24    /// on spawn and when the genuine slot set changed (map-refresh
25    /// rule, ROTATION.md).
26    genuine: HashMap<TouchPointSlot, RefIndex>,
27    /// Completion-claimed slots → the pin they currently fill.
28    /// Rebuilt from `claimed_indices` every frame - a wandering claim
29    /// follows its pair without re-solving.
30    claimed: HashMap<TouchPointSlot, RefIndex>,
31}
32
33impl PinMap {
34    /// Builds the map from a solved match: `correspondence` binds the
35    /// genuine slots, `claimed_indices` the claimed ones. Empty maps
36    /// when the match carries no correspondence/ref_points.
37    pub(crate) fn from_match(pattern_match: &PatternMatch, touches: TouchSet<'_>) -> Self {
38        let mut map = Self {
39            refs: pattern_match.ref_points.clone().unwrap_or_default(),
40            genuine: pattern_match.genuine_map(touches),
41            claimed: HashMap::new(),
42        };
43        map.update_claimed(pattern_match, touches);
44        map
45    }
46
47    /// Rebuilds the claimed half from the match's `Claim`s
48    /// (claim churn follows the current frame's bindings).
49    pub(crate) fn update_claimed(&mut self, pattern_match: &PatternMatch, touches: TouchSet<'_>) {
50        self.claimed = pattern_match
51            .claimed_indices
52            .as_slice()
53            .iter()
54            .map(|claim| (touches[claim.touch_index].slot, claim.ref_index))
55            .collect();
56    }
57
58    /// Rewrites the genuine half from a fresh solved match
59    /// (map-refresh rule: genuine slots changed → re-bind).
60    pub(crate) fn refresh_genuine(&mut self, pattern_match: &PatternMatch, touches: TouchSet<'_>) {
61        self.refs = pattern_match.ref_points.clone().unwrap_or_default();
62        self.genuine = pattern_match.genuine_map(touches);
63    }
64
65    /// The slots the genuine half binds - compared to
66    /// `matched_slots` for the refresh rule.
67    pub(crate) fn genuine_slots(&self) -> SlotSet {
68        self.genuine.keys().copied().collect()
69    }
70
71    /// Rotation of the mapped slots' current positions against the
72    /// stored reference frame - `similarity_transform` over all
73    /// bound slots present in `touches`. `None` when the match carried
74    /// no ref geometry or fewer than two bound slots are live (keep
75    /// the last good angle then).
76    pub(crate) fn rotation(&self, touches: TouchSet<'_>) -> Option<RotationRad> {
77        if self.refs.len() < 2 {
78            return None;
79        }
80        let (measured, mapping): (Vec<_>, CombinationBuf) = self
81            .genuine
82            .iter()
83            .chain(&self.claimed)
84            .filter_map(|(slot, ref_index)| {
85                touches
86                    .iter()
87                    .find(|t| t.slot == *slot)
88                    .map(|touch| (touch.pos, ref_index.as_usize()))
89            })
90            .unzip();
91        self.refs
92            .similarity_transform((&measured).into(), mapping.as_combination())
93            .map(|transform| transform.angle)
94    }
95}

112    /// Touch points `slot`, `(x, y)` - a fixed square under `angle`.
113    fn touches(angle: f64) -> Vec<TouchPoint> {
114        let (sin, cos) = angle.sin_cos();
115        [(0.0, 0.0), (40.0, 0.0), (40.0, 40.0), (0.0, 40.0)]
116            .iter()
117            .enumerate()
118            .map(|(i, &(x, y))| {
119                let offset = Vec2::<Millimeters>::new(100.0, 100.0);
120                let point = Vec2::<Millimeters>::new(x, y);
121                TouchPoint {
122                    slot: TouchPointSlot::new(i as u32),
123                    tracking_id: TrackingId::new(i as i32),
124                    pos: offset + point * cos + point.perpendicular() * sin,
125                }
126            })
127            .collect()
128    }
129
130    fn solved_match(refs: &[Vec2<Millimeters>]) -> PatternMatch {
131        PatternMatch {
132            pattern_id: PatternId::new(1),
133            pattern_name: "quad".into(),
134            center: Vec2::new(0.0, 0.0),
135            rotation_rad: RotationRad::new(0.0),
136            touch_indices: CombinationBuf::from(vec![0, 1, 2, 3]),
137            correspondence: vec![0.into(), 1.into(), 2.into(), 3.into()].into(),
138            ref_points: Some(refs.to_vec().into()),
139            partial: false,
140            claimed_indices: ClaimedIndices::default(),
141            spawn_stability: FrameCount::new(15),
142        }
143    }
144
145    /// The persisted map reproduces the planted rotation of the
146    /// bound slots (ROTATION.md testing section).
147    #[test]
148    fn rotation_reads_planted_angle() {
149        let refs = vec![
150            Vec2::new(0.0, 0.0),
151            Vec2::new(40.0, 0.0),
152            Vec2::new(40.0, 40.0),
153            Vec2::new(0.0, 40.0),
154        ];
155        let angle = 0.7;
156        let touch_points = touches(angle);
157        let map = PinMap::from_match(&solved_match(&refs), (&touch_points).into());
158        let rotation = map
159            .rotation((&touch_points).into())
160            .expect("identity-mapped square solves");
161        assert!(
162            (rotation.as_f64() - angle).abs() < 0.05,
163            "map rotation {}, planted {angle}",
164            rotation.as_f64()
165        );
166    }
167
168    /// Fewer than two bound slots gives no angle - the caller keeps
169    /// the last good value then.
170    #[test]
171    fn rotation_none_under_two_slots() {
172        let refs = vec![
173            Vec2::new(0.0, 0.0),
174            Vec2::new(40.0, 0.0),
175            Vec2::new(40.0, 40.0),
176            Vec2::new(0.0, 40.0),
177        ];
178        let mut match_ = solved_match(&refs);
179        match_.touch_indices = CombinationBuf::from(vec![0]);
180        match_.correspondence = vec![0.into()].into();
181        let touch_points = touches(0.0);
182        let map = PinMap::from_match(&match_, (&touch_points).into());
183        assert_eq!(map.genuine_slots().len(), 1);
184        assert!(map.rotation((&touch_points).into()).is_none());
185    }
