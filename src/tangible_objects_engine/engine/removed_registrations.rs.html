1//! `RemovedRegistrations` - what `Engine::remove_by_owner` tore
2//! down, so the caller can emit lifecycle events and unpersist the
3//! right store files.
4
5use tangible_objects_model::DeadZoneName;
6use tangible_objects_model::FieldSetName;

11/// The registrations removed for one owner (`UnregisterOwner` /
12/// D-Bus disconnect cleanup).
13pub struct RemovedRegistrations {
14    /// Removed patterns `(id, name)` - the name is needed to
15    /// unpersist `profiles.d`.
16    pub patterns: Vec<(PatternId, PatternName)>,
17    /// Removed field sets by name.
18    pub field_sets: Vec<FieldSetName>,
19    /// Removed dead zones by name.
20    pub dead_zones: Vec<DeadZoneName>,
21    /// Removed spells by name.
22    pub spells: Vec<SpellName>,
23}
