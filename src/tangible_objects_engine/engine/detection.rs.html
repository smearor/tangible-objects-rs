64/// Headless detection engine: slot state machine + pattern matching +
65/// frame-to-frame tracking.
66///
67/// Feed [`RawEvent`]s via [`Engine::process`]; every `SYN_REPORT` produces an
68/// [`EngineFrame`] with the frame's matches, touches and domain events, and
69/// updates the tracker.
70pub struct Engine {
71    /// Slot state machine: accumulates `ABS_MT_*` events into active
72    /// contacts between `SYN_REPORT` boundaries.
73    touch_state: TouchState,
74    /// Loaded figure patterns evaluated against every frame's touches.
75    patterns: PatternBuf,
76    /// Frame-to-frame tracker: nearest-neighbor association, EMA
77    /// smoothing, slot bindings and expiry hysteresis.
78    tracker: Tracker,
79    /// Occupancy tracking for single-contact field sets.
80    fields: FieldTracker,
81    /// Per-slot gesture trajectories and spell dispatch (SPELLS.md).
82    gestures: GestureTracker,
83    /// Raw-to-millimeter conversion derived from the device's `ABS_INFO`,
84    /// possibly overridden by the device profile. `None` on replays
85    /// without an evtest header.
86    calibration: Option<Calibration>,
87    /// Missing-pin completion configuration (claim radius, max missing
88    /// pins, persistence, suppression mode).
89    completion: CompletionConfig,
90    /// Next free pattern ID for dynamic registrations - monotonic, freed
91    /// IDs are never reused (DYNAMIC_PROFILES.md decision).
92    next_pattern_id: PatternId,
93    /// Owner (registering application) per dynamically added pattern.
94    pattern_owners: HashMap<PatternId, Option<Owner>>,
95    /// Owner per dynamically added field set, keyed by set name.
96    field_set_owners: HashMap<FieldSetName, Option<Owner>>,
97    /// Blacklisted surface positions - contacts inside are dropped at
98    /// ingest like `edge_margin` ghosts.
99    dead_zones: DeadZones,
100    /// Owner per dynamically added dead zone, keyed by zone name.
101    dead_zone_owners: HashMap<DeadZoneName, Option<Owner>>,
102    /// Owner per dynamically registered spell, keyed by spell name.
103    spell_owners: HashMap<SpellName, Option<Owner>>,
104    /// Touches dropped by dead-zone filtering on the last processed
105    /// frame - the ingest-level suppression counter.
106    dead_zone_filtered: ContactCount,
107    /// Touches dropped by edge-margin filtering on the last processed
108    /// frame - ghost contacts near the device edges.
109    edge_filtered: ContactCount,
110    /// Active contacts before ingest filtering on the last processed
111    /// frame - edge + dead-zone + surviving touches.
112    incoming_contacts: ContactCount,
113    /// Previous frame's touches - used to emit `TouchPointsUpdated` only
114    /// on actual change.
115    prev_touches: Vec<TouchPoint>,
116    /// Edge margin in millimeters — touches inside it are treated as ghost
117    /// contacts and excluded from the frame before matching.
118    edge_margin: Millimeters,
119    /// Last smoothed position per slot, keyed with the tracking ID so a
120    /// new contact on the same slot restarts at its raw position.
121    smoothed: SmoothedContacts,
122    /// A matching-relevant input changed outside the touch set
123    /// (SKIP_UNCHANGED.md): pattern registration, completion config or
124    /// edge margin. Consumed by one generic pass in `process_inner` so
125    /// a changed configuration re-runs matching even on a still board.
126    matching_dirty: bool,
127    /// Touches as of the last generic matching pass (SKIP_UNCHANGED.md
128    /// jitter fallback): the gate compares against this baseline, not
129    /// the previous frame - cumulative drift beyond
130    /// [`GENERIC_MIN_MOVEMENT`] still re-runs matching, while per-frame
131    /// panel noise below the threshold keeps it closed.
132    generic_touches: Vec<TouchPoint>,
133}
134
135/// Movement per contact that counts as "changed" for the generic
136/// matching gate (SKIP_UNCHANGED.md): above typical panel position
137/// jitter, far below meaningful figure movement (tolerances are 3-5 mm).
138/// Tune if the target panel jitters harder.
139const GENERIC_MIN_MOVEMENT: Millimeters = Millimeters::new(1.0);

182    /// Registers a spell at runtime (`Register`/`Registration::Spell`).
183    ///
184    /// Rejects an `exclusive-zone` spell whose region overlaps a
185    /// field - the zone reserves its area for gestures while the
186    /// field expects occupancy (SPELLS.md's zone/field exclusion).
187    pub fn add_spell(
188        &mut self,
189        spell: Spell,
190        owner: Option<Owner>,
191    ) -> Result<(), RegistrationError> {
192        if let SpellArming::ExclusiveZone { region } = &spell.arming {
193            for set in self.fields.sets() {
194                for field in &set.fields {
195                    if region.overlaps(&field.region) {
196                        return Err(RegistrationError::SpellZoneOverlapsField(
197                            spell.name.clone(),
198                            field.name.clone(),
199                        ));
200                    }
201                }
202            }
203        }
204        self.spell_owners.insert(spell.name.clone(), owner);
205        self.gestures.add_spell(spell);
206        Ok(())
207    }

301    /// Registers a pattern at runtime (DYNAMIC_PROFILES.md).
302    ///
303    /// The new pattern gets a fresh monotonic [`PatternId`]; its
304    /// fingerprint is collision-checked against every existing pattern -
305    /// on collision the registration is rejected and nothing changes.
306    /// Returns the assigned `PatternId`.
307    pub fn add_pattern(
308        &mut self,
309        profile: &ProfileFile,
310        owner: Option<Owner>,
311    ) -> Result<PatternId, RegistrationError> {
312        self.check_pattern_collision(profile, None)?;
313        let id = self.next_pattern_id;
314        self.next_pattern_id = id.next();
315        self.pattern_owners.insert(id, owner);
316        self.patterns.push(profile.to_pattern(id));
317        self.matching_dirty = true;
318        Ok(id)
319    }
320
321    /// Replaces a registered pattern's geometry/name while keeping its
322    /// ID (`UpdateProfile`). The collision check excludes the pattern
323    /// itself; already-tracked tangibles keep their recorded pose - the
324    /// update applies to new matches.
325    pub fn update_pattern(
326        &mut self,
327        id: PatternId,
328        profile: &ProfileFile,
329    ) -> Result<(), RegistrationError> {
330        if self.patterns.by_id(id).is_none() {
331            return Err(RegistrationError::PatternNotFound(id));
332        }
333        self.check_pattern_collision(profile, Some(id))?;
334        if let Some(pattern) = self.patterns.by_id_mut(id) {
335            *pattern = profile.to_pattern(id);
336            self.matching_dirty = true;
337        }
338        Ok(())
339    }
340
341    /// Fingerprint check of `profile` against all patterns except
342    /// `except` (used by add/update).
343    fn check_pattern_collision(
344        &self,
345        profile: &ProfileFile,
346        except: Option<PatternId>,
347    ) -> Result<(), RegistrationError> {
348        let pattern = profile.to_pattern(PatternId::new(0));
349        if let Some(points) = pattern.ref_points.as_deref()
350            && let Some(existing) = self.patterns.find_matching(points.into(), except)
351        {
352            return Err(RegistrationError::FingerprintCollision(
353                existing.name.clone(),
354            ));
355        }
356        Ok(())
357    }

359    /// Removes a pattern by ID (dynamic unregister). Its ID is never
360    /// reused. Returns `true` when a pattern was removed.
361    pub fn remove_pattern(&mut self, id: PatternId) -> bool {
362        self.pattern_owners.remove(&id);
363        let before = self.patterns.len();
364        self.patterns.retain(|pattern| pattern.id != id);
365        if self.patterns.len() != before {
366            self.matching_dirty = true;
367            true
368        } else {
369            false
370        }
371    }

431    /// Removes every pattern, field set and dead zone owned by `owner`
432    /// (disconnect cleanup). Returns `(id, name)` pairs for patterns
433    /// and names for field sets and dead zones - names let the caller
434    /// remove persisted files too.
435    pub fn remove_by_owner(&mut self, owner: &Owner) -> RemovedRegistrations {
436        let patterns: Vec<(PatternId, PatternName)> = self
437            .pattern_owners
438            .iter()
439            .filter(|(_, o)| o.as_ref() == Some(owner))
440            .filter_map(|(id, _)| {
441                self.patterns
442                    .by_id(*id)
443                    .map(|pattern| (*id, pattern.name.clone()))
444            })
445            .collect();
446        for (id, _) in &patterns {
447            self.remove_pattern(*id);
448        }
449        let set_names: Vec<FieldSetName> = self
450            .field_set_owners
451            .iter()
452            .filter(|(_, o)| o.as_ref() == Some(owner))
453            .map(|(name, _)| name.clone())
454            .collect();
455        for name in &set_names {
456            self.remove_field_set(name);
457        }
458        let zone_names: Vec<DeadZoneName> = self
459            .dead_zone_owners
460            .iter()
461            .filter(|(_, o)| o.as_ref() == Some(owner))
462            .map(|(name, _)| name.clone())
463            .collect();
464        for name in &zone_names {
465            self.remove_dead_zone(name);
466        }
467        let spell_names: Vec<SpellName> = self
468            .spell_owners
469            .iter()
470            .filter(|(_, o)| o.as_ref() == Some(owner))
471            .map(|(name, _)| name.clone())
472            .collect();
473        for name in &spell_names {
474            self.remove_spell(name);
475        }
476        RemovedRegistrations {
477            patterns,
478            field_sets: set_names,
479            dead_zones: zone_names,
480            spells: spell_names,
481        }
482    }

484    /// Adds an occupancy field set at runtime; rejected on overlapping
485    /// fields or duplicate set name.
486    pub fn add_field_set(
487        &mut self,
488        set: FieldSet,
489        owner: Option<Owner>,
490    ) -> Result<(), RegistrationError> {
491        if let Some((a, b, overlap)) = set.overlap_pair() {
492            return Err(RegistrationError::FieldsOverlap(
493                set.name.clone(),
494                a.name.clone(),
495                b.name.clone(),
496                overlap,
497            ));
498        }
499        if self
500            .fields
501            .sets()
502            .iter()
503            .any(|existing| existing.name == set.name)
504        {
505            return Err(RegistrationError::DuplicateFieldSet(set.name.clone()));
506        }
507        self.field_set_owners.insert(set.name.clone(), owner);
508        self.fields.add_set(set);
509        Ok(())
510    }

601    /// Sets the edge margin: touches closer than `margin` to the surface
602    /// bounds are dropped before matching (ghost contacts reported by some
603    /// panels at `x=0`/`y=0` or the max bounds). `0` disables the filter.
604    /// Needs a calibration — without mm bounds there is nothing to filter
605    /// against.
606    /// EMA-smooths every touch position per slot
607    /// (`CompletionConfig::smoothing_alpha`). A contact whose tracking ID
608    /// changed restarts at its raw position; dead slots are dropped so
609    /// stale smoothed state can never leak into a new contact.
610    fn smooth_touches(&mut self, touches: &mut [TouchPoint]) {
611        self.smoothed.step(self.completion.smoothing_alpha, touches);
612    }

644    /// Finds all pattern matches in a given set of touch points without
645    /// touching engine state.
646    pub fn find_matches(&self, touches: TouchSet<'_>) -> Vec<PatternMatch> {
647        self.patterns().find_matches(touches)
648    }

650    /// Feeds one raw event into the engine.
651    ///
652    /// Returns `Some` with the frame's matches, touches and domain events on
653    /// each `SYN_REPORT`, `None` for all other events.
654    ///
655    /// This is the production path: no candidate diagnostics are emitted and
656    /// [`EngineFrame::candidates`] stays empty.
657    pub fn process(&mut self, event: &RawEvent) -> Option<EngineFrame> {
658        self.process_inner(event, None)
659    }
660
661    /// Like [`Engine::process`], additionally collecting a
662    /// [`CandidateReport`] for every tested point combination - matched or
663    /// rejected - into [`EngineFrame::candidates`].
664    ///
665    /// Used by the recording pipeline (`--record`); allocation happens only
666    /// on this path.
667    pub fn process_with_diagnostics(&mut self, event: &RawEvent) -> Option<EngineFrame> {
668        self.process_inner(event, Some(&mut Vec::new()))
669    }
670
671    fn process_inner(
672        &mut self,
673        event: &RawEvent,
674        mut diagnostics: Option<&mut Vec<CandidateReport>>,
675    ) -> Option<EngineFrame> {
676        self.touch_state.process(event);
677        if !event.is_syn_report() {
678            return None;
679        }
680        let mut touches = self.touch_state.active_touches();
681        touches.sort_by_key(|touch| touch.slot);
682        self.incoming_contacts = touches.len().into();
683        let before_edge = touches.len();
684        touches.retain(|touch| !self.is_edge_ghost(touch));
685        self.edge_filtered = (before_edge - touches.len()).into();
686        let before_dead_zones = touches.len();
687        touches.retain(|touch| !self.dead_zones.contains(&touch.pos));
688        self.dead_zone_filtered = (before_dead_zones - touches.len()).into();
689        self.smooth_touches(&mut touches);
690        let spell_matches: Vec<SpellMatched> = self
691            .gestures
692            .observe_frame(&touches, self.tracker.active().0, std::time::Instant::now())
693            .into_iter()
694            .collect();
695        // Compared before matching (SKIP_UNCHANGED.md): the generic pass
696        // is a pure function of the smoothed touch set, so it only runs
697        // when it can find something new - changed contacts, a pending
698        // spawn still collecting stability frames, or a changed matcher
699        // input (`matching_dirty`). `touches_changed` is the exact diff
700        // (`TouchPointsUpdated` semantics); the generic gate uses the
701        // jitter-tolerant `significantly_changed` against the last
702        // evaluated baseline instead.
703        let touches_changed = touches != self.prev_touches;
704
705        // Slot affinity (TRACKING_STABILIZATION.md): every tracked
706        // tangible re-matches on the slots it owned in the previous
707        // frame, before generic matching runs.
708        //
709        // Pass 1 reserves *all* bound slots first - otherwise an
710        // incumbent processed early could complete-claim slots that a
711        // later incumbent owns (the claim pool must exclude every
712        // tracked tangible's bindings, not just the ones already seen).
713        let mut matches = Vec::new();
714        let mut reserved = TakenIndices::default();
715        for tangible in self.tracker.active() {
716            if let Some(slot_set) = self.tracker.slot_set(tangible.id) {
717                reserved.extend(
718                    touches
719                        .iter()
720                        .enumerate()
721                        .filter(|(_, touch)| slot_set.contains(&touch.slot))
722                        .map(|(index, _)| index),
723                );
724            }
725        }
726        // SPELLS.md's exclusive spell zones: contacts inside a
727        // registered zone reserve the area for gestures and never
728        // form NEW matches. The incumbent pass below is unaffected -
729        // a tracked tangible may be dragged through the zone.
730        let zones: Vec<Region> = self.spell_zones().collect();
731        if !zones.is_empty() {
732            reserved.extend(
733                touches
734                    .iter()
735                    .enumerate()
736                    .filter(|(_, touch)| zones.iter().any(|z| z.contains(&touch.pos)))
737                    .map(|(index, _)| index),
738            );
739        }
740
741        // Pass 2: incumbent re-match on the genuinely matched slots only
742        // (`match_set` excludes claimed pins - a wrong claim stays
743        // reserved but can never make the incumbent fail to match).
744        let touch_set: TouchSet = (&touches).into();
745        for tangible in self.tracker.active() {
746            let Some(pattern) = self.patterns.by_id(tangible.pattern_id) else {
747                continue;
748            };
749            // Disabled patterns stop confirming their incumbents - the
750            // tracker drops the track as with any missed re-match.
751            if !pattern.enabled {
752                continue;
753            }
754            let Some(match_set) = self.tracker.match_set(tangible.id) else {
755                continue;
756            };
757            let indices = touch_set.indices_of_slots(match_set);
758            let affinity_match = if indices.len() == pattern.num_points {
759                pattern.try_exact(indices.as_combination(), touch_set, &self.completion)
760            } else if indices.len() + self.completion.max_missing as usize >= pattern.num_points
761                && pattern.num_points >= 4
762            {
763                // Claim pool: every non-reserved touch PLUS this
764                // incumbent's own bound-but-unmatched slots - a slot
765                // this figure claimed last frame must stay claimable
766                // for it (reserving it from others must not unclaim it
767                // for its owner, otherwise the binding would oscillate).
768                let own_reserved: TakenIndices = self
769                    .tracker
770                    .slot_set(tangible.id)
771                    .map(|slot_set| {
772                        touch_set
773                            .indices_of_slots(slot_set)
774                            .iter()
775                            .copied()
776                            .collect()
777                    })
778                    .unwrap_or_default();
779                let pool: CandidateIndices = touches
780                    .iter()
781                    .enumerate()
782                    .filter(|(index, _)| !reserved.contains(index) || own_reserved.contains(index))
783                    .map(|(index, _)| index)
784                    .collect();
785                pattern.try_partial(
786                    indices.as_combination(),
787                    touch_set,
788                    &self.completion,
789                    pool.as_claim_pool(),
790                )
791            } else {
792                None
793            };
794            if let Some(pattern_match) = affinity_match {
795                matches.push(pattern_match);
796            }
797        }
798
799        let touches_moved = Self::significantly_changed(&self.generic_touches, &touches);
800        // A frame whose contacts are all reserved for incumbent tracks
801        // holds no unreserved candidate - the generic pass is provably
802        // empty, so movement of a fully-bound set does not re-trigger
803        // matching.
804        let has_unreserved = touches.len() > reserved.len();
805        let ran_generic =
806            self.matching_dirty || self.tracker.has_pending() || (touches_moved && has_unreserved);
807        let mut candidates_tested = CandidateCount::default();
808        if ran_generic {
809            let generic: Vec<PatternMatch> = match diagnostics.as_mut() {
810                Some(out) => {
811                    let generic = self.patterns().find_matches_with_diagnostics(
812                        touch_set,
813                        &mut **out,
814                        &reserved,
815                        &self.completion,
816                    );
817                    candidates_tested = CandidateCount::from(out.len() as u64);
818                    generic
819                }
820                None => {
821                    let mut tested = 0usize;
822                    let generic = self.patterns().find_matches_excluding_counted(
823                        touch_set,
824                        &reserved,
825                        &self.completion,
826                        &mut tested,
827                    );
828                    candidates_tested = CandidateCount::from(tested as u64);
829                    generic
830                }
831            };
832            matches.extend(generic);
833            self.generic_touches.clone_from(&touches);
834        } else if touches_moved {
835            // Skipped frame: keep the gate baseline current so the next
836            // frame does not re-diff against a stale snapshot.
837            self.generic_touches.clone_from(&touches);
838        }
839        self.matching_dirty = false;
840
841        let mut events = self
842            .tracker
843            .update(&matches, touch_set, self.completion.persist);
844        events.extend(self.fields.update(touch_set));
845        if touches_changed {
846            self.prev_touches.clone_from(&touches);
847            events.push(EngineEvent::TouchPointsUpdated);
848        }
849        debug!(
850            matches = matches.len(),
851            tracked = self.tracker.active().len(),
852            "frame processed"
853        );
854        Some(EngineFrame {
855            matches,
856            touches,
857            events,
858            candidates: diagnostics.map(std::mem::take).unwrap_or_default(),
859            ran_generic,
860            candidates_tested,
861            dead_zone_filtered: self.dead_zone_filtered,
862            spell_matches,
863            gesture_slots: self.gestures.gesture_slots(std::time::Instant::now()),
864            consumed_slots: self.gestures.take_consumed(),
865        })
866    }
867
868    /// Jitter-tolerant change detector for the generic gate
869    /// (SKIP_UNCHANGED.md): `true` when set sizes, slots, tracking IDs
870    /// or any position's movement since `prev` (the last evaluated
871    /// baseline, not the last frame) crosses [`GENERIC_MIN_MOVEMENT`].
872    /// Comparing against the baseline keeps slow cumulative drift
873    /// visible - per-frame noise below the threshold never opens the
874    /// gate, a slowly sliding figure re-triggers it once the total
875    /// movement passes ε.
876    fn significantly_changed(prev: &[TouchPoint], next: &[TouchPoint]) -> bool {
877        if prev.len() != next.len() {
878            return true;
879        }
880        prev.iter().zip(next.iter()).any(|(a, b)| {
881            a.slot != b.slot
882                || a.tracking_id != b.tracking_id
883                || a.pos.distance(&b.pos) >= GENERIC_MIN_MOVEMENT
884        })
885    }
886
887    /// Whether a touch sits inside the configured edge margin — a likely
888    /// ghost contact some panels emit at the axis bounds.
889    ///
890    /// Disabled when the margin is zero or no calibration is loaded (the mm
891    /// surface bounds are unknowable then).
892    fn is_edge_ghost(&self, touch: &TouchPoint) -> bool {
893        let Some(cal) = &self.calibration else {
894            return false;
895        };
896        let margin = self.edge_margin.as_f64();
897        if margin <= 0.0 {
898            return false;
899        }
900        let (x0, x1) = (
901            cal.x_min.as_f64() / cal.x_res.as_f64(),
902            cal.x_max.as_f64() / cal.x_res.as_f64(),
903        );
904        let (y0, y1) = (
905            cal.y_min.as_f64() / cal.y_res.as_f64(),
906            cal.y_max.as_f64() / cal.y_res.as_f64(),
907        );
908        touch.pos.x < x0 + margin
909            || touch.pos.x > x1 - margin
910            || touch.pos.y < y0 + margin
911            || touch.pos.y > y1 - margin
912    }
913}
