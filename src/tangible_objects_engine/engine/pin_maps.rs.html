12/// Persistent slot→pin bindings of all tracked tangibles
13/// (ROTATION.md): each entry is solved once at spawn, reused for
14/// rotation on every incumbent frame instead of the argmax
15/// `rotation_of` estimate.
16///
17/// The map's lifecycle is the tracker lifecycle - spawn inserts,
18/// refresh re-binds only when the genuine slot set changed, expiry
19/// removes.
20#[derive(Default)]
21pub(crate) struct PinMaps(pub HashMap<TangibleId, PinMap>);
22
23impl PinMaps {
24    /// Correspondence-fitted rotation of the tangible's bound slots -
25    /// `None` when no map exists or fewer than two bound slots are
26    /// live (the caller keeps the match's own angle then).
27    pub fn rotation_of(&self, id: TangibleId, touches: TouchSet<'_>) -> Option<RotationRad> {
28        self.0.get(&id).and_then(|map| map.rotation(touches))
29    }
30
31    /// Syncs the tangible's binding to the frame's match: inserts a
32    /// fresh binding when none exists, re-binds the genuine slots
33    /// when the genuine slot set changed (mass slot churn) - claims
34    /// always follow their current pairs inside `refresh`.
35    pub fn sync(&mut self, id: TangibleId, pattern_match: &PatternMatch, touches: TouchSet<'_>) {
36        let map = self
37            .0
38            .entry(id)
39            .or_insert_with(|| PinMap::from_match(pattern_match, touches));
40        if map.genuine_slots() != pattern_match.matched_slots(touches)
41            && !pattern_match.correspondence.is_empty()
42        {
43            map.refresh_genuine(pattern_match, touches);
44        }
45        map.update_claimed(pattern_match, touches);
46    }
47}
48
49impl Deref for PinMaps {
50    type Target = HashMap<TangibleId, PinMap>;
51
52    /// Maps deref to the underlying `HashMap` - `remove` on expiry.
53    fn deref(&self) -> &HashMap<TangibleId, PinMap> {
54        &self.0
55    }
56}
57
58impl DerefMut for PinMaps {
59    /// Mutable access to the underlying `HashMap`.
60    fn deref_mut(&mut self) -> &mut HashMap<TangibleId, PinMap> {
61        &mut self.0
62    }
63}

82    fn square() -> Vec<Vec2<Millimeters>> {
83        vec![
84            Vec2::new(0.0, 0.0),
85            Vec2::new(40.0, 0.0),
86            Vec2::new(40.0, 40.0),
87            Vec2::new(0.0, 40.0),
88        ]
89    }
90
91    fn touches(angle: f64, slots: &[u32]) -> Vec<TouchPoint> {
92        let rotation = RotationRad::new(angle);
93        square()
94            .iter()
95            .zip(slots)
96            .map(|(p, &slot)| {
97                let offset = Vec2::<Millimeters>::new(100.0, 100.0);
98                TouchPoint {
99                    slot: TouchPointSlot::new(slot),
100                    tracking_id: TrackingId::new(slot as i32),
101                    pos: offset + p.rotated(rotation),
102                }
103            })
104            .collect()
105    }
106
107    fn solved_match(refs: &[Vec2<Millimeters>], touch_indices: Vec<usize>) -> PatternMatch {
108        PatternMatch {
109            pattern_id: PatternId::new(1),
110            pattern_name: "quad".into(),
111            center: Vec2::new(0.0, 0.0),
112            rotation_rad: RotationRad::new(0.0),
113            touch_indices: CombinationBuf::from(touch_indices),
114            correspondence: RefIndices::all(refs.len().into()),
115            ref_points: Some(refs.to_vec().into()),
116            partial: false,
117            claimed_indices: ClaimedIndices::default(),
118            spawn_stability: FrameCount::new(15),
119        }
120    }
121
122    /// First `sync` inserts a fresh binding under the tangible's id.
123    #[test]
124    fn sync_inserts_on_first_match() {
125        let mut maps = PinMaps::default();
126        let id = TangibleId::new(1);
127        let touch_set = touches(0.0, &[0, 1, 2, 3]);
128        let pattern_match = solved_match(&square(), vec![0, 1, 2, 3]);
129        maps.sync(id, &pattern_match, (&touch_set).into());
130        assert_eq!(maps.get(&id).map(|m| m.genuine_slots().len()), Some(4));
131    }
132
133    /// `rotation_of` yields `None` for an unregistered tangible.
134    #[test]
135    fn rotation_of_none_for_unknown_tangible() {
136        let maps = PinMaps::default();
137        let touch_set = touches(0.0, &[0, 1, 2, 3]);
138        assert!(
139            maps.rotation_of(TangibleId::new(9), (&touch_set).into())
140                .is_none()
141        );
142    }

144    /// After `sync`, `rotation_of` fits the bound slots' planted
145    /// rotation back out of the persisted map.
146    #[test]
147    fn rotation_of_reads_planted_angle() {
148        let mut maps = PinMaps::default();
149        let id = TangibleId::new(1);
150        let spawn = touches(0.0, &[0, 1, 2, 3]);
151        let pattern_match = solved_match(&square(), vec![0, 1, 2, 3]);
152        maps.sync(id, &pattern_match, (&spawn).into());
153        let moved = touches(0.3, &[0, 1, 2, 3]);
154        let rotation = maps.rotation_of(id, (&moved).into());
155        assert!(
156            rotation.is_some_and(|r| (r.as_f64() - 0.3).abs() < 0.05),
157            "synced map must recover the planted rotation"
158        );
159    }
160
161    /// A match whose genuine slots changed re-binds the map -
162    /// `sync` runs `refresh_genuine`, not just `update_claimed`.
163    #[test]
164    fn sync_rebinds_when_genuine_slots_change() {
165        let mut maps = PinMaps::default();
166        let id = TangibleId::new(1);
167        let spawn = touches(0.0, &[0, 1, 2, 3]);
168        let first = solved_match(&square(), vec![0, 1, 2, 3]);
169        maps.sync(id, &first, (&spawn).into());
170        // Same pattern re-bound on four different slots - the touch
171        // indices stay positional; the slots behind them changed.
172        let rebound = touches(0.0, &[4, 5, 6, 7]);
173        let second = solved_match(&square(), vec![0, 1, 2, 3]);
174        maps.sync(id, &second, (&rebound).into());
175        let bound = maps.get(&id).map(|m| m.genuine_slots()).unwrap_or_default();
176        assert!(bound.contains(&TouchPointSlot::new(4)));
177        assert!(!bound.contains(&TouchPointSlot::new(0)));
178    }
