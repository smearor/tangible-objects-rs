1//! `Episode` - the facts a completed gesture episode carries into
2//! arming evaluation.
3
4use smearor_commons_geometry::Millimeters;
5use smearor_commons_geometry::Vec2;
6use tangible_objects_model::HandName;
7use tangible_objects_model::Region;
8use tangible_objects_model::TouchPointSlot;
9
10/// Whether an episode satisfied a spell's arming condition - the
11/// `Listening -> Armed` transition's per-episode evaluation.
12pub(super) struct Episode<'a> {
13    /// Completed strokes of the episode with their slots - the
14    /// slots identify what a matched spell consumed.
15    pub(super) strokes: &'a [(TouchPointSlot, Vec<Vec2<Millimeters>>)],
16    /// The episode's very first contact point - `start-spot` checks
17    /// this against its region.
18    pub(super) first_point: Option<Vec2<Millimeters>>,
19    /// `five-finger-tap` latch active during the episode.
20    pub(super) five_tap_armed: bool,
21    /// The hand the tap latch identified - a `five-finger-tap` spell
22    /// naming a `hand` only arms when the tap's constellation
23    /// matched that handprint (per-user spells, SPELLS.md).
24    pub(super) armed_hand: Option<HandName>,
25}
26
27impl Episode<'_> {
28    /// Stroke points without slot identity - the matcher input.
29    pub(super) fn polylines(&self) -> Vec<Vec<Vec2<Millimeters>>> {
30        self.strokes
31            .iter()
32            .map(|(_, points)| points.clone())
33            .collect()
34    }
35
36    /// `exclusive-zone` arming: every stroke point must lie inside
37    /// the region - leaving the zone fails the episode.
38    pub(super) fn all_points_inside(&self, region: &Region) -> bool {
39        self.strokes
40            .iter()
41            .all(|(_, stroke)| stroke.iter().all(|point| region.contains(point)))
42    }
43}
