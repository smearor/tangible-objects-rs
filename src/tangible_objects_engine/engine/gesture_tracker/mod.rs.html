1//! `GestureTracker` - accumulates per-slot trajectories and
2//! dispatches completed strokes against registered spells
3//! (SPELLS.md's recognition core).
4//!
5//! Arming (SPELLS.md phase model): a spell only matches when its
6//! declared arming condition was satisfied for the episode:
7//!
8//! - `free` - always armed (legacy/freeform gestures).
9//! - `exclusive-zone` - every stroke point must stay inside the
10//!   zone's region.
11//! - `start-spot` - the episode's first contact must land inside
12//!   the spot's region.
13//! - `five-finger-tap` - a short simultaneous five-contact tap
14//!   latches arming for a window; the next episode may draw with
15//!   fewer fingers.
16//! - `tangible-chord` - a tracked tangible of the named pattern
17//!   acts as the wand; its center path is the stroke and lifting
18//!   the tangible completes the spell.
19

51/// Minimum stroke path length for a slot to count as a gesture
52/// candidate - below it a contact is a rest or a normal touch
53/// (SPELLS.md's temporal discriminator).
54const MIN_STROKE_PATH: Millimeters = Millimeters::new(20.0);
55
56/// Minimum distance between appended trajectory samples -
57/// dedupes the position stream so the polyline stays sparse.
58const SAMPLE_STEP: Millimeters = Millimeters::new(2.0);
59
60/// Maximum spread between the five contacts' down events for a
61/// five-finger tap to count as simultaneous (SPELLS.md).
62const TAP_SPREAD: Duration = Duration::from_millis(150);
63
64/// Maximum total duration of a five-finger tap (down to last lift).
65const TAP_TOTAL: Duration = Duration::from_millis(400);
66
67/// How long a completed five-finger tap keeps the arming latch.
68const TAP_LATCH: Duration = Duration::from_secs(10);
69
70/// Number of simultaneous contacts forming a five-finger tap.
71const TAP_CONTACTS: usize = 5;
72
73/// Accumulates per-slot stroke trajectories and recognizes spells
74/// once a gesture episode completes.
75///
76/// An episode spans from the first contact until every gesture slot
77/// has lifted (`Dispatch` in SPELLS.md's phase model). The tracker
78/// evaluates each spell's [`SpellArming`] against the episode's
79/// facts - unsatisfied arming excludes the spell from matching.
80#[derive(Default)]
81pub struct GestureTracker {
82    /// Registered spells to match completed strokes against.
83    spells: Vec<Spell>,
84    /// Registered handprints - `five-finger-tap` spells naming a
85    /// `hand` only arm when the tap's constellation matched that
86    /// profile (per-user spells, SPELLS.md).
87    hands: Vec<HandProfile>,
88    /// Live stroke per slot - exists while the contact is down.
89    trajectories: HashMap<TouchPointSlot, Trajectory>,
90    /// Strokes completed during the current episode (slot lifted
91    /// while others still move), keyed by their slot so a matched
92    /// spell can name the contacts it consumed.
93    completed: Vec<(TouchPointSlot, Vec<Vec2<Millimeters>>)>,
94    /// The episode's first contact point - `start-spot` arming.
95    episode_first: Option<Vec2<Millimeters>>,
96    /// Down timestamp per live slot - five-finger-tap detection.
97    down_at: HashMap<TouchPointSlot, Instant>,
98    /// Candidate tap: the simultaneous contact group waiting for a
99    /// full lift within [`TAP_TOTAL`].
100    pending_tap: Option<PendingTap>,
101    /// `five-finger-tap` latch deadline - armed while in the future.
102    five_tap_armed_until: Option<Instant>,
103    /// The hand the latched tap matched - `None` while unlatched or
104    /// when no registered handprint matched (unidentified hands
105    /// still arm `hand: None` spells).
106    armed_hand: Option<HandName>,
107    /// Slots consumed by the last dispatched spell - the boundary
108    /// drops their withheld events instead of replaying them.
109    consumed: SlotSet,
110    /// Active spell-training session - episodes are captured as
111    /// template repetitions instead of matched (SPELLS.md).
112    training: Option<SpellTraining>,
113    /// The spell the last training session derived - the device loop
114    /// picks it up and registers it persistently.
115    trained: Option<Spell>,
116    /// Center trajectory per wand tangible, keyed by tangible ID -
117    /// only tangibles whose pattern backs a `tangible-chord` spell.
118    chords: HashMap<TangibleId, (PatternName, Trajectory)>,
119    /// Slots whose first contact landed inside an enabled field -
120    /// SPELLS.md's field exclusion: such strokes are never gesture
121    /// candidates, so they join neither trajectories, episodes nor
122    /// five-finger taps. The block holds for the contact's lifetime
123    /// even when it later leaves the field.
124    field_blocked: HashSet<TouchPointSlot>,
125    /// Best candidate's match evaluation of the last dispatched
126    /// episode - SPELLS.md's debug view (per-stroke Fréchet vs
127    /// corridor plus the finger assignment), retained for near
128    /// misses too.
129    diagnostics: Option<SpellDiagnostics>,
130}

190    /// Slots to withhold from forwarding while their episode is
191    /// armed (SPELLS.md's state-gated suppression). Only an armed
192    /// episode suppresses - an unarmed stroke (a normal drag,
193    /// pinch or scroll) must flow through untouched, so arming is
194    /// evaluated on the live trajectories, not on dispatch.
195    pub fn gesture_slots(&self, now: Instant) -> SlotSet {
196        if !self.armed_episode_active(now) {
197            return SlotSet::default();
198        }
199        self.trajectories.keys().copied().collect()
200    }

202    /// Slots consumed by a dispatched spell - their withheld events
203    /// must drop permanently, never replay (a matched stroke is the
204    /// input itself, not deferred desktop input).
205    pub fn take_consumed(&mut self) -> SlotSet {
206        std::mem::take(&mut self.consumed)
207    }

209    /// Whether any registered spell's arming is satisfied by the
210    /// in-progress episode - the live counterpart of
211    /// [`GestureTracker::episode_armed`] used for suppression.
212    ///
213    /// `free` and `tangible-chord` never arm a touch episode:
214    /// freeform strokes flow through (SPELLS.md's defer-and-replay
215    /// caveat) and chord slots are already suppressed via their
216    /// tangible binding.
217    fn armed_episode_active(&self, now: Instant) -> bool {
218        self.spells.iter().any(|spell| match &spell.arming {
219            SpellArming::Free | SpellArming::TangibleChord { .. } => false,
220            SpellArming::ExclusiveZone { region } => self
221                .trajectories
222                .values()
223                .all(|traj| traj.points.iter().all(|point| region.contains(point))),
224            SpellArming::StartSpot { region } => self
225                .episode_first
226                .is_some_and(|point| region.contains(&point)),
227            SpellArming::FiveFingerTap { hand } => {
228                self.five_tap_armed(now) && self.hand_satisfied(hand)
229            }
230        })
231    }
232
233    /// Whether the latched tap's identified hand satisfies a spell's
234    /// `hand` binding - `None` accepts any tap (identified or not).
235    fn hand_satisfied(&self, required: &Option<HandName>) -> bool {
236        required
237            .as_ref()
238            .is_none_or(|required| self.armed_hand.as_ref() == Some(required))
239    }

241    /// Feeds one frame's contacts and tracked tangibles: records
242    /// five-finger taps, accumulates wand trajectories, appends
243    /// positions for live slots, completes lifted slots, and
244    /// dispatches when an episode closes (all gesture contacts or
245    /// the wand lifted). `now` is injected for testability;
246    /// `field_sets` carries the registered fields so a stroke
247    /// starting inside an enabled one is never a gesture candidate
248    /// (SPELLS.md's field exclusion).
249    pub fn observe_frame(
250        &mut self,
251        touches: &[TouchPoint],
252        tracked: &[TrackedTangible],
253        now: Instant,
254        field_sets: &[FieldSet],
255    ) -> Option<SpellMatched> {
256        let live: HashSet<TouchPointSlot> = touches.iter().map(|t| t.slot).collect();
257        self.field_blocked.retain(|slot| live.contains(slot));
258        for touch in touches {
259            if !self.trajectories.contains_key(&touch.slot)
260                && !self.field_blocked.contains(&touch.slot)
261                && field_sets
262                    .iter()
263                    .any(|set| set.enabled && set.contains_point(&touch.pos))
264            {
265                self.field_blocked.insert(touch.slot);
266            }
267        }
268        let eligible: Vec<TouchPoint> = touches
269            .iter()
270            .filter(|touch| !self.field_blocked.contains(&touch.slot))
271            .cloned()
272            .collect();
273
274        self.track_five_tap(&eligible, now);
275        let chord_match = self.track_chords(tracked);
276
277        let lifted: Vec<TouchPointSlot> = self
278            .trajectories
279            .keys()
280            .copied()
281            .filter(|slot| !live.contains(slot))
282            .collect();
283        for slot in lifted {
284            if let Some(traj) = self.trajectories.remove(&slot)
285                && traj.path_length >= MIN_STROKE_PATH
286            {
287                self.completed.push((slot, traj.points));
288            }
289        }
290
291        for touch in &eligible {
292            let traj = self.trajectories.entry(touch.slot).or_default();
293            let Some(last) = traj.points.last() else {
294                traj.points.push(touch.pos);
295                continue;
296            };
297            let step = last.distance(&touch.pos);
298            if step >= SAMPLE_STEP {
299                traj.path_length = traj.path_length + step;
300                traj.points.push(touch.pos);
301            }
302        }
303        if self.episode_first.is_none() {
304            self.episode_first = eligible.first().map(|t| t.pos);
305        }
306
307        // Dispatch: the touch episode closes when every gesture slot
308        // has lifted. A chord dispatch wins over a same-frame touch
309        // dispatch - wand strokes are unambiguous by construction.
310        if let Some(matched) = chord_match {
311            return Some(matched);
312        }
313        if self.trajectories.is_empty() {
314            let first_point = self.episode_first.take();
315            if self.completed.is_empty() {
316                return None;
317            }
318            let strokes = std::mem::take(&mut self.completed);
319            let episode = Episode {
320                strokes: &strokes,
321                first_point,
322                five_tap_armed: self.five_tap_armed(now),
323                armed_hand: self.armed_hand.clone(),
324            };
325            if episode.five_tap_armed {
326                self.five_tap_armed_until = None;
327                self.armed_hand = None;
328            }
329            // A training session consumes the episode instead of
330            // matching it - the strokes become template repetitions
331            // (SPELLS.md's learn-by-repetition).
332            if self.training.is_some() {
333                self.capture_training(episode.polylines());
334                return None;
335            }
336            return self.dispatch(&episode);
337        }
338        None
339    }

390    /// Five-finger tap detection: five contacts down within
391    /// [`TAP_SPREAD`] and fully lifted within [`TAP_TOTAL`] latches
392    /// arming for [`TAP_LATCH`].
393    fn track_five_tap(&mut self, touches: &[TouchPoint], now: Instant) {
394        let live: HashSet<TouchPointSlot> = touches.iter().map(|t| t.slot).collect();
395        self.down_at.retain(|slot, _| live.contains(slot));
396        for touch in touches {
397            self.down_at.entry(touch.slot).or_insert(now);
398        }
399
400        if let Some(tap) = &self.pending_tap {
401            let all_lifted = tap
402                .contacts
403                .iter()
404                .all(|contact| !live.contains(&contact.slot));
405            let expired = now.duration_since(tap.started) > TAP_TOTAL;
406            if all_lifted || expired {
407                if all_lifted && !expired {
408                    // The tap's constellation identifies the hand -
409                    // per-user `five-finger-tap` spells only arm for
410                    // the handprint it matched.
411                    let points: Vec<Vec2<Millimeters>> =
412                        tap.contacts.iter().map(|contact| contact.pos).collect();
413                    self.armed_hand = self
414                        .hands
415                        .iter()
416                        .find(|hand| hand.matches(PointSet::new(&points)))
417                        .map(|hand| hand.name.clone());
418                    self.five_tap_armed_until = Some(now + TAP_LATCH);
419                }
420                self.pending_tap = None;
421            }
422            return;
423        }
424
425        // A fresh simultaneous five-contact cluster arms a pending tap.
426        let mut recent: Vec<(Instant, TouchPointSlot)> = self
427            .down_at
428            .iter()
429            .filter(|(_, at)| now.duration_since(**at) <= TAP_SPREAD)
430            .map(|(slot, at)| (*at, *slot))
431            .collect();
432        if recent.len() >= TAP_CONTACTS {
433            recent.sort_by_key(|(at, _)| *at);
434            let earliest = recent.first().map(|(at, _)| *at).unwrap_or(now);
435            let contacts: Vec<TouchPoint> = recent
436                .iter()
437                .take(TAP_CONTACTS)
438                .filter_map(|(_, slot)| touches.iter().find(|touch| touch.slot == *slot).cloned())
439                .collect();
440            self.pending_tap = Some(PendingTap {
441                started: earliest,
442                contacts,
443            });
444        }
445    }

447    /// Wand trajectories for `tangible-chord` spells: the center
448    /// path of every tracked tangible whose pattern backs a chord
449    /// spell. A disappearing tangible completes its spell episode.
450    fn track_chords(&mut self, tracked: &[TrackedTangible]) -> Option<SpellMatched> {
451        let live_ids: HashSet<TangibleId> = tracked.iter().map(|t| t.id).collect();
452        let lifted: Vec<TangibleId> = self
453            .chords
454            .keys()
455            .copied()
456            .filter(|id| !live_ids.contains(id))
457            .collect();
458        let mut matched = None;
459        for id in lifted {
460            if let Some((pattern, traj)) = self.chords.remove(&id)
461                && traj.path_length >= MIN_STROKE_PATH
462            {
463                matched = self.dispatch_chord(&pattern, &traj.points).or(matched);
464            }
465        }
466
467        // A tangible acts as wand when at least one chord spell
468        // names its pattern - or accepts any tangible (`None`).
469        let wand_for = |pattern: &PatternName| {
470            self.spells.iter().any(|s| match &s.arming {
471                SpellArming::TangibleChord { pattern: p } => {
472                    p.as_ref().is_none_or(|p| p == pattern)
473                }
474                _ => false,
475            })
476        };
477        for tangible in tracked {
478            if !wand_for(&tangible.pattern_name) {
479                continue;
480            }
481            let traj = &mut self
482                .chords
483                .entry(tangible.id)
484                .or_insert_with(|| (tangible.pattern_name.clone(), Trajectory::default()))
485                .1;
486            let Some(last) = traj.points.last() else {
487                traj.points.push(tangible.center);
488                continue;
489            };
490            let step = last.distance(&tangible.center);
491            if step >= SAMPLE_STEP {
492                traj.path_length = traj.path_length + step;
493                traj.points.push(tangible.center);
494            }
495        }
496        matched
497    }

499    /// Whether the episode satisfied a spell's arming condition -
500    /// `tangible-chord` never matches touch episodes (it dispatches
501    /// via [`GestureTracker::dispatch_chord`]).
502    fn episode_armed(&self, spell: &Spell, episode: &Episode<'_>) -> bool {
503        match &spell.arming {
504            SpellArming::Free => true,
505            SpellArming::ExclusiveZone { region } => episode.all_points_inside(region),
506            SpellArming::StartSpot { region } => episode
507                .first_point
508                .is_some_and(|point| region.contains(&point)),
509            SpellArming::FiveFingerTap { hand } => {
510                episode.five_tap_armed
511                    && hand
512                        .as_ref()
513                        .is_none_or(|required| episode.armed_hand.as_ref() == Some(required))
514            }
515            SpellArming::TangibleChord { .. } => false,
516        }
517    }

519    /// Evaluates the completed stroke set against all spells whose
520    /// arming the episode satisfied; the lowest score wins when it
521    /// sits inside the spell's corridor. A match marks the episode's
522    /// slots consumed - the boundary drops their withheld events
523    /// instead of replaying the stroke.
524    fn dispatch(&mut self, episode: &Episode<'_>) -> Option<SpellMatched> {
525        let polylines = episode.polylines();
526        let mut best: Option<(&Spell, Millimeters)> = None;
527        // The debug view keeps the closest candidate's diagnostics -
528        // a miss still reports how far each stroke was off.
529        let mut closest: Option<SpellDiagnostics> = None;
530        for spell in &self.spells {
531            if !self.episode_armed(spell, episode) {
532                continue;
533            }
534            let Some(diag) = spell.diagnostics(&polylines) else {
535                continue;
536            };
537            if diag.matched && best.is_none_or(|(_, best_score)| diag.score < best_score) {
538                best = Some((spell, diag.score));
539            }
540            if closest
541                .as_ref()
542                .is_none_or(|current| diag.score < current.score)
543            {
544                closest = Some(diag);
545            }
546        }
547        self.diagnostics = closest;
548        let (spell, score) = best?;
549        self.consumed = episode.strokes.iter().map(|(slot, _)| *slot).collect();
550        Some(SpellMatched {
551            name: spell.name.clone(),
552            center: Trajectory::centroid(&polylines),
553            score,
554        })
555    }

557    /// Wand dispatch: the lifted tangible's center path is the sole
558    /// stroke, matched against every `tangible-chord` spell naming
559    /// the tangible's pattern.
560    fn dispatch_chord(
561        &mut self,
562        pattern: &PatternName,
563        points: &[Vec2<Millimeters>],
564    ) -> Option<SpellMatched> {
565        let strokes = [points.to_vec()];
566        let mut best: Option<(&Spell, Millimeters)> = None;
567        let mut closest: Option<SpellDiagnostics> = None;
568        for spell in &self.spells {
569            let armed = match &spell.arming {
570                SpellArming::TangibleChord { pattern: p } => {
571                    p.as_ref().is_none_or(|p| p == pattern)
572                }
573                _ => false,
574            };
575            if !armed {
576                continue;
577            }
578            let Some(diag) = spell.diagnostics(&strokes) else {
579                continue;
580            };
581            if diag.matched && best.is_none_or(|(_, best_score)| diag.score < best_score) {
582                best = Some((spell, diag.score));
583            }
584            if closest
585                .as_ref()
586                .is_none_or(|current| diag.score < current.score)
587            {
588                closest = Some(diag);
589            }
590        }
591        self.diagnostics = closest;
592        let (spell, score) = best?;
593        Some(SpellMatched {
594            name: spell.name.clone(),
595            center: Trajectory::centroid(&strokes),
596            score,
597        })
598    }

624    fn touch(slot: u32, x: f64, y: f64) -> TouchPoint {
625        TouchPoint {

648    /// Draws the V gesture at `offset`: down-stroke then up-stroke
649    /// over successive frames.
650    fn draw_v(
651        tracker: &mut GestureTracker,
652        offset: (f64, f64),
653        now: Instant,
654    ) -> Option<SpellMatched> {
655        draw_v_in_fields(tracker, offset, now, &[])
656    }

658    /// [`draw_v`] variant carrying the registered field sets.
659    fn draw_v_in_fields(
660        tracker: &mut GestureTracker,
661        offset: (f64, f64),
662        now: Instant,
663        field_sets: &[tangible_objects_model::FieldSet],
664    ) -> Option<SpellMatched> {
665        let arm: Vec<(f64, f64)> = (0..=10)
666            .map(|i| (offset.0 + i as f64 * 5.0, offset.1 + i as f64 * 7.0))
667            .collect();
668        let leg: Vec<(f64, f64)> = (0..=10)
669            .map(|i| {
670                (
671                    offset.0 + 50.0 + i as f64 * 5.0,
672                    offset.1 + 70.0 - i as f64 * 7.0,
673                )
674            })
675            .collect();
676        for (x, y) in arm.iter().chain(leg.iter()) {
677            tracker.observe_frame(&[touch(0, *x, *y)], &[], now, field_sets);
678        }
679        tracker.observe_frame(&[], &[], now, field_sets)
680    }
