1//! `GestureTracker` - accumulates per-slot trajectories and
2//! dispatches completed strokes against registered spells
3//! (SPELLS.md's recognition core).
4//!
5//! Arming (SPELLS.md phase model): a spell only matches when its
6//! declared arming condition was satisfied for the episode:
7//!
8//! - `free` - always armed (legacy/freeform gestures).
9//! - `exclusive-zone` - every stroke point must stay inside the
10//!   zone's region.
11//! - `start-spot` - the episode's first contact must land inside
12//!   the spot's region.
13//! - `five-finger-tap` - a short simultaneous five-contact tap
14//!   latches arming for a window; the next episode may draw with
15//!   fewer fingers.
16//! - `tangible-chord` - a tracked tangible of the named pattern
17//!   acts as the wand; its center path is the stroke and lifting
18//!   the tangible completes the spell.
19

42/// Minimum stroke path length for a slot to count as a gesture
43/// candidate (mm) - below it a contact is a rest or a normal touch
44/// (SPELLS.md's temporal discriminator).
45const MIN_STROKE_PATH_MM: f64 = 20.0;
46
47/// Minimum distance between appended trajectory samples (mm) -
48/// dedupes the position stream so the polyline stays sparse.
49const SAMPLE_STEP_MM: f64 = 2.0;
50
51/// Maximum spread between the five contacts' down events for a
52/// five-finger tap to count as simultaneous (SPELLS.md).
53const TAP_SPREAD: Duration = Duration::from_millis(150);
54
55/// Maximum total duration of a five-finger tap (down to last lift).
56const TAP_TOTAL: Duration = Duration::from_millis(400);
57
58/// How long a completed five-finger tap keeps the arming latch.
59const TAP_LATCH: Duration = Duration::from_secs(10);
60
61/// Number of simultaneous contacts forming a five-finger tap.
62const TAP_CONTACTS: usize = 5;
63
64/// Whether an episode satisfied a spell's arming condition - the
65/// `Listening -> Armed` transition's per-episode evaluation.
66struct Episode<'a> {
67    /// Completed strokes of the episode with their slots - the
68    /// slots identify what a matched spell consumed.
69    strokes: &'a [(TouchPointSlot, Vec<Vec2<Millimeters>>)],
70    /// The episode's very first contact point - `start-spot` checks
71    /// this against its region.
72    first_point: Option<Vec2<Millimeters>>,
73    /// `five-finger-tap` latch active during the episode.
74    five_tap_armed: bool,
75}
76
77impl Episode<'_> {
78    /// Stroke points without slot identity - the matcher input.
79    fn polylines(&self) -> Vec<Vec<Vec2<Millimeters>>> {
80        self.strokes
81            .iter()
82            .map(|(_, points)| points.clone())
83            .collect()
84    }
85}

87/// Accumulates per-slot stroke trajectories and recognizes spells
88/// once a gesture episode completes.
89///
90/// An episode spans from the first contact until every gesture slot
91/// has lifted (`Dispatch` in SPELLS.md's phase model). The tracker
92/// evaluates each spell's [`SpellArming`] against the episode's
93/// facts - unsatisfied arming excludes the spell from matching.
94#[derive(Default)]
95pub struct GestureTracker {
96    /// Registered spells to match completed strokes against.
97    spells: Vec<Spell>,
98    /// Live stroke per slot - exists while the contact is down.
99    trajectories: HashMap<TouchPointSlot, Trajectory>,
100    /// Strokes completed during the current episode (slot lifted
101    /// while others still move), keyed by their slot so a matched
102    /// spell can name the contacts it consumed.
103    completed: Vec<(TouchPointSlot, Vec<Vec2<Millimeters>>)>,
104    /// The episode's first contact point - `start-spot` arming.
105    episode_first: Option<Vec2<Millimeters>>,
106    /// Down timestamp per live slot - five-finger-tap detection.
107    down_at: HashMap<TouchPointSlot, Instant>,
108    /// Candidate tap: earliest down time plus the contact group.
109    pending_tap: Option<(Instant, HashSet<TouchPointSlot>)>,
110    /// `five-finger-tap` latch deadline - armed while in the future.
111    five_tap_armed_until: Option<Instant>,
112    /// Slots consumed by the last dispatched spell - the boundary
113    /// drops their withheld events instead of replaying them.
114    consumed: SlotSet,
115    /// Active spell-training session - episodes are captured as
116    /// template repetitions instead of matched (SPELLS.md).
117    training: Option<SpellTraining>,
118    /// The spell the last training session derived - the device loop
119    /// picks it up and registers it persistently.
120    trained: Option<Spell>,
121    /// Center trajectory per wand tangible, keyed by tangible ID -
122    /// only tangibles whose pattern backs a `tangible-chord` spell.
123    chords: HashMap<TangibleId, (PatternName, Trajectory)>,
124}

139        self.spells.retain(|s| s.name != spell.name);

144    pub fn remove_spell(&mut self, name: &SpellName) -> bool {
145        let before = self.spells.len();
146        self.spells.retain(|s| &s.name != name);
147        self.spells.len() != before

150    /// The registered spells.

152        &self.spells
153    }

155    /// Live trajectories for the debug overlay.
156    pub fn trajectories(&self) -> &HashMap<TouchPointSlot, Trajectory> {
157        &self.trajectories
158    }
159
160    /// Slots to withhold from forwarding while their episode is
161    /// armed (SPELLS.md's state-gated suppression). Only an armed
162    /// episode suppresses - an unarmed stroke (a normal drag,
163    /// pinch or scroll) must flow through untouched, so arming is
164    /// evaluated on the live trajectories, not on dispatch.
165    pub fn gesture_slots(&self, now: Instant) -> SlotSet {
166        if !self.armed_episode_active(now) {
167            return SlotSet::default();
168        }
169        self.trajectories.keys().copied().collect()
170    }
171
172    /// Slots consumed by a dispatched spell - their withheld events
173    /// must drop permanently, never replay (a matched stroke is the
174    /// input itself, not deferred desktop input).
175    pub fn take_consumed(&mut self) -> SlotSet {
176        std::mem::take(&mut self.consumed)
177    }
178
179    /// Whether any registered spell's arming is satisfied by the
180    /// in-progress episode - the live counterpart of
181    /// [`GestureTracker::episode_armed`] used for suppression.
182    ///
183    /// `free` and `tangible-chord` never arm a touch episode:
184    /// freeform strokes flow through (SPELLS.md's defer-and-replay
185    /// caveat) and chord slots are already suppressed via their
186    /// tangible binding.
187    fn armed_episode_active(&self, now: Instant) -> bool {
188        self.spells.iter().any(|spell| match &spell.arming {
189            SpellArming::Free | SpellArming::TangibleChord { .. } => false,
190            SpellArming::ExclusiveZone { region } => self
191                .trajectories
192                .values()
193                .all(|traj| traj.points.iter().all(|point| region.contains(point))),
194            SpellArming::StartSpot { region } => self
195                .episode_first
196                .is_some_and(|point| region.contains(&point)),
197            SpellArming::FiveFingerTap => self.five_tap_armed(now),
198        })
199    }
200
201    /// Feeds one frame's contacts and tracked tangibles: records
202    /// five-finger taps, accumulates wand trajectories, appends
203    /// positions for live slots, completes lifted slots, and
204    /// dispatches when an episode closes (all gesture contacts or
205    /// the wand lifted). `now` is injected for testability.
206    pub fn observe_frame(
207        &mut self,
208        touches: &[TouchPoint],
209        tracked: &[TrackedTangible],
210        now: Instant,
211    ) -> Option<SpellMatched> {
212        self.track_five_tap(touches, now);
213        let chord_match = self.track_chords(tracked);
214
215        let live: HashSet<TouchPointSlot> = touches.iter().map(|t| t.slot).collect();
216        let lifted: Vec<TouchPointSlot> = self
217            .trajectories
218            .keys()
219            .copied()
220            .filter(|slot| !live.contains(slot))
221            .collect();
222        for slot in lifted {
223            if let Some(traj) = self.trajectories.remove(&slot)
224                && traj.path_length >= MIN_STROKE_PATH_MM
225            {
226                self.completed.push((slot, traj.points));
227            }
228        }
229
230        for touch in touches {
231            let traj = self.trajectories.entry(touch.slot).or_default();
232            let Some(last) = traj.points.last() else {
233                traj.points.push(touch.pos);
234                continue;
235            };
236            let step = last.distance(&touch.pos).as_f64();
237            if step >= SAMPLE_STEP_MM {
238                traj.path_length += step;
239                traj.points.push(touch.pos);
240            }
241        }
242        if self.episode_first.is_none() {
243            self.episode_first = touches.first().map(|t| t.pos);
244        }
245
246        // Dispatch: the touch episode closes when every gesture slot
247        // has lifted. A chord dispatch wins over a same-frame touch
248        // dispatch - wand strokes are unambiguous by construction.
249        if let Some(matched) = chord_match {
250            return Some(matched);
251        }
252        if self.trajectories.is_empty() {
253            let first_point = self.episode_first.take();
254            if self.completed.is_empty() {
255                return None;
256            }
257            let strokes = std::mem::take(&mut self.completed);
258            let episode = Episode {
259                strokes: &strokes,
260                first_point,
261                five_tap_armed: self.five_tap_armed(now),
262            };
263            if episode.five_tap_armed {
264                self.five_tap_armed_until = None;
265            }
266            // A training session consumes the episode instead of
267            // matching it - the strokes become template repetitions
268            // (SPELLS.md's learn-by-repetition).
269            if self.training.is_some() {
270                self.capture_training(episode.polylines());
271                return None;
272            }
273            return self.dispatch(&episode);
274        }
275        None
276    }

278    /// Feeds a completed episode into the training buffer; a full
279    /// session moves the finished spell into `trained` and disarms.
280    fn capture_training(&mut self, strokes: Vec<Vec<Vec2<Millimeters>>>) {
281        if let Some(training) = self.training.as_mut()
282            && let Some(spell) = training.capture(strokes)
283        {
284            self.trained = Some(spell);
285            self.training = None;
286        }
287    }

308    fn five_tap_armed(&self, now: Instant) -> bool {
309        self.five_tap_armed_until
310            .is_some_and(|deadline| now < deadline)
311    }

313    /// Five-finger tap detection: five contacts down within
314    /// [`TAP_SPREAD`] and fully lifted within [`TAP_TOTAL`] latches
315    /// arming for [`TAP_LATCH`].
316    fn track_five_tap(&mut self, touches: &[TouchPoint], now: Instant) {
317        let live: HashSet<TouchPointSlot> = touches.iter().map(|t| t.slot).collect();
318        self.down_at.retain(|slot, _| live.contains(slot));
319        for touch in touches {
320            self.down_at.entry(touch.slot).or_insert(now);
321        }
322
323        if let Some((started, group)) = &self.pending_tap {
324            let all_lifted = group.iter().all(|slot| !live.contains(slot));
325            let expired = now.duration_since(*started) > TAP_TOTAL;
326            if all_lifted || expired {
327                if all_lifted && !expired {
328                    self.five_tap_armed_until = Some(now + TAP_LATCH);
329                }
330                self.pending_tap = None;
331            }
332            return;
333        }
334
335        // A fresh simultaneous five-contact cluster arms a pending tap.
336        let mut recent: Vec<(Instant, TouchPointSlot)> = self
337            .down_at
338            .iter()
339            .filter(|(_, at)| now.duration_since(**at) <= TAP_SPREAD)
340            .map(|(slot, at)| (*at, *slot))
341            .collect();
342        if recent.len() >= TAP_CONTACTS {
343            recent.sort_by_key(|(at, _)| *at);
344            let earliest = recent.first().map(|(at, _)| *at).unwrap_or(now);
345            let group: HashSet<TouchPointSlot> = recent
346                .iter()
347                .take(TAP_CONTACTS)
348                .map(|(_, slot)| *slot)
349                .collect();
350            self.pending_tap = Some((earliest, group));
351        }
352    }

354    /// Wand trajectories for `tangible-chord` spells: the center
355    /// path of every tracked tangible whose pattern backs a chord
356    /// spell. A disappearing tangible completes its spell episode.
357    fn track_chords(&mut self, tracked: &[TrackedTangible]) -> Option<SpellMatched> {
358        let live_ids: HashSet<TangibleId> = tracked.iter().map(|t| t.id).collect();
359        let lifted: Vec<TangibleId> = self
360            .chords
361            .keys()
362            .copied()
363            .filter(|id| !live_ids.contains(id))
364            .collect();
365        let mut matched = None;
366        for id in lifted {
367            if let Some((pattern, traj)) = self.chords.remove(&id)
368                && traj.path_length >= MIN_STROKE_PATH_MM
369            {
370                matched = self.dispatch_chord(&pattern, &traj.points).or(matched);
371            }
372        }
373
374        // A tangible acts as wand when at least one chord spell
375        // names its pattern - or accepts any tangible (`None`).
376        let wand_for = |pattern: &PatternName| {
377            self.spells.iter().any(|s| match &s.arming {
378                SpellArming::TangibleChord { pattern: p } => {
379                    p.as_ref().is_none_or(|p| p == pattern)
380                }
381                _ => false,
382            })
383        };
384        for tangible in tracked {
385            if !wand_for(&tangible.pattern_name) {
386                continue;
387            }
388            let traj = &mut self
389                .chords
390                .entry(tangible.id)
391                .or_insert_with(|| (tangible.pattern_name.clone(), Trajectory::default()))
392                .1;
393            let Some(last) = traj.points.last() else {
394                traj.points.push(tangible.center);
395                continue;
396            };
397            let step = last.distance(&tangible.center).as_f64();
398            if step >= SAMPLE_STEP_MM {
399                traj.path_length += step;
400                traj.points.push(tangible.center);
401            }
402        }
403        matched
404    }

406    /// Whether the episode satisfied a spell's arming condition -
407    /// `tangible-chord` never matches touch episodes (it dispatches
408    /// via [`GestureTracker::dispatch_chord`]).
409    fn episode_armed(&self, spell: &Spell, episode: &Episode<'_>) -> bool {
410        match &spell.arming {
411            SpellArming::Free => true,
412            SpellArming::ExclusiveZone { region } => {
413                all_points_inside(&episode.polylines(), region)
414            }
415            SpellArming::StartSpot { region } => episode
416                .first_point
417                .is_some_and(|point| region.contains(&point)),
418            SpellArming::FiveFingerTap => episode.five_tap_armed,
419            SpellArming::TangibleChord { .. } => false,
420        }
421    }
422
423    /// Evaluates the completed stroke set against all spells whose
424    /// arming the episode satisfied; the lowest score wins when it
425    /// sits inside the spell's corridor. A match marks the episode's
426    /// slots consumed - the boundary drops their withheld events
427    /// instead of replaying the stroke.
428    fn dispatch(&mut self, episode: &Episode<'_>) -> Option<SpellMatched> {
429        let polylines = episode.polylines();
430        let mut best: Option<(&Spell, Millimeters)> = None;
431        for spell in &self.spells {
432            if !self.episode_armed(spell, episode) {
433                continue;
434            }
435            let Some(score) = spell.try_match(&polylines) else {
436                continue;
437            };
438            if score <= spell.corridor && best.is_none_or(|(_, best_score)| score < best_score) {
439                best = Some((spell, score));
440            }
441        }
442        let (spell, score) = best?;
443        self.consumed = episode.strokes.iter().map(|(slot, _)| *slot).collect();
444        Some(SpellMatched {
445            name: spell.name.clone(),
446            center: Trajectory::centroid(&polylines),
447            score,
448        })
449    }

451    /// Wand dispatch: the lifted tangible's center path is the sole
452    /// stroke, matched against every `tangible-chord` spell naming
453    /// the tangible's pattern.
454    fn dispatch_chord(
455        &self,
456        pattern: &PatternName,
457        points: &[Vec2<Millimeters>],
458    ) -> Option<SpellMatched> {
459        let strokes = [points.to_vec()];
460        let mut best: Option<(&Spell, Millimeters)> = None;
461        for spell in &self.spells {
462            let armed = match &spell.arming {
463                SpellArming::TangibleChord { pattern: p } => {
464                    p.as_ref().is_none_or(|p| p == pattern)
465                }
466                _ => false,
467            };
468            if !armed {
469                continue;
470            }
471            let Some(score) = spell.try_match(&strokes) else {
472                continue;
473            };
474            if score <= spell.corridor && best.is_none_or(|(_, best_score)| score < best_score) {
475                best = Some((spell, score));
476            }
477        }
478        let (spell, score) = best?;
479        Some(SpellMatched {
480            name: spell.name.clone(),
481            center: Trajectory::centroid(&strokes),
482            score,
483        })
484    }
485}

487/// `exclusive-zone` arming: every stroke point must lie inside the
488/// region - leaving the zone fails the episode.
489fn all_points_inside(strokes: &[Vec<Vec2<Millimeters>>], region: &Region) -> bool {
490    strokes
491        .iter()
492        .all(|stroke| stroke.iter().all(|point| region.contains(point)))
493}

540    /// Draws the V gesture at `offset`: down-stroke then up-stroke
541    /// over successive frames.
542    fn draw_v(
543        tracker: &mut GestureTracker,
544        offset: (f64, f64),
545        now: Instant,
546    ) -> Option<SpellMatched> {
547        let arm: Vec<(f64, f64)> = (0..=10)
548            .map(|i| (offset.0 + i as f64 * 5.0, offset.1 + i as f64 * 7.0))
549            .collect();
550        let leg: Vec<(f64, f64)> = (0..=10)
551            .map(|i| {
552                (
553                    offset.0 + 50.0 + i as f64 * 5.0,
554                    offset.1 + 70.0 - i as f64 * 7.0,
555                )
556            })
557            .collect();
558        for (x, y) in arm.iter().chain(leg.iter()) {
559            tracker.observe_frame(&[touch(0, *x, *y)], &[], now);
560        }
561        tracker.observe_frame(&[], &[], now)
562    }

564    #[test]
565    fn training_derives_template_and_corridor() {
566        let mut tracker = GestureTracker::new();
567        tracker.start_training(SpellName::new("learned-v"), SpellArming::Free);
568        let now = Instant::now();
569        // Three repetitions at slight offsets - each returns None
570        // (the episode is captured, not matched).
571        assert!(draw_v(&mut tracker, (100.0, 100.0), now).is_none());
572        assert!(tracker.is_training());
573        assert!(draw_v(&mut tracker, (103.0, 98.0), now).is_none());
574        assert!(draw_v(&mut tracker, (97.0, 102.0), now).is_none());
575        assert!(!tracker.is_training());

725    #[test]
726    fn tangible_chord_uses_the_wand_trajectory() {
727        let pattern = PatternName::new("wand");
728        let mut tracker = GestureTracker::new();
729        tracker.add_spell(v_spell(SpellArming::TangibleChord {
730            pattern: Some(pattern.clone()),
731        }));
732        let now = Instant::now();
733
734        let wand = |x: f64, y: f64| TrackedTangible {
735            id: TangibleId::new(7),
736            pattern_id: PatternId::new(1),
737            pattern_name: pattern.clone(),
738            center: Vec2::new(x, y),
739            rotation_rad: RotationRad::ZERO,
740            pending: false,
741            pending_frames: FrameCount::new(0),
742            spawn_stability: FrameCount::new(0),
743            frames_since_match: FrameCount::new(0),
744        };
745
746        // Drag the wand along the V.
747        let mut points: Vec<(f64, f64)> = (0..=10)
748            .map(|i| (100.0 + i as f64 * 5.0, 100.0 + i as f64 * 7.0))
749            .collect();
750        points.extend((0..=10).map(|i| (150.0 + i as f64 * 5.0, 170.0 - i as f64 * 7.0)));
751        for (x, y) in &points {
752            tracker.observe_frame(&[], &[wand(*x, *y)], now);
753        }
754        // Lift the wand - completes the spell.
755        let matched = tracker
756            .observe_frame(&[], &[], now)
757            .expect("wand V should match");
758        assert_eq!(matched.name.as_str(), "V");
759    }
