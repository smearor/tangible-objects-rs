10/// One stage of the event pipeline (EVENT_PIPELINE.md).
11///
12/// A stage owns its suppression *policy* - which slots count as
13/// suppressed this frame and how the decision evolves over frames -
14/// while [`super::EmitBoundary`] owns the mechanism (dropping, retraction,
15/// guarded withholding). The boundary merges the per-stage views through
16/// this trait; `EmitBoundary::finish` is the only consumer.
17pub(crate) trait Stage {
18    /// The stage's suppression policy for mechanism merges - stages
19    /// without a `FilterMode` concept report `Immediate` (always-on
20    /// suppression with no gate).
21    fn mode(&self) -> FilterMode;
22
23    /// Consecutive matched frames before suppression - only meaningful
24    /// for gated tangible stages; gateless stages report 0.
25    fn stability_gate(&self) -> FrameCount;
26
27    /// Whether `slot` is suppressed under this stage's policy right now -
28    /// the check the boundary applies per event.
29    fn suppresses(&self, slot: TouchPointSlot) -> bool;
30
31    /// Raw suppressed-set membership - the guarded-release decision
32    /// already knows the policy.
33    fn was_suppressed(&self, slot: TouchPointSlot) -> bool;
34
35    /// Currently suppressed slots - unioned by the boundary for
36    /// retraction.
37    fn suppressed_slots(&self) -> Box<dyn Iterator<Item = TouchPointSlot> + '_>;
38
39    /// The pattern scope this stage's `matched` set covers
40    /// (EVENT_PIPELINE.md phase 4): `None` = the general stage for
41    /// untagged patterns. Positional stages ignore `matched` entirely -
42    /// the default satisfies them. Returned by reference so the pipeline
43    /// can key the scoped-match map without cloning the name per frame.
44    fn scope(&self) -> &Option<StageName> {
45        &None
46    }
47
48    /// Advances the stage's classification by one frame.
49    fn classify(&mut self, frame: &StageFrame<'_>);
50
51    /// Releases `slot` - a lifting contact frees it for a new tangible.
52    fn release(&mut self, slot: TouchPointSlot);
53
54    /// `SetFilterMode` propagation - only tangible stages carry a
55    /// `FilterMode`; always-on stages ignore the command.
56    fn set_mode(&mut self, _mode: FilterMode, _stability: Option<FrameCount>) {}
57
58    /// Field-set propagation - only the field stage consumes it.
59    fn set_fields(&mut self, _sets: Vec<FieldSet>) {}
60
61    /// Marker for the field stage - the pipeline creates it lazily.
62    fn is_field_stage(&self) -> bool {
63        false
64    }
65
66    /// Marker for tangible stages - scoped stage lifecycle
67    /// (`StageRegistered`/`Unregistered`) applies to them only.
68    fn is_tangible_stage(&self) -> bool {
69        false
70    }
71
72    /// Whether this stage was claimed by `RegisterStage` - explicit
73    /// stages survive `sync_scopes` garbage collection.
74    fn is_explicit(&self) -> bool {
75        false
76    }
77
78    /// Claims the stage as `RegisterStage`-driven.
79    fn mark_explicit(&mut self) {}
80
81    /// The stage's counters - per-stage metrics (phase 6).
82    fn metrics(&self) -> &StageMetrics;
83}
