5/// Per-stage counters (EVENT_PIPELINE.md phase 6).
6///
7/// Each stage owns one and updates it inside `classify` - global
8/// `SessionMetrics` counters stay the aggregate; `suppressed` is the
9/// gauge answering "which stage is eating contacts right now".
10#[derive(Debug, Default, Clone)]
11pub struct StageMetrics {
12    /// Frames this stage classified.
13    pub frames: FrameCount,
14    /// Slots suppressed in the current frame (gauge, not cumulative).
15    pub suppressed: ContactCount,
16    /// Cumulative slot-suppressions - a slot suppressed again after a
17    /// release counts anew.
18    pub suppressed_total: ContactCount,
19    /// Cumulative matched slots fed to this stage's `classify`
20    /// (scope-filtered input).
21    pub matched_total: ContactCount,
22}
23
24impl StageMetrics {
25    /// Records a rebuilt per-frame suppression set: bumps `frames`,
26    /// counts the newly suppressed slots into `suppressed_total`,
27    /// refreshes the `suppressed` gauge and swaps `new` into the stage's
28    /// set - the shared tail of rebuild-style `classify` impls.
29    pub fn record_frame(&mut self, suppressed: &mut SlotSet, new: SlotSet) {
30        self.frames += 1;
31        self.suppressed_total += new.difference(suppressed).count().into();
32        self.suppressed = new.len().into();
33        *suppressed = new;
34    }
35}

37#[cfg(test)]
38mod tests {
39    use super::StageMetrics;
40    use tangible_objects_model::SlotSet;
41    use tangible_objects_model::TouchPointSlot;
42
43    /// `record_frame` bumps the frame counter, refreshes the gauge and
44    /// swaps the suppressed set.
45    #[test]
46    fn record_frame_updates_counters_and_set() {
47        let mut metrics = StageMetrics::default();
48        let mut suppressed = SlotSet::default();
49        metrics.record_frame(&mut suppressed, [TouchPointSlot::new(0)].into());
50        assert_eq!(metrics.frames.as_u64(), 1);
51        assert_eq!(metrics.suppressed.0, 1);
52        assert!(suppressed.contains(&TouchPointSlot::new(0)));
53    }
54
55    /// `suppressed_total` counts only newly suppressed slots - a slot
56    /// suppressed again the next frame does not re-count.
57    #[test]
58    fn suppressed_total_counts_only_new_slots() {
59        let mut metrics = StageMetrics::default();
60        let mut suppressed = SlotSet::default();
61        metrics.record_frame(&mut suppressed, [TouchPointSlot::new(0)].into());
62        metrics.record_frame(&mut suppressed, [TouchPointSlot::new(0)].into());
63        assert_eq!(metrics.suppressed_total.0, 1);
64        // A release + re-suppression counts anew.
65        metrics.record_frame(&mut suppressed, SlotSet::default());
66        metrics.record_frame(&mut suppressed, [TouchPointSlot::new(0)].into());
67        assert_eq!(metrics.suppressed_total.0, 2);
68    }
