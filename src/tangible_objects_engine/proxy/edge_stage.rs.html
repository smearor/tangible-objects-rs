12/// Suppresses contacts inside the edge margin - ghost contacts some
13/// panels emit at the axis bounds (EVENT_PIPELINE.md).
14///
15/// Empirically (SiS panel): placing a tangible in the middle of the
16/// surface makes the controller fabricate extra phantom contacts at the
17/// axis bounds - unmatched, so only position can catch them. They pollute
18/// matching *and* leak to the OS as ghost taps. A real finger inside the
19/// margin dies the same way - positional filtering cannot distinguish -
20/// which is why the margin is opt-in and should be kept small.
21///
22/// Stateless per frame: `classify` recomputes the suppressed set from the
23/// frame's touches - an edge contact that moves inward is unsuppressed
24/// the same frame. Always-on: edge suppression is device hygiene, not a
25/// `FilterMode` policy, so `set_mode` does not reach it and `Off` does
26/// not release edge contacts.
27///
28/// The engine's own `is_edge_ghost` stays in place - ghosts must not
29/// pollute matching either; this stage owns the emitted-stream side
30/// (until now edge ghosts leaked to the virtual device).
31#[derive(Debug)]
32pub(crate) struct EdgeStage {
33    /// Inset from the surface bounds that counts as the edge band.
34    margin: Millimeters,
35    /// Usable surface in millimeters (from the device calibration).
36    surface: Region,
37    /// Slots suppressed this frame - rebuilt by `classify`.
38    suppressed: SlotSet,
39    /// Per-stage counters - updated in `classify`.
40    metrics: StageMetrics,
41}
42
43impl EdgeStage {
44    /// Creates an edge stage suppressing contacts within `margin` of the
45    /// `surface` bounds.
46    pub(crate) fn new(margin: Millimeters, surface: Region) -> Self {
47        Self {
48            margin,
49            surface,
50            suppressed: SlotSet::default(),
51            metrics: StageMetrics::default(),
52        }
53    }
54}
55
56impl Stage for EdgeStage {
57    fn mode(&self) -> FilterMode {
58        FilterMode::Immediate
59    }
60
61    fn stability_gate(&self) -> FrameCount {
62        FrameCount::new(0)
63    }
64
65    fn suppresses(&self, slot: TouchPointSlot) -> bool {
66        self.suppressed.contains(&slot)
67    }
68
69    fn was_suppressed(&self, slot: TouchPointSlot) -> bool {
70        self.suppressed.contains(&slot)
71    }
72
73    fn suppressed_slots(&self) -> Box<dyn Iterator<Item = TouchPointSlot> + '_> {
74        Box::new(self.suppressed.iter().copied())
75    }
76
77    fn classify(&mut self, frame: &StageFrame<'_>) {
78        let inner = self.surface.inset(self.margin);
79        let suppressed = frame.slots_where(|touch| !inner.contains(&touch.pos));
80        self.metrics.record_frame(&mut self.suppressed, suppressed);
81    }
82
83    fn release(&mut self, slot: TouchPointSlot) {
84        self.suppressed.remove(&slot);
85    }
86
87    fn metrics(&self) -> &StageMetrics {
88        &self.metrics
89    }
90}

93mod tests {
94    use super::EdgeStage;
95    use super::Stage;

104
105    fn surface() -> Region {
106        Region {
107            x_min: Millimeters::new(0.0),
108            y_min: Millimeters::new(0.0),
109            x_max: Millimeters::new(100.0),
110            y_max: Millimeters::new(100.0),
111        }
112    }
113
114    fn touch(slot: u32, x: f64, y: f64) -> TouchPoint {
115        TouchPoint {
116            slot: TouchPointSlot::new(slot),
117            tracking_id: TrackingId::new(slot as i32),
118            pos: Vec2::new(x, y),
119        }
120    }
121
122    /// Contacts in the margin are ghost-suppressed; interior contacts
123    /// pass through.
124    #[test]
125    fn classify_suppresses_edge_band_only() {
126        let mut stage = EdgeStage::new(Millimeters::new(10.0), surface());
127        let matched = SlotSet::default();
128        let touches = vec![touch(0, 5.0, 50.0), touch(1, 50.0, 50.0)];
129        stage.classify(&StageFrame {
130            matched: &matched,
131            touches: (&touches).into(),
132        });
133        assert!(stage.suppresses(TouchPointSlot::new(0)));
134        assert!(!stage.suppresses(TouchPointSlot::new(1)));
135    }
136
137    /// The stage is stateless - a contact that moved into the interior
138    /// is unsuppressed the same frame.
139    #[test]
140    fn classify_unsuppresses_on_move_inward() {
141        let mut stage = EdgeStage::new(Millimeters::new(10.0), surface());
142        let matched = SlotSet::default();
143        let edge = vec![touch(0, 5.0, 50.0)];
144        stage.classify(&StageFrame {
145            matched: &matched,
146            touches: (&edge).into(),
147        });
148        let inner = vec![touch(0, 50.0, 50.0)];
149        stage.classify(&StageFrame {
150            matched: &matched,
151            touches: (&inner).into(),
152        });
153        assert!(!stage.suppresses(TouchPointSlot::new(0)));
154    }
