18/// Owned stage chain of a pipeline (`Vec<Box<dyn Stage>>`).
19///
20/// Carries both access directions the emit boundary needs: read-side
21/// merging (`merged_*`, `suppresses`, `was_suppressed`,
22/// `suppressed_slots`) through `&self` and mutation (`classify`,
23/// `release_slot`, `set_mode`) through `&mut self` - one borrow serves
24/// both, so no view juggling is needed inside `finish`.
25pub(crate) struct Stages(pub Vec<Box<dyn Stage>>);
26
27// Read-side merge - the boundary's view over all stages.
28impl Stages {
29    /// The mechanism's effective policy across all stages:
30    /// `TangiblesOnly` and `Guarded` are mechanism-visible (`Guarded`
31    /// withholds, `TangiblesOnly` drains); `Immediate`/`Gated`/`Off`
32    /// differ only in stage-side classification, not mechanism, so the
33    /// first stage's mode suffices.
34    pub fn merged_mode(&self) -> FilterMode {
35        if self
36            .iter()
37            .any(|stage| stage.mode() == FilterMode::TangiblesOnly)
38        {
39            return FilterMode::TangiblesOnly;
40        }
41        if self.iter().any(|stage| stage.mode() == FilterMode::Guarded) {
42            return FilterMode::Guarded;
43        }
44        self.first().map_or(FilterMode::default(), |s| s.mode())
45    }
46
47    /// The gate used for guarded-mode release - the gate of the (first)
48    /// guarded stage, else any stage's.
49    pub fn merged_gate(&self) -> FrameCount {
50        self.iter()
51            .find(|stage| stage.mode() == FilterMode::Guarded)
52            .or_else(|| self.first())
53            .map_or(FrameCount::new(0), |s| s.stability_gate())
54    }
55
56    /// Whether any stage suppresses `slot` under its active policy.
57    pub fn suppresses(&self, slot: TouchPointSlot) -> bool {
58        self.iter().any(|stage| stage.suppresses(slot))
59    }
60
61    /// Whether any stage holds `slot` in its suppressed set - used by the
62    /// guarded-release logic.
63    pub fn was_suppressed(&self, slot: TouchPointSlot) -> bool {
64        self.iter().any(|stage| stage.was_suppressed(slot))
65    }
66
67    /// Union of every stage's suppressed slots - deduplicated: a slot
68    /// suppressed by two stages must retract only once.
69    pub fn suppressed_slots(&self) -> impl Iterator<Item = TouchPointSlot> + '_ {
70        self.iter()
71            .flat_map(|stage| stage.suppressed_slots())
72            .collect::<HashSet<_>>()
73            .into_iter()
74    }
75}
76
77// Mutation side - per-frame classification and slot lifecycle.
78impl Stages {
79    /// Advances every stage's classification by one frame - each stage
80    /// receives the `scoped` subset keyed by its own `scope()` plus the
81    /// shared touches (EVENT_PIPELINE.md phase 4 input masking).
82    pub fn classify(&mut self, scoped: &ScopedMatches, touches: TouchSet<'_>) {
83        let empty = SlotSet::default();
84        for stage in &mut self.0 {
85            let matched = scoped.get(stage.scope()).unwrap_or(&empty);
86            stage.classify(&StageFrame { matched, touches });
87        }
88    }
89
90    /// The stage keyed by `scope`, if present - the daemon ensures one
91    /// per scope appearing on patterns.
92    pub fn find_by_scope(&self, scope: &StageName) -> Option<&dyn Stage> {
93        self.iter()
94            .find(|stage| stage.scope().as_ref() == Some(scope))
95            .map(|stage| &**stage)
96    }
97
98    /// Scopes of all tangible stages - the lifecycle-synced set the
99    /// daemon diffs against the engine's pattern scopes.
100    pub fn tangible_scopes(&self) -> impl Iterator<Item = &StageName> {
101        self.iter()
102            .filter(|stage| stage.is_tangible_stage())
103            .filter_map(|stage| stage.scope().as_ref())
104    }
105
106    /// Removes the tangible stage for `scope` - its hysteresis dies with
107    /// it (phase 4/6 lifecycle: the scope's last pattern is gone).
108    pub fn remove_tangible_stage(&mut self, scope: &StageName) -> bool {
109        let before = self.len();
110        self.0
111            .retain(|stage| !(stage.is_tangible_stage() && stage.scope().as_ref() == Some(scope)));
112        self.len() != before
113    }
114
115    /// Per-stage metric snapshots, in declaration order.
116    pub fn stage_metrics(&self) -> impl Iterator<Item = &StageMetrics> {
117        self.iter().map(|stage| stage.metrics())
118    }
119
120    /// Releases `slot` on every stage - a lifting contact frees it for a
121    /// new tangible regardless of which stage suppressed it.
122    pub fn release_slot(&mut self, slot: TouchPointSlot) {
123        for stage in &mut self.0 {
124            stage.release(slot);
125        }
126    }
127
128    /// Switches the suppression policy (`SetFilterMode` control
129    /// command): `scope` `Some` applies it to the matching stage only,
130    /// `None` broadcasts to all tangible stages - positional stages
131    /// ignore `SetFilterMode` regardless.
132    pub fn set_mode(
133        &mut self,
134        mode: FilterMode,
135        stability: Option<FrameCount>,
136        scope: Option<&StageName>,
137    ) {
138        for stage in &mut self.0 {
139            if scope.is_none() || stage.scope().as_ref() == scope {
140                stage.set_mode(mode, stability);
141            }
142        }
143    }
144
145    /// Scopes of `RegisterStage`-claimed tangible stages - these survive
146    /// `sync_scopes` garbage collection.
147    pub fn explicit_scopes(&self) -> impl Iterator<Item = &StageName> {
148        self.iter()
149            .filter(|stage| stage.is_explicit())
150            .filter_map(|stage| stage.scope().as_ref())
151    }
152
153    /// Claims the tangible stage for `scope` as `RegisterStage`-driven;
154    /// returns `false` when no such stage exists.
155    pub fn mark_stage_explicit(&mut self, scope: &StageName) -> bool {
156        let mut found = false;
157        for stage in &mut self.0 {
158            if stage.is_tangible_stage() && stage.scope().as_ref() == Some(scope) {
159                stage.mark_explicit();
160                found = true;
161            }
162        }
163        found
164    }
165
166    /// Propagates the current field sets to the field stage - other
167    /// stages ignore it.
168    pub fn set_field_sets(&mut self, sets: Vec<FieldSet>) {
169        for stage in &mut self.0 {
170            stage.set_fields(sets.clone());
171        }
172    }
173
174    /// Whether a field stage is present.
175    pub fn has_field_stage(&self) -> bool {
176        self.iter().any(|stage| stage.is_field_stage())
177    }
178}
