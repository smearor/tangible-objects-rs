11/// Suppresses every contact inside the declared zones - play-area
12/// masking (EVENT_PIPELINE.md).
13///
14/// The inverse of a profile's `regions` constraint: regions bound where
15/// a *pattern may match*, a zone stage suppresses *any* contact inside
16/// its areas regardless of classification - e.g. a figure parking zone
17/// whose pins must never reach the OS even when no profile matches.
18///
19/// Stateless per frame like [`super::EdgeStage`]: `classify` rebuilds the
20/// suppressed set; always-on, `set_mode` does not reach it.
21#[derive(Debug)]
22pub(crate) struct ZoneStage {
23    /// Masked areas in millimeters - any contact inside is suppressed.
24    zones: Vec<Region>,
25    /// Slots suppressed this frame - rebuilt by `classify`.
26    suppressed: SlotSet,
27    /// Per-stage counters - updated in `classify`.
28    metrics: StageMetrics,
29}
30
31impl ZoneStage {
32    /// Creates a zone stage suppressing contacts inside `zones`.
33    pub(crate) fn new(zones: Vec<Region>) -> Self {
34        Self {
35            zones,
36            suppressed: SlotSet::default(),
37            metrics: StageMetrics::default(),
38        }
39    }
40}
41
42impl Stage for ZoneStage {
43    fn mode(&self) -> FilterMode {
44        FilterMode::Immediate
45    }
46
47    fn stability_gate(&self) -> FrameCount {
48        FrameCount::new(0)
49    }
50
51    fn suppresses(&self, slot: TouchPointSlot) -> bool {
52        self.suppressed.contains(&slot)
53    }
54
55    fn was_suppressed(&self, slot: TouchPointSlot) -> bool {
56        self.suppressed.contains(&slot)
57    }
58
59    fn suppressed_slots(&self) -> Box<dyn Iterator<Item = TouchPointSlot> + '_> {
60        Box::new(self.suppressed.iter().copied())
61    }
62
63    fn classify(&mut self, frame: &StageFrame<'_>) {
64        let suppressed =
65            frame.slots_where(|touch| self.zones.iter().any(|zone| zone.contains(&touch.pos)));
66        self.metrics.record_frame(&mut self.suppressed, suppressed);
67    }
68
69    fn release(&mut self, slot: TouchPointSlot) {
70        self.suppressed.remove(&slot);
71    }
72
73    fn metrics(&self) -> &StageMetrics {
74        &self.metrics
75    }
76}

78#[cfg(test)]
79mod tests {
80    use super::Stage;
81    use super::ZoneStage;
82    use crate::proxy::stage_frame::StageFrame;
83    use tangible_objects_model::Millimeters;
84    use tangible_objects_model::Region;
85    use tangible_objects_model::SlotSet;
86    use tangible_objects_model::TouchPoint;
87    use tangible_objects_model::TouchPointSlot;
88    use tangible_objects_model::TrackingId;
89    use tangible_objects_model::Vec2;
90
91    fn zone() -> Region {
92        Region {
93            x_min: Millimeters::new(10.0),
94            y_min: Millimeters::new(10.0),
95            x_max: Millimeters::new(20.0),
96            y_max: Millimeters::new(20.0),
97        }
98    }
99
100    fn touch(slot: u32, x: f64, y: f64) -> TouchPoint {
101        TouchPoint {
102            slot: TouchPointSlot::new(slot),
103            tracking_id: TrackingId::new(slot as i32),
104            pos: Vec2::new(x, y),
105        }
106    }
107
108    fn classify(stage: &mut ZoneStage, touches: &[TouchPoint]) {
109        let matched = SlotSet::default();
110        stage.classify(&StageFrame {
111            matched: &matched,
112            touches: touches.into(),
113        });
114    }
115
116    /// A contact inside the zone is suppressed; one outside passes.
117    #[test]
118    fn classify_suppresses_inside_zone_only() {
119        let mut stage = ZoneStage::new(vec![zone()]);
120        let touches = vec![touch(0, 15.0, 15.0), touch(1, 50.0, 50.0)];
121        classify(&mut stage, &touches);
122        assert!(stage.suppresses(TouchPointSlot::new(0)));
123        assert!(!stage.suppresses(TouchPointSlot::new(1)));
124    }
125
126    /// The stage is stateless - a contact that moved out of the zone
127    /// is unsuppressed the same frame.
128    #[test]
129    fn classify_unsuppresses_on_move_out() {
130        let mut stage = ZoneStage::new(vec![zone()]);
131        let inside = vec![touch(0, 15.0, 15.0)];
132        classify(&mut stage, &inside);
133        let outside = vec![touch(0, 50.0, 50.0)];
134        classify(&mut stage, &outside);
135        assert!(!stage.suppresses(TouchPointSlot::new(0)));
136    }
137
138    /// `release` drops the slot from the suppressed set.
139    #[test]
140    fn release_clears_suppression() {
141        let mut stage = ZoneStage::new(vec![zone()]);
142        let touches = vec![touch(0, 15.0, 15.0)];
143        classify(&mut stage, &touches);
144        stage.release(TouchPointSlot::new(0));
145        assert!(!stage.suppresses(TouchPointSlot::new(0)));
146    }
