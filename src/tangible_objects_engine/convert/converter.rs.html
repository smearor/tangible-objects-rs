20/// Codes carrying positions that convert through the calibration map -
21/// both multitouch axes and the legacy single-touch duplicates.
22const POSITION_CODES: &[u16] = &[
23    53, // ABS_MT_POSITION_X
24    54, // ABS_MT_POSITION_Y
25    0,  // ABS_X
26    1,  // ABS_Y
27];
28
29/// Converts a `.evtest` recording of one touchscreen into the raw event
30/// stream another device would produce.
31///
32/// Position values travel raw → mm → raw through both calibrations;
33/// timestamps pass through unchanged (the source report cadence is kept -
34/// the target's `report_rate_hz` is informational). Events whose
35/// `(type, code)` the target does not emit are dropped and counted in
36/// [`ConvertStats::dropped`]; out-of-bounds positions clamp to the target
37/// axis range ([`ConvertStats::clamped`]).
38pub struct DeviceConverter {
39    /// Target device - provides the calibration, capability set and the
40    /// `evtest` header of the output artifact.
41    target: DeviceProfile,
42}
43
44impl DeviceConverter {
45    /// Creates a converter producing artifacts for `target`.
46    pub fn new(target: DeviceProfile) -> Self {
47        Self { target }
48    }
49
50    /// Converts `input` into `output`; returns conversion statistics.
51    pub fn convert(&self, input: &Path, output: &Path) -> Result<ConvertStats, EngineError> {
52        let mut replay = EvtestReplay::open(input)?;
53        let source = replay.calibration().ok_or_else(|| {
54            EngineError::InvalidInput(format!(
55                "{} has no calibration header - cannot convert to raw axis values",
56                input.display()
57            ))
58        })?;
59
60        let file = File::create(output)?;
61        let mut writer = BufWriter::new(file);
62        writer.write_all(
63            header_text(Some(&self.target.name), Some(&self.target.calibration())).as_bytes(),
64        )?;
65
66        let mut stats = ConvertStats::default();
67        while let Some((event, timestamp)) = replay.next_event_timed() {
68            stats.events += 1;
69            if event.is_syn_report() {
70                writer.write_all(format_event(&event, timestamp).as_bytes())?;
71                stats.kept += 1;
72                continue;
73            }
74            if !self.target.supports(event.event_type, event.code) {
75                stats.dropped += 1;
76                continue;
77            }
78            let converted = self.convert_event(&event, &source, &mut stats);
79            writer.write_all(format_event(&converted, timestamp).as_bytes())?;
80            stats.kept += 1;
81        }
82        writer.flush()?;
83        Ok(stats)
84    }
85
86    /// Maps one event into target raw values. Position codes go through
87    /// `raw → mm → raw`; everything else passes through unchanged.
88    fn convert_event(
89        &self,
90        event: &RawEvent,
91        source: &Calibration,
92        stats: &mut ConvertStats,
93    ) -> RawEvent {
94        if event.event_type != EventType::Abs || !POSITION_CODES.contains(&event.code.as_u16()) {
95            return *event;
96        }
97        let axis_x = event.code == ABS_MT_POSITION_X || event.code == EventCode::new(0);
98        let res = if axis_x { source.x_res } else { source.y_res };
99        let mm = Millimeters::new(event.value as f64 / res.as_f64());
100        let (raw, clamped) = self.target.to_raw(mm, axis_x);
101        if clamped {
102            stats.clamped += 1;
103        }
104        RawEvent::new(event.event_type, event.code, raw.as_f64().round() as i32)
105    }
106}
