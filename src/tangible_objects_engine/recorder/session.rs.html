19/// A running recording session: paired frame + raw sinks with labeling.
20///
21/// Bundles [`FrameRecorder`] and [`RawRecorder`] writing `<base_path>.jsonl`
22/// and `<base_path>.evtest`. The caller supplies session-relative timestamps
23/// so the same session type serves live recording (monotonic clock) and
24/// deterministic replay recording (evtest timestamps).
25///
26/// Two label channels exist: a persistent session label (`--record-label`,
27/// `SetRecordLabel`) and a temporary mark ([`Self::mark_label`]) that applies
28/// to the next N recorded frames - the interactive keypress path.
29pub struct RecordingSession {
30    /// Base path the artifacts are written to (without extension).
31    pub base_path: PathBuf,
32    /// JSONL writer for processed frames (`<base>.jsonl`).
33    frame_recorder: FrameRecorder,
34    /// evtest writer for raw events (`<base>.evtest`).
35    raw_recorder: RawRecorder,
36    /// Calibration snapshot recorded into every frame's context.
37    calibration: Option<Calibration>,
38    /// Session-wide ground-truth label (e.g. the figure being recorded).
39    label: Option<RecordLabel>,
40    /// Temporary label override for `SetRecordLabel` - takes precedence
41    /// over `label` while its frame budget lasts.
42    label_override: Option<LabelOverride>,
43    /// Frames recorded so far - the session's own counter, separate from
44    /// the frame numbers which count `SYN_REPORT`s including skipped
45    /// empty frames.
46    recorded_frames: FrameCount,
47}
48
49/// Active mark-label override: applies `label` for `remaining` more
50/// recorded frames.
51struct LabelOverride {
52    /// Temporary ground-truth label.
53    label: RecordLabel,
54    /// Remaining budget - decremented per recorded frame, saturates at zero
55    /// and expires the override.
56    remaining: RemainingFrameCount,
57}
58
59impl RecordingSession {
60    /// Starts a session writing `<base_path>.jsonl` + `<base_path>.evtest`.
61    pub fn start(
62        base_path: PathBuf,
63        device_name: Option<&str>,
64        calibration: Option<&Calibration>,
65        label: Option<RecordLabel>,
66    ) -> Result<Self> {
67        let frame_recorder = FrameRecorder::spawn(&base_path.with_extension("jsonl"))?;
68        let raw_recorder = RawRecorder::spawn(
69            &base_path.with_extension("evtest"),
70            device_name,
71            calibration,
72        )?;
73        Ok(Self {
74            base_path,
75            frame_recorder,
76            raw_recorder,
77            calibration: calibration.copied(),
78            label,
79            label_override: None,
80            recorded_frames: FrameCount::new(0),
81        })
82    }
83
84    /// Queues a raw event for the `evtest` artifact.
85    ///
86    /// `timestamp_secs` is the session-relative event time in seconds.
87    pub fn record_event(&self, event: &RawEvent, timestamp_secs: f64) {
88        self.raw_recorder.record(event, timestamp_secs);
89    }
90
91    /// Records a non-empty frame; returns whether the frame was written.
92    ///
93    /// Frames without touches and tracked tangibles are skipped per the
94    /// recording rules - the caller still counts `frame_no` for every
95    /// `SYN_REPORT` so numbering stays aligned.
96    pub fn record_frame(
97        &mut self,
98        frame_no: FrameNumber,
99        timestamp_ms: Milliseconds,
100        frame: &EngineFrame,
101        tracked: TrackedSet<'_>,
102    ) -> bool {
103        if frame.touches.is_empty() && tracked.is_empty() {
104            return false;
105        }
106        let label = self.take_label();
107        self.frame_recorder.record(
108            frame_no,
109            timestamp_ms,
110            frame,
111            tracked,
112            &FrameContext {
113                label: label.as_ref(),
114                calibration: self.calibration.as_ref(),
115                raw_dropped: self.raw_recorder.dropped_count(),
116            },
117        );
118        self.recorded_frames += 1;
119        true
120    }
121
122    /// The label currently applied to recorded frames (override or session
123    /// default).
124    pub fn label(&self) -> Option<&RecordLabel> {
125        match &self.label_override {
126            Some(over) if !over.remaining.is_expired() => Some(&over.label),
127            _ => self.label.as_ref(),
128        }
129    }
130
131    /// Changes the persistent session label (`ControlCommand::SetRecordLabel`
132    /// path). A pending [`mark_label`](Self::mark_label) override still wins
133    /// for its remaining frames.
134    pub fn set_label(&mut self, label: Option<RecordLabel>) {
135        self.label = label;
136    }
137
138    /// Temporarily applies `label` to the next `frames` recorded frames -
139    /// the interactive keypress path; afterwards the session label applies
140    /// again.
141    pub fn mark_label(&mut self, label: RecordLabel, frames: u64) {
142        self.label_override = Some(LabelOverride {
143            label,
144            remaining: RemainingFrameCount::new(frames),
145        });
146    }
147
148    /// Frames recorded so far (non-empty frames only).
149    pub fn recorded_frames(&self) -> FrameCount {
150        self.recorded_frames
151    }
152
153    /// Frames dropped because the frame sink's buffer was full.
154    pub fn dropped_frames(&self) -> FrameCount {
155        self.frame_recorder.dropped_count()
156    }
157
158    /// Raw events dropped because the raw sink's buffer was full.
159    pub fn dropped_events(&self) -> EventCount {
160        self.raw_recorder.dropped_count()
161    }
162
163    /// Drains, flushes and closes both artifacts.
164    pub fn finish(self) -> Result<()> {
165        self.frame_recorder.finish()?;
166        self.raw_recorder.finish()
167    }
168
169    /// Resolves the effective label for the next recorded frame: the mark
170    /// override while budget remains, else the session label.
171    fn take_label(&mut self) -> Option<RecordLabel> {
172        if let Some(over) = &mut self.label_override {
173            if over.remaining.is_expired() {
174                self.label_override = None;
175            } else {
176                let label = over.label.clone();
177                over.remaining -= 1;
178                return Some(label);
179            }
180        }
181        self.label.clone()
182    }
183}
