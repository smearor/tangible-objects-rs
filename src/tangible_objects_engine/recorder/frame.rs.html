25/// Bounded channel capacity between the device loop and the writer thread
26/// (~8 seconds at 120 fps).
27const FRAME_CHANNEL_CAPACITY: usize = 1024;
28
29/// Writes one JSONL [`FrameRecord`] per recorded engine frame.
30///
31/// Serialization and file writes happen on a dedicated thread behind a
32/// bounded channel: [`FrameRecorder::record`] never blocks the device loop;
33/// a full buffer drops frames and counts them in
34/// [`FrameRecorder::dropped_count`].
35pub struct FrameRecorder {
36    /// Channel to the writer thread - `None` after shutdown so the
37    /// writer's `recv` fails and the thread finishes.
38    sender: Option<SyncSender<FrameRecord>>,
39    /// Frames dropped because the channel was full; shared with the
40    /// polling side so [`Self::dropped_count`] stays readable while the
41    /// writer runs.
42    dropped: Arc<AtomicU64>,
43    /// Writer thread join handle - `None` after `finish`/`shutdown`.
44    thread: Option<JoinHandle<std::io::Result<()>>>,
45}
46
47impl FrameRecorder {
48    /// Creates the JSONL file at `path` and spawns the writer thread.
49    pub fn spawn(path: &Path) -> std::io::Result<Self> {
50        let file = File::create(path)?;
51        let (sender, receiver) = sync_channel::<FrameRecord>(FRAME_CHANNEL_CAPACITY);
52        let dropped = Arc::new(AtomicU64::new(0));
53        let thread = std::thread::spawn(move || {
54            let mut writer = BufWriter::new(file);
55            while let Ok(record) = receiver.recv() {
56                serde_json::to_writer(&mut writer, &record).map_err(std::io::Error::other)?;
57                writer.write_all(b"\n")?;
58            }
59            writer.flush()
60        });
61        Ok(Self {
62            sender: Some(sender),
63            dropped,
64            thread: Some(thread),
65        })
66    }
67
68    /// Records one engine frame.
69    ///
70    /// `frame_no` counts every `SYN_REPORT` of the session - skipped empty
71    /// frames leave gaps, keeping [`FrameNumber`] aligned with the raw stream.
72    /// `timestamp_ms` is caller-supplied (session clock live, evtest time on
73    /// replay) so replayed recordings stay deterministic.
74    /// Never blocks: a full channel drops the frame and counts it.
75    pub fn record(
76        &self,
77        frame_no: FrameNumber,
78        timestamp_ms: Milliseconds,
79        frame: &EngineFrame,
80        tracked: TrackedSet<'_>,
81        context: &FrameContext,
82    ) {
83        let record = FrameRecord {
84            schema_version: RECORD_SCHEMA_VERSION,
85            frame: frame_no,
86            timestamp_ms,
87            label: context.label.cloned(),
88            calibration: context.calibration.copied(),
89            touches: frame.touches.clone(),
90            candidates: frame.candidates.clone(),
91            tracked: tracked.iter().map(TrackedRecord::from).collect(),
92            raw_dropped: context.raw_dropped,
93        };
94        let Some(sender) = &self.sender else {
95            return;
96        };
97        match sender.try_send(record) {
98            Ok(()) => {}
99            Err(TrySendError::Full(_)) => {
100                let dropped = self.dropped.fetch_add(1, Ordering::Relaxed) + 1;
101                if dropped == 1 {
102                    warn!("frame recorder buffer full - dropping frames");
103                }
104            }
105            Err(TrySendError::Disconnected(_)) => {
106                warn!("frame recorder writer closed before stop");
107            }
108        }
109    }
110
111    /// Running count of frames dropped because the writer buffer was full.
112    pub fn dropped_count(&self) -> FrameCount {
113        FrameCount::new(self.dropped.load(Ordering::Relaxed))
114    }
115
116    /// Closes the channel, drains the writer thread and flushes the file.
117    pub fn finish(mut self) -> std::io::Result<()> {
118        self.shutdown()
119    }
120
121    fn shutdown(&mut self) -> std::io::Result<()> {
122        self.sender.take();
123        match self.thread.take() {
124            Some(thread) => match thread.join() {
125                Ok(result) => result,
126                Err(_) => Err(std::io::Error::other("frame recorder thread panicked")),
127            },
128            None => Ok(()),
129        }
130    }
131}
132
133impl Drop for FrameRecorder {
134    fn drop(&mut self) {
135        let _ = self.shutdown();
136    }
137}

157    fn temp_path(name: &str) -> std::path::PathBuf {
158        std::env::temp_dir().join(format!("robofigure-{name}-{}", std::process::id()))
159    }
160
161    #[test]
162    fn writes_one_jsonl_record_per_frame() {
163        let path = temp_path("frame-recorder.jsonl");
164        let frame = EngineFrame {
165            matches: Vec::new(),
166            touches: vec![TouchPoint {
167                slot: TouchPointSlot::new(0),
168                tracking_id: TrackingId::new(240),
169                pos: Vec2::new(192.78, 94.27),
170            }],
171            events: Vec::new(),
172            ran_generic: true,
173            candidates_tested: 0.into(),
174            dead_zone_filtered: ContactCount::default(),
175            candidates: vec![CandidateReport {
176                pattern_id: PatternId::new(5),
177                pattern_name: "TestFigur-5P".into(),
178                touch_indices: vec![0].into(),
179                matched: true,
180                partial: false,
181                measured_distances: vec![Millimeters::new(34.15)].into(),
182                deviations: vec![Deviation::from(Millimeters::new(0.83))].into(),
183                tolerance: PatternTolerance::new(3.0),
184                tolerances: Default::default(),
185                score: None,
186                match_error: Millimeters::new(5.69).into(),
187            }],
188        };
189
190        {
191            let recorder = FrameRecorder::spawn(&path).expect("spawn recorder");
192            let context = FrameContext {
193                label: None,
194                calibration: None,
195                raw_dropped: EventCount::default(),
196            };
197            recorder.record(
198                7.into(),
199                Milliseconds::new(0),
200                &frame,
201                (&[]).into(),
202                &context,
203            );
204            recorder.record(
205                8.into(),
206                Milliseconds::new(16),
207                &frame,
208                (&[]).into(),
209                &context,
210            );
211            recorder.finish().expect("finish recorder");
212        }
213
214        let content = std::fs::read_to_string(&path).expect("read recording");
215        let lines: Vec<&str> = content.lines().collect();
216        assert_eq!(lines.len(), 2);
217        let first: serde_json::Value = serde_json::from_str(lines[0]).expect("parse line");
218        assert_eq!(first["v"], 1);
219        assert_eq!(first["frame"], 7);
220        assert_eq!(first["touches"][0]["tracking_id"], 240);
221        assert_eq!(first["candidates"][0]["deviations"][0], 0.83);
222        assert!(first["label"].is_null());
223        assert!(first["calibration"].is_null());
224        let second: serde_json::Value = serde_json::from_str(lines[1]).expect("parse line");
225        assert_eq!(second["frame"], 8);
226
227        std::fs::remove_file(&path).ok();
228    }
