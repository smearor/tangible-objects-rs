33/// Default capacity of the broadcast event channel.
34///
35/// Lagging receivers drop the oldest events (`RecvError::Lagged`); a full
36/// channel never blocks the device loop.
37pub const EVENT_CHANNEL_CAPACITY: usize = 256;
38
39/// Capacity of the control command channel back to the device loop.
40pub const COMMAND_CHANNEL_CAPACITY: usize = 32;
41
42/// Daemon-side runtime: publishes frames, runs set tracking and receives
43/// control commands.
44///
45/// Created once per detection session; the cloneable [`EngineHandle`] is
46/// handed to interface servers.
47pub struct EngineRuntime {
48    /// Tangible Object Set tracker - computes set/object transitions
49    /// from each published frame's tracked tangibles.
50    set_tracker: SetTracker,
51    /// Broadcast channel for `DomainEvent`s (engine + set events);
52    /// lagging receivers lose the oldest events.
53    events: broadcast::Sender<DomainEvent>,
54    /// Watch channel carrying the latest snapshot for query-style
55    /// interface requests.
56    snapshot: watch::Sender<EngineSnapshot>,
57    /// Receiving half of the control channel - drained by the device
58    /// loop once per frame/timeout.
59    commands: mpsc::Receiver<ControlCommand>,
60    /// Device info embedded in every snapshot; set once at session
61    /// start.
62    device: Option<DeviceInfo>,
63    /// Recording status embedded in every snapshot while a session is
64    /// or was active.
65    recording: Option<RecordingStatus>,
66    /// Active completion configuration embedded in every snapshot.
67    completion: Option<CompletionConfig>,
68    /// Active field sets (static + dynamically registered) embedded in
69    /// every snapshot.
70    field_sets: Vec<FieldSet>,
71    /// Active dead zones (runtime-store set + dynamically registered)
72    /// embedded in every snapshot.
73    dead_zones: Vec<DeadZone>,
74    /// Profile list shared with the handle - registrations at runtime
75    /// must be visible to interface readers (`RwLock`, not `watch`).
76    profiles: Arc<RwLock<Vec<ProfileInfo>>>,
77    /// Accumulated session recognition metrics.
78    metrics: SessionMetrics,
79    /// User-facing message shown by the GUI (`ShowMessage` command).
80    message: Option<String>,
81    /// Whether the daemon-embedded GUI window is shown
82    /// (`SetGuiVisible` command) - only with the `gui` feature.
83    #[cfg(feature = "gui")]
84    gui_visible: bool,
85    /// Quarter turns the GUI canvas renders (`SetDisplayRotation`
86    /// command) - only with the `gui` feature.
87    #[cfg(feature = "gui")]
88    display_rotation: DisplayRotation,
89    /// Pending remote profile capture (`TrainProfile` command) -
90    /// the device loops tick it with each frame's touches.
91    training: Option<TrainingState>,
92}

95    /// Creates the runtime with a set tracker for the given definitions and
96    /// the profile info of the loaded figure patterns.
97    ///
98    /// Returns the runtime (daemon side) and the handle handed to interface
99    /// crates.
100    pub fn new(sets: Vec<TangibleSet>, profiles: Vec<ProfileInfo>) -> (Self, EngineHandle) {
101        Self::with_capacity(
102            sets,
103            profiles,
104            EVENT_CHANNEL_CAPACITY,
105            COMMAND_CHANNEL_CAPACITY,
106        )
107    }
108
109    /// Like [`Self::new`] with explicit channel capacities.
110    pub fn with_capacity(
111        sets: Vec<TangibleSet>,
112        profiles: Vec<ProfileInfo>,
113        event_capacity: usize,
114        command_capacity: usize,
115    ) -> (Self, EngineHandle) {
116        let (events, _) = broadcast::channel(event_capacity);
117        let (snapshot_tx, snapshot_rx) = watch::channel(EngineSnapshot::default());
118        let (command_tx, commands) = mpsc::channel(command_capacity);
119        let profiles = Arc::new(RwLock::new(profiles));
120        let runtime = Self {
121            set_tracker: SetTracker::new(sets.clone()),
122            events: events.clone(),
123            snapshot: snapshot_tx,
124            commands,
125            device: None,
126            recording: None,
127            completion: None,
128            field_sets: Vec::new(),
129            dead_zones: Vec::new(),
130            profiles: profiles.clone(),
131            metrics: SessionMetrics::default(),
132            message: None,
133            #[cfg(feature = "gui")]
134            gui_visible: true,
135            #[cfg(feature = "gui")]
136            display_rotation: DisplayRotation::default(),
137            training: None,
138        };
139        let handle = EngineHandle::new(events, snapshot_rx, command_tx, Arc::new(sets), profiles);
140        (runtime, handle)
141    }

190    /// Publishes a processed frame: runs the set tracker, updates the
191    /// snapshot and broadcasts engine events followed by set actions.
192    pub fn publish(&mut self, frame: &EngineFrame, tracked: TrackedSet<'_>) {
193        let actions = self.set_tracker.update(tracked);
194        let touches_changed = frame
195            .events
196            .iter()
197            .any(|e| matches!(e, EngineEvent::TouchPointsUpdated));
198        self.metrics.record(
199            (&frame.touches).into(),
200            &frame.matches,
201            tracked,
202            touches_changed,
203            frame.ran_generic,
204            frame.candidates_tested,
205        );
206        self.snapshot.send_replace(EngineSnapshot {
207            tracked: tracked.to_vec(),
208            touches: frame.touches.clone(),
209            active_sets: self.set_tracker.active_set_ids(),
210            device: self.device.clone(),
211            recording: self.recording.clone(),
212            completion: self.completion,
213            field_sets: self.field_sets.clone(),
214            dead_zones: self.dead_zones.clone(),
215            metrics: self.metrics.clone(),
216            message: self.message.clone(),
217            #[cfg(feature = "gui")]
218            gui_visible: self.gui_visible,
219        });
220        for event in &frame.events {
221            self.broadcast(DomainEvent::Engine(event.clone()));
222        }
223        for action in actions {
224            self.broadcast(DomainEvent::Set(action));
225        }
226    }
227
228    /// Publishes the current engine state without a frame - used on the
229    /// command-drain wake-up path so interface-visible state (e.g. a stopped
230    /// recording) reaches clients even while the device is idle.
231    pub fn publish_snapshot(&mut self, touches: Vec<TouchPoint>, tracked: TrackedSet<'_>) {
232        let actions = self.set_tracker.update(tracked);
233        self.snapshot.send_replace(EngineSnapshot {
234            tracked: tracked.to_vec(),
235            touches,
236            active_sets: self.set_tracker.active_set_ids(),
237            device: self.device.clone(),
238            recording: self.recording.clone(),
239            completion: self.completion,
240            field_sets: self.field_sets.clone(),
241            dead_zones: self.dead_zones.clone(),
242            metrics: self.metrics.clone(),
243            message: self.message.clone(),
244            #[cfg(feature = "gui")]
245            gui_visible: self.gui_visible,
246        });
247        for action in actions {
248            self.broadcast(DomainEvent::Set(action));
249        }
250    }

300    /// Arms remote profile capture (`TrainProfile` command) - the
301    /// device loops tick [`TrainingState`] with each frame's touches.
302    pub fn start_training(&mut self, name: &PatternName) {
303        self.training = Some(TrainingState::new(name));
304    }
305
306    /// Advances remote training with this frame's touches; returns the
307    /// captured [`ProfileFile`] once the contact set stayed stable
308    /// (and clears the request).
309    pub fn training_tick(&mut self, touches: &[TouchPoint]) -> Option<ProfileFile> {
310        let state = self.training.as_mut()?;
311        let profile = state.tick(touches)?;
312        self.training = None;
313        Some(profile)
314    }

316    /// Next pending control command from an interface, if any.
317    ///
318    /// Non-blocking; intended to be drained once per frame in the
319    /// synchronous device loop.
320    pub fn try_recv_command(&mut self) -> Result<ControlCommand, TryRecvError> {
321        self.commands.try_recv()
322    }
323
324    /// Emits a [`LifecycleEvent`] to all domain-event subscribers.
325    ///
326    /// Called by the device loop when a control command *applies* - the
327    /// event means "this actually took effect", so emit it only after the
328    /// engine accepted the change (SUBSCRIPTIONS.md).
329    pub fn emit_lifecycle(&self, event: LifecycleEvent) {
330        self.broadcast(DomainEvent::Lifecycle(event));
331    }
332
333    /// Broadcasts a domain event to all subscribers.
334    ///
335    /// Zero receivers is the expected steady state when no interface is
336    /// attached; a full channel drops the oldest event for lagging
337    /// receivers but never blocks the device loop.
338    fn broadcast(&self, event: DomainEvent) {
339        let _ = self.events.send(event);
340    }

395    #[test]
396    fn publish_broadcasts_engine_events_then_set_actions() {
397        let tracked = vec![tangible(1, 1)];
398        let (mut runtime, handle) = EngineRuntime::new(vec![set(1, &[1])], Vec::new());
399        let mut receiver = handle.subscribe();
400
401        runtime.publish(
402            &frame(vec![
403                EngineEvent::TangibleAdded(tracked[0].clone()),
404                EngineEvent::TouchPointsUpdated,
405            ]),
406            (&tracked).into(),
407        );
408
409        assert!(matches!(
410            receiver.try_recv(),
411            Ok(DomainEvent::Engine(EngineEvent::TangibleAdded(_)))
412        ));
413        assert!(matches!(
414            receiver.try_recv(),
415            Ok(DomainEvent::Engine(EngineEvent::TouchPointsUpdated))
416        ));
417        assert!(matches!(
418            receiver.try_recv(),
419            Ok(DomainEvent::Set(SetAction::SetAdded(_)))
420        ));
421        assert!(matches!(
422            receiver.try_recv(),
423            Ok(DomainEvent::Set(SetAction::ObjectAdded(_, _)))
424        ));
425        assert!(receiver.try_recv().is_err());
426    }
