21/// Control command sent by an interface back to the daemon's device loop.
22///
23/// Commands are drained by the runtime after each published frame
24/// ([`crate::EngineRuntime::try_recv_command`]); the concrete behavior is
25/// implemented where the owning functionality lands (e.g. filter mode in
26/// Phase 4, remote training with the interface crates).
27#[derive(Debug)]
28pub enum ControlCommand {
29    /// Switch the filter proxy's suppression policy
30    /// ([`FilterMode::Off`] = transparent passthrough). `stability`
31    /// overrides the consecutive-matched-frames gate when `Some`; `None`
32    /// keeps the current value.
33    SetFilterMode {
34        /// New suppression policy.
35        mode: FilterMode,
36        /// Consecutive frames a slot must match before suppression starts.
37        stability: Option<FrameCount>,
38        /// Scope of the stage to reconfigure (EVENT_PIPELINE.md phase 7):
39        /// `Some` applies the policy to that stage only, `None`
40        /// broadcasts to all tangible stages. Positional stages always
41        /// ignore `SetFilterMode`.
42        stage: Option<StageName>,
43    },
44    /// Create or claim a pipeline stage for `name` before any `stage =
45    /// "name"` pattern registers - the stage persists until
46    /// `UnregisterStage`, independent of pattern lifetimes
47    /// (EVENT_PIPELINE.md phase 7). Lets a client pre-configure a
48    /// stage's `SetFilterMode` policy.
49    RegisterStage {
50        /// Stage scope name.
51        name: StageName,
52    },
53    /// Remove the pipeline stage for `name` - its suppression state
54    /// dies with it; patterns of that scope keep matching but are no
55    /// longer stage-suppressed until a new stage exists.
56    UnregisterStage {
57        /// Stage scope name.
58        name: StageName,
59    },
60    /// Capture a new figure profile with the given name.
61    TrainProfile(PatternName),
62    /// Start a recording session writing `<base_path>.jsonl` (frames) and
63    /// `<base_path>.evtest` (raw events).
64    StartRecording {
65        /// Base path of the recording artifacts, without extension.
66        base_path: PathBuf,
67        /// Ground-truth label applied to recorded frames until changed.
68        label: Option<RecordLabel>,
69    },
70    /// Stop the active recording session.
71    StopRecording,
72    /// Change the ground-truth label applied to subsequent recorded frames.
73    SetRecordLabel(Option<RecordLabel>),
74    /// Replace the missing-pin completion configuration at runtime
75    /// (TRACKING_STABILIZATION.md). Applies to matching and suppression in
76    /// all modes from the next frame on.
77    SetCompletion(CompletionConfig),
78    /// Reset the accumulated [`crate::SessionMetrics`] counters - e.g.
79    /// between automated test runs.
80    ResetMetrics,
81    /// Display a user-facing message in the GUI ("Legen Sie Figur 10
82    /// auf") - interfaces steer the operator during test series.
83    /// `None`-equivalent empty string clears it.
84    ShowMessage(String),
85    /// Register a profile or field set at runtime (DYNAMIC_PROFILES.md).
86    /// Registrations whose fingerprint collides with an already-loaded
87    /// profile are rejected in the device loop.
88    Register {
89        /// What to register - boxed: `ProfileFile` makes this variant
90        /// large; boxing keeps `ControlCommand` small for the channel.
91        registration: Box<Registration>,
92        /// Optional application name the registration belongs to -
93        /// `UnregisterOwner` removes every registration of an owner;
94        /// D-Bus binds registrations to the caller's unique name for
95        /// automatic disconnect cleanup.
96        owner: Option<Owner>,
97        /// `Session` (default) or `Persistent` - persistent registrations
98        /// are written to the daemon's runtime store on apply.
99        lifetime: Lifetime,
100        /// Receives the applied outcome - `Ok` carries what was
101        /// registered (the profile's assigned `PatternId`), the typed
102        /// rejection on failure - so interface callers answer with the
103        /// real result instead of a fire-and-forget "queued".
104        reply: oneshot::Sender<Result<RegistrationApplied, RegistrationError>>,
105    },
106    /// Replace a registered pattern's geometry/name while keeping its
107    /// ID (`UpdateProfile`). Applies to new matches; tracked tangibles
108    /// keep their recorded pose.
109    UpdateProfile {
110        /// Pattern to replace.
111        id: PatternId,
112        /// New profile definition - boxed like `Register`.
113        profile: Box<ProfileFile>,
114    },
115    /// Remove a dynamically registered pattern by its pattern ID. The ID
116    /// is never reused.
117    UnregisterPattern {
118        /// Pattern to remove.
119        id: PatternId,
120    },
121    /// Remove a field set by name.
122    UnregisterFieldSet {
123        /// Field set to remove.
124        name: FieldSetName,
125    },
126    /// Remove a dead zone by name.
127    UnregisterDeadZone {
128        /// Dead zone to remove.
129        name: DeadZoneName,
130    },
131    /// Remove every pattern, field set and dead zone owned by `owner` -
132    /// used for explicit app cleanup and D-Bus disconnect handling.
133    UnregisterOwner {
134        /// Owner name (or the caller's unique bus name on D-Bus).
135        owner: Owner,
136    },
137    /// Show or hide the daemon-embedded GUI window (`crates/gtk`,
138    /// `--gui` mode). Part of the command vocabulary on all builds -
139    /// daemons without the `gui` feature ignore it (the runtime's
140    /// `gui_visible` state is feature-gated).
141    SetGuiVisible(bool),
142    /// Quarter-turn the GUI canvas renders (`--display-rotation`).
143    /// Same availability as [`Self::SetGuiVisible`] - headless daemons
144    /// ignore it.
145    SetDisplayRotation(DisplayRotation),
146    /// Show or hide the dead-zone overlay on the GUI canvas. Same
147    /// availability as [`Self::SetGuiVisible`] - headless daemons
148    /// ignore it.
149    SetDeadZonesVisible(bool),
150    /// Canvas background opacity (`--background-alpha`). Same
151    /// availability as [`Self::SetGuiVisible`] - headless daemons
152    /// ignore it.
153    SetBackgroundAlpha(AlphaPercent),
154    /// Show or hide the edge-margin overlay on the GUI canvas -
155    /// the area where edge-ghost contacts are ingest-filtered.
156    /// Same availability as [`Self::SetGuiVisible`].
157    SetEdgeMarginVisible(bool),
158    /// Show or hide the field-set overlay on the GUI canvas - the
159    /// occupancy regions of every registered field set. Same
160    /// availability as [`Self::SetGuiVisible`].
161    SetFieldsVisible(bool),
162    /// Show or hide the ruler overlay on the GUI canvas - pixel ticks
163    /// on one edge, millimeter ticks on the other. Same availability
164    /// as [`Self::SetGuiVisible`].
165    SetRulerVisible(bool),
166    /// Show or hide the grid overlay on the GUI canvas - a millimeter
167    /// lattice across the whole surface. Same availability as
168    /// [`Self::SetGuiVisible`].
169    SetGridVisible(bool),
170    /// Show or hide the positions overlay on the GUI canvas - mm and
171    /// pixel coordinates of tangibles, touch points and field-set
172    /// fields. Same availability as [`Self::SetGuiVisible`].
173    SetPositionsVisible(bool),
174    /// Let touch events pass through the GTK window to the desktop
175    /// beneath - except where a GUI element (control bar) is hit.
176    /// Same availability as [`Self::SetGuiVisible`]; Wayland only.
177    SetTouchPassthrough(bool),
178    /// Enable or disable a registered pattern at runtime - disabled
179    /// profiles stay registered but produce no matches and spawn no
180    /// tangibles. The GUI toggles it from the profiles card.
181    SetProfileEnabled {
182        /// Pattern to toggle.
183        id: PatternId,
184        /// New enabled state.
185        enabled: bool,
186    },
187    /// Enable or disable a registered field set at runtime - disabled
188    /// sets keep their registration but track no occupancy and
189    /// suppress nothing. The GUI toggles it from the field-sets card.
190    SetFieldSetEnabled {
191        /// Field set to toggle.
192        name: FieldSetName,
193        /// New enabled state.
194        enabled: bool,
195    },
196}

198impl ControlCommand {
199    /// Builds a `Register` command plus the receiver its caller awaits
200    /// for the applied outcome. Internal callers that never await simply
201    /// drop the receiver - `reply.send` in the device loop tolerates a
202    /// closed channel.
203    pub fn register(
204        registration: Registration,
205        owner: Option<Owner>,
206        lifetime: Lifetime,
207    ) -> (
208        Self,
209        oneshot::Receiver<Result<RegistrationApplied, RegistrationError>>,
210    ) {
211        let (reply, rx) = oneshot::channel();
212        (
213            Self::Register {
214                registration: Box::new(registration),
215                owner,
216                lifetime,
217                reply,
218            },
219            rx,
220        )
221    }
222}
