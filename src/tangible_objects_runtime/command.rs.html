19/// Control command sent by an interface back to the daemon's device loop.
20///
21/// Commands are drained by the runtime after each published frame
22/// ([`crate::EngineRuntime::try_recv_command`]); the concrete behavior is
23/// implemented where the owning functionality lands (e.g. filter mode in
24/// Phase 4, remote training with the interface crates).
25#[derive(Debug)]
26pub enum ControlCommand {
27    /// Switch the filter proxy's suppression policy
28    /// ([`FilterMode::Off`] = transparent passthrough). `stability`
29    /// overrides the consecutive-matched-frames gate when `Some`; `None`
30    /// keeps the current value.
31    SetFilterMode {
32        /// New suppression policy.
33        mode: FilterMode,
34        /// Consecutive frames a slot must match before suppression starts.
35        stability: Option<FrameCount>,
36        /// Scope of the stage to reconfigure (EVENT_PIPELINE.md phase 7):
37        /// `Some` applies the policy to that stage only, `None`
38        /// broadcasts to all tangible stages. Positional stages always
39        /// ignore `SetFilterMode`.
40        stage: Option<StageName>,
41    },
42    /// Create or claim a pipeline stage for `name` before any `stage =
43    /// "name"` pattern registers - the stage persists until
44    /// `UnregisterStage`, independent of pattern lifetimes
45    /// (EVENT_PIPELINE.md phase 7). Lets a client pre-configure a
46    /// stage's `SetFilterMode` policy.
47    RegisterStage {
48        /// Stage scope name.
49        name: StageName,
50    },
51    /// Remove the pipeline stage for `name` - its suppression state
52    /// dies with it; patterns of that scope keep matching but are no
53    /// longer stage-suppressed until a new stage exists.
54    UnregisterStage {
55        /// Stage scope name.
56        name: StageName,
57    },
58    /// Capture a new figure profile with the given name.
59    TrainProfile(PatternName),
60    /// Start a recording session writing `<base_path>.jsonl` (frames) and
61    /// `<base_path>.evtest` (raw events).
62    StartRecording {
63        /// Base path of the recording artifacts, without extension.
64        base_path: PathBuf,
65        /// Ground-truth label applied to recorded frames until changed.
66        label: Option<RecordLabel>,
67    },
68    /// Stop the active recording session.
69    StopRecording,
70    /// Change the ground-truth label applied to subsequent recorded frames.
71    SetRecordLabel(Option<RecordLabel>),
72    /// Replace the missing-pin completion configuration at runtime
73    /// (TRACKING_STABILIZATION.md). Applies to matching and suppression in
74    /// all modes from the next frame on.
75    SetCompletion(CompletionConfig),
76    /// Reset the accumulated [`crate::SessionMetrics`] counters - e.g.
77    /// between automated test runs.
78    ResetMetrics,
79    /// Display a user-facing message in the GUI ("Legen Sie Figur 10
80    /// auf") - interfaces steer the operator during test series.
81    /// `None`-equivalent empty string clears it.
82    ShowMessage(String),
83    /// Register a profile or field set at runtime (DYNAMIC_PROFILES.md).
84    /// Registrations whose fingerprint collides with an already-loaded
85    /// profile are rejected in the device loop.
86    Register {
87        /// What to register - boxed: `ProfileFile` makes this variant
88        /// large; boxing keeps `ControlCommand` small for the channel.
89        registration: Box<Registration>,
90        /// Optional application name the registration belongs to -
91        /// `UnregisterOwner` removes every registration of an owner;
92        /// D-Bus binds registrations to the caller's unique name for
93        /// automatic disconnect cleanup.
94        owner: Option<Owner>,
95        /// `Session` (default) or `Persistent` - persistent registrations
96        /// are written to the daemon's runtime store on apply.
97        lifetime: Lifetime,
98        /// Receives the applied outcome - `Ok` carries what was
99        /// registered (the profile's assigned `PatternId`), the typed
100        /// rejection on failure - so interface callers answer with the
101        /// real result instead of a fire-and-forget "queued".
102        reply: oneshot::Sender<Result<RegistrationApplied, RegistrationError>>,
103    },
104    /// Replace a registered pattern's geometry/name while keeping its
105    /// ID (`UpdateProfile`). Applies to new matches; tracked tangibles
106    /// keep their recorded pose.
107    UpdateProfile {
108        /// Pattern to replace.
109        id: PatternId,
110        /// New profile definition - boxed like `Register`.
111        profile: Box<ProfileFile>,
112    },
113    /// Remove a dynamically registered pattern by its pattern ID. The ID
114    /// is never reused.
115    UnregisterPattern {
116        /// Pattern to remove.
117        id: PatternId,
118    },
119    /// Remove a field set by name.
120    UnregisterFieldSet {
121        /// Field set to remove.
122        name: FieldSetName,
123    },
124    /// Remove every pattern and field set owned by `owner` - used for
125    /// explicit app cleanup and D-Bus disconnect handling.
126    UnregisterOwner {
127        /// Owner name (or the caller's unique bus name on D-Bus).
128        owner: Owner,
129    },
130    /// Show or hide the daemon-embedded GUI window (`crates/gtk`,
131    /// `--gui` mode). Part of the command vocabulary on all builds -
132    /// daemons without the `gui` feature ignore it (the runtime's
133    /// `gui_visible` state is feature-gated).
134    SetGuiVisible(bool),
135    /// Quarter-turn the GUI canvas renders (`--display-rotation`).
136    /// Same availability as [`Self::SetGuiVisible`] - headless daemons
137    /// ignore it.
138    SetDisplayRotation(DisplayRotation),
139}

141impl ControlCommand {
142    /// Builds a `Register` command plus the receiver its caller awaits
143    /// for the applied outcome. Internal callers that never await simply
144    /// drop the receiver - `reply.send` in the device loop tolerates a
145    /// closed channel.
146    pub fn register(
147        registration: Registration,
148        owner: Option<Owner>,
149        lifetime: Lifetime,
150    ) -> (
151        Self,
152        oneshot::Receiver<Result<RegistrationApplied, RegistrationError>>,
153    ) {
154        let (reply, rx) = oneshot::channel();
155        (
156            Self::Register {
157                registration: Box::new(registration),
158                owner,
159                lifetime,
160                reply,
161            },
162            rx,
163        )
164    }
165}
