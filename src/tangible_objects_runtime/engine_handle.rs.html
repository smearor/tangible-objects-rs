18/// Cloneable consumer-side handle for interface crates.
19///
20/// Provides event subscription, snapshot access and the control channel -
21/// everything an interface needs without engine access.
22#[derive(Clone)]
23pub struct EngineHandle {
24    /// Broadcast sender for subscribing to the domain event stream.
25    events: broadcast::Sender<DomainEvent>,
26    /// Watch receiver for the latest snapshot (`snapshot()`,
27    /// `watch_snapshot()`).
28    snapshot: watch::Receiver<EngineSnapshot>,
29    /// Sending half of the control channel - interface commands reach
30    /// the device loop through it.
31    commands: mpsc::Sender<ControlCommand>,
32    /// Static Tangible Object Set definitions loaded at session start.
33    sets: Arc<Vec<TangibleSet>>,
34    /// Profile list shared with the runtime - dynamic registrations are
35    /// published through it.
36    profiles: Arc<RwLock<Vec<ProfileInfo>>>,
37    /// Seat-session registry shared with the runtime - session
38    /// claims and roster changes are visible to interface readers
39    /// between frames (the `profiles` channel's seat counterpart).
40    seat_tracker: Arc<RwLock<SeatTracker>>,
41}
42
43impl EngineHandle {
44    /// Creates the consumer-side channel ends - called by
45    /// `EngineRuntime::with_capacity`, which pairs the halves.
46    pub(super) fn new(
47        events: broadcast::Sender<DomainEvent>,
48        snapshot: watch::Receiver<EngineSnapshot>,
49        commands: mpsc::Sender<ControlCommand>,
50        sets: Arc<Vec<TangibleSet>>,
51        profiles: Arc<RwLock<Vec<ProfileInfo>>>,
52        seat_tracker: Arc<RwLock<SeatTracker>>,
53    ) -> Self {
54        Self {
55            events,
56            snapshot,
57            commands,
58            sets,
59            profiles,
60            seat_tracker,
61        }
62    }
63
64    /// Subscribes to the domain event stream.
65    ///
66    /// Each call returns an independent receiver; lagging receivers drop
67    /// the oldest events.
68    pub fn subscribe(&self) -> broadcast::Receiver<DomainEvent> {
69        self.events.subscribe()
70    }
71
72    /// The latest published snapshot.
73    pub fn snapshot(&self) -> EngineSnapshot {
74        self.snapshot.borrow().clone()
75    }
76
77    /// Watches the snapshot for change notifications.
78    pub fn watch_snapshot(&self) -> watch::Receiver<EngineSnapshot> {
79        self.snapshot.clone()
80    }
81
82    /// The configured Tangible Object Set definitions.
83    pub fn sets(&self) -> &[TangibleSet] {
84        &self.sets
85    }
86
87    /// The claimed seat sessions with their rosters - reads the live
88    /// registry so claim/roster changes are visible between frames
89    /// (TANGIBLE_OBJECTS_TABLE_TOP_SEAT.md).
90    pub fn seat_sessions(&self) -> Vec<SeatSession> {
91        self.seat_tracker
92            .read()
93            .map(|tracker| tracker.sessions().to_vec())
94            .unwrap_or_default()
95    }
96
97    /// Info of the loaded plus dynamically registered figure profiles.
98    pub fn profiles(&self) -> Vec<ProfileInfo> {
99        self.profiles
100            .read()
101            .map(|profiles| profiles.clone())
102            .unwrap_or_default()
103    }
104
105    /// Sends a control command to the daemon's device loop.
106    pub async fn send_command(
107        &self,
108        command: ControlCommand,
109    ) -> Result<(), SendError<ControlCommand>> {
110        self.commands.send(command).await
111    }
112
113    /// Non-blocking variant of [`Self::send_command`].
114    pub fn try_send_command(
115        &self,
116        command: ControlCommand,
117    ) -> Result<(), TrySendError<ControlCommand>> {
118        self.commands.try_send(command)
119    }
120}
