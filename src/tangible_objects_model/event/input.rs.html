1//! Linux input event code constants (multitouch protocol type B).
2//!
3//! Event types are modeled by [`super::EventType`]; the constants below
4//! name the event codes interpreted by the pipeline.
5
6use super::EventCode;
7
8/// Synchronization code delimiting one input frame.
9pub const SYN_REPORT: EventCode = EventCode::new(0x00);
10
11/// `SYN_DROPPED` - the kernel's resync signal: the reader's event queue
12/// overflowed, events were dropped and the client's accumulated state
13/// may diverge from kernel truth (STALE_SLOTS root cause). The correct
14/// client response is an `EVIOCGMTSLOTS` resync, never heuristics.
15pub const SYN_DROPPED: EventCode = EventCode::new(0x03);
16
17/// Absolute multitouch code selecting the active slot.
18pub const ABS_MT_SLOT: EventCode = EventCode::new(47);
19
20/// Absolute multitouch code for the contact X position.
21pub const ABS_MT_POSITION_X: EventCode = EventCode::new(53);
22
23/// Absolute multitouch code for the contact Y position.
24pub const ABS_MT_POSITION_Y: EventCode = EventCode::new(54);
25
26/// Absolute multitouch code for the contact tracking ID (-1 removes the slot).
27pub const ABS_MT_TRACKING_ID: EventCode = EventCode::new(57);
28
29/// Legacy single-touch axis: X position of the (first) contact.
30/// Not slot-scoped - mirrors whether any contact is active.
31pub const ABS_X: EventCode = EventCode::new(0);
32
33/// Legacy single-touch axis: Y position of the (first) contact.
34pub const ABS_Y: EventCode = EventCode::new(1);
35
36/// Legacy single-touch key: pressed while at least one contact exists.
37/// Not slot-scoped.
38pub const BTN_TOUCH: EventCode = EventCode::new(330);
