11/// One floating card's contribution to the touch-passthrough input
12/// region, as last computed by the GUI's region sync.
13///
14/// Diagnostics only: comparing `position`/`rotation`/`scale` with the
15/// rasterized `footprint` bounds shows whether the surface input
16/// region actually covers the card where it renders.
17#[derive(Debug, Clone, Serialize, Deserialize)]
18pub struct CardFootprint {
19    /// Stable card identifier.
20    pub card_id: CardId,
21    /// Card center in board-layer pixels.
22    pub position: Vec2<Pixels>,
23    /// Card rotation around its center.
24    pub rotation: Rotation,
25    /// Card scale factor.
26    pub scale: ScaleFactor,
27    /// Whether the card's transform stack is GTK-visible.
28    pub visible: bool,
29    /// Rasterized footprint bounds in board-layer pixels; `None`
30    /// when the card could not be measured or rasterized - a missing
31    /// footprint makes the card completely input-dead.
32    pub footprint: Option<Rect<Pixels, i32>>,
33}
