11/// A rotation angle in radians.
12///
13/// Wraps the plain `f64` so angles cannot be confused with distances or
14/// other scalar values flowing through the pipeline.
15#[derive(Debug, Clone, Copy, PartialEq, PartialOrd, Serialize, Deserialize)]
16#[serde(transparent)]
17pub struct RotationRad(pub f64);
18
19impl RotationRad {
20    /// Creates a rotation from radians.
21    pub const fn new(radians: f64) -> Self {
22        Self(radians)
23    }
24
25    /// The angle in radians.
26    pub const fn as_f64(self) -> f64 {
27        self.0
28    }
29
30    /// The angle converted to degrees.
31    pub fn to_degrees(self) -> f64 {
32        self.0.to_degrees()
33    }
34
35    /// `(sin, cos)` of the angle - trigonomtrie for transforms and
36    /// geometry (the `f64::sin_cos` delegate).
37    pub fn sin_cos(self) -> (f64, f64) {
38        self.0.sin_cos()
39    }
40
41    /// The interior angle between two consecutive edge directions
42    /// of a positive-area polygon: `PI - signed turn` from
43    /// `previous` to `next` (sharp corners < ~PI/2).
44    pub fn interior<U>(previous: Vec2<U>, next: Vec2<U>) -> Self {
45        Self(
46            PI - previous
47                .cross::<f64>(&next)
48                .atan2(previous.dot::<f64>(&next)),
49        )
50    }
51
52    /// The shortest signed arc from `self` to `to`, in [-π, π].
53    ///
54    /// The ±PI wrap is crossed instead of going the long way around;
55    /// the sign tells the turning direction.
56    pub fn signed_delta_to(self, to: RotationRad) -> RotationRad {
57        let mut delta = to.0 - self.0;
58        while delta > PI {
59            delta -= 2.0 * PI;
60        }
61        while delta < -PI {
62            delta += 2.0 * PI;
63        }
64        RotationRad(delta)
65    }
66
67    /// Wraparound-aware EMA blend - `old + alpha * diff(new, old)`,
68    /// where `diff` is the shortest signed arc between the angles
69    /// (modulo 2π). `alpha` is the weight of `new`.
70    pub fn ema(old: RotationRad, new: RotationRad, alpha: SmoothingAlpha) -> RotationRad {
71        Self(old.0 + alpha.as_f64() * old.signed_delta_to(new).0)
72    }
73}

75impl From<f64> for RotationRad {
76    fn from(radians: f64) -> Self {
77        Self(radians)
78    }
79}
80
81impl From<RotationRad> for f64 {
82    fn from(rotation: RotationRad) -> Self {
83        rotation.0
84    }
85}

87impl Add for RotationRad {
88    type Output = RotationRad;
89
90    /// Composes two rotations.
91    fn add(self, other: RotationRad) -> RotationRad {
92        RotationRad(self.0 + other.0)
93    }
94}
95
96impl AddAssign for RotationRad {
97    /// Adds a rotation in place.
98    fn add_assign(&mut self, other: RotationRad) {
99        self.0 += other.0;
100    }
101}
102
103impl Sub for RotationRad {
104    type Output = RotationRad;
105
106    /// The signed difference between two rotations.
107    fn sub(self, other: RotationRad) -> RotationRad {
108        RotationRad(self.0 - other.0)
109    }
110}
111
112impl Neg for RotationRad {
113    type Output = RotationRad;
114
115    /// The inverse rotation.
116    fn neg(self) -> RotationRad {
117        RotationRad(-self.0)
118    }
119}

124#[cfg(test)]
125mod tests {
126    use super::RotationRad;
127    use crate::SmoothingAlpha;
128    use crate::Vec2;
129    use std::f64::consts::PI;
130
131    /// `alpha = 1` makes the blend pure `new` - history has no weight.
132    #[test]
133    fn ema_full_weight_is_new() {
134        let result = RotationRad::ema(
135            RotationRad::new(1.0),
136            RotationRad::new(2.0),
137            SmoothingAlpha::new(1.0),
138        );
139        assert_eq!(result.as_f64(), 2.0);
140    }
141
142    /// `alpha = 0` keeps `old` - a frozen pose ignores measurement.
143    #[test]
144    fn ema_zero_weight_is_old() {
145        let result = RotationRad::ema(
146            RotationRad::new(1.0),
147            RotationRad::new(2.0),
148            SmoothingAlpha::new(0.0),
149        );
150        assert_eq!(result.as_f64(), 1.0);
151    }
152
153    /// The shortest arc crosses ±π upward: `3.0 → -3.0` passes over
154    /// π, not back through 0.
155    #[test]
156    fn ema_crosses_pi_boundary_shortest_way() {
157        let result = RotationRad::ema(
158            RotationRad::new(3.0),
159            RotationRad::new(-3.0),
160            SmoothingAlpha::new(0.5),
161        );
162        assert!(result.as_f64() > 3.0);
163        assert!(result.as_f64() <= PI + 1e-9);
164    }
165
166    /// Within ±PI the arc is the plain difference.
167    #[test]
168    fn signed_delta_inside_bounds() {
169        let delta = RotationRad::new(1.0).signed_delta_to(RotationRad::new(2.0));
170        assert_eq!(delta.as_f64(), 1.0);
171    }
172
173    /// `3.0 -> -3.0` wraps across +PI (delta ~= +0.28), not -6 the
174    /// long way.
175    #[test]
176    fn signed_delta_wraps_shortest_way() {
177        let delta = RotationRad::new(3.0).signed_delta_to(RotationRad::new(-3.0));
178        assert!((delta.as_f64() - (2.0 * PI - 6.0)).abs() < 1e-9);
179    }
180
181    /// The mirrored crossing reports the negative direction.
182    #[test]
183    fn signed_delta_wraps_negative_side() {
184        let delta = RotationRad::new(-3.0).signed_delta_to(RotationRad::new(3.0));
185        assert!((delta.as_f64() - (6.0 - 2.0 * PI)).abs() < 1e-9);
186    }
187    #[test]
188    fn interior_angle_of_a_right_turn() {
189        let angle = RotationRad::interior(
190            crate::Vec2::<crate::Millimeters>::new(1.0, 0.0),
191            crate::Vec2::new(0.0, 1.0),
192        );
193        assert!((angle.as_f64() - std::f64::consts::FRAC_PI_2).abs() < 1e-9);
194    }
195}
