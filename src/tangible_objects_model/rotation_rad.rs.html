10/// A rotation angle in radians.
11///
12/// Wraps the plain `f64` so angles cannot be confused with distances or
13/// other scalar values flowing through the pipeline.
14#[derive(Debug, Clone, Copy, PartialEq, PartialOrd, Serialize, Deserialize)]
15#[serde(transparent)]
16pub struct RotationRad(pub f64);
17
18impl RotationRad {
19    /// Creates a rotation from radians.
20    pub const fn new(radians: f64) -> Self {
21        Self(radians)
22    }
23
24    /// The angle in radians.
25    pub const fn as_f64(self) -> f64 {
26        self.0
27    }
28
29    /// The angle converted to degrees.
30    pub fn to_degrees(self) -> f64 {
31        self.0.to_degrees()
32    }
33
34    /// `(sin, cos)` of the angle - trigonomtrie for transforms and
35    /// geometry (the `f64::sin_cos` delegate).
36    pub fn sin_cos(self) -> (f64, f64) {
37        self.0.sin_cos()
38    }
39
40    /// Wraparound-aware EMA blend - `old + alpha * diff(new, old)`,
41    /// where `diff` is the shortest signed arc between the angles
42    /// (modulo 2π). `alpha` is the weight of `new`.
43    pub fn ema(old: RotationRad, new: RotationRad, alpha: SmoothingAlpha) -> RotationRad {
44        let mut diff = new.0 - old.0;
45        while diff > PI {
46            diff -= 2.0 * PI;
47        }
48        while diff < -PI {
49            diff += 2.0 * PI;
50        }
51        Self(old.0 + alpha.as_f64() * diff)
52    }
53}
54
55impl From<f64> for RotationRad {
56    fn from(radians: f64) -> Self {
57        Self(radians)
58    }
59}
60
61impl From<RotationRad> for f64 {
62    fn from(rotation: RotationRad) -> Self {

64    }
65}

67impl Add for RotationRad {
68    type Output = RotationRad;
69
70    /// Composes two rotations.
71    fn add(self, other: RotationRad) -> RotationRad {
72        RotationRad(self.0 + other.0)
73    }
74}

76impl AddAssign for RotationRad {
77    /// Adds a rotation in place.
78    fn add_assign(&mut self, other: RotationRad) {
79        self.0 += other.0;
80    }
81}

83impl Sub for RotationRad {
84    type Output = RotationRad;
85
86    /// The signed difference between two rotations.
87    fn sub(self, other: RotationRad) -> RotationRad {

89    }

104#[cfg(test)]
105mod tests {
106    use super::RotationRad;
107    use crate::SmoothingAlpha;
108    use std::f64::consts::PI;
109
110    /// `alpha = 1` makes the blend pure `new` - history has no weight.
111    #[test]
112    fn ema_full_weight_is_new() {
113        let result = RotationRad::ema(
114            RotationRad::new(1.0),
115            RotationRad::new(2.0),
116            SmoothingAlpha::new(1.0),
117        );
118        assert_eq!(result.as_f64(), 2.0);
119    }
120
121    /// `alpha = 0` keeps `old` - a frozen pose ignores measurement.
122    #[test]
123    fn ema_zero_weight_is_old() {
124        let result = RotationRad::ema(
125            RotationRad::new(1.0),
126            RotationRad::new(2.0),
127            SmoothingAlpha::new(0.0),
128        );
129        assert_eq!(result.as_f64(), 1.0);
130    }
131
132    /// The shortest arc crosses ±π upward: `3.0 → -3.0` passes over
133    /// π, not back through 0.
134    #[test]
135    fn ema_crosses_pi_boundary_shortest_way() {
136        let result = RotationRad::ema(
137            RotationRad::new(3.0),
138            RotationRad::new(-3.0),
139            SmoothingAlpha::new(0.5),
140        );
141        assert!(result.as_f64() > 3.0);
142        assert!(result.as_f64() <= PI + 1e-9);
143    }
