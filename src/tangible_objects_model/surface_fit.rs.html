7/// Composed mm↔px mapping of a surface onto a display area.
8///
9/// Bundles the three pieces every renderer or hit-test needs so they
10/// cannot drift apart: the display `rotation` (the evdev stream always
11/// speaks the panel's native frame), the uniform `scale`
12/// ([`PixelsPerMm`], letterboxed so figures keep their physical
13/// aspect) and the `offset` centering the fitted surface inside the
14/// area. Built by `Calibration::surface_fit`; `map` is the render
15/// direction, `unmap` the hit-test direction.
16#[derive(Debug, Clone, Copy, PartialEq)]
17pub struct SurfaceFit {
18    /// Uniform scale applied after rotation - same factor on both
19    /// axes (uniform fit, no aspect distortion).
20    pub scale: PixelsPerMm,
21    /// Letterbox offset of the rotated surface inside the display
22    /// area, in pixels.
23    pub offset: Vec2<Pixels>,
24    /// Quarter turn applied to mm positions before scaling.
25    pub rotation: DisplayRotation,
26    /// Rotated surface size in millimeters - `(h, w)` when the quarter
27    /// turn swaps axes.
28    pub size: Vec2<Millimeters>,
29}
30
31impl SurfaceFit {
32    /// Builds the fit for a `w_mm` × `h_mm` surface inside a
33    /// `w_px` × `h_px` display area under `rotation`.
34    pub fn new(
35        w_mm: Millimeters,
36        h_mm: Millimeters,
37        w_px: Pixels,
38        h_px: Pixels,
39        rotation: DisplayRotation,
40    ) -> Self {
41        let (w_disp, h_disp) = if rotation.swaps_axes() {
42            (h_mm, w_mm)
43        } else {
44            (w_mm, h_mm)
45        };
46        let scale = if w_disp.as_f64() > 0.0 && h_disp.as_f64() > 0.0 {
47            (w_px.as_f64() / w_disp.as_f64()).min(h_px.as_f64() / h_disp.as_f64())
48        } else {
49            0.0
50        };
51        Self {
52            scale: PixelsPerMm::new(scale),
53            offset: Vec2::new(
54                (w_px.as_f64() - w_disp.as_f64() * scale) / 2.0,
55                (h_px.as_f64() - h_disp.as_f64() * scale) / 2.0,
56            ),
57            rotation,
58            size: Vec2::new(w_disp.as_f64(), h_disp.as_f64()),
59        }
60    }
61
62    /// Maps a millimeter position (panel-native frame) onto display
63    /// pixels: rotate → uniform scale → letterbox offset.
64    pub fn map(&self, pos: Vec2<Millimeters>) -> Vec2<Pixels> {
65        let rotated = self.rotation.apply(
66            pos,
67            if self.rotation.swaps_axes() {
68                Millimeters::new(self.size.y)
69            } else {
70                Millimeters::new(self.size.x)
71            },
72            if self.rotation.swaps_axes() {
73                Millimeters::new(self.size.x)
74            } else {
75                Millimeters::new(self.size.y)
76            },
77        );
78        let scaled = self.scale.scale_vec(rotated);
79        self.offset + scaled
80    }
81
82    /// Maps a display pixel back onto surface millimeters (panel-native
83    /// frame) - the hit-testing direction: inverse offset → unscale →
84    /// inverse rotation.
85    pub fn unmap(&self, px: Vec2<Pixels>) -> Vec2<Millimeters> {
86        let scaled = self.scale.unscale_vec(px - self.offset);
87        self.rotation.inverse().apply(
88            scaled,
89            Millimeters::new(self.size.x),
90            Millimeters::new(self.size.y),
91        )
92    }
93}

95#[cfg(test)]
96mod tests {
97    use super::SurfaceFit;
98    use crate::DisplayRotation;
99    use crate::Millimeters;
100    use crate::Pixels;
101    use crate::Vec2;
102
103    /// 400×300 mm surface on a 1000×1000 px display.
104    fn fit(rotation: DisplayRotation) -> SurfaceFit {
105        SurfaceFit::new(
106            Millimeters::new(400.0),
107            Millimeters::new(300.0),
108            Pixels::new(1000.0),
109            Pixels::new(1000.0),
110            rotation,
111        )
112    }
113
114    #[test]
115    fn map_scales_and_letterboxes() {
116        let fit = fit(DisplayRotation::Deg0);
117        // min(2.5, 3.33) = 2.5 px/mm, vertical letterbox 125 px
118        assert!((fit.scale.as_f64() - 2.5).abs() < 1e-9);
119        let mapped = fit.map(Vec2::<Millimeters>::new(0.0, 0.0));
120        assert!((mapped.y - 125.0).abs() < 1e-9);
121        assert!((fit.map(Vec2::<Millimeters>::new(400.0, 300.0)).x - 1000.0).abs() < 1e-9);
122    }
123
124    #[test]
125    fn unmap_inverts_map_for_every_rotation() {
126        let point = Vec2::<Millimeters>::new(111.0, 222.0);
127        for rotation in [
128            DisplayRotation::Deg0,
129            DisplayRotation::Deg90,
130            DisplayRotation::Deg180,
131            DisplayRotation::Deg270,
132        ] {
133            let fit = fit(rotation);
134            let back = fit.unmap(fit.map(point));
135            assert!(
136                (back.x - point.x).abs() < 1e-6 && (back.y - point.y).abs() < 1e-6,
137                "{rotation:?} round-trip: ({},{}) vs ({},{})",
138                point.x,
139                point.y,
140                back.x,
141                back.y
142            );
143        }
144    }
