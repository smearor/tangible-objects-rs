17/// Default debounce stability for field sets (2 frames).
18pub fn default_stability() -> FrameCount {
19    FrameCount::new(2)

22/// A named set of occupancy fields - the application-supplied board.
23///
24/// Loaded from a `.toml` field-set file:
25///
26/// ```toml
27/// name = "chess"
28/// stability = 2
29///
30/// [window]          # optional: fields are window-relative
31/// x_offset = 50.0
32/// y_offset = 30.0
33///
34/// [[fields]]
35/// name = "a1"
36/// x_min = 0.0
37/// y_min = 0.0
38/// x_max = 100.0
39/// y_max = 100.0
40/// ```
41#[derive(Debug, Clone, Serialize, Deserialize)]
42#[cfg_attr(feature = "async-graphql", derive(async_graphql::SimpleObject))]
43#[cfg_attr(feature = "cynic", derive(cynic::QueryFragment))]
44#[cfg_attr(
45    feature = "cynic",
46    cynic(
47        schema_path = "../graphql/schema.graphql",
48        graphql_type = "FieldSet",
49        no_deserialize,
50        rename_all = "camelCase"
51    )
52)]
53pub struct FieldSet {
54    /// Human-readable set name - the namespace of its fields.
55    pub name: FieldSetName,
56    /// Optional window anchor: field coordinates are interpreted
57    /// relative to the application window instead of screen-absolute.
58    #[serde(default, skip_serializing_if = "Option::is_none")]
59    pub window: Option<WindowAnchor>,
60    /// Consecutive frames required to confirm a count transition
61    /// (debouncing, symmetric for place and lift). Default 2.
62    #[serde(default = "default_stability")]
63    pub stability: FrameCount,
64    /// The occupancy cells of this set. Must be mutually disjoint -
65    /// see [`Self::is_disjoint`].
66    #[serde(rename = "fields")]
67    pub fields: Vec<Field>,
68    /// Whether contacts inside this set's fields are suppressed by the
69    /// filter proxy (they never reach the OS). Default `false` -
70    /// occupancy is informational only.
71    #[serde(default)]
72    pub suppress: bool,
73}
74
75impl FieldSet {
76    /// Whether `touch` lies inside `field` - evaluated in the field's
77    /// coordinate space: window-relative when the set carries a
78    /// `window` anchor, screen-absolute otherwise.
79    pub fn field_contains(&self, field: &Field, touch: &TouchPoint) -> bool {
80        let local = match &self.window {
81            Some(anchor) => anchor.to_local(&touch.pos),
82            None => Vec2::new(touch.pos.x, touch.pos.y),
83        };
84        field.region.contains(&local)
85    }
86
87    /// Whether all fields are mutually disjoint (no overlapping cells).
88    /// Registration of an overlapping set is rejected.
89    pub fn is_disjoint(&self) -> bool {
90        self.overlap_pair().is_none()
91    }
92
93    /// The first overlapping field pair plus its intersection region -
94    /// carries the fields (names + regions) so a rejection can say
95    /// *where* they collide, not just which.
96    pub fn overlap_pair(&self) -> Option<(&Field, &Field, Region)> {
97        for (i, a) in self.fields.iter().enumerate() {
98            for b in &self.fields[i + 1..] {
99                if let Some(overlap) = a.region.intersection(&b.region) {
100                    return Some((a, b, overlap));
101                }
102            }
103        }
104        None
105    }
106
107    /// Loads a field set from a `.toml` file.
108    pub fn load(path: &Path) -> std::io::Result<Self> {
109        let content = std::fs::read_to_string(path)?;
110        toml::from_str(&content)
111            .map_err(|error| std::io::Error::new(std::io::ErrorKind::InvalidData, error))
112    }
113
114    /// Loads all `.toml` field-set files from a directory, sorted by
115    /// filename. Unreadable or invalid files are skipped and returned
116    /// in the error list.
117    pub fn load_dir(dir: &Path) -> (Vec<Self>, Vec<(PathBuf, std::io::Error)>) {
118        let mut sets = Vec::new();
119        let mut errors = Vec::new();
120        let Ok(entries) = std::fs::read_dir(dir) else {
121            return (sets, errors);
122        };
123        let mut files: Vec<PathBuf> = entries
124            .filter_map(|entry| entry.ok())
125            .map(|entry| entry.path())
126            .filter(|path| path.extension().is_some_and(|ext| ext == "toml"))
127            .collect();
128        files.sort();
129        for file in files {
130            match Self::load(&file) {
131                Ok(set) => sets.push(set),
132                Err(error) => errors.push((file, error)),
133            }
134        }
135        (sets, errors)
136    }
137}

144    fn parse(toml: &str) -> Result<FieldSet, toml::de::Error> {

148    #[test]
149    fn toml_parses_fields_and_defaults() -> Result<(), Box<dyn std::error::Error>> {
150        let set = parse(
151            r#"
152            name = "chess"
153
154            [[fields]]
155            name = "a1"
156            x_min = 0.0
157            y_min = 0.0
158            x_max = 100.0
159            y_max = 100.0
160            "#,
161        )?;
162        assert_eq!(set.name.as_str(), "chess");
163        assert_eq!(set.stability.as_u64(), 2);
164        assert!(set.window.is_none());
165        assert_eq!(set.fields.len(), 1);
166        assert!(set.is_disjoint());
167        Ok(())
168    }

170    #[test]
171    fn window_anchor_and_normalized_fields() -> Result<(), Box<dyn std::error::Error>> {
172        let set = parse(
173            r#"
174            name = "board"
175            [window]
176            x_offset = 50.0
177            y_offset = 30.0
178            width = 400.0
179            height = 400.0
180
181            [[fields]]
182            name = "cell"
183            x_min = 0.0
184            y_min = 0.0
185            x_max = 0.5
186            y_max = 0.5
187            "#,
188        )?;
189        let anchor = set.window.ok_or("window anchor expected")?;
190        let local = anchor.to_local(&Vec2::new(250.0, 230.0));
191        assert_eq!(local, Vec2::new(0.5, 0.5));
192        assert!(set.fields[0].region.contains(&local));
193        let outside = anchor.to_local(&Vec2::new(350.0, 230.0));
194        assert!(!set.fields[0].region.contains(&outside));
195        Ok(())
196    }
