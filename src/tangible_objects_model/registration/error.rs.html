9/// Errors produced by runtime profile and field-set registration
10/// (DYNAMIC_PROFILES.md).
11///
12/// Lives in `model` so the control-command reply channel can carry the
13/// typed rejection to interface crates (GraphQL/D-Bus/MCP) instead of a
14/// pre-formatted string.
15#[derive(Debug, Error)]
16pub enum RegistrationError {
17    /// The referenced pattern ID is not registered.
18    #[error("pattern {0} not found")]
19    PatternNotFound(PatternId),
20    /// The profile's geometric fingerprint collides with an already
21    /// registered pattern - registration is rejected so the new
22    /// profile cannot shadow the incumbent.
23    #[error("fingerprint collides with registered profile '{0}'")]
24    FingerprintCollision(PatternName),
25    /// The field set contains overlapping fields - an occupancy field
26    /// must not suppress the same slot twice. Carries the offending pair
27    /// and the intersection region so the rejection explains itself in
28    /// logs and responses.
29    #[error("field set '{0}' has overlapping fields: '{1}' overlaps '{2}' (overlap: {3})")]
30    FieldsOverlap(FieldSetName, FieldName, FieldName, Region),
31    /// A field set with this name is already registered.
32    #[error("field set '{0}' already registered")]
33    DuplicateFieldSet(FieldSetName),
34    /// A dead zone with this name is already registered.
35    #[error("dead zone '{0}' already registered")]
36    DuplicateDeadZone(DeadZoneName),
37}
