10/// Errors produced by runtime profile and field-set registration
11/// (DYNAMIC_PROFILES.md).
12///
13/// Lives in `model` so the control-command reply channel can carry the
14/// typed rejection to interface crates (GraphQL/D-Bus/MCP) instead of a
15/// pre-formatted string.
16#[derive(Debug, Error)]
17pub enum RegistrationError {
18    /// The referenced pattern ID is not registered.
19    #[error("pattern {0} not found")]
20    PatternNotFound(PatternId),
21    /// The profile's geometric fingerprint collides with an already
22    /// registered pattern - registration is rejected so the new
23    /// profile cannot shadow the incumbent.
24    #[error("fingerprint collides with registered profile '{0}'")]
25    FingerprintCollision(PatternName),
26    /// The field set contains overlapping fields - an occupancy field
27    /// must not suppress the same slot twice. Carries the offending pair
28    /// and the intersection region so the rejection explains itself in
29    /// logs and responses.
30    #[error("field set '{0}' has overlapping fields: '{1}' overlaps '{2}' (overlap: {3})")]
31    FieldsOverlap(FieldSetName, FieldName, FieldName, Region),
32    /// A field set with this name is already registered.
33    #[error("field set '{0}' already registered")]
34    DuplicateFieldSet(FieldSetName),
35    /// A dead zone with this name is already registered.
36    #[error("dead zone '{0}' already registered")]
37    DuplicateDeadZone(DeadZoneName),
38    /// The spell's exclusive zone overlaps a field - the zone
39    /// reserves its area for gestures while the field expects
40    /// occupancy (SPELLS.md). Carries the spell and the offending
41    /// field so the rejection explains itself in logs and
42    /// responses.
43    #[error("spell '{0}' exclusive zone overlaps field '{1}'")]
44    SpellZoneOverlapsField(SpellName, FieldName),
45}
