9/// Static information about the input device attached to this daemon -
10/// the identity needed to interpret or convert its raw event stream.
11///
12/// Published once at session start inside [`crate::EngineSnapshot`];
13/// interfaces read it via the `device_info` query (remote recording:
14/// computer A learns B's touchscreen geometry before pulling artifacts).
15#[derive(Debug, Clone, Default, Serialize, Deserialize)]
16#[serde(rename_all = "camelCase")]
17#[cfg_attr(feature = "async-graphql", derive(async_graphql::SimpleObject))]
18#[cfg_attr(feature = "cynic", derive(cynic::QueryFragment))]
19#[cfg_attr(
20    feature = "cynic",
21    cynic(
22        schema_path = "../graphql/schema.graphql",
23        graphql_type = "DeviceInfo",
24        no_deserialize
25    )
26)]
27pub struct DeviceInfo {
28    /// `Input device name` of the evdev device (or the replayed file).
29    pub name: Option<String>,
30    /// Device units per millimeter on X (`ABS_MT_POSITION_X` resolution).
31    pub x_res: Option<UnitsPerMm>,
32    /// Device units per millimeter on Y (`ABS_MT_POSITION_Y` resolution).
33    pub y_res: Option<UnitsPerMm>,
34    /// Minimum raw X value.
35    pub x_min: Option<RawAxisValue>,
36    /// Maximum raw X value.
37    pub x_max: Option<RawAxisValue>,
38    /// Minimum raw Y value.
39    pub y_min: Option<RawAxisValue>,
40    /// Maximum raw Y value.
41    pub y_max: Option<RawAxisValue>,
42}
43
44impl DeviceInfo {
45    /// Builds the info from a device name and its calibration.
46    pub fn of(name: Option<&str>, calibration: Option<&Calibration>) -> Self {
47        match calibration {
48            Some(cal) => Self {
49                name: name.map(|s| s.to_string()),
50                x_res: Some(cal.x_res),
51                y_res: Some(cal.y_res),
52                x_min: Some(cal.x_min),
53                x_max: Some(cal.x_max),
54                y_min: Some(cal.y_min),
55                y_max: Some(cal.y_max),
56            },
57            None => Self {
58                name: name.map(|s| s.to_string()),
59                ..Default::default()
60            },
61        }
62    }
63}
