12/// Static information about the input device attached to this daemon -
13/// the identity needed to interpret or convert its raw event stream.
14///
15/// Published once at session start inside [`crate::EngineSnapshot`];
16/// interfaces read it via the `device_info` query (remote recording:
17/// computer A learns B's touchscreen geometry before pulling artifacts).
18#[derive(Debug, Clone, Default, Serialize, Deserialize)]
19#[serde(rename_all = "camelCase")]
20#[cfg_attr(feature = "async-graphql", derive(async_graphql::SimpleObject))]
21#[cfg_attr(feature = "cynic", derive(cynic::QueryFragment))]
22#[cfg_attr(
23    feature = "cynic",
24    cynic(
25        schema_path = "../graphql/schema.graphql",
26        graphql_type = "DeviceInfo",
27        no_deserialize
28    )
29)]
30pub struct DeviceInfo {
31    /// `Input device name` of the evdev device (or the replayed file).
32    pub name: Option<String>,
33    /// Device units per millimeter on X (`ABS_MT_POSITION_X` resolution).
34    pub x_res: Option<UnitsPerMm>,
35    /// Device units per millimeter on Y (`ABS_MT_POSITION_Y` resolution).
36    pub y_res: Option<UnitsPerMm>,
37    /// Minimum raw X value.
38    pub x_min: Option<RawAxisValue>,
39    /// Maximum raw X value.
40    pub x_max: Option<RawAxisValue>,
41    /// Minimum raw Y value.
42    pub y_min: Option<RawAxisValue>,
43    /// Maximum raw Y value.
44    pub y_max: Option<RawAxisValue>,
45}
46
47impl DeviceInfo {
48    /// Builds the info from a device name and its calibration.
49    pub fn of(name: Option<&str>, calibration: Option<&Calibration>) -> Self {
50        match calibration {
51            Some(cal) => Self {
52                name: name.map(|s| s.to_string()),
53                x_res: Some(cal.x_res),
54                y_res: Some(cal.y_res),
55                x_min: Some(cal.x_min),
56                x_max: Some(cal.x_max),
57                y_min: Some(cal.y_min),
58                y_max: Some(cal.y_max),
59            },
60            None => Self {
61                name: name.map(|s| s.to_string()),
62                ..Default::default()
63            },
64        }
65    }
66
67    /// Physical size of the calibrated touch surface in millimeters -
68    /// the extent `SurfaceFit` maps onto the display. `None` when the
69    /// daemon did not report a complete calibration (resolution and
70    /// range on both axes).
71    pub fn surface_mm(&self) -> Option<Vec2<Millimeters>> {
72        let width = Millimeters::new((self.x_max? - self.x_min?) / self.x_res?);
73        let height = Millimeters::new((self.y_max? - self.y_min?) / self.y_res?);
74        Some(Vec2::from_mm(width, height))
75    }
76
77    /// Physical surface bounds in millimeters - `min` at the
78    /// calibrated raw-origin offset (`Calibration::to_mm` does not
79    /// subtract `x_min`), `size` the usable extent. The enclosing
80    /// rectangle marker positions project against when a seat derives
81    /// its organizational alignment (`Seat::from_position`,
82    /// TANGIBLE_OBJECTS_TABLE_TOP_SEAT.md). `None` for an incomplete
83    /// calibration.
84    pub fn surface_bounds(&self) -> Option<Rect<Millimeters>> {
85        let (x_min, x_res, y_min, y_res) = (self.x_min?, self.x_res?, self.y_min?, self.y_res?);
86        Some(Rect::new(
87            Vec2::new(x_min / x_res, y_min / y_res),
88            self.surface_mm()?,
89        ))
90    }
91}

94mod tests {
95    use super::DeviceInfo;
96    use smearor_commons_geometry::Vec2;
97
98    #[test]
99    fn surface_mm_uses_resolution_and_bounds() {
100        let info = DeviceInfo {
101            x_res: Some(2.5.into()),
102            y_res: Some(2.5.into()),
103            x_min: Some(0.0.into()),
104            x_max: Some(3500.0.into()),
105            y_min: Some(0.0.into()),
106            y_max: Some(2000.0.into()),
107            ..Default::default()
108        };
109        let surface = info.surface_mm().expect("surface");
110        assert_eq!(surface, Vec2::from_mm(1400.0.into(), 800.0.into()));
111        assert!(DeviceInfo::default().surface_mm().is_none());
112    }
