11/// Complete profile of a touchscreen device - everything needed to convert
12/// raw `.evtest` recordings between devices (TRAINING.md device profiles).
13///
14/// The TOML shape mirrors the `evtest` capability header: axis bounds and
15/// resolutions come straight from the `ABS_MT_POSITION_X/Y` blocks,
16/// `abs_codes`/`key_codes` list every event code the device emits.
17#[derive(Debug, Clone, Serialize, Deserialize)]
18pub struct DeviceProfile {
19    /// Human-readable device name - the `Input device name:` header line.
20    pub name: String,
21    /// evdev vendor ID for auto-matching (`vendor` in the evtest header).
22    #[serde(default, skip_serializing_if = "Option::is_none")]
23    pub vendor: Option<u16>,
24    /// evdev product ID for auto-matching.
25    #[serde(default, skip_serializing_if = "Option::is_none")]
26    pub product: Option<u16>,
27    /// Device units per millimeter on X (`Resolution` under `ABS_MT_POSITION_X`).
28    pub x_res: UnitsPerMm,
29    /// Device units per millimeter on Y (`Resolution` under `ABS_MT_POSITION_Y`).
30    pub y_res: UnitsPerMm,
31    /// Minimum raw X value.
32    pub x_min: RawAxisValue,
33    /// Maximum raw X value.
34    pub x_max: RawAxisValue,
35    /// Minimum raw Y value.
36    pub y_min: RawAxisValue,
37    /// Maximum raw Y value.
38    pub y_max: RawAxisValue,
39    /// Measured physical resolution on X - used instead of `x_res` when the
40    /// firmware misreports axis density (e.g. a 65" panel whose controller
41    /// declares 7 units/mm while the real sensor grid is ~2.87).
42    #[serde(default, skip_serializing_if = "Option::is_none")]
43    pub x_res_override: Option<UnitsPerMm>,
44    /// Measured physical resolution on Y (see `x_res_override`).
45    #[serde(default, skip_serializing_if = "Option::is_none")]
46    pub y_res_override: Option<UnitsPerMm>,
47    /// `EV_ABS` codes the device emits (e.g. `53`/`54`/`57`, plus legacy
48    /// `0`/`1` where present). Converting to this device drops any source
49    /// event whose code is not listed.
50    #[serde(default, skip_serializing_if = "Vec::is_empty")]
51    pub abs_codes: Vec<u16>,
52    /// `EV_KEY` codes the device emits (e.g. `330` = `BTN_TOUCH`).
53    #[serde(default, skip_serializing_if = "Vec::is_empty")]
54    pub key_codes: Vec<u16>,
55    /// Typical report rate in Hz - informational; conversion keeps the
56    /// source timestamps.
57    #[serde(default, skip_serializing_if = "Option::is_none")]
58    pub report_rate_hz: Option<u16>,
59}
60
61impl DeviceProfile {
62    /// The affine axis map as a [`Calibration`] - the resolution override
63    /// (physical truth) wins over the firmware-reported value when present.
64    pub fn calibration(&self) -> Calibration {
65        Calibration::with_bounds(
66            self.x_res_override.unwrap_or(self.x_res),
67            self.y_res_override.unwrap_or(self.y_res),
68            self.x_min,
69            self.x_max,
70            self.y_min,
71            self.y_max,
72        )
73    }
74
75    /// The measured resolutions when the firmware misreports them - `Some`
76    /// only when the profile declares an override for *both* axes.
77    pub fn resolution_override(&self) -> Option<(UnitsPerMm, UnitsPerMm)> {
78        self.x_res_override.zip(self.y_res_override)
79    }
80
81    /// Whether the device emits `(event_type, code)` - conversion drops
82    /// events outside the profile's set.
83    pub fn supports(&self, event_type: EventType, code: EventCode) -> bool {
84        match event_type {
85            EventType::Syn => true, // SYN_REPORT is a delimiter, not a capability
86            EventType::Abs => self.abs_codes.contains(&code.as_u16()),
87            EventType::Key => self.key_codes.contains(&code.as_u16()),
88            EventType::Other(_) => false,
89        }
90    }
91
92    /// Converts a millimeter position to this device's raw axis value -
93    /// the inverse of [`Calibration::to_mm`], clamped to `[min, max]`.
94    ///
95    /// Returns the clamped raw value and whether clamping occurred (the
96    /// source geometry exceeded this device's bounds).
97    pub fn to_raw(&self, position: Millimeters, axis_x: bool) -> (RawAxisValue, bool) {
98        let (res, min, max) = if axis_x {
99            (
100                self.x_res_override.unwrap_or(self.x_res),
101                self.x_min,
102                self.x_max,
103            )
104        } else {
105            (
106                self.y_res_override.unwrap_or(self.y_res),
107                self.y_min,
108                self.y_max,
109            )
110        };
111        let raw = position.as_f64() * res.as_f64() + min.as_f64();
112        let clamped = raw.clamp(min.as_f64(), max.as_f64());
113        (
114            RawAxisValue::new(clamped),
115            (clamped - raw).abs() > f64::EPSILON,
116        )
117    }
118}

120#[cfg(test)]
121mod tests {
122    use super::DeviceProfile;
123    use crate::EventCode;
124    use crate::EventType;
125    use smearor_commons_geometry::Millimeters;
126
127    fn profile() -> DeviceProfile {
128        DeviceProfile {
129            name: "test".to_string(),
130            vendor: None,
131            product: None,
132            x_res: 8.0.into(),
133            y_res: 8.0.into(),
134            x_min: 0.0.into(),
135            x_max: 4000.0.into(),
136            y_min: 0.0.into(),
137            y_max: 4000.0.into(),
138            abs_codes: vec![53, 54, 57],
139            key_codes: vec![330],
140            report_rate_hz: Some(120),
141            x_res_override: None,
142            y_res_override: None,
143        }
144    }
145
146    #[test]
147    fn to_raw_inverts_to_mm() {
148        let profile = profile();
149        let calibration = profile.calibration();
150        // 250 mm on the source axis → 2000 raw on this device (res 8).
151        let (raw, clamped) = profile.to_raw(Millimeters::new(250.0), true);
152        assert_eq!(raw.as_f64(), 2000.0);
153        assert!(!clamped);
154        // Round-trip through to_mm.
155        let back = calibration.to_mm(smearor_commons_geometry::Vec2::new(
156            raw.as_f64(),
157            raw.as_f64(),
158        ));
159        assert!((back.x - 250.0).abs() < 0.01);
160    }
161
162    #[test]
163    fn to_raw_clamps_out_of_bounds() {
164        let profile = profile();
165        // 600 mm exceeds max (4000 raw = 500 mm) → clamped to the bound.
166        let (raw, clamped) = profile.to_raw(Millimeters::new(600.0), true);
167        assert_eq!(raw.as_f64(), 4000.0);
168        assert!(clamped);
169    }
170
171    #[test]
172    fn supports_only_listed_codes() {
173        let profile = profile();
174        assert!(profile.supports(EventType::Abs, EventCode::new(53)));
175        assert!(!profile.supports(EventType::Abs, EventCode::new(0)));
176        assert!(profile.supports(EventType::Key, EventCode::new(330)));
177        assert!(profile.supports(EventType::Syn, EventCode::new(0)));
178    }
