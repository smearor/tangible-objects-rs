1//! `PhysicalShape` - the declared physical outline of a tangible
2//! (`[shape]` section in the profile TOML), rendered by the GUI in
3//! `PhysicalGeometry`/`TouchPointsAndPhysicalGeometry` render modes.
4
5use serde::Deserialize;
6use serde::Serialize;
7use smearor_commons_geometry::Millimeters;
8use smearor_commons_geometry::Vec2;
9
10/// The physical outline of a tangible's body in the pattern's frame
11/// (millimeters relative to the tangible center, pattern
12/// orientation - the GUI rotates it by the match's `rotation_rad`).
13///
14/// Serialized with a `type` tag, e.g.:
15///
16/// ```toml
17/// [shape]
18/// type = "polygon"
19/// points = [[-30.0, -20.0], [30.0, -20.0], [30.0, 20.0], [-30.0, 20.0]]
20/// ```
21#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
22#[serde(tag = "type", rename_all = "kebab-case")]
23pub enum PhysicalShape {
24    /// Filled/stroked circle around the tangible center.
25    Circle {
26        /// Circle radius in millimeters.
27        radius: Millimeters,
28    },
29    /// Rectangle in the pattern's frame - rotates with the tangible.
30    Rectangle {
31        /// Half the rectangle width in millimeters.
32        half_width: Millimeters,
33        /// Half the rectangle height in millimeters.
34        half_height: Millimeters,
35    },
36    /// Ellipse in the pattern's frame - rotates with the tangible.
37    Ellipse {
38        /// Half the ellipse width in millimeters.
39        half_width: Millimeters,
40        /// Half the ellipse height in millimeters.
41        half_height: Millimeters,
42    },
43    /// Arbitrary closed outline - e.g. a body with a hole approximated
44    /// as a polygon; points in millimeters relative to the center.
45    Polygon {
46        /// Polygon vertices in pattern-space millimeters.
47        #[serde(with = "smearor_commons_geometry::serde_points")]
48        points: Vec<Vec2<Millimeters>>,
49    },
50}
