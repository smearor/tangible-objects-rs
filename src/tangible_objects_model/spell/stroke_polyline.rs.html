1//! `StrokePolyline` - one finger's gesture path (template or live
2//! trajectory), normalized for corridor matching.
3
4use serde::Deserialize;
5use serde::Serialize;
6use smearor_commons_geometry::Millimeters;
7use smearor_commons_geometry::Vec2;
8
9/// Samples per resampled stroke - enough to keep shape detail,
10/// cheap enough to run the Fréchet scan per frame (SPELLS.md).
11pub const STROKE_SAMPLES: usize = 64;
12
13/// A stroke as an ordered polyline in millimeters.
14///
15/// Templates live in *pattern space*: origin at the first point,
16/// optionally uniformly scaled to a reference extent. Live
17/// trajectories are normalized the same way before matching.
18#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize)]
19pub struct StrokePolyline {
20    /// Ordered stroke points in millimeters.
21    #[serde(with = "smearor_commons_geometry::serde_points")]
22    pub points: Vec<Vec2<Millimeters>>,
23}
24
25impl StrokePolyline {
26    /// Wraps points as a stroke.
27    pub fn new(points: Vec<Vec2<Millimeters>>) -> Self {
28        Self { points }
29    }
30
31    /// Resamples the stroke to `STROKE_SAMPLES` evenly spaced points
32    /// along its arc length (kills sampling-rate noise; the
33    /// $-family recognizers' canonical step).
34    pub fn resampled(&self) -> Self {
35        let count = STROKE_SAMPLES;
36        if self.points.len() < 2 {
37            return self.clone();
38        }
39        let total: f64 = self
40            .points
41            .windows(2)
42            .map(|pair| pair[0].distance(&pair[1]).as_f64())
43            .sum();
44        if total < 1e-6 {
45            return self.clone();
46        }
47        let step = total / (count - 1) as f64;
48        let mut out = Vec::with_capacity(count);
49        let mut previous = self.points[0];
50        out.push(previous);
51        let mut walked = 0.0;
52        for point in &self.points[1..] {
53            let mut seg = previous.distance(point).as_f64();
54            while walked + seg >= step && out.len() < count {
55                let t = (step - walked) / seg;
56                let interpolated = Vec2::<Millimeters>::new(
57                    previous.x + (point.x - previous.x) * t,
58                    previous.y + (point.y - previous.y) * t,
59                );
60                out.push(interpolated);
61                previous = interpolated;
62                walked = 0.0;
63                seg = previous.distance(point).as_f64();
64            }
65            walked += seg;
66            previous = *point;
67        }
68        while out.len() < count {
69            out.push(*self.points.last().unwrap_or(&previous));
70        }
71        Self::new(out)
72    }
73
74    /// Translates the stroke so its first point sits at the origin
75    /// (position-free matching - a spell is drawable anywhere).
76    pub fn translated_to_origin(&self) -> Self {
77        let Some(first) = self.points.first() else {
78            return self.clone();
79        };
80        Self::new(
81            self.points
82                .iter()
83                .map(|p| Vec2::<Millimeters>::new(p.x - first.x, p.y - first.y))
84                .collect(),
85        )
86    }
87
88    /// Uniformly scales the stroke so its extent (max distance from
89    /// the origin point) becomes `reference` millimeters.
90    pub fn scaled_to(&self, reference: Millimeters) -> Self {
91        let extent = self
92            .points
93            .iter()
94            .map(|p| p.distance(&Vec2::ZERO).as_f64())
95            .fold(0.0_f64, f64::max);
96        if extent < 1e-6 {
97            return self.clone();
98        }
99        let factor = reference.as_f64() / extent;
100        Self::new(
101            self.points
102                .iter()
103                .map(|p| Vec2::<Millimeters>::new(p.x * factor, p.y * factor))
104                .collect(),
105        )
106    }
107
108    /// The normalization applied to both template and trajectory
109    /// before corridor matching: translate to origin, optionally
110    /// scale to the template's reference extent.
111    pub fn normalized(&self, reference: Millimeters, scale: bool) -> Self {
112        let translated = self.translated_to_origin();
113        if scale {
114            translated.scaled_to(reference)
115        } else {
116            translated
117        }
118    }
119
120    /// Discrete Fréchet distance to `other` - the maximum leash the
121    /// worst point-pair needs when both paths are walked with
122    /// monotonic progress (SPELLS.md's corridor primitive: tolerates
123    /// speed differences while enforcing direction and shape).
124    pub fn frechet(&self, other: &Self) -> Millimeters {
125        let n = self.points.len();
126        let m = other.points.len();
127        if n == 0 || m == 0 {
128            return Millimeters::new(f64::INFINITY);
129        }
130        // Rolling rows - O(n*m) time, O(m) space.
131        let mut prev = vec![f64::INFINITY; m];
132        let mut cur = vec![f64::INFINITY; m];
133        for i in 0..n {
134            for (j, o) in other.points.iter().enumerate() {
135                let d = self.points[i].distance(o).as_f64();
136                cur[j] = if i == 0 && j == 0 {
137                    d
138                } else {
139                    let best = (if i > 0 { prev[j] } else { f64::INFINITY })
140                        .min(if j > 0 { prev[j - 1] } else { f64::INFINITY })
141                        .min(if j > 0 { cur[j - 1] } else { f64::INFINITY });
142                    d.max(best)
143                };
144            }
145            std::mem::swap(&mut prev, &mut cur);
146        }
147        Millimeters::new(prev[m - 1])
148    }
149}

151#[cfg(test)]
152mod tests {
153    use super::StrokePolyline;
154    use smearor_commons_geometry::Vec2;
155
156    fn stroke(coords: &[(f64, f64)]) -> StrokePolyline {
157        StrokePolyline::new(coords.iter().map(|&(x, y)| Vec2::new(x, y)).collect())
158    }
159
160    #[test]
161    fn identical_strokes_have_zero_frechet() {
162        let a = stroke(&[(0.0, 0.0), (10.0, 0.0), (20.0, 0.0)]);
163        let resampled_a = a.resampled();
164        assert!(resampled_a.frechet(&resampled_a).as_f64() < 1e-9);
165    }
166
167    #[test]
168    fn same_shape_elsewhere_matches_after_translation() {
169        let template = stroke(&[(0.0, 0.0), (10.0, 0.0)])
170            .resampled()
171            .translated_to_origin();
172        let shifted = stroke(&[(100.0, 50.0), (110.0, 50.0)])
173            .resampled()
174            .translated_to_origin();
175        assert!(template.frechet(&shifted).as_f64() < 0.5);
176    }
177
178    #[test]
179    fn different_shape_fails_corridor() {
180        let a = stroke(&[(0.0, 0.0), (50.0, 0.0)]).resampled();
181        let b = stroke(&[(0.0, 0.0), (0.0, 50.0)]).resampled();
182        assert!(a.frechet(&b).as_f64() > 20.0);
183    }
184
185    #[test]
186    fn scaled_copy_matches_with_scale_normalization() {
187        let a = stroke(&[(0.0, 0.0), (10.0, 5.0), (20.0, 0.0)]).resampled();
188        let big = stroke(&[(0.0, 0.0), (20.0, 10.0), (40.0, 0.0)]).resampled();
189        let an = a.normalized(20.0.into(), true);
190        let bn = big.normalized(20.0.into(), true);
191        assert!(an.frechet(&bn).as_f64() < 1.0);
192    }
193
194    #[test]
195    fn direction_matters() {
196        let a = stroke(&[(0.0, 0.0), (50.0, 0.0)]).resampled();
197        let b = stroke(&[(50.0, 0.0), (0.0, 0.0)]).resampled();
198        // Reversed direction: Frechet enforces monotonic progress -
199        // the distance is the stroke length, not zero.
200        assert!(a.frechet(&b).as_f64() > 40.0);
201    }
