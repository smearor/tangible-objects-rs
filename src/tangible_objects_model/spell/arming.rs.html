11/// How a spell is armed - every spell declares exactly one arming
12/// type (`[arming]` TOML section with a `type` tag).
13///
14/// The armed state is what justifies suppression: while the state
15/// machine is `Armed`/`Active`, the spell's contacts route to the
16/// gesture tracker and never reach the virtual device.
17#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
18#[serde(tag = "type", rename_all = "kebab-case")]
19pub enum SpellArming {
20    /// A reserved surface region - contacts inside it are always
21    /// gesture candidates; no field may overlap it and new
22    /// tangibles never match inside, while already-tracked
23    /// tangibles may be dragged through.
24    ExclusiveZone {
25        /// The reserved rectangle in surface millimeters.
26        region: Region,
27    },
28    /// A small region that arms the spell when the gesture's first
29    /// contact lands on it - screen corner, window edge, or a
30    /// button an application registered.
31    StartSpot {
32        /// The arming rectangle in surface millimeters.
33        region: Region,
34    },
35    /// A brief whole-hand tap (~5 simultaneous contacts inside a
36    /// tight time window) arms a short window; the gesture itself
37    /// may use fewer fingers.
38    FiveFingerTap,
39    /// A registered tangible acts as the wand - armed while the
40    /// tangible is tracked, its center pose is the stroke, lift
41    /// completes the spell.
42    TangibleChord {
43        /// The wand's pattern - `None` accepts any tracked
44        /// tangible.
45        pattern: Option<PatternName>,
46    },
47    /// No explicit trigger - the optional fallback that uses
48    /// prefix-gated defer-and-replay (SPELLS.md's pinch/rotate
49    /// caveat applies).
50    Free,
51}
