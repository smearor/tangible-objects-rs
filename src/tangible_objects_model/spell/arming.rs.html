12/// How a spell is armed - every spell declares exactly one arming
13/// type (`[arming]` TOML section with a `type` tag).
14///
15/// The armed state is what justifies suppression: while the state
16/// machine is `Armed`/`Active`, the spell's contacts route to the
17/// gesture tracker and never reach the virtual device.
18#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
19#[serde(tag = "type", rename_all = "kebab-case")]
20pub enum SpellArming {
21    /// A reserved surface region - contacts inside it are always
22    /// gesture candidates; no field may overlap it and new
23    /// tangibles never match inside, while already-tracked
24    /// tangibles may be dragged through.
25    ExclusiveZone {
26        /// The reserved rectangle in surface millimeters.
27        region: Region,
28    },
29    /// A small region that arms the spell when the gesture's first
30    /// contact lands on it - screen corner, window edge, or a
31    /// button an application registered.
32    StartSpot {
33        /// The arming rectangle in surface millimeters.
34        region: Region,
35    },
36    /// A brief whole-hand tap (~5 simultaneous contacts inside a
37    /// tight time window) arms a short window; the gesture itself
38    /// may use fewer fingers.
39    FiveFingerTap,
40    /// A registered tangible acts as the wand - armed while the
41    /// tangible is tracked, its center pose is the stroke, lift
42    /// completes the spell.
43    TangibleChord {
44        /// The wand's pattern - `None` accepts any tracked
45        /// tangible.
46        pattern: Option<PatternName>,
47    },
48    /// No explicit trigger - the optional fallback that uses
49    /// prefix-gated defer-and-replay (SPELLS.md's pinch/rotate
50    /// caveat applies).
51    Free,
52}
53
54impl FromStr for SpellArming {
55    type Err = toml::de::Error;
56
57    /// Parses the `[arming]` TOML table - `type = "<kebab-kind>"`
58    /// plus the fields the armed variants carry (e.g.
59    /// `type = "start-spot"` + `[region]`). An empty string yields
60    /// [`SpellArming::Free`].
61    fn from_str(s: &str) -> Result<Self, Self::Err> {
62        if s.is_empty() {
63            return Ok(Self::Free);
64        }
65        toml::from_str(s)
66    }
67}
