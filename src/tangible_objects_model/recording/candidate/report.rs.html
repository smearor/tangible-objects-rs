15/// Diagnostic record of one tested point combination against a pattern.
16///
17/// Emitted by the engine's opt-in matching diagnostics for every candidate -
18/// including rejected ones - so deviation distributions, per-pair tolerances
19/// and collision risks can be analyzed offline.
20#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
21pub struct CandidateReport {
22    /// Pattern the combination was tested against.
23    pub pattern_id: PatternId,
24    /// Name of the figure profile the combination was tested against.
25    pub pattern_name: PatternName,
26    /// Indices into the frame's `touches` forming this combination.
27    pub touch_indices: CombinationBuf,
28    /// Whether the combination matched the pattern within `tolerance`.
29    pub matched: bool,
30    /// Whether this was a partial attempt (`num_points - 1` contacts, one pin
31    /// missing). Partial candidates compare against a subset fingerprint.
32    pub partial: bool,
33    /// Sorted pairwise distances measured on the combination's points.
34    /// Index `i` corresponds to the pattern's `sorted_distances[i]` (or the
35    /// subset fingerprint for partial candidates).
36    pub measured_distances: Distances,
37    /// Absolute deviation `|measured[i] - reference[i]|` per distance index,
38    /// parallel to `measured_distances`.
39    pub deviations: crate::Deviations,
40    /// Scalar tolerance applied during the test (also the fallback for
41    /// partial candidates, whose re-indexed fingerprint cannot use
42    /// per-index tolerances).
43    pub tolerance: PatternTolerance,
44    /// Per-index tolerances applied during the test - only present when the
45    /// pattern configured `tolerances`; absent means `tolerance` applied to
46    /// every index. `#[serde(default)]` keeps older recordings loadable.
47    #[serde(default, skip_serializing_if = "PatternTolerances::is_empty")]
48    pub tolerances: PatternTolerances,
49    /// Normalized deviation score (`max_i dev[i] / tolerance[i]`), recorded
50    /// for matched and rejected candidates. `None` in older recordings - the
51    /// analysis recomputes it from `deviations` and `tolerances`.
52    #[serde(default, skip_serializing_if = "Option::is_none")]
53    pub score: Option<MatchScore>,
54    /// Total absolute deviation across all distances (lower is better).
55    /// Recorded for rejected candidates too - near-misses feed the
56    /// collision analysis.
57    pub match_error: MatchError,
58}
59
60impl CandidateReport {
61    /// The tolerance this candidate would need to match: the largest
62    /// per-index deviation.
63    ///
64    /// Basis of the collision analysis - for a rejected candidate it is the
65    /// tolerance at which it would have become a false positive. An empty
66    /// deviation vector (degenerate candidate) needs zero tolerance.
67    pub fn required_tolerance(&self) -> PatternTolerance {
68        self.deviations
69            .iter()
70            .copied()
71            .max_by(Deviation::total_cmp)
72            .map(|deviation| deviation.as_millimeters().into())
73            .unwrap_or_default()
74    }
75
76    /// Effective tolerance for distance index `i` - the recorded per-index
77    /// value when present, else the scalar `tolerance`.
78    pub fn tolerance_at(&self, index: usize) -> PatternTolerance {
79        self.tolerances
80            .get(index)
81            .copied()
82            .unwrap_or(self.tolerance)
83    }
84
85    /// Normalized deviation score, recomputed from `deviations` and the
86    /// effective tolerances. Falls back to the recorded `score` when
87    /// `deviations` is empty; used by the offline threshold sweep so old and
88    /// new recordings evaluate identically.
89    pub fn normalized_score(&self) -> MatchScore {
90        let computed = self
91            .deviations
92            .iter()
93            .enumerate()
94            .map(|(i, dev)| dev.as_f64() / self.tolerance_at(i).as_f64())
95            .reduce(f64::max);
96        match (self.score, computed) {
97            (_, Some(score)) => MatchScore::new(score),
98            (Some(recorded), None) => recorded,
99            (None, None) => MatchScore::new(0.0),
100        }
101    }
102
103    /// Whether a ground-truth label refers to this candidate's pattern.
104    ///
105    /// Labels carry either a profile name (`"TestFigur-4P"`) or a numeric
106    /// pattern ID (`"4"`) - both forms are produced by `--record-label` and
107    /// the interface APIs.
108    pub fn matches_label(&self, label: &RecordLabel) -> bool {
109        let label = label.as_str();
110        self.pattern_name.as_str() == label || self.pattern_id.0.to_string() == label
111    }
112}
