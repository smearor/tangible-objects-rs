15/// A 2D point or vector tagged with its coordinate unit `U`.
16///
17/// The phantom unit distinguishes positions in millimeters
18/// (`Vec2<Millimeters>`), raw device units (`Vec2<RawAxisValue>`) and screen
19/// pixels (`Vec2<Pixels>`) so they cannot be mixed up. Components stay plain
20/// `f64`; the unit only labels which coordinate space the point lives in.
21/// Serializes as a flat `{"x": .., "y": ..}` map; the unit marker is skipped.
22/// With the `async-graphql` feature, `Vec2<Millimeters>` is exposed as the
23/// GraphQL object type `Point`.
24#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
25#[cfg_attr(feature = "async-graphql", derive(async_graphql::SimpleObject))]
26#[cfg_attr(
27    feature = "async-graphql",
28    graphql(concrete(name = "Point", params(crate::Millimeters)))
29)]
30pub struct Vec2<U> {
31    /// X coordinate in the unit `U`.
32    pub x: f64,
33    /// Y coordinate in the unit `U`.
34    pub y: f64,
35    /// Phantom marker tagging the coordinate unit; not serialized.
36    #[serde(skip)]
37    #[cfg_attr(feature = "async-graphql", graphql(skip))]
38    unit: PhantomData<U>,
39}
40
41impl<U> Vec2<U> {
42    /// The origin `(0, 0)` in the unit `U`.
43    pub const ZERO: Self = Self {
44        x: 0.0,
45        y: 0.0,
46        unit: PhantomData,
47    };
48
49    /// Creates a new 2D point in the unit `U`.
50    pub fn new(x: f64, y: f64) -> Self {
51        Self {
52            x,
53            y,
54            unit: PhantomData,
55        }
56    }
57
58    /// Euclidean distance to another point in the same unit.
59    pub fn distance(&self, other: &Vec2<U>) -> U
60    where
61        U: From<f64>,
62    {
63        U::from(((self.x - other.x).powi(2) + (self.y - other.y).powi(2)).sqrt())
64    }
65
66    /// Dot product `x·x + y·y` of two vectors in the same unit.
67    ///
68    /// The caller picks the product type `T` (like [`Vec2::distance`]'s
69    /// unit return): `f64` for unit-free math or a unit type such as
70    /// [`crate::SquareMillimeters`] for `Vec2<Millimeters>` products.
71    pub fn dot<T: From<f64>>(&self, other: &Vec2<U>) -> T {
72        T::from(self.x * other.x + self.y * other.y)
73    }
74
75    /// 2D determinant `x·y' − y·x'` (z component of the 3D cross
76    /// product): positive when `other` turns left from `self`, negative
77    /// right; zero for (anti-)parallel vectors.
78    ///
79    /// Same caller-chosen product type as [`Vec2::dot`].
80    pub fn cross<T: From<f64>>(&self, other: &Vec2<U>) -> T {
81        T::from(self.x * other.y - self.y * other.x)
82    }
83
84    /// Euclidean magnitude of the vector, in the unit `U`.
85    pub fn magnitude(&self) -> U
86    where
87        U: From<f64>,
88    {
89        self.distance(&Vec2::ZERO)
90    }
91
92    /// The vector rotated 90° counterclockwise: `(-y, x)` - the
93    /// perpendicular used for offset and normal computations.
94    pub fn perpendicular(&self) -> Self {
95        Vec2::new(-self.y, self.x)
96    }
97
98    /// The unit vector pointing the same way; a vector shorter
99    /// than `1e-9` normalizes to itself.
100    pub fn normalized(&self) -> Self {
101        let magnitude = self.x.hypot(self.y);
102        if magnitude < 1e-9 {
103            return Vec2::new(self.x, self.y);
104        }
105        Vec2::new(self.x / magnitude, self.y / magnitude)
106    }
107
108    /// Geometric centroid of a non-empty point set.
109    ///
110    /// Returns [`Vec2::new(0.0, 0.0)`] for an empty slice instead of panicking.
111    pub fn centroid(points: &[Vec2<U>]) -> Self {
112        let n = points.len() as f64;
113        if n == 0.0 {
114            return Vec2::new(0.0, 0.0);
115        }
116        Vec2::new(
117            points.iter().map(|p| p.x).sum::<f64>() / n,
118            points.iter().map(|p| p.y).sum::<f64>() / n,
119        )
120    }
121}

222/// Serde adapter for `#[serde(with)]`: `Vec2<U>` as a compact
223/// `[x, y]` pair instead of the default `{"x": .., "y": ..}` map.
224///
225/// Dataset-style formats like figure profiles or game maps often
226/// prefer the tighter pair shape; downstream crates use this module as
227/// `#[serde(with = "tangible_objects_model::serde_pair")]`.
228pub mod serde_pair {
229    use super::Vec2;
230    use serde::Deserialize;
231    use serde::Deserializer;
232    use serde::Serialize;
233    use serde::Serializer;
234
235    /// Serializes the point as `[x, y]`.
236    pub fn serialize<U, S: Serializer>(v: &Vec2<U>, serializer: S) -> Result<S::Ok, S::Error> {
237        [v.x, v.y].serialize(serializer)
238    }
239
240    /// Deserializes `[x, y]` into a `Vec2` in the unit `U`.
241    pub fn deserialize<'de, U, D: Deserializer<'de>>(deserializer: D) -> Result<Vec2<U>, D::Error> {
242        <[f64; 2]>::deserialize(deserializer).map(Vec2::from)
243    }
244}

272    fn dot_and_cross_of_perpendiculars() {
273        let right = Vec2::<Millimeters>::new(1.0, 0.0);
274        let up = Vec2::<Millimeters>::new(0.0, 1.0);
275        let dot: f64 = right.dot(&up);
276        assert!(dot.abs() < 1e-9);
277        let (cross, reverse): (f64, f64) = (right.cross(&up), up.cross(&right));
278        assert!((cross - 1.0).abs() < 1e-9);
279        assert!((reverse - -1.0).abs() < 1e-9);
280        let area: crate::SquareMillimeters = right.cross(&up);
281        assert!((area.as_f64() - 1.0).abs() < 1e-9);
282    }

299    /// Vectors add, subtract, negate and scale component-wise.
300    #[test]
301    fn arithmetic_ops() {
302        let a = Vec2::<Millimeters>::new(10.0, 4.0);
303        let b = Vec2::<Millimeters>::new(2.0, 1.0);

309        assert_eq!(a / 2.0, Vec2::new(5.0, 2.0));

312    /// `magnitude` measures the vector's length; `perpendicular`
313    /// turns it 90° left.
314    #[test]
315    fn magnitude_and_perpendicular() {
316        let vector = Vec2::<Millimeters>::new(3.0, 4.0);
317        assert_eq!(vector.magnitude(), Millimeters::new(5.0));
318        assert_eq!(vector.perpendicular(), Vec2::new(-4.0, 3.0));
319        assert_eq!(vector.normalized(), Vec2::new(0.6, 0.8));
320        assert_eq!(Vec2::<Millimeters>::ZERO.normalized(), Vec2::ZERO);
321    }
322
323    /// `normalized` yields a unit vector in the same direction.
324    #[test]
325    fn normalized_preserves_direction() {
326        let vector = Vec2::<Millimeters>::new(-6.0, 8.0);
327        let unit = vector.normalized();
328        assert!((unit.x.hypot(unit.y) - 1.0).abs() < 1e-9);
329        // no component along the perpendicular - the direction is unchanged
330        assert!(unit.dot::<f64>(&vector.perpendicular()).abs() < 1e-9);
331        // scaling the unit vector back up reproduces the original
332        assert!(
333            (unit * vector.magnitude().as_f64() - vector)
334                .magnitude()
335                .as_f64()
336                < 1e-9
337        );
338    }

340    /// Vectors below the `1e-9` threshold normalize to themselves -
341    /// no direction can be read off a degenerate vector.
342    #[test]
343    fn normalized_keeps_degenerate_vectors() {
344        assert_eq!(Vec2::<Millimeters>::ZERO.normalized(), Vec2::ZERO);
345        let tiny = Vec2::<Millimeters>::new(1e-12, 0.0);
346        assert_eq!(tiny.normalized(), tiny);
347        // just above the threshold still normalizes to unit length
348        let small = Vec2::<Millimeters>::new(1e-6, 0.0);
349        assert_eq!(small.normalized(), Vec2::new(1.0, 0.0));
350    }
