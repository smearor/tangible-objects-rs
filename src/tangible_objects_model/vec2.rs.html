8/// A 2D point or vector tagged with its coordinate unit `U`.
9///
10/// The phantom unit distinguishes positions in millimeters
11/// (`Vec2<Millimeters>`), raw device units (`Vec2<RawAxisValue>`) and screen
12/// pixels (`Vec2<Pixels>`) so they cannot be mixed up. Components stay plain
13/// `f64`; the unit only labels which coordinate space the point lives in.
14/// Serializes as a flat `{"x": .., "y": ..}` map; the unit marker is skipped.
15/// With the `async-graphql` feature, `Vec2<Millimeters>` is exposed as the
16/// GraphQL object type `Point`.
17#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
18#[cfg_attr(feature = "async-graphql", derive(async_graphql::SimpleObject))]
19#[cfg_attr(
20    feature = "async-graphql",
21    graphql(concrete(name = "Point", params(crate::Millimeters)))
22)]
23pub struct Vec2<U> {
24    /// X coordinate in the unit `U`.
25    pub x: f64,
26    /// Y coordinate in the unit `U`.
27    pub y: f64,
28    /// Phantom marker tagging the coordinate unit; not serialized.
29    #[serde(skip)]
30    #[cfg_attr(feature = "async-graphql", graphql(skip))]
31    unit: PhantomData<U>,
32}
33
34impl<U> Vec2<U> {
35    /// Creates a new 2D point in the unit `U`.
36    pub fn new(x: f64, y: f64) -> Self {
37        Self {
38            x,
39            y,
40            unit: PhantomData,
41        }
42    }
43
44    /// Euclidean distance to another point in the same unit.
45    pub fn distance(&self, other: &Vec2<U>) -> U
46    where
47        U: From<f64>,
48    {
49        U::from(((self.x - other.x).powi(2) + (self.y - other.y).powi(2)).sqrt())
50    }
51
52    /// Geometric centroid of a non-empty point set.
53    ///
54    /// Returns [`Vec2::new(0.0, 0.0)`] for an empty slice instead of panicking.
55    pub fn centroid(points: &[Vec2<U>]) -> Self {
56        let n = points.len() as f64;
57        if n == 0.0 {
58            return Vec2::new(0.0, 0.0);
59        }
60        Vec2::new(
61            points.iter().map(|p| p.x).sum::<f64>() / n,
62            points.iter().map(|p| p.y).sum::<f64>() / n,
63        )
64    }
65}

67impl<U> From<Vec2<U>> for [f64; 2] {
68    /// The point as `[x, y]`.
69    fn from(v: Vec2<U>) -> Self {
70        [v.x, v.y]
71    }
72}

74impl<U> From<[f64; 2]> for Vec2<U> {
75    /// The pair as a `Vec2` in the unit `U`.
76    fn from([x, y]: [f64; 2]) -> Self {
77        Self::new(x, y)
78    }
79}

91    ) {
92        let mut builder = builder;

94        let _ = builder.select_field::<schema::__fields::Point::y, f64>();
95    }

98/// Serde adapter for `#[serde(with)]`: `Vec2<U>` as a compact
99/// `[x, y]` pair instead of the default `{"x": .., "y": ..}` map.
100///
101/// Dataset-style formats like figure profiles or game maps often
102/// prefer the tighter pair shape; downstream crates use this module as
103/// `#[serde(with = "tangible_objects_model::serde_pair")]`.
104pub mod serde_pair {
105    use super::Vec2;
106    use serde::Deserialize;
107    use serde::Deserializer;
108    use serde::Serialize;
109    use serde::Serializer;
110
111    /// Serializes the point as `[x, y]`.
112    pub fn serialize<U, S: Serializer>(v: &Vec2<U>, serializer: S) -> Result<S::Ok, S::Error> {
113        [v.x, v.y].serialize(serializer)
114    }
115
116    /// Deserializes `[x, y]` into a `Vec2` in the unit `U`.
117    pub fn deserialize<'de, U, D: Deserializer<'de>>(deserializer: D) -> Result<Vec2<U>, D::Error> {
118        <[f64; 2]>::deserialize(deserializer).map(Vec2::from)
119    }
120}

122#[cfg(test)]
123mod tests {
124    use super::Vec2;
125    use crate::Millimeters;
126
127    #[test]
128    fn distance_of_axis_aligned_points() {
129        let a = Vec2::<Millimeters>::new(0.0, 0.0);
130        let b = Vec2::<Millimeters>::new(3.0, 4.0);
131        assert!((a.distance(&b).as_f64() - 5.0).abs() < 1e-9);
132    }
133
134    #[test]
135    fn centroid_of_triangle() {
136        let points = [
137            Vec2::<Millimeters>::new(0.0, 0.0),
138            Vec2::<Millimeters>::new(6.0, 0.0),
139            Vec2::<Millimeters>::new(0.0, 6.0),
140        ];
141        let c = Vec2::centroid(&points);
142        assert!((c.x - 2.0).abs() < 1e-9);
143        assert!((c.y - 2.0).abs() < 1e-9);
144    }
145
146    #[test]
147    fn centroid_of_empty_set_is_origin() {
148        let empty: Vec<Vec2<Millimeters>> = Vec::new();
149        assert_eq!(Vec2::centroid(&empty), Vec2::new(0.0, 0.0));
150    }
151
152    #[test]
153    fn array_conversions_roundtrip() {
154        let v = Vec2::<Millimeters>::new(1.5, -2.25);
155        let arr: [f64; 2] = v.into();
156        assert_eq!(arr, [1.5, -2.25]);
157        assert_eq!(Vec2::<Millimeters>::from(arr), v);
158    }
159
160    #[test]
161    fn serde_pair_serializes_as_pair() {
162        #[derive(Debug, PartialEq, serde::Serialize, serde::Deserialize)]
163        struct Row {
164            #[serde(with = "crate::serde_pair")]
165            pos: Vec2<Millimeters>,
166        }
167        let row = Row {
168            pos: Vec2::new(80.0, 220.0),
169        };
170        let json = serde_json::to_string(&row).unwrap_or_default();
171        assert_eq!(json, r#"{"pos":[80.0,220.0]}"#);
172        assert_eq!(serde_json::from_str::<Row>(&json).ok(), Some(row));
173    }
