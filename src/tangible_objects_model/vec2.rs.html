10/// A 2D point or vector tagged with its coordinate unit `U`.
11///
12/// The phantom unit distinguishes positions in millimeters
13/// (`Vec2<Millimeters>`), raw device units (`Vec2<RawAxisValue>`) and screen
14/// pixels (`Vec2<Pixels>`) so they cannot be mixed up. Components stay plain
15/// `f64`; the unit only labels which coordinate space the point lives in.
16/// Serializes as a flat `{"x": .., "y": ..}` map; the unit marker is skipped.
17/// With the `async-graphql` feature, `Vec2<Millimeters>` is exposed as the
18/// GraphQL object type `Point`.
19#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
20#[cfg_attr(feature = "async-graphql", derive(async_graphql::SimpleObject))]
21#[cfg_attr(
22    feature = "async-graphql",
23    graphql(concrete(name = "Point", params(crate::Millimeters)))
24)]
25pub struct Vec2<U> {
26    /// X coordinate in the unit `U`.
27    pub x: f64,
28    /// Y coordinate in the unit `U`.
29    pub y: f64,
30    /// Phantom marker tagging the coordinate unit; not serialized.
31    #[serde(skip)]
32    #[cfg_attr(feature = "async-graphql", graphql(skip))]
33    unit: PhantomData<U>,
34}
35
36impl<U> Vec2<U> {
37    /// The origin `(0, 0)` in the unit `U`.
38    pub const ZERO: Self = Self {
39        x: 0.0,
40        y: 0.0,
41        unit: PhantomData,
42    };
43
44    /// Creates a new 2D point in the unit `U`.
45    pub fn new(x: f64, y: f64) -> Self {
46        Self {
47            x,
48            y,
49            unit: PhantomData,
50        }
51    }
52
53    /// Euclidean distance to another point in the same unit.
54    pub fn distance(&self, other: &Vec2<U>) -> U
55    where
56        U: From<f64>,
57    {
58        U::from(((self.x - other.x).powi(2) + (self.y - other.y).powi(2)).sqrt())
59    }
60
61    /// Dot product `x·x + y·y` of two vectors in the same unit.
62    ///
63    /// The caller picks the product type `T` (like [`Vec2::distance`]'s
64    /// unit return): `f64` for unit-free math or a unit type such as
65    /// [`crate::SquareMillimeters`] for `Vec2<Millimeters>` products.
66    pub fn dot<T: From<f64>>(&self, other: &Vec2<U>) -> T {
67        T::from(self.x * other.x + self.y * other.y)
68    }
69
70    /// 2D determinant `x·y' − y·x'` (z component of the 3D cross
71    /// product): positive when `other` turns left from `self`, negative
72    /// right; zero for (anti-)parallel vectors.
73    ///
74    /// Same caller-chosen product type as [`Vec2::dot`].
75    pub fn cross<T: From<f64>>(&self, other: &Vec2<U>) -> T {
76        T::from(self.x * other.y - self.y * other.x)
77    }
78
79    /// Geometric centroid of a non-empty point set.
80    ///
81    /// Returns [`Vec2::new(0.0, 0.0)`] for an empty slice instead of panicking.
82    pub fn centroid(points: &[Vec2<U>]) -> Self {
83        let n = points.len() as f64;
84        if n == 0.0 {
85            return Vec2::new(0.0, 0.0);
86        }
87        Vec2::new(
88            points.iter().map(|p| p.x).sum::<f64>() / n,
89            points.iter().map(|p| p.y).sum::<f64>() / n,
90        )
91    }
92}

122#[cfg(feature = "cynic")]
123impl<U> cynic::QueryFragment for Vec2<U> {
124    type SchemaType = schema::Point;
125    type VariablesFields = ();
126    const TYPE: Option<&'static str> = Some("Point");
127
128    /// `Point { x y }` - the unit `U` is a compile-time tag only and
129    /// carries no wire representation.
130    fn query(
131        builder: cynic::queries::SelectionBuilder<'_, Self::SchemaType, Self::VariablesFields>,
132    ) {
133        let mut builder = builder;
134        let _ = builder.select_field::<schema::__fields::Point::x, f64>();
135        let _ = builder.select_field::<schema::__fields::Point::y, f64>();
136    }
137}

139/// Serde adapter for `#[serde(with)]`: `Vec2<U>` as a compact
140/// `[x, y]` pair instead of the default `{"x": .., "y": ..}` map.
141///
142/// Dataset-style formats like figure profiles or game maps often
143/// prefer the tighter pair shape; downstream crates use this module as
144/// `#[serde(with = "tangible_objects_model::serde_pair")]`.
145pub mod serde_pair {
146    use super::Vec2;
147    use serde::Deserialize;
148    use serde::Deserializer;
149    use serde::Serialize;
150    use serde::Serializer;
151
152    /// Serializes the point as `[x, y]`.
153    pub fn serialize<U, S: Serializer>(v: &Vec2<U>, serializer: S) -> Result<S::Ok, S::Error> {
154        [v.x, v.y].serialize(serializer)
155    }
156
157    /// Deserializes `[x, y]` into a `Vec2` in the unit `U`.
158    pub fn deserialize<'de, U, D: Deserializer<'de>>(deserializer: D) -> Result<Vec2<U>, D::Error> {
159        <[f64; 2]>::deserialize(deserializer).map(Vec2::from)
160    }
161}

169    #[test]
170    fn distance_of_axis_aligned_points() {
171        let a = Vec2::<Millimeters>::new(0.0, 0.0);
172        let b = Vec2::<Millimeters>::new(3.0, 4.0);
173        assert!((a.distance(&b).as_f64() - 5.0).abs() < 1e-9);
174    }
175
176    #[test]
177    fn centroid_of_triangle() {
178        let points = [
179            Vec2::<Millimeters>::new(0.0, 0.0),
180            Vec2::<Millimeters>::new(6.0, 0.0),
181            Vec2::<Millimeters>::new(0.0, 6.0),
182        ];
183        let c = Vec2::centroid(&points);
184        assert!((c.x - 2.0).abs() < 1e-9);
185        assert!((c.y - 2.0).abs() < 1e-9);
186    }
187
188    #[test]
189    fn dot_and_cross_of_perpendiculars() {
190        let right = Vec2::<Millimeters>::new(1.0, 0.0);
191        let up = Vec2::<Millimeters>::new(0.0, 1.0);
192        let dot: f64 = right.dot(&up);
193        assert!(dot.abs() < 1e-9);
194        let (cross, reverse): (f64, f64) = (right.cross(&up), up.cross(&right));
195        assert!((cross - 1.0).abs() < 1e-9);
196        assert!((reverse - -1.0).abs() < 1e-9);
197        let area: crate::SquareMillimeters = right.cross(&up);
198        assert!((area.as_f64() - 1.0).abs() < 1e-9);
199    }
200
201    #[test]
202    fn dot_and_cross_of_diagonal() {
203        let diagonal = Vec2::<Millimeters>::new(1.0, 1.0);
204        let axis = Vec2::<Millimeters>::new(1.0, 0.0);
205        let (dot, cross): (f64, f64) = (diagonal.dot(&axis), diagonal.cross(&axis));
206        assert!((dot - 1.0).abs() < 1e-9);
207        assert!((cross - -1.0).abs() < 1e-9);
208    }
