12/// Event kind a client may request via `eventStream(kinds: ...)`.
13///
14/// Mirrors [`EventKind`] minus `LAGGED` - the lag control event is never
15/// requestable, it is emitted whenever this subscription drops events
16/// regardless of the `kinds` filter (SUBSCRIPTIONS.md).
17#[derive(Debug, Copy, Clone, Eq, PartialEq)]
18#[cfg_attr(not(feature = "cynic"), derive(Serialize, Deserialize))]
19#[cfg_attr(not(feature = "cynic"), serde(rename_all = "SCREAMING_SNAKE_CASE"))]
20#[cfg_attr(feature = "async-graphql", derive(async_graphql::Enum))]
21#[cfg_attr(feature = "cynic", derive(cynic::Enum))]
22#[cfg_attr(
23    feature = "cynic",
24    cynic(
25        schema_path = "../graphql/schema.graphql",
26        rename_all = "SCREAMING_SNAKE_CASE"

28)]
29pub enum StreamEvent {
30    /// A new tangible was detected and is now tracked.
31    TangibleAdded,
32    /// A tracked tangible's pose changed.
33    TangibleMoved,
34    /// A tracked tangible expired.
35    TangibleRemoved,
36    /// The set of active touch points changed - high-rate (~120 Hz),
37    /// respect `maxHz`.
38    TouchPointsUpdated,
39    /// An occupancy field became occupied.
40    FieldPlaced,
41    /// An occupancy field became empty.
42    FieldLifted,
43    /// An occupancy field's debounced contact count changed.
44    FieldCountChanged,
45    /// All mandatory set members became present (TOS-ADD).
46    SetAdded,
47    /// The last mandatory set member disappeared (TOS-REM).
48    SetRemoved,
49    /// A member tangible appeared in an active set (TOS-OBJ-ADD).
50    ObjectAdded,
51    /// A member tangible disappeared (TOS-OBJ-REM).
52    ObjectRemoved,
53    /// A member tangible's pose changed in an active set (TOS-OBJ-MOV).
54    ObjectMoved,
55    /// A seat was added to a seat session.
56    SeatAdded,
57    /// An anchored seat's derived pose changed (marker moved).
58    SeatMoved,
59    /// A seat was removed from its session.
60    SeatRemoved,
61    /// A marker tangible bound to a seat (first bind or rebind).
62    SeatMarkerBound,
63    /// A seat's marker tangible stopped being tracked.
64    SeatMarkerAbsent,
65    /// A seat session was claimed by an application.
66    SessionClaimed,
67    /// A seat session was released.
68    SessionReleased,
69    /// A pattern was registered or replaced at runtime.
70    ProfileRegistered,
71    /// A registered pattern was removed.
72    ProfileUnregistered,
73    /// A field set was registered at runtime.
74    FieldSetRegistered,
75    /// A field set was removed.
76    FieldSetUnregistered,
77    /// A dead zone was registered at runtime.
78    DeadZoneRegistered,
79    /// A dead zone was removed.
80    DeadZoneUnregistered,
81    /// A scoped tangible stage was created on the pipeline.
82    StageRegistered,
83    /// A scoped tangible stage was removed.
84    StageUnregistered,
85    /// The completion configuration changed.
86    CompletionChanged,
87    /// The filter-proxy suppression policy changed.
88    FilterModeChanged,
89    /// A recording session started.
90    RecordingStarted,
91    /// A recording session stopped.
92    RecordingStopped,
93    /// A file-triggered config reload took effect.
94    ConfigReloaded,
95}

97impl StreamEvent {
98    /// The [`EventKind`] emitted for this request kind - a 1:1 mapping;
99    /// request and emitted kinds share the same vocabulary.
100    pub fn kind(self) -> EventKind {
101        match self {
102            Self::TangibleAdded => EventKind::TangibleAdded,
103            Self::TangibleMoved => EventKind::TangibleMoved,
104            Self::TangibleRemoved => EventKind::TangibleRemoved,
105            Self::TouchPointsUpdated => EventKind::TouchPointsUpdated,
106            Self::FieldPlaced => EventKind::FieldPlaced,
107            Self::FieldLifted => EventKind::FieldLifted,
108            Self::FieldCountChanged => EventKind::FieldCountChanged,
109            Self::SetAdded => EventKind::SetAdded,
110            Self::SetRemoved => EventKind::SetRemoved,
111            Self::ObjectAdded => EventKind::ObjectAdded,
112            Self::ObjectRemoved => EventKind::ObjectRemoved,
113            Self::ObjectMoved => EventKind::ObjectMoved,
114            Self::SeatAdded => EventKind::SeatAdded,
115            Self::SeatMoved => EventKind::SeatMoved,
116            Self::SeatRemoved => EventKind::SeatRemoved,
117            Self::SeatMarkerBound => EventKind::SeatMarkerBound,
118            Self::SeatMarkerAbsent => EventKind::SeatMarkerAbsent,
119            Self::SessionClaimed => EventKind::SessionClaimed,
120            Self::SessionReleased => EventKind::SessionReleased,
121            Self::ProfileRegistered => EventKind::ProfileRegistered,
122            Self::ProfileUnregistered => EventKind::ProfileUnregistered,
123            Self::FieldSetRegistered => EventKind::FieldSetRegistered,
124            Self::StageRegistered => EventKind::StageRegistered,
125            Self::StageUnregistered => EventKind::StageUnregistered,
126            Self::FieldSetUnregistered => EventKind::FieldSetUnregistered,
127            Self::DeadZoneRegistered => EventKind::DeadZoneRegistered,
128            Self::DeadZoneUnregistered => EventKind::DeadZoneUnregistered,
129            Self::CompletionChanged => EventKind::CompletionChanged,
130            Self::FilterModeChanged => EventKind::FilterModeChanged,
131            Self::RecordingStarted => EventKind::RecordingStarted,
132            Self::RecordingStopped => EventKind::RecordingStopped,
133            Self::ConfigReloaded => EventKind::ConfigReloaded,
134        }
135    }
136}

138impl FromStr for StreamEvent {
139    type Err = StreamEventParseError;
140
141    /// Parses a request kind - case-insensitive `SCREAMING_SNAKE` wire
142    /// names (`"tangible_added"`, `"TANGIBLE_ADDED"` both work; the CLI's
143    /// comma-separated `--kinds` values).
144    fn from_str(s: &str) -> Result<Self, Self::Err> {
145        match s.to_uppercase().as_str() {
146            "TANGIBLE_ADDED" => Ok(Self::TangibleAdded),
147            "TANGIBLE_MOVED" => Ok(Self::TangibleMoved),
148            "TANGIBLE_REMOVED" => Ok(Self::TangibleRemoved),
149            "TOUCH_POINTS_UPDATED" => Ok(Self::TouchPointsUpdated),
150            "FIELD_PLACED" => Ok(Self::FieldPlaced),
151            "FIELD_LIFTED" => Ok(Self::FieldLifted),
152            "FIELD_COUNT_CHANGED" => Ok(Self::FieldCountChanged),
153            "SET_ADDED" => Ok(Self::SetAdded),
154            "SET_REMOVED" => Ok(Self::SetRemoved),
155            "OBJECT_ADDED" => Ok(Self::ObjectAdded),
156            "OBJECT_REMOVED" => Ok(Self::ObjectRemoved),
157            "OBJECT_MOVED" => Ok(Self::ObjectMoved),
158            "SEAT_ADDED" => Ok(Self::SeatAdded),
159            "SEAT_MOVED" => Ok(Self::SeatMoved),
160            "SEAT_REMOVED" => Ok(Self::SeatRemoved),
161            "SEAT_MARKER_BOUND" => Ok(Self::SeatMarkerBound),
162            "SEAT_MARKER_ABSENT" => Ok(Self::SeatMarkerAbsent),
163            "SESSION_CLAIMED" => Ok(Self::SessionClaimed),
164            "SESSION_RELEASED" => Ok(Self::SessionReleased),
165            "PROFILE_REGISTERED" => Ok(Self::ProfileRegistered),
166            "PROFILE_UNREGISTERED" => Ok(Self::ProfileUnregistered),
167            "FIELD_SET_REGISTERED" => Ok(Self::FieldSetRegistered),
168            "FIELD_SET_UNREGISTERED" => Ok(Self::FieldSetUnregistered),
169            "DEAD_ZONE_REGISTERED" => Ok(Self::DeadZoneRegistered),
170            "DEAD_ZONE_UNREGISTERED" => Ok(Self::DeadZoneUnregistered),
171            "STAGE_REGISTERED" => Ok(Self::StageRegistered),
172            "STAGE_UNREGISTERED" => Ok(Self::StageUnregistered),
173            "COMPLETION_CHANGED" => Ok(Self::CompletionChanged),
174            "FILTER_MODE_CHANGED" => Ok(Self::FilterModeChanged),
175            "RECORDING_STARTED" => Ok(Self::RecordingStarted),
176            "RECORDING_STOPPED" => Ok(Self::RecordingStopped),
177            "CONFIG_RELOADED" => Ok(Self::ConfigReloaded),
178            _ => Err(StreamEventParseError::new(s)),
179        }
180    }
