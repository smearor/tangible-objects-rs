12/// Event kind a client may request via `eventStream(kinds: ...)`.
13///
14/// Mirrors [`EventKind`] minus `LAGGED` - the lag control event is never
15/// requestable, it is emitted whenever this subscription drops events
16/// regardless of the `kinds` filter (SUBSCRIPTIONS.md).
17#[derive(Debug, Copy, Clone, Eq, PartialEq)]
18#[cfg_attr(not(feature = "cynic"), derive(Serialize, Deserialize))]
19#[cfg_attr(not(feature = "cynic"), serde(rename_all = "SCREAMING_SNAKE_CASE"))]
20#[cfg_attr(feature = "async-graphql", derive(async_graphql::Enum))]
21#[cfg_attr(feature = "cynic", derive(cynic::Enum))]
22#[cfg_attr(
23    feature = "cynic",
24    cynic(
25        schema_path = "../graphql/schema.graphql",
26        rename_all = "SCREAMING_SNAKE_CASE"
27    )
28)]
29pub enum StreamEvent {
30    /// A new tangible was detected and is now tracked.
31    TangibleAdded,
32    /// A tracked tangible's pose changed.
33    TangibleMoved,
34    /// A tracked tangible expired.
35    TangibleRemoved,
36    /// The set of active touch points changed - high-rate (~120 Hz),
37    /// respect `maxHz`.
38    TouchPointsUpdated,
39    /// An occupancy field became occupied.
40    FieldPlaced,
41    /// An occupancy field became empty.
42    FieldLifted,
43    /// An occupancy field's debounced contact count changed.
44    FieldCountChanged,
45    /// All mandatory set members became present (TOS-ADD).
46    SetAdded,
47    /// The last mandatory set member disappeared (TOS-REM).
48    SetRemoved,
49    /// A member tangible appeared in an active set (TOS-OBJ-ADD).
50    ObjectAdded,
51    /// A member tangible disappeared (TOS-OBJ-REM).
52    ObjectRemoved,
53    /// A member tangible's pose changed in an active set (TOS-OBJ-MOV).
54    ObjectMoved,
55    /// A pattern was registered or replaced at runtime.
56    ProfileRegistered,
57    /// A registered pattern was removed.
58    ProfileUnregistered,
59    /// A field set was registered at runtime.
60    FieldSetRegistered,
61    /// A field set was removed.
62    FieldSetUnregistered,
63    /// A dead zone was registered at runtime.
64    DeadZoneRegistered,
65    /// A dead zone was removed.
66    DeadZoneUnregistered,
67    /// A scoped tangible stage was created on the pipeline.
68    StageRegistered,
69    /// A scoped tangible stage was removed.
70    StageUnregistered,
71    /// The completion configuration changed.
72    CompletionChanged,
73    /// The filter-proxy suppression policy changed.
74    FilterModeChanged,
75    /// A recording session started.
76    RecordingStarted,
77    /// A recording session stopped.
78    RecordingStopped,
79    /// A file-triggered config reload took effect.
80    ConfigReloaded,
81}

83impl StreamEvent {
84    /// The [`EventKind`] emitted for this request kind - a 1:1 mapping;
85    /// request and emitted kinds share the same vocabulary.
86    pub fn kind(self) -> EventKind {
87        match self {
88            Self::TangibleAdded => EventKind::TangibleAdded,
89            Self::TangibleMoved => EventKind::TangibleMoved,
90            Self::TangibleRemoved => EventKind::TangibleRemoved,
91            Self::TouchPointsUpdated => EventKind::TouchPointsUpdated,
92            Self::FieldPlaced => EventKind::FieldPlaced,
93            Self::FieldLifted => EventKind::FieldLifted,
94            Self::FieldCountChanged => EventKind::FieldCountChanged,
95            Self::SetAdded => EventKind::SetAdded,
96            Self::SetRemoved => EventKind::SetRemoved,
97            Self::ObjectAdded => EventKind::ObjectAdded,
98            Self::ObjectRemoved => EventKind::ObjectRemoved,
99            Self::ObjectMoved => EventKind::ObjectMoved,
100            Self::ProfileRegistered => EventKind::ProfileRegistered,
101            Self::ProfileUnregistered => EventKind::ProfileUnregistered,
102            Self::FieldSetRegistered => EventKind::FieldSetRegistered,
103            Self::StageRegistered => EventKind::StageRegistered,
104            Self::StageUnregistered => EventKind::StageUnregistered,
105            Self::FieldSetUnregistered => EventKind::FieldSetUnregistered,
106            Self::DeadZoneRegistered => EventKind::DeadZoneRegistered,
107            Self::DeadZoneUnregistered => EventKind::DeadZoneUnregistered,
108            Self::CompletionChanged => EventKind::CompletionChanged,
109            Self::FilterModeChanged => EventKind::FilterModeChanged,
110            Self::RecordingStarted => EventKind::RecordingStarted,
111            Self::RecordingStopped => EventKind::RecordingStopped,
112            Self::ConfigReloaded => EventKind::ConfigReloaded,
113        }
114    }
115}

117impl FromStr for StreamEvent {
118    type Err = StreamEventParseError;
119
120    /// Parses a request kind - case-insensitive `SCREAMING_SNAKE` wire
121    /// names (`"tangible_added"`, `"TANGIBLE_ADDED"` both work; the CLI's
122    /// comma-separated `--kinds` values).
123    fn from_str(s: &str) -> Result<Self, Self::Err> {
124        match s.to_uppercase().as_str() {
125            "TANGIBLE_ADDED" => Ok(Self::TangibleAdded),
126            "TANGIBLE_MOVED" => Ok(Self::TangibleMoved),
127            "TANGIBLE_REMOVED" => Ok(Self::TangibleRemoved),
128            "TOUCH_POINTS_UPDATED" => Ok(Self::TouchPointsUpdated),
129            "FIELD_PLACED" => Ok(Self::FieldPlaced),
130            "FIELD_LIFTED" => Ok(Self::FieldLifted),
131            "FIELD_COUNT_CHANGED" => Ok(Self::FieldCountChanged),
132            "SET_ADDED" => Ok(Self::SetAdded),
133            "SET_REMOVED" => Ok(Self::SetRemoved),
134            "OBJECT_ADDED" => Ok(Self::ObjectAdded),
135            "OBJECT_REMOVED" => Ok(Self::ObjectRemoved),
136            "OBJECT_MOVED" => Ok(Self::ObjectMoved),
137            "PROFILE_REGISTERED" => Ok(Self::ProfileRegistered),
138            "PROFILE_UNREGISTERED" => Ok(Self::ProfileUnregistered),
139            "FIELD_SET_REGISTERED" => Ok(Self::FieldSetRegistered),
140            "FIELD_SET_UNREGISTERED" => Ok(Self::FieldSetUnregistered),
141            "DEAD_ZONE_REGISTERED" => Ok(Self::DeadZoneRegistered),
142            "DEAD_ZONE_UNREGISTERED" => Ok(Self::DeadZoneUnregistered),
143            "STAGE_REGISTERED" => Ok(Self::StageRegistered),
144            "STAGE_UNREGISTERED" => Ok(Self::StageUnregistered),
145            "COMPLETION_CHANGED" => Ok(Self::CompletionChanged),
146            "FILTER_MODE_CHANGED" => Ok(Self::FilterModeChanged),
147            "RECORDING_STARTED" => Ok(Self::RecordingStarted),
148            "RECORDING_STOPPED" => Ok(Self::RecordingStopped),
149            "CONFIG_RELOADED" => Ok(Self::ConfigReloaded),
150            _ => Err(StreamEventParseError::new(s)),
151        }
152    }
153}
