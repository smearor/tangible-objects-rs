6/// Domain event emitted by the engine for a completed input frame.
7///
8/// Collected in [`crate::EngineFrame::events`]; consumed by interface crates
9/// (GraphQL subscriptions, D-Bus signals, MCP notifications).
10#[derive(Debug, Clone)]
11pub enum EngineEvent {
12    /// The set of active touch points changed since the previous frame.
13    TouchPointsUpdated,
14    /// A new tangible was detected and is now tracked.
15    TangibleAdded(TrackedTangible),
16    /// A tracked tangible's smoothed pose was updated by a match.
17    TangibleMoved(TrackedTangible),
18    /// A tracked tangible expired after exceeding its match-less frame budget.
19    TangibleRemoved(TrackedTangible),
20    /// A field became occupied (debounced `placed` transition, count 0 -> n).
21    FieldPlaced {
22        /// Name of the owning field set.
23        set: FieldSetName,
24        /// Name of the occupied field.
25        field: FieldName,
26    },
27    /// A field became empty (debounced `lifted` transition, count n -> 0).
28    FieldLifted {
29        /// Name of the owning field set.
30        set: FieldSetName,
31        /// Name of the vacated field.
32        field: FieldName,
33    },
34    /// A field's debounced contact count changed (multi-contact fields).
35    FieldCountChanged {
36        /// Name of the owning field set.
37        set: FieldSetName,
38        /// Name of the field.
39        field: FieldName,
40        /// Confirmed contact count inside the field.
41        count: ContactCount,
42    },
43}
