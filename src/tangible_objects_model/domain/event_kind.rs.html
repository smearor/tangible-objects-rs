21// Under `cynic` the derive owns serde - the wire form is the same
22// SCREAMING_SNAKE either way, so the feature switch is invisible on
23// the wire.
24pub enum EventKind {
25    /// The set of active touch points changed.
26    TouchPointsUpdated,
27    /// A new tangible was detected and is now tracked.
28    TangibleAdded,
29    /// A tracked tangible's pose changed.
30    TangibleMoved,
31    /// A tracked tangible expired.
32    TangibleRemoved,
33    /// All mandatory set members became present (TOS-ADD).
34    SetAdded,
35    /// The last mandatory set member disappeared (TOS-REM).
36    SetRemoved,
37    /// A member tangible appeared in an active set (TOS-OBJ-ADD).
38    ObjectAdded,
39    /// A member tangible disappeared (TOS-OBJ-REM).
40    ObjectRemoved,
41    /// A member tangible's pose changed in an active set (TOS-OBJ-MOV).
42    ObjectMoved,
43    /// An occupancy field became occupied (debounced placed).
44    FieldPlaced,
45    /// An occupancy field became empty (debounced lifted).
46    FieldLifted,
47    /// An occupancy field's debounced contact count changed.
48    FieldCountChanged,
49    /// A pattern was registered or replaced at runtime.
50    ProfileRegistered,
51    /// A registered pattern was removed.
52    ProfileUnregistered,
53    /// A field set was registered at runtime.
54    FieldSetRegistered,
55    /// A field set was removed.
56    FieldSetUnregistered,
57    /// A scoped tangible stage was created on the pipeline.
58    StageRegistered,
59    /// A scoped tangible stage was removed.
60    StageUnregistered,
61    /// The completion configuration changed.
62    CompletionChanged,
63    /// The filter-proxy suppression policy changed.
64    FilterModeChanged,
65    /// A recording session started.
66    RecordingStarted,
67    /// A recording session stopped (`count` carries the frames written).
68    RecordingStopped,
69    /// A file-triggered config reload took effect.
70    ConfigReloaded,
71    /// Events were skipped because this subscription lagged - a control
72    /// event, never requestable via `kinds`; `skipped` carries the count.
73    Lagged,
74}

76/// `snake_case` text name of the kind - the wire form stays
77/// `SCREAMING_SNAKE_CASE` (serde/cynic); `Display` is what log lines and
78/// event descriptions print.
79impl std::fmt::Display for EventKind {
80    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
81        let name = match self {
82            Self::TouchPointsUpdated => "touch_points_updated",
83            Self::TangibleAdded => "tangible_added",
84            Self::TangibleMoved => "tangible_moved",
85            Self::TangibleRemoved => "tangible_removed",
86            Self::SetAdded => "set_added",
87            Self::SetRemoved => "set_removed",
88            Self::ObjectAdded => "object_added",
89            Self::ObjectRemoved => "object_removed",
90            Self::ObjectMoved => "object_moved",
91            Self::FieldPlaced => "field_placed",
92            Self::FieldLifted => "field_lifted",
93            Self::FieldCountChanged => "field_count_changed",
94            Self::ProfileRegistered => "profile_registered",
95            Self::ProfileUnregistered => "profile_unregistered",
96            Self::FieldSetRegistered => "field_set_registered",
97            Self::FieldSetUnregistered => "field_set_unregistered",
98            Self::StageRegistered => "stage_registered",
99            Self::StageUnregistered => "stage_unregistered",
100            Self::CompletionChanged => "completion_changed",
101            Self::FilterModeChanged => "filter_mode_changed",
102            Self::RecordingStarted => "recording_started",
103            Self::RecordingStopped => "recording_stopped",
104            Self::ConfigReloaded => "config_reloaded",
105            Self::Lagged => "lagged",
106        };
107        f.write_str(name)
108    }
109}
