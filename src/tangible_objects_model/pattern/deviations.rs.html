11/// Per-index absolute deviations `|measured[i] - reference[i]|`
12/// between a measured fingerprint and a reference fingerprint -
13/// parallel to [`crate::Distances`], not a fingerprint itself.
14///
15/// A deviations vector has no ordering invariant (its order is the
16/// fingerprint's distance index order) and its values are deltas, not
17/// distances - the distinction a `Distances`/`Deviations` split makes
18/// visible.
19#[derive(Debug, Clone, PartialEq, Default, Serialize, Deserialize)]
20#[serde(transparent)]
21pub struct Deviations(pub Vec<Deviation>);
22
23impl Deviations {
24    /// The deviation slice.
25    pub fn as_slice(&self) -> &[Deviation] {
26        &self.0
27    }
28
29    /// Highest normalized score - `max_i deviation[i] / tolerance(i)`.
30    ///
31    /// `tolerance_at` resolves the tolerance of deviation `i` (per-index
32    /// `Tolerances` or a widened partial constant); `0.0` on empty input.
33    /// `<= 1.0` means the candidate matched.
34    pub fn max_score(&self, tolerance_at: impl Fn(usize) -> Millimeters) -> MatchScore {
35        self.iter()
36            .enumerate()
37            .map(|(i, dev)| MatchScore::normalized(dev.as_millimeters(), tolerance_at(i)))
38            .reduce(MatchScore::max)
39            .unwrap_or_default()
40    }
41
42    /// Total absolute deviation across all indices - the candidate's
43    /// match error (lower is better).
44    pub fn total(&self) -> MatchError {
45        MatchError::new(Millimeters::new(
46            self.0.iter().map(|deviation| deviation.as_f64()).sum(),
47        ))
48    }
49}
50
51impl From<Vec<Deviation>> for Deviations {
52    /// A deviation vector IS a deviations set.
53    fn from(deviations: Vec<Deviation>) -> Self {
54        Self(deviations)
55    }
56}
57
58impl Deref for Deviations {
59    type Target = [Deviation];
60
61    /// Deviations view their slice transparently.
62    fn deref(&self) -> &[Deviation] {
63        &self.0
64    }
65}

67#[cfg(test)]
68mod tests {
69    use super::Deviations;
70    use crate::Deviation;
71    use crate::Millimeters;
72
73    /// Per-index tolerances: `deviation[i] / tolerance(i)` - the worst
74    /// (highest) normalized score wins.
75    #[test]
76    fn max_score_uses_per_index_tolerance() {
77        let deviations = Deviations::from(vec![
78            Deviation::new(Millimeters::new(5.0)),
79            Deviation::new(Millimeters::new(10.0)),
80            Deviation::new(Millimeters::new(2.0)),
81        ]);
82        // Index-wise: 5/10, 10/20, 2/0.5 → 0.5, 0.5, 4.0 → max = 4.0.
83        let score = deviations.max_score(|i| match i {
84            0 => Millimeters::new(10.0),
85            1 => Millimeters::new(20.0),
86            _ => Millimeters::new(0.5),
87        });
88        assert_eq!(score.as_f64(), 4.0);
89    }
90
91    /// A constant tolerance closure is the partial-matching case:
92    /// `dev / tol` against the same widened baseline.
93    #[test]
94    fn max_score_constant_tolerance() {
95        let deviations = Deviations::from(vec![
96            Deviation::new(Millimeters::new(3.0)),
97            Deviation::new(Millimeters::new(9.0)),
98            Deviation::new(Millimeters::new(4.5)),
99        ]);
100        let score = deviations.max_score(|_| Millimeters::new(4.5));
101        assert_eq!(score.as_f64(), 2.0);
102    }
103
104    /// An empty vector aggregates to the `max` identity `0.0`.
105    #[test]
106    fn max_score_empty_is_zero() {
107        let score = Deviations::default().max_score(|_| Millimeters::new(1.0));
108        assert_eq!(score.as_f64(), 0.0);
109    }
