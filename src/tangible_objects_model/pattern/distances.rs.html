9/// Sorted pairwise-distance fingerprint of a point set.
10///
11/// Invariant: elements are sorted ascending - matching compares
12/// fingerprints index-by-index against `tolerance_at(i)`. A *deviation*
13/// vector is a different thing (element-wise error, not sorted); it
14/// stays `Vec<Millimeters>`.
15#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
16#[serde(transparent)]
17pub struct Distances(pub Vec<Millimeters>);
18
19impl Distances {
20    /// Sorted pairwise distances of a point set - the fingerprint used
21    /// for matching.
22    pub fn of(points: PointSet<'_>) -> Self {
23        let mut distances = Vec::new();
24        for (index, a) in points.iter().enumerate() {
25            for b in &points[index + 1..] {
26                distances.push(a.distance(b));
27            }
28        }
29        distances.sort_by(Millimeters::total_cmp);
30        Self(distances)
31    }
32
33    /// Largest absolute per-index deviation against another fingerprint.
34    /// Returns [`f64::MAX`] millimeters for empty or length-mismatched
35    /// fingerprints - a maximal, never-matching collision distance.
36    pub fn max_deviation(&self, other: &Self) -> crate::Deviation {
37        if self.len() != other.len() || self.is_empty() {
38            return Millimeters::new(f64::MAX).into();
39        }
40        self.iter()
41            .zip(other.iter())
42            .map(|(a, b)| (*a - *b).abs())
43            .map(crate::Deviation::from)
44            .max_by(|a, b| a.total_cmp(b))
45            .unwrap_or(Millimeters::new(f64::MAX).into())
46    }
47
48    /// Absolute per-index deviations against another fingerprint;
49    /// empty when the lengths differ.
50    pub fn deviations(&self, other: &Self) -> Deviations {
51        if self.len() != other.len() {
52            return Deviations::default();
53        }
54        self.iter()
55            .zip(other.iter())
56            .map(|(a, b)| (*a - *b).abs().into())
57            .collect::<Vec<crate::Deviation>>()
58            .into()
59    }
60}
