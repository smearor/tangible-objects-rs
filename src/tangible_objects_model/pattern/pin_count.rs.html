8/// Number of contact pins a figure must have - a structural property of
9/// the pattern, not a measured observation.
10///
11/// Kept distinct from [`crate::ContactCount`]: `num_points` describes
12/// what the figure *is*, `ContactCount` counts what the screen *sees*.
13/// `Ord` derives `min`/`max` for subset-size arithmetic.
14#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
15#[serde(transparent)]
16pub struct PinCount(pub usize);

35    /// All `missing`-element pin subsets of `0..self` - the reference
36    /// positions a partial fingerprint skips. Returned as positional
37    /// combinations (ref indices, not touch indices).
38    pub fn skip_combinations(self, missing: usize) -> Combinations {
39        let mut result = Vec::new();
40        let mut combo = Vec::with_capacity(missing);
41        enumerate_subsets(self.0, missing, 0, &mut combo, &mut result);
42        result.into()
43    }
44}
45
46/// Subset backtracking of `skip_combinations`: `combo` accumulates
47/// `missing` positions from `start..n`, recursing to `start + 1`.
48fn enumerate_subsets(
49    n: usize,
50    missing: usize,
51    start: usize,
52    combo: &mut Vec<usize>,
53    out: &mut Vec<crate::CombinationBuf>,
54) {
55    if combo.len() == missing {
56        out.push(combo.clone().into());
57        return;
58    }
59    for i in start..n {
60        combo.push(i);
61        enumerate_subsets(n, missing, i + 1, combo, out);
62        combo.pop();
63    }
64}

127#[cfg(test)]
128mod tests {
129    use super::PinCount;
130
131    /// C(4, 1) enumerates the four singleton skips.
132    #[test]
133    fn skip_combinations_one_of_four() {
134        let skips = PinCount::new(4).skip_combinations(1);
135        let combos: Vec<&[usize]> = skips
136            .iter()
137            .map(|c| c.as_combination().as_slice())
138            .collect();
139        assert_eq!(combos, vec![&[0], &[1], &[2], &[3]]);
140    }
141
142    /// C(4, 2) enumerates six distinct pairs.
143    #[test]
144    fn skip_combinations_two_of_four() {
145        let skips = PinCount::new(4).skip_combinations(2);
146        assert_eq!(skips.len(), 6);
147        let combos: Vec<&[usize]> = skips
148            .iter()
149            .map(|c| c.as_combination().as_slice())
150            .collect();
151        assert_eq!(
152            combos,
153            vec![&[0, 1], &[0, 2], &[0, 3], &[1, 2], &[1, 3], &[2, 3]]
154        );
155    }
156
157    /// C(n, n) is exactly the full-set combination.
158    #[test]
159    fn skip_combinations_all_is_full_set() {
160        let skips = PinCount::new(3).skip_combinations(3);
161        assert_eq!(skips.len(), 1);
162        assert_eq!(skips[0].as_combination().as_slice(), &[0, 1, 2]);
163    }
164
165    /// C(n, 0) is exactly the empty combination.
166    #[test]
167    fn skip_combinations_zero_is_empty() {
168        let skips = PinCount::new(4).skip_combinations(0);
169        assert_eq!(skips.len(), 1);
170        assert!(skips[0].as_combination().as_slice().is_empty());
171    }
