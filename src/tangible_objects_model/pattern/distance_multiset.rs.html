1use smearor_commons_geometry::Millimeters;
2use smearor_commons_geometry::Vec2;
3/// Sorted distance multiset of one point to all other points in its
4/// set - the per-point fingerprint correspondence solving compares
5/// (ROTATION.md). Two equal multisets mean the two points occupy the
6/// same role in their respective geometries.
7pub struct DistanceMultiset(pub Vec<Millimeters>);
8
9impl DistanceMultiset {
10    /// The multiset of `points[index]` - sorted distances to every
11    /// other point in the set.
12    pub fn of(points: &[Vec2<Millimeters>], index: usize) -> Self {
13        let mut dists: Vec<Millimeters> = points
14            .iter()
15            .enumerate()
16            .filter(|(j, _)| *j != index)
17            .map(|(_, other)| points[index].distance(other))
18            .collect();
19        dists.sort_by(Millimeters::total_cmp);
20        Self(dists)
21    }
22
23    /// Whether `other` has an equal multiset - same length and every
24    /// pairwise distance within `tolerance` of ours.
25    pub fn matches_within(&self, other: &DistanceMultiset, tolerance: Millimeters) -> bool {
26        self.0.len() == other.0.len()
27            && self
28                .0
29                .iter()
30                .zip(other.0.iter())
31                .all(|(a, b)| (*a - *b).abs() <= tolerance)
32    }
33}
