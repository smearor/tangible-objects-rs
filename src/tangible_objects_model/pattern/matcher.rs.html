99    fn find_matches_inner(
100        self,
101        touches: TouchSet<'_>,
102        diagnostics: &mut impl CandidateDiagnostics,
103        exclude: &TakenIndices,
104        completion: &CompletionConfig,
105    ) -> (Vec<PatternMatch>, usize) {
106        let mut matches = Vec::new();
107        let mut tested = 0usize;
108        let mut used_indices = TakenIndices::default();
109
110        // Sort by pattern ID for a deterministic evaluation order.
111        let mut sorted_patterns: Vec<&TangiblePattern> = self.iter().collect();
112        sorted_patterns.sort_by_key(|p| p.id);
113
114        for pattern in sorted_patterns {
115            // A missing pin may still match via the partial (num_points - 1)
116            // reference - patterns with stored reference points stay eligible
117            // one touch short of the full pin count.
118            let partial_possible = pattern.ref_points.is_some()
119                && pattern.num_points >= 4
120                && touches.len() + 1 >= pattern.num_points;
121            if touches.len() < pattern.num_points && !partial_possible {
122                continue;
123            }
124
125            // Indices of touch points not yet assigned to a figure and not
126            // reserved by an incumbent slot binding (slot affinity).
127            let taken = used_indices.union(exclude);
128            let available = touches.available_indices(&taken);
129            let pos_of = |i: usize| touches[i].pos;
130
131            if available.len() < pattern.num_points
132                && !(partial_possible && available.len() + 1 >= pattern.num_points)
133            {
134                continue;
135            }
136
137            let mut best = BestScore::default();
138
139            for combo in combinations_within(
140                &available,
141                pattern.num_points.as_usize(),
142                &pos_of,
143                pattern.footprint(),
144            ) {
145                tested += 1;
146                let points = touches.positions_of(&combo);
147
148                let matched =
149                    pattern.matches((&points).into()) && pattern.accepts((&points).into());
150                diagnostics.report(|| {
151                    exact_report(pattern, points.as_point_set(), (&combo).into(), matched)
152                });
153                if matched {
154                    // Score-based ranking (TRAINING.md 3.2): the normalized
155                    // max-deviation score - equivalent to total error under a
156                    // scalar tolerance, decisive under per-index tolerances.
157                    let score = pattern.match_score((&points).into());
158                    let candidate = pattern.build_match(
159                        points.as_point_set(),
160                        combo.as_combination(),
161                        false,
162                        Default::default(),
163                    );
164                    best.consider(candidate, score);
165                }
166            }
167
168            // Partial match: with reference points, try N-k points when a
169            // figure loses contacts (e.g. a tilting object). `max_missing`
170            // bounds how many pins may be missing (1 or 2); 4+ pin figures
171            // only. A successful partial may then *claim* the predicted
172            // missing pins from still-unassigned touches (completion).
173            //
174            // Spawn gate: partials may only *create* a tangible for
175            // `spawn_on_partial` patterns - reduced contact sets of other
176            // figures must not spawn phantoms (SPAWN_AS_PARTIAL.md).
177            if best.is_none()
178                && pattern.spawn_on_partial
179                && pattern.ref_points.is_some()
180                && available.len() + completion.max_missing as usize >= pattern.num_points
181                && pattern.num_points >= 4
182            {
183                let max_missing = (completion.max_missing as usize).min(pattern.num_points - 1);
184                for missing in 1..=max_missing {
185                    let partial_n = pattern.num_points - missing;
186                    for combo in
187                        combinations_within(&available, partial_n, &pos_of, pattern.footprint())
188                    {
189                        tested += 1;
190                        let points = touches.positions_of(&combo);
191
192                        let result = pattern
193                            .matches_partial_missing((&points).into(), missing)
194                            .filter(|_| pattern.accepts((&points).into()));
195                        diagnostics.report(|| {
196                            partial_report(
197                                pattern,
198                                points.as_point_set(),
199                                (&combo).into(),
200                                result.is_some(),
201                            )
202                        });
203                        if let Some(matched) = result {
204                            // Partial candidates share the same (scalar × 1.5)
205                            // scale - ranking by error is ranking by
206                            // normalized score here.
207                            let score = MatchScore::normalized(
208                                matched.match_error.as_millimeters(),
209                                pattern.partial_tolerance(),
210                            );
211                            let free = available.without(combo.as_combination());
212                            let claimed = pattern.claim_missing(
213                                &matched.kept_indices,
214                                points.as_point_set(),
215                                (&combo).into(),
216                                free.as_claim_pool(),
217                                touches,
218                                completion,
219                            );
220                            let mut candidate = pattern.build_match(
221                                points.as_point_set(),
222                                combo.as_combination(),
223                                true,
224                                claimed,
225                            );
226                            candidate.resolve_correspondence(
227                                pattern,
228                                points.as_point_set(),
229                                &matched.kept_indices,
230                            );
231                            best.consider(candidate, score);
232                        }
233                    }
234                }
235            }
236
237            if let Some(mut pattern_match) = best.into_value() {
238                // Exact winners solve their pin correspondence once
239                // (ROTATION.md); partials solved it at build time, where
240                // the kept reference subset is still in scope.
241                if !pattern_match.partial {
242                    let points = touches.positions_of(&pattern_match.touch_indices);
243                    let kept_refs = RefIndices::all(pattern.num_points);
244                    pattern_match.resolve_correspondence(
245                        pattern,
246                        points.as_point_set(),
247                        &kept_refs,
248                    );
249                }
250                // Matched pins and completion-claimed pins are both consumed -
251                // a claimed point belongs to this figure this frame even when
252                // `persist` is off (persistence only governs future binding).
253                for index in pattern_match.all_indices() {
254                    used_indices.insert(index);
255                }
256                matches.push(pattern_match);
257            }
258        }
259
260        (matches, tested)
261    }

264/// Builds a [`CandidateReport`] for an exact (full-pin) combination.
265fn exact_report(
266    pattern: &TangiblePattern,
267    points: PointSet<'_>,
268    combo: Combination<'_>,
269    matched: bool,
270) -> CandidateReport {
271    let measured_distances = points.distances();
272    let deviations = measured_distances.deviations(&pattern.sorted_distances);
273    CandidateReport {
274        pattern_id: pattern.id,
275        pattern_name: pattern.name.clone(),
276        touch_indices: combo.into(),
277        matched,
278        partial: false,
279        match_error: deviations.total(),
280        score: Some(pattern.score_of(&deviations)),
281        tolerances: pattern.tolerances.clone(),
282        measured_distances,
283        deviations,
284        tolerance: pattern.tolerance,
285    }
286}

288/// Builds a [`CandidateReport`] for a partial (`num_points - 1`) combination.
289///
290/// Deviations are reported against the best-fitting one-pin-dropped reference
291/// subset - the closest geometric explanation, matched or not.
292fn partial_report(
293    pattern: &TangiblePattern,
294    points: PointSet<'_>,
295    combo: Combination<'_>,
296    matched: bool,
297) -> CandidateReport {
298    let measured_distances = points.distances();
299    let (deviations, match_error) = (0..pattern.num_points.as_usize())
300        .filter_map(|skip| pattern.partial_reference(skip.into()))
301        .map(|reference| {
302            let deviations = TangiblePattern::deviations(&measured_distances, &reference.distances);
303            let error = deviations.total();
304            (deviations, error)
305        })
306        .min_by(|a, b| a.1.total_cmp(&b.1))
307        .unwrap_or_else(|| (Default::default(), Millimeters::new(0.0).into()));
308
309    let score = deviations.max_score(|_| pattern.partial_tolerance());
310
311    CandidateReport {
312        pattern_id: pattern.id,
313        pattern_name: pattern.name.clone(),
314        touch_indices: combo.into(),
315        matched,
316        partial: true,
317        measured_distances,
318        deviations,
319        tolerance: pattern.tolerance,
320        tolerances: Default::default(),
321        score: Some(score),
322        match_error,
323    }
324}

326/// DFS context for [`combinations_within`] (SPATIAL_CLUSTERING.md):
327/// the fixed enumeration inputs - item pool, arity, position lookup
328/// and the pattern's span bound - kept out of the recursion's
329/// parameter list.
330struct SpanSearch<'a> {
331    /// Item pool to draw indices from (touch indices of the frame).
332    items: &'a [usize],
333    /// Combination arity - how many indices each candidate holds.
334    arity: usize,
335    /// Position lookup: item (touch index) -> surface position in mm.
336    pos_of: &'a dyn Fn(usize) -> Vec2<Millimeters>,
337    /// Pairwise-diameter bound - extending beyond prunes the branch.
338    max_span: Millimeters,
339}
340
341impl SpanSearch<'_> {
342    /// Depth-first combination walk: `start` is the next item position,
343    /// `combo`/`span` the accumulating prefix and its pairwise diameter.
344    fn descend(
345        &self,
346        start: usize,
347        combo: &mut Vec<usize>,
348        span: &mut Millimeters,
349        result: &mut Vec<CombinationBuf>,
350    ) {
351        if combo.len() == self.arity {
352            result.push(combo.clone().into());
353            return;
354        }
355        let needed = self.arity - combo.len();
356        for (offset, &item) in self.items[start..].iter().enumerate() {
357            if self.items.len() - (start + offset) < needed {
358                break;
359            }
360            let pos = (self.pos_of)(item);
361            let new_span = combo
362                .iter()
363                .map(|&c| pos.distance(&(self.pos_of)(c)))
364                .fold(*span, |acc, d| if d > acc { d } else { acc });
365            if new_span > self.max_span {
366                continue;
367            }
368            let old_span = *span;
369            *span = new_span;
370            combo.push(item);
371            self.descend(start + offset + 1, combo, span, result);
372            combo.pop();
373            *span = old_span;
374        }
375    }
376}

378/// Span-pruned k-combination enumeration (SPATIAL_CLUSTERING.md):
379/// `pos_of` maps each item to its position; a partial combination
380/// whose pairwise diameter exceeds `max_span` prunes the whole branch -
381/// adding points never shrinks a diameter, so every superposition
382/// would fail the fingerprint anyway. Emits the same lexicographic
383/// order as [`combinations`], minus provably impossible candidates.
384fn combinations_within(
385    items: &[usize],
386    k: usize,
387    pos_of: &impl Fn(usize) -> Vec2<Millimeters>,
388    max_span: Millimeters,
389) -> Combinations {
390    if k == 0 {
391        return Combinations::one_empty();
392    }
393    let mut result = Vec::new();
394    SpanSearch {
395        items,
396        arity: k,
397        pos_of,
398        max_span,
399    }
400    .descend(
401        0,
402        &mut Vec::with_capacity(k),
403        &mut Millimeters::default(),
404        &mut result,
405    );
406    result.into()
407}

409/// Generates all k-element index combinations of `items` - the
410/// unpruned baseline kept for tests.
411#[cfg(test)]
412fn combinations(items: &[usize], k: usize) -> Combinations {
413    if k == 0 {
414        return Combinations::one_empty();
415    }
416    if items.is_empty() || items.len() < k {
417        return Combinations::default();
418    }
419
420    let mut result = Combinations::new();
421    let head = items[0];
422    let tail = &items[1..];
423
424    for mut combo in combinations(tail, k - 1) {
425        combo.0.insert(0, head);
426        result.push(combo);
427    }
428    result.extend(combinations(tail, k));
429
430    result
431}
