33/// A tangible figure pattern defined by invariant pairwise distances.
34///
35/// For n points the fingerprint consists of all `n*(n-1)/2` pairwise
36/// distances, sorted ascending. Matching compares a measured distance set
37/// against the fingerprint within a tolerance, which makes the fingerprint
38/// position- and rotation-invariant.
39#[derive(Debug, Clone)]
40pub struct TangiblePattern {
41    /// Runtime ID assigned when the profile directory is loaded.
42    pub id: PatternId,
43    /// Human-readable figure name (from the profile file).
44    pub name: PatternName,
45    /// Whether a partial match may spawn this tangible
46    /// (SPAWN_AS_PARTIAL.md); the incumbent path ignores the flag.
47    pub spawn_on_partial: bool,
48    /// Frames a partial spawn must hold before the tangible leaves
49    /// pending state.
50    pub spawn_stability: FrameCount,
51    /// Number of contact pins the figure must have.
52    pub num_points: PinCount,
53    /// Sorted pairwise distances of the reference point set (the fingerprint).
54    pub sorted_distances: Distances,
55    /// Maximum allowed deviation per distance during matching.
56    pub tolerance: PatternTolerance,
57    /// Optional per-distance tolerances, parallel to `sorted_distances`.
58    /// When empty (or length-mismatched), the scalar `tolerance` applies to
59    /// every index - the recorded `p99` suggestions from the analysis crate
60    /// are written here.
61    pub tolerances: PatternTolerances,
62    /// Original reference point positions, if stored in the profile.
63    /// Required for partial matching when one contact pin is missing.
64    pub ref_points: Option<RefPoints>,
65    /// Detection regions restricting where this figure may match
66    /// (PROFILE_REGIONS.md). Empty = anywhere on the surface.
67    pub regions: Vec<Region>,
68    /// Pipeline stage this pattern suppresses through
69    /// (EVENT_PIPELINE.md). `None` = the general stage.
70    pub stage: Option<StageName>,
71}

73impl TangiblePattern {
74    /// Creates a pattern from known reference point positions.
75    pub fn from_points(
76        id: impl Into<PatternId>,
77        name: &str,
78        points: PointSet<'_>,
79        tolerance: impl Into<PatternTolerance>,
80    ) -> Self {
81        Self {
82            id: id.into(),
83            name: name.into(),
84            spawn_on_partial: false,
85            spawn_stability: DEFAULT_SPAWN_STABILITY,
86            num_points: PinCount::new(points.len()),
87            sorted_distances: points.distances(),
88            tolerance: tolerance.into(),
89            tolerances: PatternTolerances::default(),
90            ref_points: Some(points.to_vec().into()),
91            regions: Vec::new(),
92            stage: None,
93        }
94    }

96    /// Whether `points` may match spatially: every point must lie inside
97    /// the union of this pattern's [`Region`]s. With no regions configured
98    /// every position is accepted.
99    pub fn accepts(&self, points: PointSet<'_>) -> bool {
100        self.regions.is_empty()
101            || points
102                .iter()
103                .all(|point| self.regions.iter().any(|region| region.contains(point)))
104    }

106    /// The effective tolerance for distance index `i`: the per-index value
107    /// when `tolerances` is configured, else the scalar `tolerance`.
108    pub fn tolerance_at(&self, index: usize) -> PatternTolerance {
109        self.tolerances
110            .get(index)
111            .copied()
112            .unwrap_or(self.tolerance)
113    }

115    /// Checks whether a set of measured points matches this pattern exactly.
116    ///
117    /// Each deviation is compared against its per-index tolerance
118    /// ([`Self::tolerance_at`]); the scalar `tolerance` is the fallback when
119    /// no `tolerances` vector is configured.
120    pub fn matches(&self, points: PointSet<'_>) -> bool {
121        if points.len() != self.num_points {
122            return false;
123        }
124        let measured = points.distances();
125        measured.len() == self.sorted_distances.len()
126            && measured
127                .iter()
128                .zip(self.sorted_distances.iter())
129                .enumerate()
130                .all(|(i, (m, t))| (*m - *t).abs() <= self.tolerance_at(i).as_millimeters())
131    }
132
133    /// The pattern's spatial footprint - the largest span a matching
134    /// candidate's pairwise diameter may reach, `max_ij(s_ij + tol_ij)`
135    /// (SPATIAL_CLUSTERING.md). Combinations spanning more than this can
136    /// never satisfy the distance fingerprint; the matcher prunes them
137    /// before evaluation.
138    pub fn footprint(&self) -> Millimeters {
139        self.sorted_distances
140            .iter()
141            .enumerate()
142            .map(|(i, s)| *s + self.tolerance_at(i).as_millimeters())
143            .max_by(Millimeters::total_cmp)
144            .unwrap_or_default()
145    }
146
147    /// Normalized deviation score of a measured fingerprint: the largest
148    /// deviation-to-tolerance ratio across all distance indices.
149    ///
150    /// `score <= 1.0` is equivalent to [`Self::matches`] on the same point
151    /// set - but the score additionally ranks candidates (lower is closer)
152    /// and quantifies near-misses, which the score-based matcher and the
153    /// offline analysis use.
154    pub fn match_score(&self, points: PointSet<'_>) -> MatchScore {
155        self.score_of(&points.distances().deviations(&self.sorted_distances))
156    }

158    /// Builds the [`PatternMatch`] for a matched point combination:
159    /// centroid center, `rotation_of` estimate (replaced later for
160    /// exact winners via [`PatternMatch::resolve_correspondence`]),
161    /// and the `(partial)` name marker on reduced-fingerprint matches.
162    pub fn build_match(
163        &self,
164        points: PointSet<'_>,
165        combo: Combination<'_>,
166        partial: bool,
167        claimed: ClaimedIndices,
168    ) -> PatternMatch {
169        let center = Vec2::centroid(points.as_slice());
170        let name = if partial {
171            format!("{} (partial)", self.name).into()
172        } else {
173            self.name.clone()
174        };
175        PatternMatch {
176            pattern_id: self.id,
177            pattern_name: name,
178            spawn_stability: self.spawn_stability,
179            center,
180            rotation_rad: points.rotation_of(&center),
181            touch_indices: combo.into(),
182            correspondence: RefIndices::default(),
183            ref_points: None,
184            partial,
185            claimed_indices: claimed,
186        }
187    }
188
189    /// Incumbent re-match on bound slots (TRACKING_STABILIZATION.md):
190    /// `combo` is verified against the affinity-scaled score
191    /// (`match_score <= factor`) - the slot binding already proves
192    /// identity, so motion distortion must not lose the match.
193    pub fn try_exact(
194        &self,
195        combo: Combination<'_>,
196        touches: TouchSet<'_>,
197        completion: &CompletionConfig,
198    ) -> Option<PatternMatch> {
199        let points = touches.positions_of(combo.as_slice());
200        (self.match_score((&points).into()).as_f64() <= completion.affinity_factor.as_f64()
201            && self.accepts((&points).into()))
202        .then(|| self.build_match(points.as_point_set(), combo, false, Default::default()))
203    }
204
205    /// Like [`Self::try_exact`] for a partial (`num_points - 1`)
206    /// combination; also claims the predicted missing pin(s) from
207    /// `pool` when completion is enabled.
208    pub fn try_partial(
209        &self,
210        combo: Combination<'_>,
211        touches: TouchSet<'_>,
212        completion: &CompletionConfig,
213        pool: ClaimPool<'_>,
214    ) -> Option<PatternMatch> {
215        let points = touches.positions_of(combo.as_slice());
216        let missing = self.num_points.as_usize().checked_sub(combo.len())?;
217        self.matches_partial_missing_scaled((&points).into(), missing, completion.affinity_factor)
218            .filter(|_| self.accepts((&points).into()))
219            .map(|matched| {
220                let claimed = self.claim_missing(
221                    &matched.kept_indices,
222                    points.as_point_set(),
223                    combo,
224                    pool,
225                    touches,
226                    completion,
227                );
228                self.build_match(points.as_point_set(), combo, true, claimed)
229            })
230    }

232    /// Claims the predicted positions of a partial match's missing
233    /// pins (TRACKING_STABILIZATION.md).
234    ///
235    /// `kept_refs` are the reference indices that matched;
236    /// `points`/`combo` the measured points. For every missing
237    /// reference pin the expected position is predicted via
238    /// [`Self::predict_missing`] and the nearest free touch inside
239    /// `radius_factor` × tolerance is claimed.
240    pub fn claim_missing(
241        &self,
242        kept_refs: &RefIndices,
243        points: PointSet<'_>,
244        combo: Combination<'_>,
245        pool: ClaimPool<'_>,
246        touches: TouchSet<'_>,
247        completion: &CompletionConfig,
248    ) -> ClaimedIndices {
249        let radius = self.tolerance.as_millimeters() * completion.radius_factor;
250        let mut claimed: Vec<Claim> = Vec::new();
251        let mut taken: HashSet<usize> = combo.iter().copied().collect();
252        for missing in kept_refs.complement(self.num_points) {
253            let Some(predicted) = self.predict_missing(kept_refs, points, missing) else {
254                continue;
255            };
256            let nearest = pool
257                .iter()
258                .filter(|index| !taken.contains(index))
259                .min_by(|&&a, &&b| {
260                    touches[a]
261                        .pos
262                        .distance(&predicted)
263                        .total_cmp(&touches[b].pos.distance(&predicted))
264                });
265            if let Some(&index) =
266                nearest.filter(|&&index| touches[index].pos.distance(&predicted) <= radius)
267            {
268                claimed.push(Claim::new(index, missing));
269                taken.insert(index);
270            }
271        }
272        claimed.into()
273    }

275    /// Score of an already-computed deviation vector - `max_i dev[i] /
276    /// tolerance_at(i)`; 0.0 for empty or length-mismatched input.
277    pub fn score_of(&self, deviations: &Deviations) -> MatchScore {
278        deviations.max_score(|i| self.tolerance_at(i).as_millimeters())
279    }

281    /// Total absolute deviation across all distances (lower is better).
282    pub fn match_error(&self, points: PointSet<'_>) -> MatchError {
283        points
284            .distances()
285            .deviations(&self.sorted_distances)
286            .total()
287    }

289    /// Absolute per-index deviation between a measured and a reference

291    pub fn deviations(measured: &Distances, reference: &Distances) -> Deviations {

295    /// Widened acceptance tolerance for partial (missing-pin) matches -
296    /// a reduced fingerprint needs more slack than an exact one
297    /// (SPAWN_AS_PARTIAL.md): `self.tolerance` unscaled.
298    pub fn partial_tolerance(&self) -> Millimeters {
299        self.tolerance.partial(AffinityFactor::neutral())
300    }

302    /// Fingerprint of the reference point set with pin `skip` removed.
303    ///
304    /// Returns the kept reference indices and their sorted pairwise
305    /// distances; `None` when no reference points are stored, their count
306    /// differs from `num_points` or `skip` is out of range.
307    pub fn partial_reference(&self, skip: RefIndex) -> Option<PartialReference> {
308        self.partial_reference_multi(&vec![skip].into())
309    }
310
311    /// Like [`Self::partial_reference`] with `skips` pins removed -
312    /// needed for `num_points - 2` completion candidates.
313    pub fn partial_reference_multi(&self, skips: &RefIndices) -> Option<PartialReference> {
314        let ref_points = self.ref_points.as_ref()?;
315        if ref_points.len() != self.num_points
316            || skips.iter().any(|skip| skip.as_usize() >= self.num_points)
317            || skips.is_empty()
318        {
319            return None;
320        }
321        let subset: Vec<Vec2<Millimeters>> = ref_points
322            .iter()
323            .enumerate()
324            .filter(|(index, _)| !skips.contains(&RefIndex::new(*index)))
325            .map(|(_, point)| *point)
326            .collect();
327        let indices = skips.complement(self.num_points);
328        Some(PartialReference {
329            kept_indices: indices,
330            distances: PointSet::new(&subset).distances(),
331        })
332    }

334    /// Partial match: tests whether `points` (`num_points - 1` contacts) fit a
335    /// subset of the reference points with one pin missing (e.g. a tilting
336    /// object). Every possibility of dropping one reference point is tried.
337    /// The scalar tolerance is raised by 50% to account for the higher
338    /// uncertainty - per-index `tolerances` cannot apply here because
339    /// dropping a pin re-indexes the distance fingerprint.
340    ///
341    /// Returns the matched reference indices and the total error.
342    pub fn matches_partial(&self, points: PointSet<'_>) -> Option<PartialMatch> {
343        self.matches_partial_missing(points, 1)
344    }
345
346    /// Generalized partial match with `missing` dropped pins (1 or 2).
347    /// The returned indices are the *kept* reference points, ordered like
348    /// the reference - the complement are the missing pins, used by
349    /// missing-pin completion to predict their expected positions.
350    pub fn matches_partial_missing(
351        &self,
352        points: PointSet<'_>,
353        missing: usize,
354    ) -> Option<PartialMatch> {
355        let ref_pts = self.ref_points.as_ref()?;
356        if self.num_points == 0
357            || missing == 0
358            || missing >= self.num_points
359            || points.len() != self.num_points - missing
360            || ref_pts.len() != self.num_points
361        {
362            return None;
363        }
364        self.matches_partial_missing_scaled(points, missing, AffinityFactor::neutral())
365    }
366
367    /// Like [`Self::matches_partial_missing`], with the partial tolerance
368    /// scaled by `tol_scale` - incumbent re-matching relaxes the threshold
369    /// (slot binding already proves identity); `1.0` keeps the default.
370    pub fn matches_partial_missing_scaled(
371        &self,
372        points: PointSet<'_>,
373        missing: usize,
374        tol_scale: AffinityFactor,
375    ) -> Option<PartialMatch> {
376        self.ref_points.as_ref()?;
377        let partial_tolerance = self.tolerance.partial(tol_scale);
378        let measured = points.distances();
379        let mut best: Option<PartialMatch> = None;
380
381        for skips in self.num_points.skip_combinations(missing) {
382            let Some(reference) = self.partial_reference_multi(&skips.ref_indices()) else {
383                continue;
384            };
385
386            let deviations = Self::deviations(&measured, &reference.distances);
387            if deviations.len() == reference.distances.len()
388                && deviations
389                    .iter()
390                    .all(|d| d.as_f64() <= partial_tolerance.as_f64())
391            {
392                let candidate = PartialMatch {
393                    kept_indices: reference.kept_indices,
394                    match_error: deviations.total(),
395                };
396                if best
397                    .as_ref()
398                    .is_none_or(|b| candidate.match_error < b.match_error)
399                {
400                    best = Some(candidate);
401                }
402            }
403        }
404        best
405    }

407    /// Predicts the screen position of a missing reference pin from a
408    /// partial match's geometry.
409    ///
410    /// `kept_refs` are the reference indices that matched (from
411    /// [`Self::matches_partial_missing`]); `measured` are the frame
412    /// positions of the kept pins in the same order. The missing pin's
413    /// expected position is estimated by fitting a rigid transform
414    /// (rotation + translation, no scale) ref→measured.
415    ///
416    /// Returns `None` when the correspondence between kept reference and
417    /// measured points is ambiguous within `tolerance`.
418    pub fn predict_missing(
419        &self,
420        kept_refs: &RefIndices,
421        measured: PointSet<'_>,
422        missing: RefIndex,
423    ) -> Option<Vec2<Millimeters>> {
424        let ref_points = self.ref_points.as_ref()?;
425        if missing.as_usize() >= self.num_points
426            || kept_refs.len() != measured.len()
427            || kept_refs.len() < 2
428            || kept_refs.iter().any(|i| i.as_usize() >= self.num_points)
429        {
430            return None;
431        }
432        let kept_ref_points = kept_refs.points_of(ref_points);
433        // Try every correspondence consistent with the distance
434        // multisets (symmetric figures admit several) and keep the
435        // transform with the smallest residual.
436        let mut best: Option<(RigidTransform, Millimeters)> = None;
437        for mapping in kept_ref_points.correspondences(measured, self.tolerance) {
438            let Some(transform) =
439                kept_ref_points.similarity_transform(measured, mapping.as_combination())
440            else {
441                continue;
442            };
443            let residual = Millimeters::new(
444                transform
445                    .residual(&kept_ref_points, measured, mapping.as_combination())
446                    .sqrt(),
447            );
448            if best.as_ref().is_none_or(|(_, r)| residual < *r) {
449                best = Some((transform, residual));
450            }
451        }
452        best.map(|(transform, _)| transform.apply(ref_points[missing.as_usize()]))
453    }

519        // Fingerprint [50, 50, ~70.7]; indices 1–2 generous, index 0 strict.
520        let mut pattern = TangiblePattern::from_points(1, "triangle", (&triangle()).into(), 1.0);
521        pattern.tolerances = vec![1.0.into(), 20.0.into(), 20.0.into()].into();
522        // Stretched third contact: deviations [0, ~15, ~11] - the shortest
