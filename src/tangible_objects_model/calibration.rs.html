13/// Device calibration: converts raw device coordinates into millimeters.
14///
15/// `x_res`/`y_res` are the `resolution` fields of `ABS_MT_POSITION_X/Y`
16/// (units per millimeter) read from evdev `ABS_INFO` or an `evtest` header.
17#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
18pub struct Calibration {
19    /// Device units per millimeter on the X axis.
20    pub x_res: UnitsPerMm,
21    /// Device units per millimeter on the Y axis.
22    pub y_res: UnitsPerMm,
23    /// Minimum raw X value reported by the device.
24    pub x_min: RawAxisValue,
25    /// Maximum raw X value reported by the device.
26    pub x_max: RawAxisValue,
27    /// Minimum raw Y value reported by the device.
28    pub y_min: RawAxisValue,
29    /// Maximum raw Y value reported by the device.
30    pub y_max: RawAxisValue,
31}
32
33impl Calibration {
34    /// Creates a calibration without bounds information.
35    pub fn new(x_res: impl Into<UnitsPerMm>, y_res: impl Into<UnitsPerMm>) -> Self {
36        Self {
37            x_res: x_res.into(),
38            y_res: y_res.into(),
39            x_min: RawAxisValue::new(0.0),
40            x_max: RawAxisValue::new(0.0),
41            y_min: RawAxisValue::new(0.0),
42            y_max: RawAxisValue::new(0.0),
43        }
44    }
45
46    /// Creates a calibration including the device's raw axis bounds.
47    pub fn with_bounds(
48        x_res: impl Into<UnitsPerMm>,
49        y_res: impl Into<UnitsPerMm>,
50        x_min: impl Into<RawAxisValue>,
51        x_max: impl Into<RawAxisValue>,
52        y_min: impl Into<RawAxisValue>,
53        y_max: impl Into<RawAxisValue>,
54    ) -> Self {
55        Self {
56            x_res: x_res.into(),
57            y_res: y_res.into(),
58            x_min: x_min.into(),
59            x_max: x_max.into(),
60            y_min: y_min.into(),
61            y_max: y_max.into(),
62        }
63    }
64
65    /// Converts a raw device coordinate into millimeters.
66    pub fn to_mm(&self, raw: Vec2<RawAxisValue>) -> Vec2<Millimeters> {
67        Vec2::new(raw.x / self.x_res.as_f64(), raw.y / self.y_res.as_f64())
68    }
69
70    /// Physical width of the touch surface in millimeters.
71    pub fn width_mm(&self) -> Millimeters {
72        if self.x_res.as_f64() > 0.0 {
73            Millimeters::new((self.x_max - self.x_min) / self.x_res)
74        } else {
75            Millimeters::new(0.0)
76        }
77    }
78
79    /// Physical height of the touch surface in millimeters.
80    pub fn height_mm(&self) -> Millimeters {
81        if self.y_res.as_f64() > 0.0 {
82            Millimeters::new((self.y_max - self.y_min) / self.y_res)
83        } else {
84            Millimeters::new(0.0)
85        }
86    }
87
88    /// The whole usable surface as a [`crate::Region`] in millimeters -
89    /// the bounds edge-ghost and zone filtering refer to.
90    pub fn surface_region(&self) -> Region {
91        Region {
92            x_min: Millimeters::new(self.x_min.as_f64() / self.x_res.as_f64()),
93            y_min: Millimeters::new(self.y_min.as_f64() / self.y_res.as_f64()),
94            x_max: Millimeters::new(self.x_max.as_f64() / self.x_res.as_f64()),
95            y_max: Millimeters::new(self.y_max.as_f64() / self.y_res.as_f64()),
96        }
97    }
98
99    /// The [`SurfaceFit`] mapping this calibrated surface onto a
100    /// `screen_w` × `screen_h` display area under `rotation` - the
101    /// shared mm↔px transform renderers *and* hit-tests use (the fit's
102    /// `unmap` is the pixel→millimeter direction card hit-testing and
103    /// safe-area footprints need).
104    pub fn surface_fit(&self, area: Vec2<Pixels>, rotation: DisplayRotation) -> SurfaceFit {
105        SurfaceFit::new(
106            Vec2::from_mm(self.width_mm(), self.height_mm()),
107            area,
108            rotation,
109        )
110    }
111
112    /// Maps a millimeter coordinate onto screen pixels.
113    ///
114    /// `rotation` first turns the position into the displayed orientation
115    /// (the evdev stream always speaks the panel's native frame); the
116    /// result is then scaled *uniformly* - the largest factor fitting the
117    /// surface into the display `area`, letterboxed centered, so figures
118    /// keep their physical aspect instead of stretching when panel and
119    /// screen aspects differ. Returns `(0.0, 0.0)` when no
120    /// bounds/resolution are available.
121    pub fn mm_to_screen(
122        &self,
123        pos: Vec2<Millimeters>,
124        area: Vec2<Pixels>,
125        rotation: DisplayRotation,
126    ) -> Vec2<Pixels> {
127        if self.width_mm().as_f64() <= 0.0 || self.height_mm().as_f64() <= 0.0 {
128            return Vec2::ZERO;
129        }
130        let local =
131            pos - Vec2::<Millimeters>::new(self.x_min / self.x_res, self.y_min / self.y_res);
132        self.surface_fit(area, rotation).map(local)
133    }
134}

136#[cfg(test)]
137mod tests {
138    use super::Calibration;
139    use crate::RawAxisValue;
140    use crate::Vec2;
141
142    #[test]
143    fn raw_coordinate_converts_to_mm() {
144        let cal = Calibration::new(4.0, 2.0);
145        let mm = cal.to_mm(Vec2::<RawAxisValue>::new(40.0, 20.0));
146        assert!((mm.x - 10.0).abs() < 1e-9);
147        assert!((mm.y - 10.0).abs() < 1e-9);
148    }
149
150    #[test]
151    fn bounds_yield_physical_dimensions() {
152        let cal = Calibration::with_bounds(10.0, 10.0, 0.0, 2000.0, 0.0, 1000.0);
153        assert!((cal.width_mm().as_f64() - 200.0).abs() < 1e-9);
154        assert!((cal.height_mm().as_f64() - 100.0).abs() < 1e-9);
155    }
156
157    #[test]
158    fn missing_resolution_yields_zero_dimensions() {
159        let cal = Calibration::new(0.0, 0.0);
160        assert_eq!(cal.width_mm().as_f64(), 0.0);
161        assert_eq!(cal.height_mm().as_f64(), 0.0);
162    }
