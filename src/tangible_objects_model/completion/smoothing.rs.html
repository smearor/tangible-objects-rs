9/// Per-slot position smoothing factor (TRACKING_STABILIZATION.md): EMA
10/// weight of the current raw position (`pos = alpha*raw +
11/// (1-alpha)*prev`). Averages out the differential-latency distortion
12/// panels report during motion. `0.0` disables smoothing, `1.0` uses
13/// raw positions.
14#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
15#[serde(transparent)]
16pub struct SmoothingAlpha(f64);
17
18impl SmoothingAlpha {
19    /// Creates a smoothing factor.
20    pub const fn new(alpha: f64) -> Self {
21        Self(alpha)
22    }
23
24    /// The EMA weight of the raw position.
25    pub fn as_f64(&self) -> f64 {
26        self.0
27    }
28
29    /// Whether running the smoothing algorithm changes anything -
30    /// `0.0` freezes the position (disabled), `1.0` passes raw
31    /// positions through (identity); only `0 < alpha < 1` blends.
32    pub fn is_effective(&self) -> bool {
33        self.0 > 0.0 && self.0 < 1.0
34    }
35
36    /// The complementary weight `1 − alpha` - the previous value's
37    /// share of an EMA blend (`pos * alpha + prev * alpha.complement()`).
38    pub fn complement(self) -> Self {
39        Self(1.0 - self.0)
40    }
41}
42
43impl<U> Mul<SmoothingAlpha> for Vec2<U> {
44    type Output = Vec2<U>;
45
46    /// Scales the vector by the weight (`v * alpha`).
47    fn mul(self, alpha: SmoothingAlpha) -> Vec2<U> {
48        self * alpha.as_f64()
49    }
50}

52impl Default for SmoothingAlpha {
53    fn default() -> Self {
54        Self(0.6)
55    }
56}

58impl From<f64> for SmoothingAlpha {

62}

64impl Display for SmoothingAlpha {

66        write!(f, "{}", self.0)
67    }
68}
