29// Under `cynic` the derive owns serde (SCREAMING_SNAKE wire form);
30// without it serde keeps the snake_case form used by replay CLIs.
31pub enum AnalysisMode {
32    /// Fold the recorded generic candidates - fast, but blind to frames
33    /// whose contacts ran through the incumbent track path.
34    #[default]
35    Candidates,
36    /// Re-evaluate every same-size contact subset of each frame against
37    /// the pattern fingerprints - covers incumbent-tracked frames.
38    Contacts,
39}
40
41impl FromStr for AnalysisMode {
42    type Err = String;
43
44    /// Parses `"candidates"` or `"contacts"`.
45    fn from_str(value: &str) -> Result<Self, Self::Err> {
46        match value {
47            "candidates" => Ok(Self::Candidates),
48            "contacts" => Ok(Self::Contacts),
49            other => Err(format!("unknown analysis mode: {other}")),
50        }
51    }
52}
53
54impl Display for AnalysisMode {
55    /// Writes `"candidates"` or `"contacts"`.
56    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
57        f.write_str(match self {
58            Self::Candidates => "candidates",
59            Self::Contacts => "contacts",
60        })
61    }
62}
