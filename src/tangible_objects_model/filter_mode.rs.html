9/// Suppression policy of the filter proxy (FILTER_PROXY_VIRTUAL.md).
10///
11/// Controls when a slot classified as a tangible becomes invisible to the
12/// operating system and how unclassified new contacts are treated.
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14#[cfg_attr(not(feature = "cynic"), derive(Serialize, Deserialize))]
15#[cfg_attr(feature = "async-graphql", derive(async_graphql::Enum))]
16#[cfg_attr(feature = "cynic", derive(cynic::Enum))]
17#[cfg_attr(
18    feature = "cynic",
19    cynic(
20        schema_path = "../graphql/schema.graphql",
21        rename_all = "SCREAMING_SNAKE_CASE"
22    )
23)]
24// Under `cynic` the derive owns serde (SCREAMING_SNAKE wire form
25// matching the schema); without it, serde serializes the Rust variant
26// names (legacy wire form used by recordings and snapshots).
27pub enum FilterMode {
28    /// Proxy disabled: every event passes through unchanged.
29    Off,
30    /// Suppress slots on the first matched frame. Zero leak for real
31    /// tangibles, but a one-frame phantom match (fingers coincidentally
32    /// forming a profile) eats the contact for the full hysteresis window.
33    Immediate,
34    /// Suppress only after `STABILITY_GATE` consecutive matched frames -
35    /// phantom matches never suppress; a tangible leaks ~1 frame on
36    /// placement.
37    Gated,
38    /// Hold every new contact's events until the first frame evaluation
39    /// classifies it: tangible contacts never reach the system at all,
40    /// finger contacts appear ~1 frame late but complete. Safest policy.
41    Guarded,
42    /// Suppress every contact - tangible and finger alike. Tangibles are
43    /// still detected internally (engine, interfaces, tracked poses), but
44    /// the virtual device emits no touch events at all. For setups where
45    /// only figure state matters and touch input must not disturb the UI.
46    TangiblesOnly,
47}
48
49impl Default for FilterMode {
50    /// The default policy balances tangible invisibility against finger
51    /// latency.
52    fn default() -> Self {
53        Self::Guarded
54    }
55}
56
57impl std::fmt::Display for FilterMode {
58    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
59        let name = match self {
60            Self::Off => "off",
61            Self::Immediate => "immediate",
62            Self::Gated => "gated",
63            Self::Guarded => "guarded",
64            Self::TangiblesOnly => "tangibles-only",
65        };
66        write!(f, "{name}")
67    }
68}
69
70impl std::str::FromStr for FilterMode {
71    type Err = FilterModeParseError;
72
73    fn from_str(value: &str) -> Result<Self, Self::Err> {
74        match value {
75            "off" => Ok(Self::Off),
76            "immediate" => Ok(Self::Immediate),
77            "gated" => Ok(Self::Gated),
78            "guarded" => Ok(Self::Guarded),
79            "tangibles-only" | "tangibles_only" => Ok(Self::TangiblesOnly),
80            _ => Err(FilterModeParseError::new(value)),
81        }
82    }
83}
