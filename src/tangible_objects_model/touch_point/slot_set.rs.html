8/// A set of slot identities - which slots a stage binds, emits,
9/// suppresses or matched this frame.
10///
11/// The collection counterpart of [`TouchPointSlot`]: `matched_slots` /
12/// `slots_of` of a match, a stage's `suppressed`/`emitted` mask, or a
13/// tracker's bound-slot map. The parallel of [`crate::RefIndices`]
14/// for slot identities.
15#[derive(Debug, Clone, PartialEq, Eq, Default)]
16pub struct SlotSet(pub HashSet<TouchPointSlot>);
17
18impl FromIterator<TouchPointSlot> for SlotSet {
19    /// Collecting slots yields a slot set.
20    fn from_iter<T: IntoIterator<Item = TouchPointSlot>>(iter: T) -> Self {
21        Self(iter.into_iter().collect())
22    }
23}
24
25impl From<HashSet<TouchPointSlot>> for SlotSet {
26    /// A hash set IS a slot set.
27    fn from(slots: HashSet<TouchPointSlot>) -> Self {
28        Self(slots)
29    }
30}
31
32impl<const N: usize> From<[TouchPointSlot; N]> for SlotSet {
33    /// An array of slots IS a slot set - the `HashSet::from` idiom.
34    fn from(slots: [TouchPointSlot; N]) -> Self {
35        slots.into_iter().collect()
36    }
37}
38
39impl Deref for SlotSet {
40    type Target = HashSet<TouchPointSlot>;
41
42    /// Slot sets deref to the underlying `HashSet` - `contains`,
43    /// `len`, `iter`, `extend` stay available.
44    fn deref(&self) -> &HashSet<TouchPointSlot> {
45        &self.0
46    }
47}
48
49impl DerefMut for SlotSet {
50    /// Mutable access to the underlying `HashSet`.
51    fn deref_mut(&mut self) -> &mut HashSet<TouchPointSlot> {
52        &mut self.0
53    }
54}

56impl<'a> IntoIterator for &'a SlotSet {
57    type Item = &'a TouchPointSlot;
58    type IntoIter = Iter<'a, TouchPointSlot>;
59
60    /// Iterating a slot set is iterating its `HashSet`.
61    fn into_iter(self) -> Self::IntoIter {
62        self.0.iter()
63    }
