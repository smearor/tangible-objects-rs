14/// A physical place at the tabletop inside a [`crate::SeatSession`]
15/// (TANGIBLE_OBJECTS_TABLE_TOP_SEAT.md).
16///
17/// Marker-backed seats (`marker: Some`) anchor to the tracked tangible
18/// matching that profile - `pose` is re-derived from it every frame
19/// and `tangible` carries the bound instance. Virtual seats
20/// (`marker: None`) keep the application-provided pose. `state`
21/// reports the anchor's health: an `Absent` seat retains its last
22/// pose until the marker returns or the application removes the seat.
23///
24/// A seat is a *place*, not a person: who sits there - display name,
25/// player color, rights - stays in the application's domain.
26#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
27#[serde(rename_all = "camelCase")]
28#[cfg_attr(feature = "async-graphql", derive(async_graphql::SimpleObject))]
29#[cfg_attr(feature = "mcp", derive(schemars::JsonSchema))]
30pub struct TableTopSeat {
31    /// Stable seat ID - unique across sessions, never reused.
32    pub id: SeatId,
33    /// Application-facing seat name, unique within its session.
34    pub name: SeatName,
35    /// Session this seat belongs to.
36    pub session: SeatSessionName,
37    /// Application owning the session - mirrors `session.owner` so the
38    /// Seats card and event payloads carry it without a join.
39    pub owner: Owner,
40    /// Marker profile anchoring this seat - the figure profile whose
41    /// tracked instances drive `pose`. `None` = virtual seat.
42    pub marker: Option<PatternId>,
43    /// Currently bound marker tangible instance - `None` while the
44    /// marker is absent or for virtual seats.
45    pub tangible: Option<TangibleId>,
46    /// Organizational plus radial pose: table side, edge offset and
47    /// free-angle orientation. Marker-derived while anchored,
48    /// application-provided while virtual, last anchored pose while
49    /// absent.
50    #[serde(with = "pose_serde")]
51    #[cfg_attr(feature = "mcp", schemars(with = "SeatPose"))]
52    pub pose: SeatPose,
53    /// Marker anchor state.
54    pub state: SeatMarkerState,
55}

57/// Serde `with` helpers for [`TableTopSeat::pose`]: the GraphQL wire
58/// spells enum variants SCREAMING_SNAKE (`BOTTOM`) while `TableSide`'s
59/// canonical serde form is lowercase (`bottom`). Decoding through a
60/// wire mirror accepts both spellings; serialization keeps the
61/// canonical lowercase form `SeatPose` already emits.
62mod pose_serde {
63    use serde::Deserialize;
64    use serde::Deserializer;
65    use serde::Serialize;
66    use serde::Serializer;
67    use smearor_commons_geometry::Fraction;
68    use smearor_commons_geometry::RotationRad;
69    use smearor_commons_layout::Seat;
70    use smearor_commons_layout::SeatPose;
71    use smearor_commons_layout::TableSide;
72
73    /// `Seat` wire mirror - `side` decodes through [`WireSide`].
74    #[derive(Deserialize)]
75    struct WireSeat {
76        /// Table edge.
77        side: WireSide,
78        /// Position along the edge axis.
79        offset: Option<Fraction>,
80    }
81
82    /// `SeatPose` wire mirror.
83    #[derive(Deserialize)]
84    struct WirePose {
85        /// Organizational alignment.
86        seat: WireSeat,
87        /// Radial orientation.
88        orientation: RotationRad,
89    }
90
91    /// `TableSide` accepting its GraphQL (`BOTTOM`) and canonical
92    /// serde (`bottom`) spellings.
93    #[derive(Deserialize)]
94    #[serde(rename_all = "lowercase")]
95    enum WireSide {
96        /// Bottom edge.
97        #[serde(alias = "BOTTOM")]
98        Bottom,
99        /// Left edge.
100        #[serde(alias = "LEFT")]
101        Left,
102        /// Top edge.
103        #[serde(alias = "TOP")]
104        Top,
105        /// Right edge.
106        #[serde(alias = "RIGHT")]
107        Right,
108    }
109
110    impl From<WireSide> for TableSide {
111        fn from(side: WireSide) -> Self {
112            match side {
113                WireSide::Bottom => Self::Bottom,
114                WireSide::Left => Self::Left,
115                WireSide::Top => Self::Top,
116                WireSide::Right => Self::Right,
117            }
118        }
119    }
120
121    /// Serializes through `SeatPose`'s own impl - canonical lowercase.
122    pub(super) fn serialize<S: Serializer>(
123        pose: &SeatPose,
124        serializer: S,
125    ) -> Result<S::Ok, S::Error> {
126        pose.serialize(serializer)
127    }
128
129    /// Deserializes through the wire mirror - accepts `BOTTOM` and
130    /// `bottom` spellings of `TableSide`.
131    pub(super) fn deserialize<'de, D: Deserializer<'de>>(
132        deserializer: D,
133    ) -> Result<SeatPose, D::Error> {
134        let wire = WirePose::deserialize(deserializer)?;
135        Ok(SeatPose::new(
136            Seat {
137                side: wire.seat.side.into(),
138                offset: wire.seat.offset,
139            },
140            wire.orientation,
141        ))
142    }
143}

151    /// Selects the eight schema fields - `pose` expands into its
152    /// nested `seat { side offset }` and `orientation` leaves, decoded
153    /// by `SeatPose`'s serde impl (the smearor type cannot carry a
154    /// cynic derive).
155    fn query(
156        builder: cynic::queries::SelectionBuilder<'_, Self::SchemaType, Self::VariablesFields>,
157    ) {
158        let mut builder = builder;
159        let _ = builder.select_field::<schema::__fields::TableTopSeat::id, schema::SeatId>();
160        let _ = builder.select_field::<schema::__fields::TableTopSeat::name, schema::SeatName>();
161        let _ = builder
162            .select_field::<schema::__fields::TableTopSeat::session, schema::SeatSessionName>();
163        let _ = builder.select_field::<schema::__fields::TableTopSeat::owner, schema::Owner>();
164        let _ = builder
165            .select_field::<schema::__fields::TableTopSeat::marker, Option<schema::PatternId>>();
166        let _ = builder
167            .select_field::<schema::__fields::TableTopSeat::tangible, Option<schema::TangibleId>>();
168        let mut pose =
169            builder.select_field::<schema::__fields::TableTopSeat::pose, schema::SeatPose>();
170        let mut pose_fields = pose.select_children::<()>();
171        let mut seat = pose_fields.select_field::<schema::__fields::SeatPose::seat, schema::Seat>();
172        let mut seat_fields = seat.select_children::<()>();
173        let _ = seat_fields.select_field::<schema::__fields::Seat::side, schema::TableSide>();
174        let _ =
175            seat_fields.select_field::<schema::__fields::Seat::offset, Option<schema::Fraction>>();
176        let _ = pose_fields
177            .select_field::<schema::__fields::SeatPose::orientation, schema::RotationRad>();
178        let _ = builder
179            .select_field::<schema::__fields::TableTopSeat::state, schema::SeatMarkerState>();
180    }
