14/// A physical place at the tabletop inside a [`crate::SeatSession`]
15/// (TANGIBLE_OBJECTS_TABLE_TOP_SEAT.md).
16///
17/// Marker-backed seats (`marker: Some`) anchor to the tracked tangible
18/// matching that profile - `pose` is re-derived from it every frame
19/// and `tangible` carries the bound instance. Virtual seats
20/// (`marker: None`) keep the application-provided pose. `state`
21/// reports the anchor's health: an `Absent` seat retains its last
22/// pose until the marker returns or the application removes the seat.
23///
24/// A seat is a *place*, not a person: who sits there - display name,
25/// player color, rights - stays in the application's domain.
26#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
27#[serde(rename_all = "camelCase")]
28#[cfg_attr(feature = "async-graphql", derive(async_graphql::SimpleObject))]
29#[cfg_attr(feature = "mcp", derive(schemars::JsonSchema))]
30pub struct TableTopSeat {
31    /// Stable seat ID - unique across sessions, never reused.
32    pub id: SeatId,
33    /// Application-facing seat name, unique within its session.
34    pub name: SeatName,
35    /// Session this seat belongs to.
36    pub session: SeatSessionName,
37    /// Application owning the session - mirrors `session.owner` so the
38    /// Seats card and event payloads carry it without a join.
39    pub owner: Owner,
40    /// Marker profile anchoring this seat - the figure profile whose
41    /// tracked instances drive `pose`. `None` = virtual seat.
42    pub marker: Option<PatternId>,
43    /// Currently bound marker tangible instance - `None` while the
44    /// marker is absent or for virtual seats.
45    pub tangible: Option<TangibleId>,
46    /// Organizational plus radial pose: table side, edge offset and
47    /// free-angle orientation. Marker-derived while anchored,
48    /// application-provided while virtual, last anchored pose while
49    /// absent.
50    pub pose: SeatPose,
51    /// Marker anchor state.
52    pub state: SeatMarkerState,
53}

56impl cynic::QueryFragment for TableTopSeat {
57    type SchemaType = schema::TableTopSeat;
58    type VariablesFields = ();
59    const TYPE: Option<&'static str> = Some("TableTopSeat");
60
61    /// Selects the eight schema fields - `pose` expands into its
62    /// nested `seat { side offset }` and `orientation` leaves, decoded
63    /// by `SeatPose`'s serde impl (the smearor type cannot carry a
64    /// cynic derive).
65    fn query(
66        builder: cynic::queries::SelectionBuilder<'_, Self::SchemaType, Self::VariablesFields>,
67    ) {
68        let mut builder = builder;
69        let _ = builder.select_field::<schema::__fields::TableTopSeat::id, schema::SeatId>();
70        let _ = builder.select_field::<schema::__fields::TableTopSeat::name, schema::SeatName>();
71        let _ = builder
72            .select_field::<schema::__fields::TableTopSeat::session, schema::SeatSessionName>();
73        let _ = builder.select_field::<schema::__fields::TableTopSeat::owner, schema::Owner>();
74        let _ = builder
75            .select_field::<schema::__fields::TableTopSeat::marker, Option<schema::PatternId>>();
76        let _ = builder
77            .select_field::<schema::__fields::TableTopSeat::tangible, Option<schema::TangibleId>>();
78        let mut pose =
79            builder.select_field::<schema::__fields::TableTopSeat::pose, schema::SeatPose>();
80        let mut pose_fields = pose.select_children::<()>();
81        let mut seat = pose_fields.select_field::<schema::__fields::SeatPose::seat, schema::Seat>();
82        let mut seat_fields = seat.select_children::<()>();
83        let _ = seat_fields.select_field::<schema::__fields::Seat::side, schema::TableSide>();
84        let _ =
85            seat_fields.select_field::<schema::__fields::Seat::offset, Option<schema::Fraction>>();
86        let _ = pose_fields
87            .select_field::<schema::__fields::SeatPose::orientation, schema::RotationRad>();
88        let _ = builder
89            .select_field::<schema::__fields::TableTopSeat::state, schema::SeatMarkerState>();
90    }
