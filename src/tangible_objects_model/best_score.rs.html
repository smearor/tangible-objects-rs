1use crate::MatchScore;
2
3/// The best-scoring candidate seen so far during an enumeration -
4/// `None` until the first candidate is considered.
5///
6/// Score-based ranking (TRAINING.md 3.2): the lowest score wins, so a
7/// later candidate replaces the incumbent only when it strictly
8/// improves. `S` is the ranking measure - `MatchScore` for
9/// tolerance-normalized candidate selection (the matcher's
10/// `PatternMatch` ranking), a raw `f64` residual for transform fits.
11/// Shared by the engine's match selection and the deviation analysis'
12/// best-subset evaluation.
13#[derive(Debug)]
14pub struct BestScore<T, S = MatchScore>(Option<(T, S)>);
15
16impl<T, S> Default for BestScore<T, S> {
17    /// Starts with no best - the first `consider` always wins.
18    fn default() -> Self {
19        Self(None)
20    }
21}
22
23impl<T, S: PartialOrd> BestScore<T, S> {
24    /// Keeps `value` when it scores strictly better than the current
25    /// best.
26    pub fn consider(&mut self, value: T, score: S) {
27        let better = self
28            .0
29            .as_ref()
30            .is_none_or(|(_, best_score)| score < *best_score);
31        if better {
32            self.0 = Some((value, score));
33        }
34    }
35
36    /// Whether no candidate has been considered yet.
37    pub fn is_none(&self) -> bool {
38        self.0.is_none()
39    }
40
41    /// Consumes the selection and returns the winning value with its
42    /// score.
43    pub fn into_inner(self) -> Option<(T, S)> {
44        self.0
45    }
46
47    /// Consumes the selection and returns the winning value.
48    pub fn into_value(self) -> Option<T> {
49        self.0.map(|(value, _)| value)
50    }
51}
