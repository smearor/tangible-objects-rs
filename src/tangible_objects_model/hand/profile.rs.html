19/// A named hand constellation - serialized from
20/// `hands.d/*.toml` like [`crate::Spell`]:
21///
22/// ```toml
23/// name = "alice"
24/// tolerance = 8.0
25/// points = [[100.0, 300.0], [160.0, 280.0], [210.0, 320.0], [255.0, 340.0], [300.0, 310.0]]
26/// ```
27///
28/// Matching reuses the profile fingerprint: the sorted pairwise
29/// distances of the observed tap must deviate at most `tolerance`
30/// per index - rotation- and translation-invariant like every
31/// `sorted_sides` comparison.
32#[derive(Debug, Clone, Serialize, Deserialize)]
33pub struct HandProfile {
34    /// Identity - addresses the handprint for unregister and the
35    /// store, and names the binding `five-finger-tap` arming
36    /// requires.
37    pub name: HandName,
38    /// Matching tolerance per pairwise distance in millimeters.
39    pub tolerance: Millimeters,
40    /// Sorted pairwise distances of `points` - the fingerprint a
41    /// tap's measured fingerprint is compared index-by-index
42    /// against (same metric as `ProfileFile::sorted_sides`).
43    pub sorted_sides: Distances,
44    /// The recorded contact constellation in surface millimeters.
45    #[serde(with = "smearor_commons_geometry::serde_points")]
46    pub points: Vec<Vec2<Millimeters>>,
47}
48
49impl HandProfile {
50    /// Creates a handprint from measured contact positions -
51    /// computes the sorted pairwise-distance fingerprint like
52    /// [`crate::ProfileFile::from_points`].
53    pub fn from_points(
54        name: impl Into<HandName>,
55        points: &[Vec2<Millimeters>],
56        tolerance: Millimeters,
57    ) -> Self {
58        Self {
59            name: name.into(),
60            tolerance,
61            sorted_sides: PointSet::new(points).distances(),
62            points: points.to_vec(),
63        }
64    }
65
66    /// Whether `constellation` is this hand's constellation: same
67    /// point count and every pairwise distance within `tolerance`
68    /// of the fingerprint (per-index, like the pattern matcher).
69    pub fn matches(&self, constellation: PointSet<'_>) -> bool {
70        constellation
71            .distances()
72            .max_deviation(&self.sorted_sides)
73            .as_millimeters()
74            <= self.tolerance
75    }
76
77    /// Loads a single handprint `.toml` file.
78    pub fn load(path: &Path) -> std::io::Result<Self> {
79        let content = std::fs::read_to_string(path)?;
80        toml::from_str(&content)
81            .map_err(|error| std::io::Error::new(std::io::ErrorKind::InvalidData, error))
82    }
83
84    /// Loads all `.toml` handprint files from a directory, sorted
85    /// by filename. Unreadable or invalid files are skipped and
86    /// returned in the error list.
87    pub fn load_dir(dir: &Path) -> (Vec<Self>, Vec<(PathBuf, std::io::Error)>) {
88        let mut hands = Vec::new();
89        let mut errors = Vec::new();
90        let Ok(entries) = std::fs::read_dir(dir) else {
91            return (hands, errors);
92        };
93        let mut files: Vec<PathBuf> = entries
94            .filter_map(|entry| entry.ok())
95            .map(|entry| entry.path())
96            .filter(|path| path.extension().is_some_and(|ext| ext == "toml"))
97            .collect();
98        files.sort();
99        for file in files {
100            match Self::load(&file) {
101                Ok(hand) => hands.push(hand),
102                Err(error) => errors.push((file, error)),
103            }
104        }
105        (hands, errors)
106    }
107
108    /// Saves the handprint as a TOML file (runtime-store
109    /// persistence, `hands.d`).
110    pub fn save(&self, path: &Path) -> std::io::Result<()> {
111        let content = toml::to_string_pretty(self)
112            .map_err(|error| std::io::Error::new(std::io::ErrorKind::InvalidData, error))?;
113        std::fs::write(path, content)
114    }
115}

117#[cfg(test)]
118mod tests {
119    use super::HandProfile;
120    use smearor_commons_geometry::Millimeters;
121    use smearor_commons_geometry::Vec2;
122
123    fn hand_points() -> Vec<Vec2<Millimeters>> {
124        // A splayed five-finger constellation (~real hand span).
125        vec![
126            Vec2::new(0.0, 40.0),
127            Vec2::new(35.0, 15.0),
128            Vec2::new(65.0, 0.0),
129            Vec2::new(95.0, 20.0),
130            Vec2::new(110.0, 55.0),
131        ]
132    }
133
134    #[test]
135    fn same_constellation_matches_shifted() {
136        let hand = HandProfile::from_points("alice", &hand_points(), Millimeters::new(8.0));
137        // Translated + slightly jittered - still Alice's hand.
138        let shifted: Vec<Vec2<Millimeters>> = hand_points()
139            .iter()
140            .map(|point| *point + Vec2::new(200.0, 100.0))
141            .collect();
142        assert!(hand.matches((&shifted).into()));
143    }
144
145    #[test]
146    fn different_constellation_rejects() {
147        let hand = HandProfile::from_points("alice", &hand_points(), Millimeters::new(8.0));
148        // A cramped five-contact cluster is a different hand.
149        let tiny: Vec<Vec2<Millimeters>> = hand_points()
150            .iter()
151            .map(|point| Vec2::new(point.x * 0.4, point.y * 0.4))
152            .collect();
153        assert!(!hand.matches((&tiny).into()));
154    }
155
156    #[test]
157    fn wrong_point_count_rejects() {
158        let hand = HandProfile::from_points("alice", &hand_points(), Millimeters::new(8.0));
159        assert!(!hand.matches((&hand_points()[..4]).into()));
160    }
161
162    #[test]
163    fn toml_round_trip() -> Result<(), Box<dyn std::error::Error>> {
164        let hand = HandProfile::from_points("alice", &hand_points(), Millimeters::new(8.0));
165        let serialized = toml::to_string_pretty(&hand)?;
166        let loaded: HandProfile = toml::from_str(&serialized)?;
167        assert_eq!(loaded.name.as_str(), "alice");
168        assert_eq!(loaded.points.len(), 5);
169        assert_eq!(loaded.sorted_sides, hand.sorted_sides);
170        Ok(())
171    }
