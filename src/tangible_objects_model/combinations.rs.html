6/// The materialized set of all candidate combinations - every
7/// k-element index subset of an index range.
8///
9/// Matching enumerates one [`crate::Combination`] per contained
10/// [`CombinationBuf`]; the inner vectors are candidates, the outer
11/// collection is the enumeration itself.
12#[derive(Debug, Clone, PartialEq, Eq, Default)]
13pub struct Combinations(pub Vec<CombinationBuf>);
14
15impl Combinations {
16    /// An empty combination set.
17    pub fn new() -> Self {
18        Self::default()
19    }
20    /// Number of combinations in the set - C(n, k).
21    pub fn len(&self) -> usize {
22        self.0.len()
23    }
24
25    /// Whether the set is empty (n < k or empty input).
26    pub fn is_empty(&self) -> bool {
27        self.0.is_empty()
28    }
29
30    /// The single empty combination - the C(n, 0) base case of
31    /// combination enumeration.
32    pub fn one_empty() -> Self {
33        Self(vec![CombinationBuf(Vec::new())])
34    }
35}
36
37impl From<Vec<CombinationBuf>> for Combinations {
38    /// A combination list IS a combinations set.
39    fn from(combinations: Vec<CombinationBuf>) -> Self {
40        Self(combinations)
41    }
42}
43
44impl FromIterator<Vec<usize>> for Combinations {
45    /// Collecting index vectors yields the combinations set.
46    fn from_iter<T: IntoIterator<Item = Vec<usize>>>(iter: T) -> Self {
47        Self(iter.into_iter().map(CombinationBuf).collect())
48    }
49}

51impl IntoIterator for Combinations {
52    type Item = CombinationBuf;
53    type IntoIter = std::vec::IntoIter<CombinationBuf>;
54
55    /// Iterating enumerates the owned combinations.
56    fn into_iter(self) -> Self::IntoIter {
57        self.0.into_iter()
58    }
59}

61impl<'a> IntoIterator for &'a Combinations {
62    type Item = &'a CombinationBuf;
63    type IntoIter = std::slice::Iter<'a, CombinationBuf>;
64
65    /// Borrowed iteration over the combinations.
66    fn into_iter(self) -> Self::IntoIter {

68    }

71impl Deref for Combinations {

74    /// Combination sets view their vector transparently.
