16/// Runtime store for `Lifetime::Persistent` registrations:
17/// `~/.config/tangible-objects/profiles.d/` resp. `fields.d/`.
18pub(crate) fn runtime_store_dir(kind: &str) -> Option<PathBuf> {
19    completion_config_path()
20        .and_then(|path| path.parent().map(Path::to_path_buf))
21        .map(|base| base.join(kind))
22}

24/// Persists a registered profile/field set into the runtime store.
25pub(crate) fn persist_registration(registration: &Registration) {
26    let Some(store) = runtime_store_dir(match registration {
27        Registration::Pattern(_) => "profiles.d",
28        Registration::FieldSet(_) => "fields.d",
29        Registration::DeadZone(_) => "dead_zones.d",
30        Registration::Spell(_) => "spells.d",
31        Registration::HandProfile(_) => "hands.d",
32    }) else {
33        return;
34    };
35    let (name, result) = match registration {
36        Registration::Pattern(profile) => (
37            profile.name.as_str().to_string(),
38            std::fs::create_dir_all(&store).and_then(|_| {
39                profile
40                    .save(&store.join(format!("{}.toml", profile.name)))
41                    .map_err(std::io::Error::other)
42            }),
43        ),
44        Registration::FieldSet(set) => (
45            set.name.as_str().to_string(),
46            std::fs::create_dir_all(&store).and_then(|_| {
47                std::fs::write(
48                    store.join(format!("{}.toml", set.name)),
49                    toml::to_string_pretty(set).unwrap_or_default(),
50                )
51            }),
52        ),
53        Registration::DeadZone(zone) => (
54            zone.name.as_str().to_string(),
55            std::fs::create_dir_all(&store).and_then(|_| {
56                std::fs::write(
57                    store.join(format!("{}.toml", zone.name)),
58                    toml::to_string_pretty(zone).unwrap_or_default(),
59                )
60            }),
61        ),
62        Registration::Spell(spell) => (
63            spell.name.as_str().to_string(),
64            std::fs::create_dir_all(&store)
65                .and_then(|_| spell.save(&store.join(format!("{}.toml", spell.name)))),
66        ),
67        Registration::HandProfile(hand) => (
68            hand.name.as_str().to_string(),
69            std::fs::create_dir_all(&store)
70                .and_then(|_| hand.save(&store.join(format!("{}.toml", hand.name)))),
71        ),
72    };
73    match result {
74        Ok(()) => info!(name, "registration persisted"),
75        Err(e) => warn!(error = %e, name, "persist failed"),
76    }
77}

79/// Reconciles `dead_zones.d` with the engine's live dead-zone set -
80/// the dead-zones card's Save button: every registered zone gets a
81/// store file, entries for zones no longer registered are removed.
82/// Session adds and removals commit together in one shot.
83/// Returns the number of files written.
84pub(crate) fn save_dead_zones(engine: &Engine) -> usize {
85    let Some(store) = runtime_store_dir("dead_zones.d") else {
86        return 0;
87    };
88    let zones = engine.dead_zones();
89    let mut written = 0usize;
90    for zone in zones {
91        let name = zone.name.as_str();
92        match std::fs::create_dir_all(&store).and_then(|_| {
93            std::fs::write(
94                store.join(format!("{name}.toml")),
95                toml::to_string_pretty(zone).unwrap_or_default(),
96            )
97        }) {
98            Ok(()) => {
99                written += 1;
100                info!(name, "dead zone persisted");
101            }
102            Err(e) => warn!(error = %e, name, "persist failed"),
103        }
104    }
105    if let Ok(entries) = std::fs::read_dir(&store) {
106        for entry in entries.flatten() {
107            let path = entry.path();
108            let stale = path.extension().is_some_and(|ext| ext == "toml")
109                && path
110                    .file_stem()
111                    .and_then(|stem| stem.to_str())
112                    .is_some_and(|stem| !zones.iter().any(|zone| zone.name.as_str() == stem));
113            if stale {
114                match std::fs::remove_file(&path) {
115                    Ok(()) => info!(path = %path.display(), "dead zone store entry removed"),
116                    Err(e) => warn!(error = %e, path = %path.display(), "unpersist failed"),
117                }
118            }
119        }
120    }
121    written
122}

124/// Rewrites each loaded profile's `enabled`/`color` from the live
125/// engine state - the profiles card's Save button persists the
126/// runtime overrides (`SetProfileEnabled`/`SetProfileColor`) back to
127/// disk. `renames` carries the card's name-entry edits as
128/// `(old name, new name)` pairs: both the file's `name` field and the
129/// file itself move to `<new-name>.toml`. Session-only patterns
130/// without a profile file are skipped.
131/// Returns the number of rewritten files.
132pub(crate) fn save_profile_states(
133    engine: &Engine,
134    profiles_dir: Option<&Path>,
135    renames: &[(PatternName, PatternName)],
136) -> usize {
137    let dir = profiles_dir.unwrap_or_else(|| Path::new("profiles"));
138    let mut loaded = ProfileFile::load_dir(dir);
139    if let Some(store) = runtime_store_dir("profiles.d") {
140        let mut stored = ProfileFile::load_dir(&store);
141        loaded.profiles.append(&mut stored.profiles);
142        loaded.errors.append(&mut stored.errors);
143    }
144    for (path, error) in &loaded.errors {
145        warn!(path = %path.display(), %error, "profile unreadable - skipped on save");
146    }
147    let patterns = engine.patterns();
148    let mut written = 0usize;
149    for (path, mut profile) in loaded.profiles {
150        // Apply the rename before the name-keyed pattern lookup - the
151        // engine was already renamed, so the live name matches the new
152        // file name.
153        let moved = renames
154            .iter()
155            .find(|(old, _)| *old == profile.name)
156            .map(|(_, new)| {
157                profile.name = new.clone();
158                path.with_file_name(format!("{}.toml", new.as_str()))
159            });
160        let target = moved.clone().unwrap_or_else(|| path.clone());
161        let Some(pattern) = patterns.iter().find(|pattern| pattern.name == profile.name) else {
162            continue;
163        };
164        if moved.is_none() && profile.enabled == pattern.enabled && profile.color == pattern.color {
165            continue;
166        }
167        profile.enabled = pattern.enabled;
168        profile.color = pattern.color;
169        match profile.save(&target) {
170            Ok(()) => {
171                written += 1;
172                if moved.is_some()
173                    && let Err(error) = std::fs::remove_file(&path)
174                {
175                    warn!(%error, path = %path.display(), "stale profile file removal failed");
176                }
177                info!(path = %target.display(), enabled = pattern.enabled, "profile state saved");
178            }
179            Err(error) => warn!(%error, path = %target.display(), "profile save failed"),
180        }
181    }
182    written
183}

185/// Deletes a persisted registration file, if it exists.
186pub(crate) fn unpersist(store_kind: &str, name: &str) {
187    if let Some(path) = runtime_store_dir(store_kind).map(|dir| dir.join(format!("{name}.toml")))
188        && path.exists()
189        && let Err(e) = std::fs::remove_file(&path)
190    {
191        warn!(error = %e, path = %path.display(), "unpersist failed");
192    }
193}

206    fn temp_dir(test_name: &str) -> std::path::PathBuf {
207        let dir = std::env::temp_dir().join(format!(
208            "robofigure-store-{test_name}-{}-{}",
209            std::process::id(),
210            std::time::SystemTime::now()
211                .duration_since(std::time::UNIX_EPOCH)
212                .map(|d| d.as_nanos())
213                .unwrap_or_default()
214        ));
215        std::fs::create_dir_all(&dir).expect("temp dir");
216        dir
217    }

219    #[test]
220    fn save_profile_states_writes_enabled_and_color_back() {
221        let dir = temp_dir("enabled");
222        let points = vec![
223            Vec2::new(0.0, 0.0),
224            Vec2::new(50.0, 0.0),
225            Vec2::new(0.0, 50.0),
226        ];
227        let mut file = ProfileFile::from_points("fig", &points, 5.0);
228        file.enabled = false;
229        file.save(&dir.join("fig.toml")).expect("seed profile");
230        let mut engine = Engine::new(vec![file.to_pattern(PatternId::new(1))]);
231        let color = FigureColor::new(1.0, 0.0, 0.0);
232
233        assert_eq!(save_profile_states(&engine, Some(&dir), &[]), 0);
234        engine.set_pattern_enabled(PatternId::new(1), true);
235        engine.set_pattern_color(PatternId::new(1), color);
236
237        assert_eq!(save_profile_states(&engine, Some(&dir), &[]), 1);
238        let loaded = ProfileFile::load_dir(&dir)
239            .profiles
240            .pop()
241            .map(|(_, profile)| profile)
242            .expect("profile reloads");
243        assert!(loaded.enabled);
244        assert_eq!(loaded.color, Some(color));
245        std::fs::remove_dir_all(&dir).ok();
246    }
