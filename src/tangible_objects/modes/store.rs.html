15/// Runtime store for `Lifetime::Persistent` registrations:
16/// `~/.config/tangible-objects/profiles.d/` resp. `fields.d/`.
17pub(crate) fn runtime_store_dir(kind: &str) -> Option<PathBuf> {
18    completion_config_path()
19        .and_then(|path| path.parent().map(Path::to_path_buf))
20        .map(|base| base.join(kind))
21}

23/// Persists a registered profile/field set into the runtime store.
24pub(crate) fn persist_registration(registration: &Registration) {
25    let Some(store) = runtime_store_dir(match registration {
26        Registration::Pattern(_) => "profiles.d",
27        Registration::FieldSet(_) => "fields.d",
28        Registration::DeadZone(_) => "dead_zones.d",
29        Registration::Spell(_) => "spells.d",
30    }) else {
31        return;
32    };
33    let (name, result) = match registration {
34        Registration::Pattern(profile) => (
35            profile.name.as_str().to_string(),
36            std::fs::create_dir_all(&store).and_then(|_| {
37                profile
38                    .save(&store.join(format!("{}.toml", profile.name)))
39                    .map_err(std::io::Error::other)
40            }),
41        ),
42        Registration::FieldSet(set) => (
43            set.name.as_str().to_string(),
44            std::fs::create_dir_all(&store).and_then(|_| {
45                std::fs::write(
46                    store.join(format!("{}.toml", set.name)),
47                    toml::to_string_pretty(set).unwrap_or_default(),
48                )
49            }),
50        ),
51        Registration::DeadZone(zone) => (
52            zone.name.as_str().to_string(),
53            std::fs::create_dir_all(&store).and_then(|_| {
54                std::fs::write(
55                    store.join(format!("{}.toml", zone.name)),
56                    toml::to_string_pretty(zone).unwrap_or_default(),
57                )
58            }),
59        ),
60        Registration::Spell(spell) => (
61            spell.name.as_str().to_string(),
62            std::fs::create_dir_all(&store)
63                .and_then(|_| spell.save(&store.join(format!("{}.toml", spell.name)))),
64        ),
65    };
66    match result {
67        Ok(()) => info!(name, "registration persisted"),
68        Err(e) => warn!(error = %e, name, "persist failed"),
69    }
70}

72/// Rewrites each loaded profile's `enabled`/`color` from the live
73/// engine state - the profiles card's Save button persists the
74/// runtime overrides (`SetProfileEnabled`/`SetProfileColor`) back to
75/// disk. Session-only patterns without a profile file are skipped.
76/// Returns the number of rewritten files.
77pub(crate) fn save_profile_states(engine: &Engine, profiles_dir: Option<&Path>) -> usize {
78    let dir = profiles_dir.unwrap_or_else(|| Path::new("profiles"));
79    let mut loaded = ProfileFile::load_dir(dir);
80    if let Some(store) = runtime_store_dir("profiles.d") {
81        let mut stored = ProfileFile::load_dir(&store);
82        loaded.profiles.append(&mut stored.profiles);
83        loaded.errors.append(&mut stored.errors);
84    }
85    for (path, error) in &loaded.errors {
86        warn!(path = %path.display(), %error, "profile unreadable - skipped on save");
87    }
88    let patterns = engine.patterns();
89    let mut written = 0usize;
90    for (path, mut profile) in loaded.profiles {
91        let Some(pattern) = patterns.iter().find(|pattern| pattern.name == profile.name) else {
92            continue;
93        };
94        if profile.enabled == pattern.enabled && profile.color == pattern.color {
95            continue;
96        }
97        profile.enabled = pattern.enabled;
98        profile.color = pattern.color;
99        match profile.save(&path) {
100            Ok(()) => {
101                written += 1;
102                info!(path = %path.display(), enabled = pattern.enabled, "profile state saved");
103            }
104            Err(error) => warn!(%error, path = %path.display(), "profile save failed"),
105        }
106    }
107    written
108}

110/// Deletes a persisted registration file, if it exists.
111pub(crate) fn unpersist(store_kind: &str, name: &str) {
112    if let Some(path) = runtime_store_dir(store_kind).map(|dir| dir.join(format!("{name}.toml")))
113        && path.exists()
114        && let Err(e) = std::fs::remove_file(&path)
115    {
116        warn!(error = %e, path = %path.display(), "unpersist failed");
117    }
118}

130    fn temp_dir(test_name: &str) -> std::path::PathBuf {
131        let dir = std::env::temp_dir().join(format!(
132            "robofigure-store-{test_name}-{}-{}",
133            std::process::id(),
134            std::time::SystemTime::now()
135                .duration_since(std::time::UNIX_EPOCH)
136                .map(|d| d.as_nanos())
137                .unwrap_or_default()
138        ));
139        std::fs::create_dir_all(&dir).expect("temp dir");
140        dir
141    }
142
143    #[test]
144    fn save_profile_states_writes_enabled_and_color_back() {
145        let dir = temp_dir("enabled");
146        let points = vec![
147            Vec2::new(0.0, 0.0),
148            Vec2::new(50.0, 0.0),
149            Vec2::new(0.0, 50.0),
150        ];
151        let mut file = ProfileFile::from_points("fig", &points, 5.0);
152        file.enabled = false;
153        file.save(&dir.join("fig.toml")).expect("seed profile");
154        let mut engine = Engine::new(vec![file.to_pattern(PatternId::new(1))]);
155        let color = FigureColor::new(1.0, 0.0, 0.0);
156
157        assert_eq!(save_profile_states(&engine, Some(&dir)), 0);
158        engine.set_pattern_enabled(PatternId::new(1), true);
159        engine.set_pattern_color(PatternId::new(1), color);
160
161        assert_eq!(save_profile_states(&engine, Some(&dir)), 1);
162        let loaded = ProfileFile::load_dir(&dir)
163            .profiles
164            .pop()
165            .map(|(_, profile)| profile)
166            .expect("profile reloads");
167        assert!(loaded.enabled);
168        assert_eq!(loaded.color, Some(color));
169        std::fs::remove_dir_all(&dir).ok();
170    }
