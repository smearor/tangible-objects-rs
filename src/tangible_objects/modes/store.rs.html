15/// Runtime store for `Lifetime::Persistent` registrations:
16/// `~/.config/tangible-objects/profiles.d/` resp. `fields.d/`.
17pub(crate) fn runtime_store_dir(kind: &str) -> Option<PathBuf> {
18    completion_config_path()
19        .and_then(|path| path.parent().map(Path::to_path_buf))
20        .map(|base| base.join(kind))
21}
22
23/// Persists a registered profile/field set into the runtime store.
24pub(crate) fn persist_registration(registration: &Registration) {
25    let Some(store) = runtime_store_dir(match registration {
26        Registration::Pattern(_) => "profiles.d",
27        Registration::FieldSet(_) => "fields.d",
28        Registration::DeadZone(_) => "dead_zones.d",
29        Registration::Spell(_) => "spells.d",
30    }) else {
31        return;
32    };
33    let (name, result) = match registration {
34        Registration::Pattern(profile) => (
35            profile.name.as_str().to_string(),
36            std::fs::create_dir_all(&store).and_then(|_| {
37                profile
38                    .save(&store.join(format!("{}.toml", profile.name)))
39                    .map_err(std::io::Error::other)
40            }),
41        ),
42        Registration::FieldSet(set) => (
43            set.name.as_str().to_string(),
44            std::fs::create_dir_all(&store).and_then(|_| {
45                std::fs::write(
46                    store.join(format!("{}.toml", set.name)),
47                    toml::to_string_pretty(set).unwrap_or_default(),
48                )
49            }),
50        ),
51        Registration::DeadZone(zone) => (
52            zone.name.as_str().to_string(),
53            std::fs::create_dir_all(&store).and_then(|_| {
54                std::fs::write(
55                    store.join(format!("{}.toml", zone.name)),
56                    toml::to_string_pretty(zone).unwrap_or_default(),
57                )
58            }),
59        ),
60        Registration::Spell(spell) => (
61            spell.name.as_str().to_string(),
62            std::fs::create_dir_all(&store)
63                .and_then(|_| spell.save(&store.join(format!("{}.toml", spell.name)))),
64        ),
65    };
66    match result {
67        Ok(()) => info!(name, "registration persisted"),
68        Err(e) => warn!(error = %e, name, "persist failed"),
69    }
70}

72/// Reconciles `dead_zones.d` with the engine's live dead-zone set -
73/// the dead-zones card's Save button: every registered zone gets a
74/// store file, entries for zones no longer registered are removed.
75/// Session adds and removals commit together in one shot.
76/// Returns the number of files written.
77pub(crate) fn save_dead_zones(engine: &Engine) -> usize {
78    let Some(store) = runtime_store_dir("dead_zones.d") else {
79        return 0;
80    };
81    let zones = engine.dead_zones();
82    let mut written = 0usize;
83    for zone in zones {
84        let name = zone.name.as_str();
85        match std::fs::create_dir_all(&store).and_then(|_| {
86            std::fs::write(
87                store.join(format!("{name}.toml")),
88                toml::to_string_pretty(zone).unwrap_or_default(),
89            )
90        }) {
91            Ok(()) => {
92                written += 1;
93                info!(name, "dead zone persisted");
94            }
95            Err(e) => warn!(error = %e, name, "persist failed"),
96        }
97    }
98    if let Ok(entries) = std::fs::read_dir(&store) {
99        for entry in entries.flatten() {
100            let path = entry.path();
101            let stale = path.extension().is_some_and(|ext| ext == "toml")
102                && path
103                    .file_stem()
104                    .and_then(|stem| stem.to_str())
105                    .is_some_and(|stem| !zones.iter().any(|zone| zone.name.as_str() == stem));
106            if stale {
107                match std::fs::remove_file(&path) {
108                    Ok(()) => info!(path = %path.display(), "dead zone store entry removed"),
109                    Err(e) => warn!(error = %e, path = %path.display(), "unpersist failed"),
110                }
111            }
112        }
113    }
114    written
115}

117/// Rewrites each loaded profile's `enabled`/`color` from the live
118/// engine state - the profiles card's Save button persists the
119/// runtime overrides (`SetProfileEnabled`/`SetProfileColor`) back to
120/// disk. Session-only patterns without a profile file are skipped.
121/// Returns the number of rewritten files.
122pub(crate) fn save_profile_states(engine: &Engine, profiles_dir: Option<&Path>) -> usize {
123    let dir = profiles_dir.unwrap_or_else(|| Path::new("profiles"));
124    let mut loaded = ProfileFile::load_dir(dir);
125    if let Some(store) = runtime_store_dir("profiles.d") {
126        let mut stored = ProfileFile::load_dir(&store);
127        loaded.profiles.append(&mut stored.profiles);
128        loaded.errors.append(&mut stored.errors);
129    }
130    for (path, error) in &loaded.errors {
131        warn!(path = %path.display(), %error, "profile unreadable - skipped on save");
132    }
133    let patterns = engine.patterns();
134    let mut written = 0usize;
135    for (path, mut profile) in loaded.profiles {
136        let Some(pattern) = patterns.iter().find(|pattern| pattern.name == profile.name) else {
137            continue;
138        };
139        if profile.enabled == pattern.enabled && profile.color == pattern.color {
140            continue;
141        }
142        profile.enabled = pattern.enabled;
143        profile.color = pattern.color;
144        match profile.save(&path) {
145            Ok(()) => {
146                written += 1;
147                info!(path = %path.display(), enabled = pattern.enabled, "profile state saved");
148            }
149            Err(error) => warn!(%error, path = %path.display(), "profile save failed"),
150        }
151    }
152    written
153}

155/// Deletes a persisted registration file, if it exists.
156pub(crate) fn unpersist(store_kind: &str, name: &str) {
157    if let Some(path) = runtime_store_dir(store_kind).map(|dir| dir.join(format!("{name}.toml")))
158        && path.exists()
159        && let Err(e) = std::fs::remove_file(&path)
160    {
161        warn!(error = %e, path = %path.display(), "unpersist failed");
162    }
163}

175    fn temp_dir(test_name: &str) -> std::path::PathBuf {
176        let dir = std::env::temp_dir().join(format!(
177            "robofigure-store-{test_name}-{}-{}",
178            std::process::id(),
179            std::time::SystemTime::now()
180                .duration_since(std::time::UNIX_EPOCH)
181                .map(|d| d.as_nanos())
182                .unwrap_or_default()
183        ));
184        std::fs::create_dir_all(&dir).expect("temp dir");
185        dir
186    }

188    #[test]
189    fn save_profile_states_writes_enabled_and_color_back() {
190        let dir = temp_dir("enabled");
191        let points = vec![
192            Vec2::new(0.0, 0.0),
193            Vec2::new(50.0, 0.0),
194            Vec2::new(0.0, 50.0),
195        ];
196        let mut file = ProfileFile::from_points("fig", &points, 5.0);
197        file.enabled = false;
198        file.save(&dir.join("fig.toml")).expect("seed profile");
199        let mut engine = Engine::new(vec![file.to_pattern(PatternId::new(1))]);
200        let color = FigureColor::new(1.0, 0.0, 0.0);
201
202        assert_eq!(save_profile_states(&engine, Some(&dir)), 0);
203        engine.set_pattern_enabled(PatternId::new(1), true);
204        engine.set_pattern_color(PatternId::new(1), color);
205
206        assert_eq!(save_profile_states(&engine, Some(&dir)), 1);
207        let loaded = ProfileFile::load_dir(&dir)
208            .profiles
209            .pop()
210            .map(|(_, profile)| profile)
211            .expect("profile reloads");
212        assert!(loaded.enabled);
213        assert_eq!(loaded.color, Some(color));
214        std::fs::remove_dir_all(&dir).ok();
215    }
