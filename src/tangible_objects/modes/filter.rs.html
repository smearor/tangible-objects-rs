35/// Filter-proxy mode (FILTER_PROXY_VIRTUAL.md): grabs the real device and
36/// re-emits its event stream on a uinput clone, minus the slots a matched
37/// tangible occupies.
38///
39/// Only exact matches suppress - partial candidates carry a widened
40/// tolerance that would eat finger touches resembling a profile. A matched
41/// slot stays suppressed across transient match loss until its tracking
42/// contact ends or the filter's hysteresis expires (Figur7/8-style flappy
43/// pins must not leak to the OS). `SetFilterMode` control commands switch
44/// the suppression policy live (off/immediate/gated/guarded).
45pub fn run_filter(
46    path: &Path,
47    patterns: PatternSet<'_>,
48    interfaces: Interfaces,
49    stability: Option<FrameCount>,
50    legacy: Option<LegacyMode>,
51    config: DetectionConfig,
52    device_profiles: Option<&Path>,
53) -> Result<()> {
54    let mut device = TouchDevice::open(path).into_diagnostic()?;
55    if config.import_slots {
56        device.queue_state_import();
57    }
58    info!(device = %path.display(), "filter proxy");
59    if let Some(name) = device.name() {
60        info!("device: {name}");
61    }
62
63    let mut engine = Engine::new(patterns.to_vec());
64    let calibration = effective_calibration(&device, device_profiles);
65    if let Some(cal) = calibration {
66        info!(
67            "calibration: X={:.1} units/mm, Y={:.1} units/mm",
68            cal.x_res, cal.y_res
69        );
70        engine.set_calibration(cal);
71    }
72    if let Some(mm) = config.edge {
73        engine.set_edge_margin(mm);
74        info!("edge filter: {mm} mm");
75    }
76    engine.set_fields(config.fields);
77    engine.set_dead_zones(config.dead_zones);
78    engine.set_completion(config.completion);
79
80    let mut virtual_device = VirtualTouchscreen::create(device.inner()).into_diagnostic()?;
81    device.grab().into_diagnostic()?;
82    info!("EVIOCGRAB active - all events flow through the proxy");
83    info!("uinput clone created - tangibles are suppressed");
84    info!("Ctrl+C to quit\n");
85
86    let profiles: Vec<ProfileInfo> = patterns.iter().map(ProfileInfo::from).collect();
87    let (mut runtime, handle) = EngineRuntime::new(Vec::new(), profiles);
88    runtime.set_completion(config.completion);
89    runtime.set_field_sets(engine.field_sets().to_vec());
90    runtime.set_dead_zones(engine.dead_zones().to_vec());
91    spawn_interfaces(&handle, &interfaces);
92    // Report the calibration the engine actually uses (profile override
93    // applied), not the firmware-reported one - clients size their field
94    // sets from these millimeter bounds.
95    runtime.set_device_info(DeviceInfo::of(device.name(), engine.calibration().as_ref()));
96
97    let mut pipeline = Pipeline::new();
98    // Edge ghosts leak to the virtual device unless suppressed - the
99    // edge stage drops contacts inside the margin (the engine still
100    // filters them from matching via `set_edge_margin` above).
101    if let (Some(margin), Some(cal)) = (config.edge, calibration) {
102        pipeline.push_edge_stage(margin, cal.surface_region());
103    }
104    if !config.zones.is_empty() {
105        pipeline.push_zone_stage(config.zones.clone());
106        info!("zone masking: {} region(s)", config.zones.len());
107    }
108    // A stage exists per scope tagged on any pattern - lazily kept in
109    // sync as registrations introduce new scope names.
110    sync_stages(&mut runtime, &mut pipeline, &engine);
111    // `suppress`-flagged field sets live in the pipeline's field stage.
112    pipeline.set_field_sets(engine.field_sets().to_vec());
113    if let Some(gate) = stability {
114        pipeline.set_mode(
115            tangible_objects_model::FilterMode::default(),
116            Some(gate),
117            None,
118        );
119        info!("stability gate: {gate} frames");
120    }
121    if let Some(legacy_mode) = legacy {
122        pipeline.set_legacy_mode(legacy_mode);
123        info!("legacy events (BTN_TOUCH/ABS_X/Y): {legacy_mode}");
124    }
125    let mut frame: u64 = 0;
126    let device_fd = device.as_raw_fd();
127    loop {
128        // Commands run on every loop turn - including poll timeouts - so a
129        // queued SetFilterMode applies while the device is idle.
130        while let Ok(command) = runtime.try_recv_command() {
131            match command {
132                ControlCommand::SetFilterMode {
133                    mode,
134                    stability,
135                    stage,
136                } => {
137                    pipeline.set_mode(mode, stability, stage.as_ref());
138                    runtime.emit_lifecycle(LifecycleEvent::FilterModeChanged(mode));
139                    match &stage {
140                        Some(stage) => info!("filter mode: {mode} (stage {stage})"),
141                        None => info!("filter mode: {mode}"),
142                    }
143                }
144                ControlCommand::RegisterStage { name } => {
145                    if pipeline.register_stage(name.clone()) {
146                        info!(%name, "stage registered");
147                        runtime.emit_lifecycle(LifecycleEvent::StageRegistered { name });
148                    }
149                }
150                ControlCommand::UnregisterStage { name } => {
151                    if pipeline.unregister_stage(&name) {
152                        info!(%name, "stage removed");
153                        runtime.emit_lifecycle(LifecycleEvent::StageUnregistered { name });
154                    }
155                }
156                ControlCommand::SetCompletion(config) => {
157                    engine.set_completion(config);
158                    runtime.set_completion(config);
159                    runtime.emit_lifecycle(LifecycleEvent::CompletionChanged(config));
160                    info!(
161                        "completion: mode={} radius={} max-missing={} persist={}",
162                        config.mode, config.radius_factor, config.max_missing, config.persist
163                    );
164                }
165                ControlCommand::TrainProfile(name) => {
166                    runtime.start_training(&name);
167                    runtime.set_message(Some(format!(
168                        "Training '{name}': place the figure and hold still"
169                    )));
170                }
171                other => {
172                    apply_registration(&mut runtime, &mut engine, other);
173                    // Registrations may introduce or retire scope names
174                    // and field sets - keep the pipeline in sync.
175                    sync_stages(&mut runtime, &mut pipeline, &engine);
176                    pipeline.set_field_sets(engine.field_sets().to_vec());
177                }
178            }
179        }
180
181        let mut pfd = libc::pollfd {
182            fd: device_fd,
183            events: libc::POLLIN,
184            revents: 0,
185        };
186        let ready = unsafe { libc::poll(&mut pfd, 1, 250) };
187        if ready < 0 {
188            if unsafe { *libc::__errno_location() } == libc::EINTR {
189                continue;
190            }
191            error!("poll error - filter loop exits");
192            break;
193        }
194        if ready == 0 {
195            // Interfaces (GraphQL/D-Bus/MCP) still see the live state
196            // while the device idles - same wake-up path as --live.
197            runtime.publish_snapshot(engine.touches(), engine.tracked());
198            continue;
199        }
200
201        let events = match device.fetch_events() {
202            Ok(events) => events,
203            Err(e) => {
204                error!(error = %e, "device read failed - filter loop exits");
205                break;
206            }
207        };
208        for raw in events {
209            let engine_frame = engine.process(&raw);
210            pipeline.push(raw);
211            let Some(engine_frame) = engine_frame else {
212                continue;
213            };
214            frame += 1;
215            // Feed the interface-visible engine state: the set tracker
216            // generates field events and snapshots carry touches,
217            // tangibles, metrics and the device calibration.
218            runtime.publish(&engine_frame, engine.tracked());
219
220            // Exact matches always suppress; under CompletionMode::Exact a
221            // completed partial suppresses its matched *and* claimed slots.
222            // Scoped by stage: a match of a `stage = "x"` pattern feeds
223            // only that stage's classification (EVENT_PIPELINE.md).
224            let completion_exact = engine.completion().mode == CompletionMode::Exact;
225            let mut scoped: HashMap<Option<StageName>, SlotSet> = HashMap::new();
226            for matched in &engine_frame.matches {
227                if !matched.is_partial()
228                    || (completion_exact && !matched.claimed_indices.is_empty())
229                {
230                    let scope = engine
231                        .pattern(matched.pattern_id)
232                        .and_then(|pattern| pattern.stage.clone());
233                    scoped.entry(scope).or_default().extend(
234                        matched
235                            .all_indices()
236                            .map(|index| engine_frame.touches[index].slot),
237                    );
238                }
239            }
240            // `suppress`-enabled field sets suppress via the pipeline's
241            // field stage (SINGLE_CONTACT_FIELD.md) - the daemon merge is
242            // gone.
243            let out = pipeline.finish_scoped(&scoped, (&engine_frame.touches).into());
244            virtual_device.emit(&out).into_diagnostic()?;
245            // The real suppression view is the union across all pipeline
246            // stages - including the stateless edge/zone/field stages,
247            // which `scoped` alone does not cover.
248            let suppressed = pipeline.suppressed_slots().count();
249
250            // A completed capture registers the profile persistently -
251            // its stage scope may need the pipeline sync.
252            tick_training(&mut runtime, &mut engine, &engine_frame.touches);
253            sync_stages(&mut runtime, &mut pipeline, &engine);
254
255            let forwarded = engine_frame.touches.len().saturating_sub(suppressed);
256            print!(
257                "\rFrame {frame} | {} touch | {} figure | {} suppressed | {} forwarded | {} sets | {} fields | {} occupied     ",
258                engine_frame.touches.len(),
259                engine_frame.matches.len(),
260                suppressed,
261                forwarded,
262                engine.field_sets().len(),
263                engine.field_count(),
264                engine.occupied_field_count(),
265            );
266            std::io::stdout().flush().ok();
267        }
268    }
269    Ok(())
270}

272/// Keeps the pipeline's tangible stages in sync with the engine's
273/// pattern scopes and reports the changes: a `stage = "name"` profile
274/// creates its stage (`StageRegistered`), a scope whose last pattern
275/// went away drops the stage with its hysteresis (`StageUnregistered`).
276fn sync_stages(runtime: &mut EngineRuntime, pipeline: &mut Pipeline, engine: &Engine) {
277    let scopes: HashSet<StageName> = engine.stage_scopes().cloned().collect();
278    let changes = pipeline.sync_scopes(&scopes);
279    for scope in changes.registered {
280        info!(%scope, "stage registered");
281        runtime.emit_lifecycle(LifecycleEvent::StageRegistered { name: scope });
282    }
283    for scope in changes.unregistered {
284        info!(%scope, "stage removed");
285        runtime.emit_lifecycle(LifecycleEvent::StageUnregistered { name: scope });
286    }
