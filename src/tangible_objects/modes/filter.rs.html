38/// Filter-proxy mode (FILTER_PROXY_VIRTUAL.md): grabs the real device and
39/// re-emits its event stream on a uinput clone, minus the slots a matched
40/// tangible occupies.
41///
42/// Only exact matches suppress - partial candidates carry a widened
43/// tolerance that would eat finger touches resembling a profile. A matched
44/// slot stays suppressed across transient match loss until its tracking
45/// contact ends or the filter's hysteresis expires (Figur7/8-style flappy
46/// pins must not leak to the OS). `SetFilterMode` control commands switch
47/// the suppression policy live (off/immediate/gated/guarded).
48#[allow(clippy::too_many_arguments)]
49pub fn run_filter(
50    path: &Path,
51    patterns: PatternSet<'_>,
52    interfaces: Interfaces,
53    stability: Option<FrameCount>,
54    legacy: Option<LegacyMode>,
55    config: DetectionConfig,
56    device_profiles: Option<&Path>,
57    record_base: Option<&Path>,
58) -> Result<()> {
59    let mut device = TouchDevice::open(path).into_diagnostic()?;
60    if config.import_slots {
61        device.queue_state_import();
62    }
63    info!(device = %path.display(), "filter proxy");
64    if let Some(name) = device.name() {
65        info!("device: {name}");
66    }
67
68    let mut engine = Engine::new(patterns.to_vec());
69    let calibration = effective_calibration(&device, device_profiles);
70    if let Some(cal) = calibration {
71        info!(
72            "calibration: X={:.1} units/mm, Y={:.1} units/mm",
73            cal.x_res, cal.y_res
74        );
75        engine.set_calibration(cal);
76    }
77    if let Some(mm) = config.edge {
78        engine.set_edge_margin(mm);
79        info!("edge filter: {mm} mm");
80    }
81    engine.set_fields(config.fields);
82    engine.set_dead_zones(config.dead_zones);
83    engine.set_completion(config.completion);
84
85    let mut virtual_device = VirtualTouchscreen::create(device.inner()).into_diagnostic()?;
86    // `--record <base>` starts the paired input/output capture at
87    // startup; `StartRecording`/`StopRecording` control commands do
88    // the same at runtime (MCP/GraphQL/D-Bus).
89    let mut proxy_recording = record_base
90        .map(|base| ProxyRecording::start(base, device.name(), engine.calibration().as_ref(), None))
91        .transpose()
92        .into_diagnostic()?;
93    device.grab().into_diagnostic()?;
94    info!("EVIOCGRAB active - all events flow through the proxy");
95    info!("uinput clone created - tangibles are suppressed");
96    info!("Ctrl+C to quit\n");
97
98    let profiles: Vec<ProfileInfo> = patterns.iter().map(ProfileInfo::from).collect();
99    let (mut runtime, handle) = EngineRuntime::new(Vec::new(), profiles);
100    runtime.set_completion(config.completion);
101    runtime.set_field_sets(engine.field_sets().to_vec());
102    runtime.set_dead_zones(engine.dead_zones().to_vec());
103    spawn_interfaces(&handle, &interfaces);
104    // Report the calibration the engine actually uses (profile override
105    // applied), not the firmware-reported one - clients size their field
106    // sets from these millimeter bounds.
107    runtime.set_device_info(DeviceInfo::of(device.name(), engine.calibration().as_ref()));
108    runtime.set_recording_status(proxy_recording.as_ref().map(ProxyRecording::status));
109
110    let mut pipeline = Pipeline::new();
111    // Edge ghosts leak to the virtual device unless suppressed - the
112    // edge stage drops contacts inside the margin (the engine still
113    // filters them from matching via `set_edge_margin` above).
114    if let (Some(margin), Some(cal)) = (config.edge, calibration) {
115        pipeline.push_edge_stage(margin, cal.surface_region());
116    }
117    if !config.zones.is_empty() {
118        pipeline.push_zone_stage(config.zones.clone());
119        info!("zone masking: {} region(s)", config.zones.len());
120    }
121    // A stage exists per scope tagged on any pattern - lazily kept in
122    // sync as registrations introduce new scope names.
123    sync_stages(&mut runtime, &mut pipeline, &engine);
124    // `suppress`-flagged field sets live in the pipeline's field stage.
125    pipeline.set_field_sets(engine.field_sets().to_vec());
126    if let Some(gate) = stability {
127        pipeline.set_mode(
128            tangible_objects_model::FilterMode::default(),
129            Some(gate),
130            None,
131        );
132        info!("stability gate: {gate} frames");
133    }
134    if let Some(legacy_mode) = legacy {
135        pipeline.set_legacy_mode(legacy_mode);
136        info!("legacy events (BTN_TOUCH/ABS_X/Y): {legacy_mode}");
137    }
138    let mut frame: u64 = 0;
139    let device_fd = device.as_raw_fd();
140    loop {
141        // Commands run on every loop turn - including poll timeouts - so a
142        // queued SetFilterMode applies while the device is idle.
143        while let Ok(command) = runtime.try_recv_command() {
144            match command {
145                ControlCommand::SetFilterMode {
146                    mode,
147                    stability,
148                    stage,
149                } => {
150                    pipeline.set_mode(mode, stability, stage.as_ref());
151                    runtime.emit_lifecycle(LifecycleEvent::FilterModeChanged(mode));
152                    match &stage {
153                        Some(stage) => info!("filter mode: {mode} (stage {stage})"),
154                        None => info!("filter mode: {mode}"),
155                    }
156                }
157                ControlCommand::RegisterStage { name } => {
158                    if pipeline.register_stage(name.clone()) {
159                        info!(%name, "stage registered");
160                        runtime.emit_lifecycle(LifecycleEvent::StageRegistered { name });
161                    }
162                }
163                ControlCommand::UnregisterStage { name } => {
164                    if pipeline.unregister_stage(&name) {
165                        info!(%name, "stage removed");
166                        runtime.emit_lifecycle(LifecycleEvent::StageUnregistered { name });
167                    }
168                }
169                ControlCommand::SetCompletion(config) => {
170                    engine.set_completion(config);
171                    runtime.set_completion(config);
172                    runtime.emit_lifecycle(LifecycleEvent::CompletionChanged(config));
173                    info!(
174                        "completion: mode={} radius={} max-missing={} persist={}",
175                        config.mode, config.radius_factor, config.max_missing, config.persist
176                    );
177                }
178                ControlCommand::TrainProfile(name) => {
179                    runtime.start_training(&name);
180                    runtime.set_message(Some(format!(
181                        "Training '{name}': place the figure and hold still"
182                    )));
183                }
184                ControlCommand::StartRecording { base_path, label } => {
185                    if proxy_recording.is_some() {
186                        warn!("recording already active - StartRecording ignored");
187                    } else {
188                        match ProxyRecording::start(
189                            &base_path,
190                            device.name(),
191                            engine.calibration().as_ref(),
192                            label,
193                        ) {
194                            Ok(recording) => {
195                                runtime.set_recording_status(Some(recording.status()));
196                                proxy_recording = Some(recording);
197                                runtime.emit_lifecycle(LifecycleEvent::RecordingStarted);
198                            }
199                            Err(e) => warn!(error = %e, "recording could not be started"),
200                        }
201                    }
202                }
203                ControlCommand::StopRecording => match proxy_recording.take() {
204                    Some(recording) => {
205                        let frames = recording.frames();
206                        if let Err(e) = recording.finish() {
207                            warn!(error = %e, "recording flush failed");
208                        }
209                        runtime.set_recording_status(None);
210                        runtime.emit_lifecycle(LifecycleEvent::RecordingStopped { frames });
211                        info!(%frames, "proxy recording stopped");
212                    }
213                    None => warn!("no recording active - StopRecording ignored"),
214                },
215                ControlCommand::SetRecordLabel(label) => match &mut proxy_recording {
216                    Some(recording) => {
217                        recording.set_label(label);
218                        runtime.set_recording_status(Some(recording.status()));
219                    }
220                    None => warn!("no recording active - SetRecordLabel ignored"),
221                },
222                other => {
223                    apply_registration(&mut runtime, &mut engine, other);
224                    // Registrations may introduce or retire scope names
225                    // and field sets - keep the pipeline in sync.
226                    sync_stages(&mut runtime, &mut pipeline, &engine);
227                    pipeline.set_field_sets(engine.field_sets().to_vec());
228                }
229            }
230        }
231
232        let mut pfd = libc::pollfd {
233            fd: device_fd,
234            events: libc::POLLIN,
235            revents: 0,
236        };
237        let ready = unsafe { libc::poll(&mut pfd, 1, 250) };
238        if ready < 0 {
239            if unsafe { *libc::__errno_location() } == libc::EINTR {
240                continue;
241            }
242            error!("poll error - filter loop exits");
243            break;
244        }
245        if ready == 0 {
246            // Interfaces (GraphQL/D-Bus/MCP) still see the live state
247            // while the device idles - same wake-up path as --live.
248            runtime.publish_snapshot(engine.touches(), engine.tracked());
249            continue;
250        }
251
252        let events = match device.fetch_events() {
253            Ok(events) => events,
254            Err(e) => {
255                error!(error = %e, "device read failed - filter loop exits");
256                break;
257            }
258        };
259        for raw in events {
260            if let Some(recording) = &proxy_recording {
261                recording.record_in(&raw);
262            }
263            let engine_frame = engine.process(&raw);
264            pipeline.push(raw);
265            let Some(engine_frame) = engine_frame else {
266                continue;
267            };
268            frame += 1;
269            // Feed the interface-visible engine state: the set tracker
270            // generates field events and snapshots carry touches,
271            // tangibles, metrics and the device calibration.
272            runtime.publish(&engine_frame, engine.tracked());
273
274            // Exact matches always suppress; under CompletionMode::Exact a
275            // completed partial suppresses its matched *and* claimed slots.
276            // Scoped by stage: a match of a `stage = "x"` pattern feeds
277            // only that stage's classification (EVENT_PIPELINE.md).
278            let completion_exact = engine.completion().mode == CompletionMode::Exact;
279            let mut scoped: HashMap<Option<StageName>, SlotSet> = HashMap::new();
280            for matched in &engine_frame.matches {
281                if !matched.is_partial()
282                    || (completion_exact && !matched.claimed_indices.is_empty())
283                {
284                    let scope = engine
285                        .pattern(matched.pattern_id)
286                        .and_then(|pattern| pattern.stage.clone());
287                    scoped.entry(scope).or_default().extend(
288                        matched
289                            .all_indices()
290                            .map(|index| engine_frame.touches[index].slot),
291                    );
292                }
293            }
294            // `suppress`-enabled field sets suppress via the pipeline's
295            // field stage (SINGLE_CONTACT_FIELD.md) - the daemon merge is
296            // gone.
297            let out = pipeline.finish_scoped(&scoped, (&engine_frame.touches).into());
298            virtual_device.emit(&out).into_diagnostic()?;
299            if let Some(recording) = &mut proxy_recording {
300                recording.record_out(out.iter().map(RawEvent::from));
301                runtime.set_recording_status(Some(recording.status()));
302            }
303            // The real suppression view is the union across all pipeline
304            // stages - including the stateless edge/zone/field stages,
305            // which `scoped` alone does not cover.
306            let suppressed = pipeline.suppressed_slots().count();
307
308            // A completed capture registers the profile persistently -
309            // its stage scope may need the pipeline sync.
310            tick_training(&mut runtime, &mut engine, &engine_frame.touches);
311            sync_stages(&mut runtime, &mut pipeline, &engine);
312
313            let forwarded = engine_frame.touches.len().saturating_sub(suppressed);
314            print!(
315                "\rFrame {frame} | {} touch | {} figure | {} suppressed | {} forwarded | {} sets | {} fields | {} occupied | {} dead zones | {} filtered     ",
316                engine_frame.touches.len(),
317                engine_frame.matches.len(),
318                suppressed,
319                forwarded,
320                engine.field_set_count(),
321                engine.field_count(),
322                engine.occupied_field_count(),
323                engine.dead_zones().len(),
324                engine.dead_zone_filtered(),
325            );
326            std::io::stdout().flush().ok();
327        }
328    }
329    if let Some(recording) = proxy_recording
330        && let Err(error) = recording.finish()
331    {
332        error!(%error, "proxy recording could not be finalized");
333    }
334    Ok(())
335}

337/// Keeps the pipeline's tangible stages in sync with the engine's
338/// pattern scopes and reports the changes: a `stage = "name"` profile
339/// creates its stage (`StageRegistered`), a scope whose last pattern
340/// went away drops the stage with its hysteresis (`StageUnregistered`).
341fn sync_stages(runtime: &mut EngineRuntime, pipeline: &mut Pipeline, engine: &Engine) {
342    let scopes: HashSet<StageName> = engine.stage_scopes().cloned().collect();
343    let changes = pipeline.sync_scopes(&scopes);
344    for scope in changes.registered {
345        info!(%scope, "stage registered");
346        runtime.emit_lifecycle(LifecycleEvent::StageRegistered { name: scope });
347    }
348    for scope in changes.unregistered {
349        info!(%scope, "stage removed");
350        runtime.emit_lifecycle(LifecycleEvent::StageUnregistered { name: scope });
351    }
