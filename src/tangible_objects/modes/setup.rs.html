23/// Effective calibration for a live device.
24///
25/// ABS_INFO (`device.calibration()`) is authoritative for axis bounds, but
26/// when the device's profile declares a resolution override the measured
27/// physical resolution wins - some panels misreport their units/mm in
28/// firmware (e.g. the SiS panel reporting 7/12 while driving a 65" display).
29/// Returns the reported calibration when the device has no profile or the
30/// profile has no override; when the firmware reports no resolution at all,
31/// the profile's documented values are the only source. `None` only when
32/// neither source exists.
33pub fn effective_calibration(
34    device: &TouchDevice,
35    device_profiles: Option<&Path>,
36) -> Option<Calibration> {
37    let reported = device.calibration();
38    let mut registry = DeviceRegistry::builtin();
39    if let Some(dir) = device_profiles {
40        registry = registry.with_dir(dir);
41    }
42    let id = device.inner().input_id();
43    let profile = registry.find_by_ids(id.vendor(), id.product());
44    match (reported, profile) {
45        // Firmware reports a resolution - it wins unless the profile
46        // declares a measured override (panels that misreport units/mm).
47        (Some(reported), Some(profile)) => match profile.resolution_override() {
48            Some((x_res, y_res)) => {
49                info!(
50                    "resolution override from device profile: X={:.3}, Y={:.3} units/mm",
51                    x_res.as_f64(),
52                    y_res.as_f64()
53                );
54                Some(Calibration::with_bounds(
55                    x_res,
56                    y_res,
57                    reported.x_min,
58                    reported.x_max,
59                    reported.y_min,
60                    reported.y_max,
61                ))
62            }
63            None => Some(reported),
64        },
65        // No firmware resolution at all - the profile's documented
66        // values (including overrides) are the only source left.
67        (None, Some(profile)) => Some(profile.calibration()),
68        (Some(reported), None) => Some(reported),
69        (None, None) => None,
70    }
71}

73/// Loads figure patterns from a profile directory (default: `profiles/`).
74pub fn load_patterns(profiles_dir: Option<&str>) -> PatternBuf {
75    let dir = profiles_dir.unwrap_or("profiles");
76    let mut dir_result = ProfileFile::load_dir(Path::new(dir));
77    // `Lifetime::Persistent` registrations live in the runtime store and
78    // reload with the static set.
79    if let Some(store) = runtime_store_dir("profiles.d")
80        && store.is_dir()
81    {
82        let mut stored = ProfileFile::load_dir(&store);
83        dir_result.profiles.append(&mut stored.profiles);
84        dir_result.errors.append(&mut stored.errors);
85    }
86    for (_, profile) in &dir_result.profiles {
87        if profile.enabled {
88            info!(
89                "profile loaded: {} ({} points, {} distances)",
90                profile.name,
91                profile.num_points,
92                profile.sorted_sides.len()
93            );
94        } else {
95            info!("profile disabled: {}", profile.name);
96        }
97    }
98    for (path, error) in &dir_result.errors {
99        warn!(path = %path.display(), error = %error, "profile could not be loaded");
100    }
101    let patterns = dir_result.into_patterns();
102    if patterns.is_empty() {
103        warn!("no profiles found in {dir} - create one with --train");
104    }
105    patterns
106}

108/// Default path of the completion config file:
109/// `$XDG_CONFIG_HOME/tangible-objects/completion.toml`, falling back to
110/// `~/.config/tangible-objects/completion.toml`.
111pub fn completion_config_path() -> Option<PathBuf> {
112    let base = std::env::var_os("XDG_CONFIG_HOME")
113        .map(PathBuf::from)
114        .or_else(|| std::env::var_os("HOME").map(|home| PathBuf::from(home).join(".config")))?;
115    Some(base.join("tangible-objects").join("completion.toml"))
116}

118/// Loads the completion config from [`completion_config_path`]; falls
119/// back to [`CompletionConfig::default`] when the file is absent or
120/// unreadable. CLI `--completion-*` flags override individual fields on
121/// top of this.
122pub fn load_completion() -> CompletionConfig {
123    let Some(path) = completion_config_path() else {
124        return CompletionConfig::default();
125    };
126    match std::fs::read_to_string(&path) {
127        Ok(contents) => match toml::from_str::<CompletionConfig>(&contents) {
128            Ok(config) => {
129                info!(
130                    "completion config loaded: {} (mode={} radius={} max-missing={} persist={})",
131                    path.display(),
132                    config.mode,
133                    config.radius_factor,
134                    config.max_missing,
135                    config.persist
136                );
137                config
138            }
139            Err(e) => {
140                warn!(path = %path.display(), error = %e, "completion config unreadable - defaults used");
141                CompletionConfig::default()
142            }
143        },
144        Err(_) => CompletionConfig::default(),
145    }
146}

148/// Loads the startup dead zones from the runtime store
149/// (`dead_zones.d/` under the config directory - `Lifetime::Persistent`
150/// registrations reload with the daemon). Invalid files are reported
151/// and skipped.
152pub fn load_dead_zones() -> Vec<DeadZone> {
153    let mut zones = Vec::new();
154    if let Some(store) = runtime_store_dir("dead_zones.d")
155        && store.is_dir()
156    {
157        let (stored, errors) = DeadZone::load_dir(&store);
158        for (file, error) in &errors {
159            warn!(file = %file.display(), %error, "dead zone could not be loaded");
160        }
161        zones.extend(stored);
162    }
163    zones
164}

184/// Loads occupancy field sets from a file or directory (`--fields`).
185///
186/// A `.toml` file loads one set, a directory loads all sets inside.
187/// Invalid files are reported and skipped.
188pub fn load_field_sets(path: Option<&str>) -> Vec<FieldSet> {
189    let mut sets = Vec::new();
190    // Persisted field sets (`Lifetime::Persistent`) reload with the
191    // static `--fields` set.
192    if let Some(store) = runtime_store_dir("fields.d")
193        && store.is_dir()
194    {
195        let (stored, errors) = FieldSet::load_dir(&store);
196        for (file, error) in &errors {
197            warn!(file = %file.display(), %error, "field set could not be loaded");
198        }
199        sets.extend(stored);
200    }
201    let Some(path) = path else {
202        return sets;
203    };
204    let path = Path::new(path);
205    let file_sets = if path.is_dir() {
206        let (file_sets, errors) = FieldSet::load_dir(path);
207        for (file, error) in &errors {
208            warn!(file = %file.display(), %error, "field set could not be loaded");
209        }
210        file_sets
211    } else {
212        match FieldSet::load(path) {
213            Ok(set) => vec![set],
214            Err(error) => {
215                warn!(path = %path.display(), %error, "field set could not be loaded");
216                Vec::new()
217            }
218        }
219    };
220    sets.extend(file_sets);
221    for set in &sets {
222        info!(
223            "field set loaded: {} ({} fields)",
224            set.name,
225            set.fields.len()
226        );
227    }
228    sets
229}

231/// Resolves the device argument: an explicit `--device` path wins;
232/// otherwise auto-detect via `vendor:product` against the device registry
233/// (builtin + `--device-profiles` dir) - succeeds only when exactly one
234/// known touchscreen is attached.
235pub fn resolve_device(device: Option<String>, profiles_dir: Option<&Path>) -> Result<PathBuf> {
236    if let Some(path) = device {
237        return Ok(PathBuf::from(path));
238    }
239    let mut registry = DeviceRegistry::builtin();
240    if let Some(dir) = profiles_dir {
241        registry = registry.with_dir(dir);
242    }
243    let entry = TouchDevice::detect(&registry).into_diagnostic()?;
244    let profile = registry
245        .find_by_ids(entry.vendor, entry.product)
246        .expect("detected entry matches a profile");
247    info!(
248        "device detected: {} ({}) -> {}",
249        profile.name,
250        entry.name,
251        entry.path.display()
252    );
253    Ok(entry.path)
254}

256/// Lists all `/dev/input/eventX` devices with names and multitouch markers.
257pub fn list_devices() {
258    let entries = TouchDevice::enumerate();
259    if entries.is_empty() {
260        warn!("no input devices found in /dev/input/");
261        return;
262    }
263    for entry in entries {
264        println!("{entry}");
265    }
