1//! `ProxyRecording` - the paired input/output event capture of the
2//! filter proxy (`--record` / `StartRecording` on `--filter`).
3
4use std::io;
5use std::path::Path;
6use std::time::Instant;
7
8use tangible_objects_engine::RawRecorder;
9use tangible_objects_model::Calibration;
10use tangible_objects_model::EventCount;
11use tangible_objects_model::FrameCount;
12use tangible_objects_model::RawEvent;
13use tangible_objects_model::RecordLabel;
14use tangible_objects_model::RecordingStatus;
15
16/// Paired recording of the filter proxy: `<base>.evtest` captures the
17/// raw input stream and `<base>.out.evtest` the emitted OS-visible
18/// stream after suppression. Both files carry session-relative
19/// timestamps in the same clock base, so `out.ts - in.ts` measures the
20/// proxy's per-event latency.
21pub(crate) struct ProxyRecording {
22    /// Raw input events before engine and pipeline.
23    in_recorder: RawRecorder,
24    /// The batch emitted on the virtual device per frame.
25    out_recorder: RawRecorder,
26    /// Clock origin shared by both streams.
27    session_start: Instant,
28    /// Emitted frames so far - batches containing a `SYN_REPORT`.
29    frames: FrameCount,
30    /// Ground-truth label carried by the interface-facing status.
31    label: Option<RecordLabel>,
32}
33
34impl ProxyRecording {
35    /// Starts the pair under `base` (`<base>.evtest` +
36    /// `<base>.out.evtest`).
37    pub(crate) fn start(
38        base: &Path,
39        device_name: Option<&str>,
40        calibration: Option<&Calibration>,
41        label: Option<RecordLabel>,
42    ) -> io::Result<Self> {
43        let in_path = base.with_extension("evtest");
44        let out_path = base.with_extension("out.evtest");
45        let in_recorder = RawRecorder::spawn(&in_path, device_name, calibration)?;
46        let out_recorder = RawRecorder::spawn(&out_path, device_name, calibration)?;
47        tracing::info!(
48            in_path = %in_path.display(),
49            out_path = %out_path.display(),
50            "proxy recording active"
51        );
52        Ok(Self {
53            in_recorder,
54            out_recorder,
55            session_start: Instant::now(),
56            frames: FrameCount::new(0),
57            label,
58        })
59    }
60
61    /// Queues one raw input event on the `in` side.
62    pub(crate) fn record_in(&self, event: &RawEvent) {
63        self.in_recorder
64            .record(event, self.session_start.elapsed().as_secs_f64());
65    }
66
67    /// Queues the emitted batch on the `out` side; a `SYN_REPORT`
68    /// inside the batch counts a written frame.
69    pub(crate) fn record_out(&mut self, events: impl Iterator<Item = RawEvent>) {
70        let timestamp_secs = self.session_start.elapsed().as_secs_f64();
71        for raw in events {
72            if raw.is_syn_report() {
73                self.frames += 1;
74            }
75            self.out_recorder.record(&raw, timestamp_secs);
76        }
77    }
78
79    /// Frames written to the out stream so far.
80    pub(crate) fn frames(&self) -> FrameCount {
81        self.frames
82    }
83
84    /// Sets the ground-truth label.
85    pub(crate) fn set_label(&mut self, label: Option<RecordLabel>) {
86        self.label = label;
87    }
88
89    /// Live status for `recording_status` and `EngineSnapshot`.
90    ///
91    /// `recorded_frames` counts emitted `SYN_REPORT` batches; the raw
92    /// `dropped_events` counter aggregates both sides since the in
93    /// and out sinks share the same drop semantics.
94    pub(crate) fn status(&self) -> RecordingStatus {
95        RecordingStatus {
96            active: true,
97            label: self.label.clone(),
98            recorded_frames: self.frames,
99            dropped_frames: FrameCount::new(0),
100            dropped_events: EventCount::new(
101                self.in_recorder.dropped_count().as_u64()
102                    + self.out_recorder.dropped_count().as_u64(),
103            ),
104        }
105    }
106
107    /// Finalizes both files (flush + writer join).
108    pub(crate) fn finish(self) -> io::Result<()> {
109        self.in_recorder.finish()?;
110        self.out_recorder.finish()
111    }
112}
