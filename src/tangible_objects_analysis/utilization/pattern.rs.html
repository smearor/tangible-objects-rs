9/// Utilization statistics of one pattern across the recording.
10#[derive(Debug, Serialize, Deserialize)]
11pub struct PatternUtilization {
12    /// ID of the figure profile; `None` for label-only rows (a label that
13    /// never produced a candidate, e.g. a figure whose pin count exceeded
14    /// the recorded touch count - or a misspelled label).
15    pub pattern_id: Option<PatternId>,
16    /// Name of the figure profile (or the raw label for label-only rows).
17    pub pattern_name: String,
18    /// Frames with at least one matched full (non-partial) candidate.
19    pub matched_frames: FrameCount,
20    /// Frames where the pattern was matched by a partial candidate only.
21    pub partial_matched_frames: FrameCount,
22    /// Frames carrying this pattern's ground-truth label.
23    pub labeled_frames: FrameCount,
24    /// Labeled frames without any matched candidate for this pattern -
25    /// the figure lay on the surface but was not detected.
26    pub false_negative_frames: FrameCount,
27    /// Deviation distribution per distance index - the labeled pattern's
28    /// best candidate per labeled frame (matched or not; index `i`
29    /// corresponds to `sorted_distances[i]`).
30    pub per_distance: Vec<Percentiles>,
31    /// Labeled presence frames binned by `required_tolerance / tolerance`:
32    /// `[<25 %, 25–50 %, 50–75 %, 75–100 %]`.
33    pub bands: RatioBands,
34    /// Labeled presence frames whose best candidate exceeds the configured
35    /// tolerance (ratio `>= 1.0`) - counted separately because the bands
36    /// only cover `< 1.0`.
37    pub over_tolerance_frames: FrameCount,
38}
