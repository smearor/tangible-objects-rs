6/// Per-distance-index deviation statistics of one pattern over a
7/// recording - the causal "why did the match fail" answer.
8///
9/// `index i` refers to the pattern's `sorted_distances[i]`, i.e. a
10/// concrete pin pair; a high `blocker` count pinpoints the unstable
11/// pin pair that most often pushed the deviation over tolerance.
12#[derive(Debug, Serialize, Deserialize)]
13pub struct DistanceStats {
14    /// Distance index into the pattern's sorted fingerprint.
15    pub index: usize,
16    /// Frames this index was measured (full candidates only).
17    pub samples: FrameCount,
18    /// Mean absolute deviation in millimeters.
19    pub mean: Millimeters,
20    /// Worst observed deviation in millimeters.
21    pub max: Millimeters,
22    /// 99th percentile deviation in millimeters.
23    pub p99: Millimeters,
24    /// Frames where this index exceeded its effective tolerance
25    /// (per-index `tolerances[i]` when configured, else the scalar).
26    pub over_tolerance: FrameCount,
27    /// Frames where this index was the **only** over-tolerance
28    /// distance - the direct cause of the miss.
29    pub sole_blocker: FrameCount,
30    /// Frames where this index was among the over-tolerance
31    /// distances.
32    pub blocker: FrameCount,
33}
