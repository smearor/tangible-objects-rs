4use tangible_objects_model::CandidateReport;
5use tangible_objects_model::FrameNumber;
6use tangible_objects_model::FrameRecord;
7use tangible_objects_model::MatchScore;

10/// One dataset row: a recorded candidate with its ground-truth target.
11///
12/// Serializes as one JSONL object per line.
13#[derive(Debug, Serialize, Deserialize)]
14pub struct DatasetRow {
15    /// Frame the candidate was tested in.
16    pub frame: FrameNumber,
17    /// Whether the candidate's pattern is the frame's ground-truth label.
18    pub is_label: bool,
19    /// Whether the candidate matched under the recorded tolerances.
20    pub matched: bool,
21    /// Whether this was a partial (`num_points - 1`) candidate.
22    pub partial: bool,
23    /// Pattern the candidate was tested against.
24    pub pattern_id: PatternId,
25    /// Normalized deviation score (`max_i dev[i] / tolerance[i]`).
26    pub score: MatchScore,
27    /// Per-index normalized deviations `dev[i] / tolerance[i]` - the
28    /// classifier features; `score` is their maximum.
29    pub features: Vec<MatchScore>,
30}
31
32impl DatasetRow {
33    /// Builds the dataset row of a candidate within a labeled frame.
34    pub fn of(frame: &FrameRecord, label: &str, candidate: &CandidateReport) -> Self {
35        Self {
36            frame: frame.frame,
37            is_label: candidate.matches_label(&label.into()),
38            matched: candidate.matched,
39            partial: candidate.partial,
40            pattern_id: candidate.pattern_id,
41            score: candidate.normalized_score(),
42            features: candidate
43                .deviations
44                .iter()
45                .enumerate()
46                .map(|(i, dev)| MatchScore::new(dev.as_f64() / candidate.tolerance_at(i).as_f64()))
47                .collect(),
48        }
49    }
50}
