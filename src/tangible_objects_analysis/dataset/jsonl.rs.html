9/// Labeled dataset over a whole recording - one row per candidate of every
10/// labeled frame (TRAINING.md 3.3).
11///
12/// Serialized as JSONL - the same line format as the recordings themselves:
13/// typed fields instead of CSV's untyped, escaping- and locale-sensitive
14/// text cells. Unlabeled frames contribute no rows: without a ground-truth
15/// target the features are unlearnable.
16#[derive(Debug)]
17pub struct Dataset {
18    /// Collected candidate rows.
19    pub rows: Vec<DatasetRow>,
20    /// Width of the widest feature vector in the dataset.
21    pub feature_width: usize,
22}
23
24impl Dataset {
25    /// Collects the dataset from labeled frames of a recording.
26    pub fn of(recording: &Recording) -> Self {
27        let mut rows = Vec::new();
28        for frame in recording.frames() {
29            let Some(label) = &frame.label else { continue };
30            for candidate in &frame.candidates {
31                if candidate.deviations.is_empty() {
32                    continue;
33                }
34                rows.push(DatasetRow::of(frame, label.as_str(), candidate));
35            }
36        }
37        let feature_width = rows.iter().map(|row| row.features.len()).max().unwrap_or(0);
38        Self {
39            rows,
40            feature_width,
41        }
42    }
43
44    /// Writes the dataset as JSONL - one [`DatasetRow`] per line.
45    pub fn write_jsonl(&self, path: &Path) -> Result<(), AnalysisError> {
46        write_json_lines(path, &self.rows).map_err(|source| AnalysisError::Io {
47            path: path.to_path_buf(),
48            source,
49        })
50    }
51}
