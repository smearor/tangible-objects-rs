19/// Histogram bucket edges for contact lifetimes (in frames).
20const LIFETIME_EDGES: [FrameCount; 5] = [
21    FrameCount::new(1),
22    FrameCount::new(5),
23    FrameCount::new(15),
24    FrameCount::new(60),
25    FrameCount::new(120),
26];
27
28/// Contact-lifetime and dropout statistics of a recording (TRAINING.md 2.3).
29///
30/// Built from the `tracking_id` time series (`touches`) and the tracker's
31/// `frames_since_match` hysteresis (`tracked`). Per-pin attribution is not
32/// derivable from recordings - `touch_indices` is an unordered combination,
33/// not a pin mapping - so instability is reported per contact, not per pin.
34#[derive(Debug, Serialize, Deserialize)]
35pub struct FlappyReport {
36    /// Number of distinct tracking IDs observed.
37    pub contacts: ContactCount,
38    /// Percentiles of contact lifetimes in frames (first→last appearance).
39    pub lifetimes: Option<Percentiles>,
40    /// Lifetime histogram; the last bucket is open-ended.
41    pub lifetime_buckets: Vec<LifetimeBucket>,
42    /// Dropout events: runs of `frames_since_match > 0` inside a tracked
43    /// tangible's lifetime - the figure lost its match but stayed tracked.
44    pub dropout_events: EventCount,
45    /// Percentiles of dropout run lengths in frames.
46    pub dropout_lengths: Option<Percentiles>,
47    /// Partial-match share per pattern.
48    pub partial_share: Vec<PartialShare>,
49}

51impl FlappyReport {
52    /// Computes the flappy-contact report for a loaded recording.
53    pub fn analyze(recording: &Recording) -> Self {
54        // Contact lifetimes from the tracking_id time series.
55        let mut first_seen: BTreeMap<TrackingId, FrameNumber> = BTreeMap::new();
56        let mut lifetimes: BTreeMap<TrackingId, FrameCount> = BTreeMap::new();
57        for frame in recording.frames() {
58            for touch in &frame.touches {
59                let id = touch.tracking_id;
60                let first = first_seen.entry(id).or_insert(frame.frame);
61                lifetimes.insert(
62                    id,
63                    FrameCount::new(frame.frame.as_u64() - first.as_u64() + 1),
64                );
65            }
66        }
67
68        let lifetime_values: Vec<f64> = lifetimes.values().map(|v| v.as_u64() as f64).collect();
69        let mut lifetime_buckets: Vec<LifetimeBucket> = LIFETIME_EDGES
70            .iter()
71            .map(|edge| LifetimeBucket {
72                upper_bound: Some(*edge),
73                count: ContactCount::new(0),
74            })
75            .collect();
76        lifetime_buckets.push(LifetimeBucket {
77            upper_bound: None,
78            count: ContactCount::new(0),
79        });
80        for &lifetime in lifetimes.values() {
81            let index = LIFETIME_EDGES
82                .iter()
83                .position(|edge| lifetime <= *edge)
84                .unwrap_or(LIFETIME_EDGES.len());
85            lifetime_buckets[index].count += ContactCount::new(1);
86        }
87
88        // Dropout runs inside each tracked tangible's lifetime.
89        let mut dropout_events = EventCount::new(0);
90        let mut dropout_lengths = Vec::new();
91        let mut runs: BTreeMap<u32, u64> = BTreeMap::new();
92        for frame in recording.frames() {
93            for tracked in &frame.tracked {
94                if tracked.frames_since_match > FrameCount::new(0) {
95                    *runs.entry(tracked.id.0).or_insert(0) += 1;
96                } else if let Some(length) = runs.remove(&tracked.id.0) {
97                    dropout_events += 1;
98                    dropout_lengths.push(length as f64);
99                }
100            }
101        }
102        for (_, length) in runs {
103            dropout_events += 1;
104            dropout_lengths.push(length as f64);
105        }
106
107        // Partial-match share per pattern.
108        let mut share: BTreeMap<PatternName, PartialShare> = BTreeMap::new();
109        for frame in recording.frames() {
110            for candidate in &frame.candidates {
111                if !candidate.matched {
112                    continue;
113                }
114                let entry = share
115                    .entry(candidate.pattern_name.clone())
116                    .or_insert_with(|| PartialShare {
117                        pattern_name: candidate.pattern_name.clone(),
118                        partial_matches: CandidateCount::new(0),
119                        matched_total: CandidateCount::new(0),
120                    });
121                entry.matched_total += 1;
122                if candidate.partial {
123                    entry.partial_matches += 1;
124                }
125            }
126        }
127
128        Self {
129            contacts: ContactCount::new(lifetimes.len() as u32),
130            lifetimes: Percentiles::of(lifetime_values.iter().copied()),
131            lifetime_buckets,
132            dropout_events,
133            dropout_lengths: Percentiles::of(dropout_lengths.iter().copied()),
134            partial_share: share.into_values().collect(),
135        }
136    }
137}

139impl Display for FlappyReport {
140    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
141        writeln!(f, "== Flappy contacts == ({} contacts)", self.contacts)?;
142        if let Some(lifetimes) = self.lifetimes {
143            writeln!(
144                f,
145                "  lifetime frames: p50={:.0} p95={:.0} p99={:.0} max={:.0} (mean {:.1})",
146                lifetimes.p50, lifetimes.p95, lifetimes.p99, lifetimes.max, lifetimes.mean
147            )?;
148        }
149        for bucket in &self.lifetime_buckets {
150            let label = match bucket.upper_bound {
151                Some(edge) => format!("<= {edge}"),
152                None => format!("> {}", LIFETIME_EDGES[LIFETIME_EDGES.len() - 1]),
153            };
154            writeln!(f, "    {label}: {}", bucket.count)?;
155        }
156        writeln!(f, "  dropout events: {}", self.dropout_events)?;
157        if let Some(lengths) = self.dropout_lengths {
158            writeln!(
159                f,
160                "    lengths: p50={:.0} p95={:.0} max={:.0}",
161                lengths.p50, lengths.p95, lengths.max
162            )?;
163        }
164        for share in &self.partial_share {
165            writeln!(
166                f,
167                "  {}: {}/{} partial matches ({:.1}%)",
168                share.pattern_name,
169                share.partial_matches,
170                share.matched_total,
171                share.fraction() * 100.0
172            )?;
173        }
174        Ok(())
175    }
