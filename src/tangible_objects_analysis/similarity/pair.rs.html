16/// Collision assessment of one pattern pair.
17///
18/// `min_max_deviation` is the smallest achievable maximum deviation
19/// over all same-size subset fingerprints (`contact_count` pins kept)
20/// - the geometric phantom surface. A value ≤ tolerance means a real
21///   contact set can satisfy both fingerprints.
22#[derive(Debug, Serialize, Deserialize)]
23pub struct SimilarityPair {
24    /// First profile of the pair.
25    pub a: PatternName,
26    /// Second profile of the pair.
27    pub b: PatternName,
28    /// Contact count of the closest subset pairing - `num_points` means
29    /// the exact fingerprints collide.
30    pub contact_count: ContactCount,
31    /// Smallest achievable maximum deviation across subset pairings.
32    pub min_max_deviation: Deviation,
33    /// Risk classification derived from `min_max_deviation` and the
34    /// pair's tolerances.
35    pub risk: MatchRisk,
36}
37
38impl SimilarityPair {
39    /// Finds the closest fingerprint collision between two patterns over
40    /// all same-size subset pairings (k = min(n_a, n_b) down to 3).
41    ///
42    /// Returns `None` when no comparable fingerprints exist (different
43    /// sizes and no `ref_points` to derive subsets).
44    pub fn closest(a: &TangiblePattern, b: &TangiblePattern) -> Option<Self> {
45        let collision = Self::closest_collision(a, b)?;
46        Some(Self {
47            a: a.name.clone(),
48            b: b.name.clone(),
49            contact_count: collision.contact_count,
50            min_max_deviation: collision.min_max_deviation,
51            risk: Self::risk_of(collision.min_max_deviation, a.tolerance, b.tolerance),
52        })
53    }
54
55    /// Finds the closest fingerprint collision between two patterns over
56    /// all same-size subset pairings.
57    ///
58    /// `contact_count == num_points` means even the exact fingerprints
59    /// nearly collide.
60    fn closest_collision(a: &TangiblePattern, b: &TangiblePattern) -> Option<SubsetCollision> {
61        if let (Some(points_a), Some(points_b)) = (a.ref_points.as_ref(), b.ref_points.as_ref()) {
62            let max_k = a.num_points.min(b.num_points).as_usize();
63            let mut best: Option<SubsetCollision> = None;
64            for k in (3..=max_k).rev() {
65                for keep_a in combinations(a.num_points.as_usize(), k) {
66                    let fp_a = subset_fingerprint(a, points_a, &keep_a);
67                    for keep_b in combinations(b.num_points.as_usize(), k) {
68                        let fp_b = subset_fingerprint(b, points_b, &keep_b);
69                        let deviation = fp_a.max_deviation(&fp_b);
70                        if best
71                            .as_ref()
72                            .is_none_or(|c| deviation < c.min_max_deviation)
73                        {
74                            best = Some(SubsetCollision {
75                                contact_count: ContactCount::new(k as u32),
76                                min_max_deviation: deviation,
77                            });
78                        }
79                    }
80                }
81            }
82            return best;
83        }
84
85        // Fallback without ref_points: exact fingerprint comparison when
86        // both patterns carry the same number of distances.
87        if a.sorted_distances.len() == b.sorted_distances.len() {
88            return Some(SubsetCollision {
89                contact_count: ContactCount::new(a.num_points.max(b.num_points).as_u32()),
90                min_max_deviation: a.sorted_distances.max_deviation(&b.sorted_distances),
91            });
92        }
93        None
94    }
95
96    /// Classifies the risk: `Exact` within the stricter tolerance, else
97    /// `Partial` within 1.5x tolerance, else `Safe`.
98    fn risk_of(
99        deviation: Deviation,
100        tol_a: PatternTolerance,
101        tol_b: PatternTolerance,
102    ) -> MatchRisk {
103        let tolerance = tol_a.as_f64().max(tol_b.as_f64());
104        if deviation.as_f64() <= tolerance {
105            MatchRisk::Exact
106        } else if deviation.as_f64() <= tolerance * 1.5 {
107            MatchRisk::Partial
108        } else {
109            MatchRisk::Safe
110        }
111    }
112}

114/// Sorted pairwise distances of the kept reference points - `k ==
115/// num_points` falls back to the full sorted fingerprint (the
116/// `partial_reference_multi` path needs at least one skipped pin).
117fn subset_fingerprint(
118    pattern: &TangiblePattern,
119    points: &[Vec2<Millimeters>],
120    keep: &[usize],
121) -> Distances {
122    if keep.len() == pattern.num_points {
123        return pattern.sorted_distances.clone();
124    }
125    let subset: Vec<Vec2<Millimeters>> = keep.iter().map(|&i| points[i]).collect();
126    PointSet::new(&subset).distances()
127}

129/// All `k`-element index subsets of `n` points (lexicographic order).
130fn combinations(n: usize, k: usize) -> Vec<Vec<usize>> {
131    if k == 0 || k > n {
132        return Vec::new();
133    }
134    let mut result = Vec::new();
135    let mut combo: Vec<usize> = (0..k).collect();
136    loop {
137        result.push(combo.clone());
138        let Some(pos) = (0..k).rev().find(|&i| combo[i] < n - k + i) else {
139            break;
140        };
141        combo[pos] += 1;
142        for i in pos + 1..k {
143            combo[i] = combo[i - 1] + 1;
144        }
145    }
146    result
147}
