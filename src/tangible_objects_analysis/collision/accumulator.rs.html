1use smearor_commons_geometry::Millimeters;

5/// Per-cell accumulation state of the collision matrix.
6pub(crate) struct Accumulator {
7    /// Labeled frames where at least one candidate of this pattern existed.
8    pub frames_tested: FrameCount,
9    /// Frames where the pattern's best candidate matched (false positive).
10    pub collided_frames: FrameCount,
11    /// Per-frame minimal
12    /// [`required_tolerance`](tangible_objects_model::CandidateReport::required_tolerance)
13    /// samples.
14    pub required: Vec<Millimeters>,
15    /// The candidate pattern's configured tolerance.
16    pub tolerance: PatternTolerance,
17}
18
19impl Accumulator {
20    /// Starts accumulating for a cell; the pattern tolerance is fixed for the
21    /// whole session and taken from the first observed candidate.
22    pub fn new(tolerance: PatternTolerance) -> Self {
23        Self {
24            frames_tested: FrameCount::new(0),
25            collided_frames: FrameCount::new(0),
26            required: Vec::new(),
27            tolerance,
28        }
29    }
30}
