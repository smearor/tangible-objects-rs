12/// A loaded frame recording: validated JSONL lines of schema `"v": 1`.
13///
14/// Loading rejects malformed lines and unknown schema versions up front -
15/// every returned frame is a parseable, version-checked [`FrameRecord`].
16#[derive(Debug)]
17pub struct Recording {
18    frames: Vec<FrameRecord>,
19}
20
21impl Recording {
22    /// Loads and validates a `.jsonl` recording produced by `--record`.
23    ///
24    /// Empty lines are skipped; every other line must parse as a
25    /// [`FrameRecord`] with schema version [`RECORD_SCHEMA_VERSION`].
26    pub fn load(path: &Path) -> Result<Self, AnalysisError> {
27        let file = File::open(path).map_err(|source| AnalysisError::Io {
28            path: path.to_path_buf(),
29            source,
30        })?;
31        let mut frames = Vec::new();
32        for (index, line) in BufReader::new(file).lines().enumerate() {
33            let line_no = index + 1;
34            let line = line.map_err(|source| AnalysisError::Io {
35                path: PathBuf::from(path),
36                source,
37            })?;
38            if line.trim().is_empty() {
39                continue;
40            }
41            let frame: FrameRecord =
42                serde_json::from_str(&line).map_err(|source| AnalysisError::Parse {
43                    line: line_no,
44                    source,
45                })?;
46            if frame.schema_version != RECORD_SCHEMA_VERSION {
47                return Err(AnalysisError::UnsupportedVersion {
48                    line: line_no,
49                    found: frame.schema_version,
50                    expected: RECORD_SCHEMA_VERSION,
51                });
52            }
53            frames.push(frame);
54        }
55        Ok(Self { frames })
56    }
57
58    /// All recorded frames in file order.
59    pub fn frames(&self) -> &[FrameRecord] {
60        &self.frames
61    }
62}

64#[cfg(test)]
65mod tests {
66    use super::Recording;
67    use crate::AnalysisError;
68
69    fn temp_path(name: &str) -> std::path::PathBuf {
70        std::env::temp_dir().join(format!("robofigure-analysis-{name}-{}", std::process::id()))
71    }
72
73    fn write_lines(path: &std::path::Path, lines: &[&str]) -> std::io::Result<()> {
74        std::fs::write(path, lines.join("\n") + "\n")
75    }
76
77    #[test]
78    fn loads_valid_recording() -> Result<(), Box<dyn std::error::Error>> {
79        let path = temp_path("valid.jsonl");
80        write_lines(
81            &path,
82            &[
83                r#"{"v":1,"frame":0,"timestamp_ms":0,"label":null,"calibration":null,"touches":[],"candidates":[],"tracked":[],"raw_dropped":0}"#,
84                "",
85                r#"{"v":1,"frame":1,"timestamp_ms":8,"label":"fig","calibration":null,"touches":[],"candidates":[],"tracked":[],"raw_dropped":0}"#,
86            ],
87        )?;
88        let recording = Recording::load(&path)?;
89        assert_eq!(recording.frames().len(), 2);
90        assert_eq!(recording.frames()[1].frame.0, 1);
91        std::fs::remove_file(&path).ok();
92        Ok(())
93    }
94
95    #[test]
96    fn rejects_unknown_schema_version() -> Result<(), Box<dyn std::error::Error>> {
97        let path = temp_path("bad-version.jsonl");
98        write_lines(
99            &path,
100            &[
101                r#"{"v":2,"frame":0,"timestamp_ms":0,"label":null,"calibration":null,"touches":[],"candidates":[],"tracked":[],"raw_dropped":0}"#,
102            ],
103        )?;
104        assert!(matches!(
105            Recording::load(&path),
106            Err(AnalysisError::UnsupportedVersion {
107                line: 1,
108                found: 2,
109                ..
110            })
111        ));
112        std::fs::remove_file(&path).ok();
113        Ok(())
114    }
115
116    #[test]
117    fn rejects_malformed_line() -> Result<(), Box<dyn std::error::Error>> {
118        let path = temp_path("malformed.jsonl");
119        write_lines(&path, &["{not json"])?;
120        assert!(matches!(
121            Recording::load(&path),
122            Err(AnalysisError::Parse { line: 1, .. })
123        ));
124        std::fs::remove_file(&path).ok();

126    }
