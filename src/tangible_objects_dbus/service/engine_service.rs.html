1//! `EngineService` - the D-Bus service object served at
2//! `/org/tangibleobjects/Engine`; the `#[interface]` impl lives in
3//! `crate::service::interface`, method bodies in `crate::service::api`.
4
5use std::collections::HashMap;
6use std::collections::HashSet;
7use std::sync::Arc;
8use std::sync::Mutex;
9
10use tangible_objects_model::Owner;
11use tangible_objects_runtime::EngineHandle;
12use zbus::message::Header;
13use zbus::names::UniqueName;
14
15/// Maps a caller's unique bus name to the [`Owner`]s it registered
16/// under - drives automatic cleanup when the name vanishes.
17pub type OwnerSessions = Arc<Mutex<HashMap<String, HashSet<Owner>>>>;
18
19/// D-Bus service object served at `/org/tangibleobjects/Engine`.
20///
21/// Methods read the latest snapshot or forward control commands to the
22/// device loop; signals are emitted via
23/// [`emit_signal`](crate::service::signals::emit_signal).
24pub struct EngineService {
25    /// Runtime handle shared by all interface methods.
26    pub(crate) handle: EngineHandle,
27    /// Owner sessions for disconnect cleanup (`NameOwnerChanged`).
28    pub(crate) owner_sessions: OwnerSessions,
29}
30
31impl EngineService {
32    /// Creates the service for the given runtime handle and the shared
33    /// owner-session map watched by [`serve`](crate::serve).
34    pub fn new(handle: EngineHandle, owner_sessions: OwnerSessions) -> Self {
35        Self {
36            handle,
37            owner_sessions,
38        }
39    }
40
41    /// Resolves the effective owner for a register call: the explicit
42    /// `owner`, else the caller's unique bus name. Records the
43    /// owner under the sender for disconnect cleanup.
44    pub(crate) fn session_owner(&self, owner: Owner, header: &Header<'_>) -> Owner {
45        let sender = header
46            .sender()
47            .map(UniqueName::as_str)
48            .unwrap_or_default()
49            .to_string();
50        let effective = if owner.as_str().is_empty() {
51            Owner::new(sender.clone())
52        } else {
53            owner
54        };
55        if !sender.is_empty()
56            && let Ok(mut sessions) = self.owner_sessions.lock()
57        {
58            sessions
59                .entry(sender)
60                .or_default()
61                .insert(effective.clone());
62        }
63        effective
64    }
65}
