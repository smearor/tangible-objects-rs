1//! GUI methods - `set_gui_visible`, `set_display_rotation`,
2//! `set_dead_zones_visible`.
3
4use crate::service::engine_service::EngineService;
5use tangible_objects_model::DisplayRotation;
6use tangible_objects_runtime::ControlCommand;
7use zbus::fdo;
8
9/// Shows or hides the daemon-embedded GUI window (`--gui` mode) -
10/// ignored on headless daemons.
11pub(crate) async fn set_gui_visible(service: &EngineService, visible: bool) -> fdo::Result<()> {
12    service
13        .handle
14        .send_command(ControlCommand::SetGuiVisible(visible))
15        .await
16        .map_err(|_| fdo::Error::Failed("device loop is not running".to_string()))
17}
18
19/// Quarter-turn the GUI canvas renders (`--gui` mode): `"0"`, `"90"`,
20/// `"180"` or `"270"` degrees clockwise from the viewer's perspective -
21/// ignored on headless daemons.
22pub(crate) async fn set_display_rotation(
23    service: &EngineService,
24    rotation: &str,
25) -> fdo::Result<()> {
26    let parsed: DisplayRotation = rotation.parse().map_err(fdo::Error::InvalidArgs)?;
27    service
28        .handle
29        .send_command(ControlCommand::SetDisplayRotation(parsed))
30        .await
31        .map_err(|_| fdo::Error::Failed("device loop is not running".to_string()))
32}
