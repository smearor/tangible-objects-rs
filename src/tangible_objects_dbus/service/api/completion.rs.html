12/// Active missing-pin completion configuration as a flat tuple
13/// (`radius_factor`, `max_missing`, `persist`, `mode`,
14/// `affinity_factor`, `smoothing_alpha`) - empty strings are
15/// impossible here; when nothing was published yet all values are
16/// defaults.
17pub(crate) async fn completion(service: &EngineService) -> (f64, u32, bool, String, f64, f64) {
18    let config = service.handle.snapshot().completion.unwrap_or_default();
19    (
20        config.radius_factor,
21        config.max_missing,
22        config.persist,
23        config.mode.to_string(),
24        config.affinity_factor.as_f64(),
25        config.smoothing_alpha.as_f64(),
26    )
27}
28
29/// Replaces the missing-pin completion configuration (`mode` is
30/// `match` or `exact`; `max_missing` must be 1 or 2; `affinity`
31/// widens incumbent tolerances, `smoothing` is the position EMA
32/// alpha).
33pub(crate) async fn set_completion(
34    service: &EngineService,
35    radius_factor: f64,
36    max_missing: u32,
37    persist: bool,
38    mode: &str,
39    affinity: AffinityFactor,
40    smoothing: SmoothingAlpha,
41) -> fdo::Result<()> {
42    let mode = mode
43        .parse::<CompletionMode>()
44        .map_err(|_| fdo::Error::InvalidArgs("mode must be one of: match, exact".to_string()))?;
45    if !(1..=2).contains(&max_missing) {
46        return Err(fdo::Error::InvalidArgs(
47            "max_missing must be 1 or 2".to_string(),
48        ));
49    }
50    let config = CompletionConfig {
51        radius_factor,
52        max_missing,
53        persist,
54        mode,
55        affinity_factor: affinity,
56        smoothing_alpha: smoothing,
57    };
58    service
59        .handle
60        .send_command(ControlCommand::SetCompletion(config))
61        .await
62        .map_err(|_| fdo::Error::Failed("device loop is not running".to_string()))
63}
