21/// Well-known bus name of the daemon's D-Bus service.
22pub const BUS_NAME: &str = "org.tangibleobjects.Engine";
23
24/// Object path of the engine service.
25pub const OBJECT_PATH: &str = "/org/tangibleobjects/Engine";
26
27/// Serves the D-Bus interface on the session bus and forwards domain
28/// events as signals until the device loop stops.
29pub async fn serve(handle: EngineHandle) -> zbus::Result<()> {
30    let owner_sessions: OwnerSessions = Default::default();
31    let connection = connection::Builder::session()?
32        .name(BUS_NAME)?
33        .serve_at(
34            OBJECT_PATH,
35            EngineService::new(handle.clone(), owner_sessions.clone()),
36        )?
37        .build()
38        .await?;
39    info!(
40        bus = BUS_NAME,
41        path = OBJECT_PATH,
42        "D-Bus service listening"
43    );
44
45    // Disconnect cleanup: when a caller's unique bus name vanishes,
46    // unregister everything that name had registered (session-scoped).
47    let proxy = DBusProxy::new(&connection).await?;
48    let mut owner_changes = proxy.receive_name_owner_changed().await?;
49    let cleanup_handle = handle.clone();
50    tokio::spawn(async move {
51        use futures_lite::StreamExt;
52
53        while let Some(signal) = owner_changes.next().await {
54            let Ok(args) = signal.args() else {
55                continue;
56            };
57            if args.new_owner().is_some() {
58                continue;
59            }
60            let name = args.name().to_string();
61            let owners: Vec<Owner> = owner_sessions
62                .lock()
63                .map(|mut sessions| {
64                    sessions
65                        .remove(&name)
66                        .unwrap_or_default()
67                        .into_iter()
68                        .collect()
69                })
70                .unwrap_or_default();
71            for owner in owners {
72                info!(%owner, %name, "D-Bus caller vanished - unregistering");
73                let _ = cleanup_handle
74                    .send_command(ControlCommand::UnregisterOwner { owner })
75                    .await;
76            }
77        }
78    });
79
80    let emitter = SignalEmitter::new(&connection, OBJECT_PATH)?;
81    let mut events = handle.subscribe();
82    loop {
83        match events.recv().await {
84            Ok(event) => emit_signal(&emitter, &event).await,
85            Err(RecvError::Lagged(skipped)) => {
86                warn!(skipped, "D-Bus signal forwarder lagged");
87            }
88            Err(RecvError::Closed) => {
89                info!("event stream closed, stopping D-Bus signals");
90                break;
91            }
92        }
93    }
94
95    // Keep the connection alive even after the event stream ends so
96    // methods stay callable.
97    pending::<()>().await;
98    Ok(())
99}

101/// Spawns the D-Bus service on a dedicated thread with its own tokio
102/// runtime, keeping the daemon's synchronous device loop untouched.
103pub fn spawn(handle: EngineHandle) -> JoinHandle<()> {
104    std::thread::spawn(move || match Runtime::new() {
105        Ok(runtime) => {
106            if let Err(err) = runtime.block_on(serve(handle)) {
107                error!(%err, "D-Bus service failed");
108            }
109        }
110        Err(err) => error!(%err, "failed to start D-Bus runtime"),
111    })
112}
