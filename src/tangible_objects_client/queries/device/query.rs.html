1//! `DeviceInfoQuery` - the cynic fragment for the `deviceInfo` query.
2
3use cynic::QueryFragment;
4use tangible_objects_model::DeviceInfo;
5
6/// Root-field wrapper for the `deviceInfo` query - cynic derives the
7/// document and checks every selected field against `schema.graphql`
8/// at compile time. The fields are selection markers only - the
9/// response decodes through the model's `Deserialize`.
10#[allow(dead_code)]
11#[derive(QueryFragment)]
12#[cynic(
13    graphql_type = "QueryRoot",
14    schema_path = "../graphql/schema.graphql",
15    schema_module = "tangible_objects_model::schema"
16)]
17pub(crate) struct DeviceInfoQuery {
18    /// `deviceInfo` field.
19    #[cynic(rename = "deviceInfo")]
20    device_info: Option<DeviceInfo>,
21}
22
23#[cfg(test)]
24mod tests {
25    use super::DeviceInfoQuery;
26
27    /// The cynic-derived document selects every `DeviceInfo` field -
28    /// compile-time field check plus this string assert as the visible
29    /// proof.
30    #[test]
31    fn device_info_document() {
32        let query = cynic::Operation::<DeviceInfoQuery>::query(())
33            .query
34            .to_string();
35        for field in [
36            "deviceInfo",
37            "name",
38            "xRes",
39            "yRes",
40            "xMin",
41            "xMax",
42            "yMin",
43            "yMax",
44        ] {
45            assert!(query.contains(field), "missing {field} in: {query}");
46        }
47    }
