1//! `TouchPointsQuery` - the cynic fragment for the `touchPoints`
2//! query.
3
4use cynic::QueryFragment;
5use tangible_objects_model::TouchPoint;
6
7/// Root-field wrapper for the `touchPoints` query - cynic derives the
8/// document and checks every selected field against `schema.graphql`
9/// at compile time. The fields are selection markers only - the
10/// response decodes through the model's `Deserialize`.
11#[allow(dead_code)]
12#[derive(QueryFragment)]
13#[cynic(
14    graphql_type = "QueryRoot",
15    schema_path = "../graphql/schema.graphql",
16    schema_module = "tangible_objects_model::schema"
17)]
18pub(crate) struct TouchPointsQuery {
19    /// `touchPoints` field.
20    #[cynic(rename = "touchPoints")]
21    touch_points: Vec<TouchPoint>,
22}
23
24#[cfg(test)]
25mod tests {
26    use super::TouchPointsQuery;
27
28    /// The cynic-derived document selects every `TouchPoint` field.
29    #[test]
30    fn touch_points_document() {
31        let query = cynic::Operation::<TouchPointsQuery>::query(())
32            .query
33            .to_string();
34        for field in ["touchPoints", "slot", "trackingId", "pos", "x", "y"] {
35            assert!(query.contains(field), "missing {field} in: {query}");
36        }
37    }
