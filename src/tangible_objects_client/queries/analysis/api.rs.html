4use std::path::Path;
5
6use tangible_objects_analysis::AnalysisReport;
7use tangible_objects_analysis::SimilarityReport;
8use tangible_objects_model::AnalysisMode;
9
10use crate::ClientError;
11use crate::GraphQLClient;
12use crate::queries::analysis::artifact::query::RecordingArtifactQuery;
13use crate::queries::analysis::artifact::variables::RecordingArtifactVariables;
14use crate::queries::analysis::recording::query::AnalyzeRecordingQuery;
15use crate::queries::analysis::recording::variables::AnalyzeRecordingVariables;
16use crate::queries::analysis::similarity::query::AnalyzeSimilarityQuery;
17use crate::queries::analysis::similarity::variables::AnalyzeSimilarityVariables;
18
19impl GraphQLClient {
20    /// Deviation analysis over a recording (`analyzeRecording` query -
21    /// `Json<AnalysisReport>` passthrough).
22    pub async fn analyze_recording(
23        &self,
24        path: &Path,
25        mode: AnalysisMode,
26        profiles_dir: Option<&Path>,
27    ) -> Result<AnalysisReport, ClientError> {
28        self.call_query::<AnalyzeRecordingQuery, _, _>(
29            "analyzeRecording",
30            AnalyzeRecordingVariables {
31                path: path.to_string_lossy().into_owned(),
32                mode: Some(mode),
33                profiles_dir: profiles_dir.map(|dir| dir.to_string_lossy().into_owned()),
34            },
35        )
36        .await
37    }
38
39    /// Pairwise profile-collision report (`analyzeSimilarity` query -
40    /// `Json<SimilarityReport>` passthrough).
41    pub async fn analyze_similarity(
42        &self,
43        profiles_dir: Option<&Path>,
44    ) -> Result<SimilarityReport, ClientError> {
45        self.call_query::<AnalyzeSimilarityQuery, _, _>(
46            "analyzeSimilarity",
47            AnalyzeSimilarityVariables {
48                profiles_dir: profiles_dir.map(|dir| dir.to_string_lossy().into_owned()),
49            },
50        )
51        .await
52    }
53
54    /// Recording artifact content (`recordingArtifact` query) - raw
55    /// file text, legitimately an untyped `String`.
56    pub async fn recording_artifact(&self, path: &Path) -> Result<String, ClientError> {
57        self.call_query::<RecordingArtifactQuery, _, _>(
58            "recordingArtifact",
59            RecordingArtifactVariables {
60                path: path.to_string_lossy().into_owned(),
61            },
62        )
63        .await
64    }
