1//! Client-side configuration file -
2//! `~/.config/tangible-objects/client.toml`.
3
4use std::path::PathBuf;
5
6use serde::Deserialize;
7
8/// Contents of the client config file - one shared file for `cli` and
9/// `tui` (the daemon's own config lives separately).
10#[derive(Debug, Default, Deserialize)]
11pub struct ClientConfig {
12    /// Daemon endpoint (`http://host:port/graphql`).
13    pub endpoint: Option<String>,
14}
15
16impl ClientConfig {
17    /// `~/.config/tangible-objects/client.toml` - `XDG_CONFIG_HOME`
18    /// aware; `None` when no config base resolves.
19    pub fn config_path() -> Option<PathBuf> {
20        let base = std::env::var_os("XDG_CONFIG_HOME")
21            .map(PathBuf::from)
22            .or_else(|| std::env::var_os("HOME").map(|home| PathBuf::from(home).join(".config")))?;
23        Some(base.join("tangible-objects").join("client.toml"))
24    }
25
26    /// Loads the config from [`ClientConfig::config_path`]; `None` when
27    /// the file is absent or unreadable.
28    pub fn load() -> Option<Self> {
29        let path = Self::config_path()?;
30        let text = std::fs::read_to_string(path).ok()?;
31        toml::from_str(&text).ok()
32    }
33}
