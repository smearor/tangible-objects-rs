1//! Errors the [`crate::GraphQLClient`] can raise.
2
3use thiserror::Error;
4
5/// Failure modes of client calls - transport, protocol and GraphQL-level
6/// errors stay distinguishable (GraphQL `errors` arrays keep their
7/// structure, not a stringified blob).
8#[derive(Debug, Error)]
9pub enum ClientError {
10    /// HTTP request failed (unreachable host, 4xx/5xx, body decode).
11    #[error("HTTP request failed: {0}")]
12    Http(#[from] reqwest::Error),
13    /// WebSocket handshake or frame handling failed - boxed: the
14    /// tungstenite error enum is ~140 bytes and would dominate
15    /// `size_of::<ClientError>`.
16    #[error("WebSocket error: {0}")]
17    Ws(Box<tokio_tungstenite::tungstenite::Error>),
18    /// `graphql-transport-ws` subscription failure reported by
19    /// `graphql-ws-client` - handshake, keepalive timeout, GraphQL
20    /// `error` frames and frame decode all surface here.
21    #[error("subscription transport error: {0}")]
22    Subscription(#[from] graphql_ws_client::Error),
23    /// Malformed handshake - no `connection_ack`, or a `next` message
24    /// without the subscription's payload.
25    #[error("graphql-transport-ws protocol violation: {0}")]
26    Protocol(String),
27    /// The endpoint returned a GraphQL `errors` array (parse/validation
28    /// errors, resolver failures).
29    #[error("GraphQL errors: {0}")]
30    Graphql(serde_json::Value),
31    /// A response decoded but carried no `data` payload (absent or
32    /// `null`).
33    #[error("response contained no data")]
34    NoData,
35    /// The `data` object did not contain the expected operation field
36    /// - the server never ran the selection.
37    #[error("response data has no field '{0}'")]
38    FieldNotFound(String),
39    /// A `data` field failed to deserialize into the expected type.
40    #[error("response decode failed: {0}")]
41    Decode(serde_json::Error),
42}
43
44impl From<tokio_tungstenite::tungstenite::Error> for ClientError {
45    /// Wraps the transport error boxed (see [`ClientError::Ws`]).
46    fn from(e: tokio_tungstenite::tungstenite::Error) -> Self {
47        Self::Ws(Box::new(e))
48    }
49}
