1//! `RegisterSpellInput` - the cynic input object for
2//! `registerSpell`, a wire DTO of [`Spell`].
3
4use cynic::InputObject;
5use smearor_commons_geometry::Millimeters;
6use tangible_objects_model::Spell;
7use tangible_objects_model::SpellName;
8
9use crate::mutations::spell::register::arming_input::SpellArmingInput;
10
11/// The `RegisterSpellInput` input object - wire DTO of [`Spell`].
12#[derive(InputObject)]
13#[cynic(
14    graphql_type = "RegisterSpellInput",
15    schema_path = "../graphql/schema.graphql",
16    schema_module = "tangible_objects_model::schema",
17    rename_all = "camelCase"
18)]
19pub(crate) struct RegisterSpellInput {
20    /// Spell name (identity for unregister and the runtime store).
21    name: SpellName,
22    /// Max per-stroke deviation in millimeters; `None` = daemon
23    /// default.
24    corridor: Option<Millimeters>,
25    /// Whether the gesture may be drawn larger/smaller than the
26    /// template; `None` = daemon default.
27    scale_invariant: Option<bool>,
28    /// Whether the spell matches; `None` = daemon default.
29    enabled: Option<bool>,
30    /// Per-finger template strokes - `strokes[i]` is finger i's
31    /// flat polyline `[x0, y0, x1, y1, ...]` in millimeters.
32    strokes: Vec<Vec<Millimeters>>,
33    /// How the gesture is initiated; `None` = `free`.
34    arming: Option<SpellArmingInput>,
35}
36
37impl From<&Spell> for RegisterSpellInput {
38    /// Projects the spell into the wire input shape - every
39    /// [`StrokePolyline`] flattens its `Vec2` points into the
40    /// `[x0, y0, x1, y1, ...]` wire form.
41    fn from(spell: &Spell) -> Self {
42        Self {
43            name: spell.name.clone(),
44            corridor: Some(spell.corridor),
45            scale_invariant: Some(spell.scale_invariant),
46            enabled: Some(spell.enabled),
47            strokes: spell
48                .strokes
49                .iter()
50                .map(|stroke| {
51                    stroke
52                        .points
53                        .iter()
54                        .flat_map(|point| [Millimeters::new(point.x), Millimeters::new(point.y)])
55                        .collect()
56                })
57                .collect(),
58            arming: Some((&spell.arming).into()),
59        }
60    }
61}
