1//! `SpellArmingInput` - the cynic input object for the spell's
2//! arming trigger, a wire DTO of [`SpellArming`].
3
4use cynic::InputObject;
5use tangible_objects_model::HandName;
6use tangible_objects_model::PatternName;
7use tangible_objects_model::SpellArming;
8
9use crate::mutations::spell::register::arming_kind::SpellArmingKind;
10use crate::mutations::spell::register::region_input::RegionInput;
11
12/// The `SpellArmingInput` input object - the `kind` tag plus the
13/// fields the armed variants carry (`region` for zone/spot arming,
14/// `pattern` for the tangible wand).
15#[derive(InputObject)]
16#[cynic(
17    graphql_type = "SpellArmingInput",
18    schema_path = "../graphql/schema.graphql",
19    schema_module = "tangible_objects_model::schema",
20    rename_all = "camelCase"
21)]
22pub(crate) struct SpellArmingInput {
23    /// Which trigger arms the spell.
24    kind: SpellArmingKind,
25    /// Bounding box for `ExclusiveZone`/`StartSpot` arming.
26    region: Option<RegionInput>,
27    /// Wand pattern for `TangibleChord` arming; `None` = any
28    /// tangible.
29    pattern: Option<PatternName>,
30    /// Handprint for `FiveFingerTap` arming; `None` = any
31    /// whole-hand tap arms the spell.
32    hand: Option<HandName>,
33}
34
35impl From<&SpellArming> for SpellArmingInput {
36    /// Projects the arming enum into the tagged input shape.
37    fn from(arming: &SpellArming) -> Self {
38        match arming {
39            SpellArming::Free => Self {
40                kind: SpellArmingKind::Free,
41                region: None,
42                pattern: None,
43                hand: None,
44            },
45            SpellArming::ExclusiveZone { region } => Self {
46                kind: SpellArmingKind::ExclusiveZone,
47                region: Some(region.into()),
48                pattern: None,
49                hand: None,
50            },
51            SpellArming::StartSpot { region } => Self {
52                kind: SpellArmingKind::StartSpot,
53                region: Some(region.into()),
54                pattern: None,
55                hand: None,
56            },
57            SpellArming::FiveFingerTap { hand } => Self {
58                kind: SpellArmingKind::FiveFingerTap,
59                region: None,
60                pattern: None,
61                hand: hand.clone(),
62            },
63            SpellArming::TangibleChord { pattern } => Self {
64                kind: SpellArmingKind::TangibleChord,
65                region: None,
66                pattern: pattern.clone(),
67                hand: None,
68            },
69        }
70    }
71}
