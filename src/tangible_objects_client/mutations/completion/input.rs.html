1//! `CompletionInput` - the cynic input object for `setCompletion`,
2//! a wire DTO of [`CompletionConfig`].
3
4use smearor_commons_geometry::SmoothingAlpha;
5use tangible_objects_model::AffinityFactor;
6use tangible_objects_model::CompletionConfig;
7use tangible_objects_model::CompletionMode;
8
9/// The `CompletionInput` input object - wire DTO of
10/// [`CompletionConfig`] (`max_missing` narrows to the `Int` scalar).
11#[derive(cynic::InputObject)]
12#[cynic(
13    graphql_type = "CompletionInput",
14    schema_path = "../graphql/schema.graphql",
15    schema_module = "tangible_objects_model::schema",
16    rename_all = "camelCase"
17)]
18pub(crate) struct CompletionInput {
19    /// Claim radius as a multiple of the pattern tolerance.
20    radius_factor: f64,
21    /// Maximum number of missing pins a partial may complete.
22    max_missing: i32,
23    /// Whether claimed slots join the slot binding.
24    persist: bool,
25    /// Whether a completed partial counts as match or exact.
26    mode: CompletionMode,
27    /// Incumbent re-match tolerance widening factor.
28    affinity_factor: AffinityFactor,
29    /// Per-slot position smoothing alpha.
30    smoothing_alpha: SmoothingAlpha,
31}
32
33impl From<&CompletionConfig> for CompletionInput {
34    /// Projects the domain config into the wire input shape.
35    fn from(config: &CompletionConfig) -> Self {
36        Self {
37            radius_factor: config.radius_factor,
38            max_missing: config.max_missing as i32,
39            persist: config.persist,
40            mode: config.mode,
41            affinity_factor: config.affinity_factor,
42            smoothing_alpha: config.smoothing_alpha,
43        }
44    }
45}
