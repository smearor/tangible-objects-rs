37/// A seat mutation as the application intends it - the pieces of
38/// `UpdateSeatInput` that are meaningful to callers, folded onto the
39/// wire input in [`GraphQLClient::update_seat`].
40#[derive(Debug, Default)]
41pub struct SeatUpdate {
42    /// New seat name - `None` keeps the current one.
43    pub name: Option<SeatName>,
44    /// Rebinds the marker profile - `None` keeps the current one.
45    /// Takes precedence over [`Self::clear_marker`].
46    pub marker: Option<PatternId>,
47    /// Clears the marker binding so the seat turns virtual.
48    pub clear_marker: bool,
49    /// New pose - `None` keeps the current one.
50    pub pose: Option<SeatPose>,
51}
52
53impl GraphQLClient {
54    /// Claims a seat session for `owner` (`claimSeatSession`) -
55    /// idempotent for the same owner, rejected with
56    /// `SESSION_ALREADY_OWNED` for a foreign claim. The session name
57    /// is the roster's identity on every interface.
58    pub async fn claim_seat_session(
59        &self,
60        name: SeatSessionName,
61        owner: &Owner,
62    ) -> Result<ClaimSeatSessionResult, ClientError> {
63        let fragment: ClaimSeatSessionResultFragment = self
64            .call_mutation::<ClaimSeatSession, _, _>(
65                "claimSeatSession",
66                ClaimSeatSessionVariables {
67                    name,
68                    owner: owner.clone(),
69                },
70            )
71            .await?;
72        fragment.into_result()
73    }
74
75    /// Releases `owner`'s claim on the session
76    /// (`releaseSeatSession`) - its seats are removed with it.
77    pub async fn release_seat_session(
78        &self,
79        name: SeatSessionName,
80        owner: &Owner,
81    ) -> Result<ReleaseSeatSessionResult, ClientError> {
82        let fragment: ReleaseSeatSessionResultFragment = self
83            .call_mutation::<ReleaseSeatSession, _, _>(
84                "releaseSeatSession",
85                ReleaseSeatSessionVariables {
86                    name,
87                    owner: owner.clone(),
88                },
89            )
90            .await?;
91        fragment.into_result()
92    }
93
94    /// Adds a seat to a claimed session (`addSeat`) - `marker` set
95    /// anchors the seat to that profile's tracked tangibles; `marker:
96    /// None` creates a virtual seat at `pose`.
97    pub async fn add_seat(
98        &self,
99        session: SeatSessionName,
100        name: SeatName,
101        marker: Option<PatternId>,
102        pose: Option<SeatPose>,
103        owner: &Owner,
104    ) -> Result<AddSeatResult, ClientError> {
105        let fragment: AddSeatResultFragment = self
106            .call_mutation::<AddSeat, _, _>(
107                "addSeat",
108                AddSeatVariables {
109                    input: AddSeatInput {
110                        session,
111                        name,
112                        marker,
113                        pose: pose.map(Into::into),
114                    },
115                    owner: owner.clone(),
116                },
117            )
118            .await?;
119        fragment.into_result()
120    }
121
122    /// Mutates a seat addressed by its globally unique ID
123    /// (`updateSeat`) - `None` fields keep their value.
124    pub async fn update_seat(
125        &self,
126        id: SeatId,
127        update: SeatUpdate,
128        owner: &Owner,
129    ) -> Result<UpdateSeatResult, ClientError> {
130        let input = UpdateSeatInput {
131            name: update.name,
132            marker: update.marker,
133            clear_marker: update.clear_marker.then_some(true),
134            pose: update.pose.map(Into::into),
135        };
136        let fragment: UpdateSeatResultFragment = self
137            .call_mutation::<UpdateSeat, _, _>(
138                "updateSeat",
139                UpdateSeatVariables {
140                    id,
141                    input,
142                    owner: owner.clone(),
143                },
144            )
145            .await?;
146        fragment.into_result()
147    }
148
149    /// Removes a seat from its session (`removeSeat`).
150    pub async fn remove_seat(
151        &self,
152        id: SeatId,
153        owner: &Owner,
154    ) -> Result<RemoveSeatResult, ClientError> {
155        let fragment: RemoveSeatResultFragment = self
156            .call_mutation::<RemoveSeat, _, _>(
157                "removeSeat",
158                RemoveSeatVariables {
159                    id,
160                    owner: owner.clone(),
161                },
162            )
163            .await?;
164        fragment.into_result()
165    }
