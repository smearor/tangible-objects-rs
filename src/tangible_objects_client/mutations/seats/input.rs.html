1//! Wire input objects for the seat mutations - `AddSeatInput` and
2//! `UpdateSeatInput` as cynic DTOs of the schema's input objects.
3//!
4//! `Seat`/`SeatPose`/`TableSide` live in `smearor-commons-layout` and
5//! cannot carry cynic derives (the orphan rule blocks `cynic::Enum`
6//! on the external `TableSide`), so the nested inputs are wire
7//! shapes converted from the shared types.
8
9use smearor_commons_geometry::Fraction;
10use smearor_commons_geometry::RotationRad;
11use smearor_commons_layout::Seat;
12use smearor_commons_layout::SeatPose;
13use smearor_commons_layout::TableSide;
14use tangible_objects_model::PatternId;
15use tangible_objects_model::SeatName;
16use tangible_objects_model::SeatSessionName;
17
18/// `TableSide` wire enum - the schema enum values mirrored so cynic
19/// can serialize the organizational side inside `SeatInput`.
20#[derive(cynic::Enum, Clone, Copy)]
21#[cynic(
22    graphql_type = "TableSide",
23    schema_path = "../graphql/schema.graphql",
24    schema_module = "tangible_objects_model::schema",
25    rename_all = "SCREAMING_SNAKE_CASE"
26)]
27pub(crate) enum TableSideInput {
28    /// Bottom edge.
29    Bottom,
30    /// Left edge.
31    Left,
32    /// Top edge.
33    Top,
34    /// Right edge.
35    Right,
36}
37
38impl From<TableSide> for TableSideInput {
39    fn from(side: TableSide) -> Self {
40        match side {
41            TableSide::Bottom => Self::Bottom,
42            TableSide::Left => Self::Left,
43            TableSide::Top => Self::Top,
44            TableSide::Right => Self::Right,
45        }
46    }
47}

49/// `SeatInput` wire shape - organizational seat inside `SeatPoseInput`.
50#[derive(cynic::InputObject)]
51#[cynic(
52    graphql_type = "SeatInput",
53    schema_path = "../graphql/schema.graphql",
54    schema_module = "tangible_objects_model::schema",
55    rename_all = "camelCase"
56)]
57pub(crate) struct SeatInput {
58    /// The table edge the seat faces.
59    side: TableSideInput,
60    /// Position along the edge axis - `None` leaves distribution to
61    /// the strategy.
62    offset: Option<Fraction>,
63}
64
65impl From<Seat> for SeatInput {
66    fn from(seat: Seat) -> Self {
67        Self {
68            side: seat.side.into(),
69            offset: seat.offset,
70        }
71    }
72}

74/// `SeatPoseInput` wire shape - `seat` plus free radial orientation.
75#[derive(cynic::InputObject)]
76#[cynic(
77    graphql_type = "SeatPoseInput",
78    schema_path = "../graphql/schema.graphql",
79    schema_module = "tangible_objects_model::schema",
80    rename_all = "camelCase"
81)]
82pub(crate) struct SeatPoseInput {
83    /// Organizational alignment.
84    seat: SeatInput,
85    /// Radial orientation in radians.
86    orientation: RotationRad,
87}
88
89impl From<SeatPose> for SeatPoseInput {
90    fn from(pose: SeatPose) -> Self {
91        Self {
92            seat: pose.seat.into(),
93            orientation: pose.orientation,
94        }
95    }
96}

98/// `AddSeatInput` wire shape - session, name, optional marker and pose.
99#[derive(cynic::InputObject)]
100#[cynic(
101    graphql_type = "AddSeatInput",
102    schema_path = "../graphql/schema.graphql",
103    schema_module = "tangible_objects_model::schema",
104    rename_all = "camelCase"
105)]
106pub(crate) struct AddSeatInput {
107    /// Session receiving the seat.
108    pub(crate) session: SeatSessionName,
109    /// Seat name - unique within the session.
110    pub(crate) name: SeatName,
111    /// Marker profile anchoring the seat - `None` = virtual.
112    pub(crate) marker: Option<PatternId>,
113    /// Initial pose - placeholder for marker-backed, applied for
114    /// virtual seats.
115    pub(crate) pose: Option<SeatPoseInput>,
116}
117
118/// `UpdateSeatInput` wire shape - `None` fields keep their value.
119#[derive(cynic::InputObject, Default)]
120#[cynic(
121    graphql_type = "UpdateSeatInput",
122    schema_path = "../graphql/schema.graphql",
123    schema_module = "tangible_objects_model::schema",
124    rename_all = "camelCase"
125)]
126pub(crate) struct UpdateSeatInput {
127    /// New seat name.
128    pub(crate) name: Option<SeatName>,
129    /// Rebinds the marker profile.
130    pub(crate) marker: Option<PatternId>,
131    /// Clears the marker binding - the seat turns virtual.
132    pub(crate) clear_marker: Option<bool>,
133    /// New pose.
134    pub(crate) pose: Option<SeatPoseInput>,
135}
