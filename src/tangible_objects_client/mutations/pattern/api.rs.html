15use crate::mutations::pattern::unregister::mutation::UnregisterPattern;

22impl GraphQLClient {
23    /// Registers a profile at runtime (`registerPattern`) - the
24    /// `ProfileFile`'s geometry is flattened into the InputObject's
25    /// `[x0, y0, x1, y1, ...]` form. The result union carries the
26    /// assigned pattern ID or the typed rejection.
27    pub async fn register_pattern(
28        &self,
29        profile: &ProfileFile,
30        owner: Option<&Owner>,
31        lifetime: Lifetime,
32    ) -> Result<RegisterPatternResult, ClientError> {
33        let fragment: RegisterPatternResultFragment = self
34            .call_mutation::<RegisterPattern, _, _>(
35                "registerPattern",
36                RegisterPatternVariables {
37                    input: profile.into(),
38                    owner: owner.cloned(),
39                    lifetime: Some(lifetime),
40                },
41            )
42            .await?;
43        fragment.into_result()
44    }
45
46    /// Replaces a registered pattern's geometry (`updateProfile`).
47    pub async fn update_profile(
48        &self,
49        id: PatternId,
50        profile: &ProfileFile,
51    ) -> Result<bool, ClientError> {
52        self.call_mutation::<UpdateProfile, _, _>(
53            "updateProfile",
54            UpdateProfileVariables {
55                id,
56                input: profile.into(),
57            },
58        )
59        .await
60    }
61
62    /// Removes a registered pattern (`unregisterPattern`).
63    pub async fn unregister_pattern(&self, id: PatternId) -> Result<bool, ClientError> {
64        self.call_mutation::<UnregisterPattern, _, _>(
65            "unregisterPattern",
66            UnregisterPatternVariables { id },
67        )
68        .await
69    }
70
71    /// Drops every registration of `owner` (`unregisterOwner`).
72    pub async fn unregister_owner(&self, owner: &Owner) -> Result<bool, ClientError> {
73        self.call_mutation::<UnregisterOwner, _, _>(
74            "unregisterOwner",
75            UnregisterOwnerVariables {
76                owner: owner.clone(),
77            },
78        )
79        .await
80    }
