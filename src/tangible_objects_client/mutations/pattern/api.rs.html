31impl GraphQLClient {
32    /// Registers a profile at runtime (`registerPattern`) - the
33    /// `ProfileFile`'s geometry is flattened into the InputObject's
34    /// `[x0, y0, x1, y1, ...]` form. The result union carries the
35    /// assigned pattern ID or the typed rejection.
36    pub async fn register_pattern(
37        &self,
38        profile: &ProfileFile,
39        owner: Option<&Owner>,
40        lifetime: Lifetime,
41    ) -> Result<RegisterPatternResult, ClientError> {
42        let fragment: RegisterPatternResultFragment = self
43            .call_mutation::<RegisterPattern, _, _>(
44                "registerPattern",
45                RegisterPatternVariables {
46                    input: profile.into(),
47                    owner: owner.cloned(),
48                    lifetime: Some(lifetime),
49                },
50            )
51            .await?;
52        fragment.into_result()
53    }
54
55    /// Replaces a registered pattern's geometry (`updateProfile`).
56    pub async fn update_profile(
57        &self,
58        id: PatternId,
59        profile: &ProfileFile,
60    ) -> Result<bool, ClientError> {
61        self.call_mutation::<UpdateProfile, _, _>(
62            "updateProfile",
63            UpdateProfileVariables {
64                id,
65                input: profile.into(),
66            },
67        )
68        .await
69    }
70
71    /// Claims an owner-less profile for `owner` (`claimProfile`) - the
72    /// application adopts a statically loaded or released profile so
73    /// `unregister_owner`/disconnect cleanup releases it back to owner-less instead of unregistering it. The profile is
74    /// addressed by ID - `lookup_profile`'s match carries it; profile
75    /// names are not unique. `enable` activates the profile in the
76    /// same operation (cleanup reverts that activation). The result
77    /// union carries the claimed profile or the typed rejection
78    /// (`PATTERN_NOT_FOUND`, `PROFILE_ALREADY_OWNED`).
79    pub async fn claim_profile(
80        &self,
81        id: PatternId,
82        owner: &Owner,
83        enable: bool,
84    ) -> Result<ClaimProfileResult, ClientError> {
85        let fragment: ClaimProfileResultFragment = self
86            .call_mutation::<ClaimProfile, _, _>(
87                "claimProfile",
88                ClaimProfileVariables {
89                    id,
90                    owner: owner.clone(),
91                    enable: Some(enable),
92                },
93            )
94            .await?;
95        fragment.into_result()
96    }
97
98    /// Releases `owner`'s claim on a profile (`releaseProfile`) - the
99    /// profile stays registered and keeps matching; it only leaves the
100    /// owner's cleanup scope. `disable` deactivates the profile in
101    /// the same operation - the counterpart of `claim_profile`'s
102    /// `enable`. The result union carries the released profile or the
103    /// typed rejection (`PATTERN_NOT_FOUND`, `PROFILE_NOT_OWNED_BY`).
104    pub async fn release_profile(
105        &self,
106        id: PatternId,
107        owner: &Owner,
108        disable: bool,
109    ) -> Result<ReleaseProfileResult, ClientError> {
110        let fragment: ReleaseProfileResultFragment = self
111            .call_mutation::<ReleaseProfile, _, _>(
112                "releaseProfile",
113                ReleaseProfileVariables {
114                    id,
115                    owner: owner.clone(),
116                    disable: Some(disable),
117                },
118            )
119            .await?;
120        fragment.into_result()
121    }
122
123    /// Removes a registered pattern (`unregisterPattern`).
124    pub async fn unregister_pattern(&self, id: PatternId) -> Result<bool, ClientError> {
125        self.call_mutation::<UnregisterPattern, _, _>(
126            "unregisterPattern",
127            UnregisterPatternVariables { id },
128        )
129        .await
130    }
131
132    /// Drops every registration of `owner` (`unregisterOwner`).
133    pub async fn unregister_owner(&self, owner: &Owner) -> Result<bool, ClientError> {
134        self.call_mutation::<UnregisterOwner, _, _>(
135            "unregisterOwner",
136            UnregisterOwnerVariables {
137                owner: owner.clone(),
138            },
139        )
140        .await
141    }
