1//! `RegisterPatternInput` - the `registerPattern`/`updateProfile`
2//! input object, a figure defined by its pin positions.
3
4use async_graphql::Error;
5use async_graphql::Result;
6use smearor_commons_geometry::Millimeters;
7use smearor_commons_geometry::Vec2;
8use tangible_objects_model::FrameCount;
9use tangible_objects_model::PatternName;
10use tangible_objects_model::PatternTolerance;
11use tangible_objects_model::ProfileFile;
12use tangible_objects_model::StageName;
13
14/// Input for `registerPattern`: a figure defined by its pin positions.
15#[derive(async_graphql::InputObject)]
16pub(crate) struct RegisterPatternInput {
17    /// Figure name.
18    name: PatternName,
19    /// Flat pin coordinates `[x0, y0, x1, y1, ...]` in millimeters.
20    points: Vec<Millimeters>,
21    /// Scalar matching tolerance in millimeters.
22    tolerance_mm: PatternTolerance,
23    /// Whether a partial match may spawn this figure
24    /// (SPAWN_AS_PARTIAL.md); `null`/`false` = exact-only spawns.
25    spawn_on_partial: Option<bool>,
26    /// Frames a partial spawn must hold before confirmation;
27    /// `null` = 15.
28    spawn_stability: Option<FrameCount>,
29    /// Pipeline stage this pattern suppresses through
30    /// (EVENT_PIPELINE.md); `null` = the general stage.
31    stage: Option<StageName>,
32}
33
34impl TryFrom<&RegisterPatternInput> for ProfileFile {
35    type Error = Error;
36
37    /// Projects the wire input into the domain profile file.
38    fn try_from(input: &RegisterPatternInput) -> Result<Self> {
39        let mut profile = ProfileFile::from_points(
40            input.name.as_str(),
41            &input.points_vec2()?,
42            input.tolerance_mm,
43        );
44        profile.spawn_on_partial = input.spawn_on_partial.unwrap_or(false);
45        profile.stage = input.stage.clone();
46        if let Some(stability) = input.spawn_stability {
47            profile.spawn_stability = stability;
48        }
49        Ok(profile)
50    }
51}
52
53impl RegisterPatternInput {
54    /// Parses the flat `[x0, y0, x1, y1, ...]` list into [`Vec2`]s.
55    fn points_vec2(&self) -> Result<Vec<Vec2<Millimeters>>> {
56        if !self.points.len().is_multiple_of(2) || self.points.is_empty() {
57            return Err(Error::new(
58                "points must be a non-empty even-length flat list",
59            ));
60        }
61        Ok(self
62            .points
63            .as_chunks::<2>()
64            .0
65            .iter()
66            .map(|pair| Vec2::new(pair[0].as_f64(), pair[1].as_f64()))
67            .collect())
68    }
69}
