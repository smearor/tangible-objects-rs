1//! `AddSeatInput` - the `addSeat` input object: session, seat name,
2//! optional marker profile and optional initial pose.
3
4use smearor_commons_layout::SeatPose;
5use tangible_objects_model::PatternId;
6use tangible_objects_model::SeatName;
7use tangible_objects_model::SeatSessionName;
8
9/// Input for `addSeat`.
10#[derive(async_graphql::InputObject)]
11pub(crate) struct AddSeatInput {
12    /// Session receiving the seat - must exist and be owned by the
13    /// mutating owner.
14    pub(crate) session: SeatSessionName,
15    /// Seat name - unique within the session.
16    pub(crate) name: SeatName,
17    /// Marker profile anchoring the seat - `null` creates a virtual
18    /// seat (`pose` is then its applied pose; `null` pose falls back
19    /// to the session-default pose).
20    pub(crate) marker: Option<PatternId>,
21    /// Initial pose - the applied pose for virtual seats, the
22    /// placeholder until the marker binds for marker-backed seats.
23    pub(crate) pose: Option<SeatPose>,
24}
