1//! `RegisterHandProfileInput` - the `registerHandProfile` input
2//! object, a wire DTO of [`HandProfile`] (SPELLS.md's per-user
3//! spells).
4
5use async_graphql::Error;
6use async_graphql::Result;
7use smearor_commons_geometry::Millimeters;
8use smearor_commons_geometry::Vec2;
9use tangible_objects_model::HandName;
10use tangible_objects_model::HandProfile;
11
12/// The handprint tolerance default in millimeters when a
13/// registration leaves it unset.
14const DEFAULT_HAND_TOLERANCE_MM: f64 = 10.0;
15
16/// Input for `registerHandProfile`: a named five-contact
17/// constellation a `five-finger-tap` arming can bind spells to.
18#[derive(async_graphql::InputObject)]
19pub(crate) struct RegisterHandProfileInput {
20    /// Hand name (identity for unregister and the runtime store -
21    /// e.g. the user's name).
22    name: HandName,
23    /// Max per-distance deviation in millimeters; `null` = 10 mm.
24    tolerance: Option<Millimeters>,
25    /// The recorded contact constellation - flat `[x0, y0, x1, y1,
26    /// ...]` pairs in millimeters (same shape as
27    /// `registerPattern.points`).
28    points: Vec<Millimeters>,
29}
30
31impl TryFrom<&RegisterHandProfileInput> for HandProfile {
32    type Error = Error;
33
34    /// Projects the wire input into the domain handprint - validates
35    /// the point arity.
36    fn try_from(input: &RegisterHandProfileInput) -> Result<Self> {
37        if input.points.len() < 2 || !input.points.len().is_multiple_of(2) {
38            return Err(Error::new("hand point list must be flat [x, y] pairs"));
39        }
40        let (pairs, _) = input.points.as_chunks::<2>();
41        let points: Vec<Vec2<Millimeters>> = pairs
42            .iter()
43            .map(|pair| Vec2::from_mm(pair[0], pair[1]))
44            .collect();
45        Ok(Self::from_points(
46            input.name.clone(),
47            &points,
48            input
49                .tolerance
50                .unwrap_or(Millimeters::new(DEFAULT_HAND_TOLERANCE_MM)),
51        ))
52    }
53}
