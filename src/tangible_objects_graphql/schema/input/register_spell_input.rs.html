1//! `RegisterSpellInput` - the `registerSpell` input object, a wire
2//! DTO of [`Spell`] (SPELLS.md).

4use async_graphql::Error;
5use async_graphql::Result;
6use smearor_commons_geometry::Millimeters;
7use smearor_commons_geometry::Vec2;
8use tangible_objects_model::FingerCount;
9use tangible_objects_model::Spell;
10use tangible_objects_model::SpellArming;
11use tangible_objects_model::SpellName;
12use tangible_objects_model::StrokePolyline;
13
14use super::spell_arming_input::SpellArmingInput;

16/// The corridor default in millimeters when a registration leaves it
17/// unset - matches the `spells.d` example value.
18const DEFAULT_SPELL_CORRIDOR_MM: f64 = 12.0;
19
20/// Input for `registerSpell`: a gesture defined by per-finger
21/// template strokes.
22#[derive(async_graphql::InputObject)]
23pub(crate) struct RegisterSpellInput {
24    /// Spell name (identity for unregister and the runtime store).
25    name: SpellName,
26    /// Max per-stroke deviation in millimeters; `null` = 12 mm.
27    corridor: Option<Millimeters>,
28    /// Whether the gesture may be drawn larger/smaller than the
29    /// template; `null` = true.
30    scale_invariant: Option<bool>,
31    /// Whether the spell matches; `null` = true.
32    enabled: Option<bool>,
33    /// Per-finger template strokes - `strokes[i]` is finger i's flat
34    /// polyline `[x0, y0, x1, y1, ...]` in millimeters (same shape as
35    /// `registerPattern.points`). The count defines `fingers`.
36    strokes: Vec<Vec<Millimeters>>,
37    /// How the gesture is initiated; `null` = `free`.
38    arming: Option<SpellArmingInput>,
39}
40
41impl TryFrom<&RegisterSpellInput> for Spell {
42    type Error = Error;
43
44    /// Projects the wire input into the domain spell - validates the
45    /// stroke count and point arity.
46    fn try_from(input: &RegisterSpellInput) -> Result<Self> {
47        if input.strokes.is_empty() || input.strokes.len() > 5 {
48            return Err(Error::new("spell needs 1..=5 stroke templates"));
49        }
50        let mut strokes = Vec::with_capacity(input.strokes.len());
51        for stroke in &input.strokes {
52            if stroke.len() < 2 {
53                return Err(Error::new("stroke needs at least 2 points"));
54            }
55            if stroke.len() % 2 != 0 {
56                return Err(Error::new("stroke point list must be flat [x, y] pairs"));
57            }
58            let (pairs, _) = stroke.as_chunks::<2>();
59            let points = pairs
60                .iter()
61                .map(|pair| Vec2::from_mm(pair[0], pair[1]))
62                .collect();
63            strokes.push(StrokePolyline::new(points));
64        }
65        Ok(Self {
66            name: input.name.clone(),
67            fingers: FingerCount::new(strokes.len()),
68            corridor: input
69                .corridor
70                .unwrap_or(Millimeters::new(DEFAULT_SPELL_CORRIDOR_MM)),
71            scale_invariant: input.scale_invariant.unwrap_or(true),
72            enabled: input.enabled.unwrap_or(true),
73            strokes,
74            arming: input
75                .arming
76                .as_ref()
77                .map(SpellArming::try_from)
78                .transpose()?
79                .unwrap_or(SpellArming::Free),
80        })
81    }
82}
