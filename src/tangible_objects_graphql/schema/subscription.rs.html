18/// Live domain event stream.
19///
20/// Streams are deltas only - a client that needs initial state runs the
21/// `snapshot`/`profiles` queries first, then subscribes (SUBSCRIPTIONS.md:
22/// no snapshot push on connect). Broadcast lag surfaces as a `LAGGED`
23/// control event carrying `skipped`, never as a silent gap or failure.
24pub struct SubscriptionRoot;
25
26impl SubscriptionRoot {
27    /// Shared stream setup: subscribe to the broadcast channel, translate
28    /// lag into `LAGGED` events, then run the server-side filter
29    /// (kind + ids + `TOUCH_POINTS_UPDATED` throttle).
30    fn stream(handle: &EngineHandle, filter: EventFilter) -> impl Stream<Item = TangibleEvent> {
31        let mut filter = filter;
32        BroadcastStream::new(handle.subscribe())
33            .filter_map(|result| match result {
34                Ok(event) => Some(TangibleEvent::from(&event)),
35                Err(BroadcastStreamRecvError::Lagged(skipped)) => {
36                    Some(TangibleEvent::lagged(EventCount::from(skipped)))
37                }
38            })
39            .filter(move |event| filter.accepts(event))
40    }
41
42    /// Validates the `kinds` filter: an empty list would silently mean
43    /// "everything", which the schema rejects instead.
44    fn check_kinds(kinds: &[StreamEvent]) -> Result<()> {
45        if kinds.is_empty() {
46            return Err(async_graphql::Error::new(
47                "kinds must not be empty - request at least one StreamEvent kind",
48            ));
49        }
50        Ok(())
51    }
52}
53
54#[Subscription]
55impl SubscriptionRoot {
56    /// The single subscription entry point: streams the requested
57    /// `StreamEvent` kinds, narrowed by `patternId`/`setId`; `maxHz`
58    /// throttles the high-rate `TOUCH_POINTS_UPDATED` kind (default 30).
59    async fn event_stream(
60        &self,
61        ctx: &Context<'_>,
62        kinds: Vec<StreamEvent>,
63        pattern_id: Option<PatternId>,
64        set_id: Option<TangibleSetId>,
65        max_hz: Option<u32>,
66    ) -> Result<impl Stream<Item = TangibleEvent>> {
67        Self::check_kinds(&kinds)?;
68        let handle = ctx.data::<EngineHandle>()?;
69        Ok(Self::stream(
70            handle,
71            EventFilter::new(&kinds, pattern_id, set_id, None, max_hz),
72        ))
73    }
74
75    /// Tangible lifecycle + motion stream - `kinds` preset to
76    /// `TANGIBLE_ADDED`/`TANGIBLE_MOVED`/`TANGIBLE_REMOVED`; `patternId`
77    /// narrows to one figure.
78    async fn tangible_events(
79        &self,
80        ctx: &Context<'_>,
81        pattern_id: Option<PatternId>,
82    ) -> Result<impl Stream<Item = TangibleEvent>> {
83        let handle = ctx.data::<EngineHandle>()?;
84        Ok(Self::stream(
85            handle,
86            EventFilter::new(
87                &[
88                    StreamEvent::TangibleAdded,
89                    StreamEvent::TangibleMoved,
90                    StreamEvent::TangibleRemoved,
91                ],
92                pattern_id,
93                None,
94                None,
95                None,
96            ),
97        ))
98    }
99
100    /// Occupancy field stream: `FIELD_PLACED`/`FIELD_LIFTED`/
101    /// `FIELD_COUNT_CHANGED`, narrowed to one field set when `set` names it.
102    async fn field_events(
103        &self,
104        ctx: &Context<'_>,
105        set: Option<FieldSetName>,
106    ) -> Result<impl Stream<Item = TangibleEvent>> {
107        let handle = ctx.data::<EngineHandle>()?;
108        Ok(Self::stream(
109            handle,
110            EventFilter::new(
111                &[
112                    StreamEvent::FieldPlaced,
113                    StreamEvent::FieldLifted,
114                    StreamEvent::FieldCountChanged,
115                ],
116                None,
117                None,
118                set,
119                None,
120            ),
121        ))
122    }
123
124    /// Streams every domain event unfiltered.
125    #[graphql(deprecation = "use eventStream(kinds: [...]) - an explicit filter documents intent")]
126    async fn events(&self, ctx: &Context<'_>) -> Result<impl Stream<Item = TangibleEvent>> {
127        let handle = ctx.data::<EngineHandle>()?;
128        Ok(Self::stream(
129            handle,
130            EventFilter::new(
131                &[
132                    StreamEvent::TangibleAdded,
133                    StreamEvent::TangibleMoved,
134                    StreamEvent::TangibleRemoved,
135                    StreamEvent::TouchPointsUpdated,
136                    StreamEvent::FieldPlaced,
137                    StreamEvent::FieldLifted,
138                    StreamEvent::FieldCountChanged,
139                    StreamEvent::SetAdded,
140                    StreamEvent::SetRemoved,
141                    StreamEvent::ObjectAdded,
142                    StreamEvent::ObjectRemoved,
143                    StreamEvent::ObjectMoved,
144                    StreamEvent::ProfileRegistered,
145                    StreamEvent::ProfileUnregistered,
146                    StreamEvent::FieldSetRegistered,
147                    StreamEvent::FieldSetUnregistered,
148                    StreamEvent::CompletionChanged,
149                    StreamEvent::FilterModeChanged,
150                    StreamEvent::RecordingStarted,
151                    StreamEvent::RecordingStopped,
152                    StreamEvent::ConfigReloaded,
153                ],
154                None,
155                None,
156                None,
157                None,
158            ),
159        ))
160    }
