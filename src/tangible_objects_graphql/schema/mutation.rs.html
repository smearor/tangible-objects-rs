42    /// Switches the filter proxy's suppression policy
43    /// (`off`/`immediate`/`gated`/`guarded`/`tangiblesOnly`); `stability`
44    /// overrides the consecutive-matched-frames gate, `null` keeps it.
45    /// `stage` narrows the switch to that scope's tangible stage,
46    /// `null` broadcasts to all.
47    async fn set_filter_mode(
48        &self,
49        ctx: &Context<'_>,
50        mode: FilterMode,
51        stability: Option<FrameCount>,
52        stage: Option<StageName>,
53    ) -> Result<bool> {
54        ctx.data::<EngineHandle>()?
55            .send_command(ControlCommand::SetFilterMode {
56                mode,
57                stability,
58                stage,
59            })
60            .await
61            .map_err(|_| Error::new("control channel closed: device loop is not running"))?;
62        Ok(true)
63    }

214    /// Registers a figure profile at runtime (DYNAMIC_PROFILES.md). The
215    /// result union carries the assigned pattern ID or the typed
216    /// rejection (`FINGERPRINT_COLLISION`, `PATTERN_NOT_FOUND`).
217    async fn register_pattern(
218        &self,
219        ctx: &Context<'_>,
220        input: RegisterPatternInput,
221        owner: Option<Owner>,
222        lifetime: Option<Lifetime>,
223    ) -> Result<RegisterPatternResult> {
224        let profile = ProfileFile::try_from(&input)?;
225        let (command, rx) = ControlCommand::register(
226            Registration::Pattern(profile),
227            owner,
228            lifetime.unwrap_or_default(),
229        );
230        ctx.data::<EngineHandle>()?
231            .send_command(command)
232            .await
233            .map_err(|_| Error::new("control channel closed: device loop is not running"))?;
234        match rx.await {
235            Ok(Ok(RegistrationApplied::Profile(pattern_id))) => {
236                Ok(RegisterPatternResult::Registered(PatternRegistered {
237                    pattern_id,
238                }))
239            }
240            Ok(Ok(_)) => Err(Error::new(
241                "device loop returned a mismatched registration result",
242            )),
243            Ok(Err(error)) => Ok(RegisterPatternResult::Rejected(error.into())),
244            Err(_) => Err(Error::new("device loop closed the registration reply")),
245        }
246    }
