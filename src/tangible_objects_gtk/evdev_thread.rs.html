38/// Consecutive same-size frames before a point set counts as stable

42/// Field-set card rows for `SharedState` - set name, registration
43/// owner, enabled flag and field count per registered set.
44fn field_set_rows(engine: &Engine) -> Vec<FieldSetRow> {
45    engine
46        .field_sets()
47        .iter()
48        .map(|set| FieldSetRow {
49            name: set.name.clone(),
50            owner: engine.field_set_owner(&set.name).cloned(),
51            enabled: set.enabled,
52            field_count: set.fields.len(),
53        })
54        .collect()
55}

57/// Device loop thread: opens the evdev device, runs the engine per
58/// `SYN_REPORT` frame, publishes snapshots/events through the runtime
59/// and writes the GUI's [`SharedState`]. Daemon semantics (interfaces,
60/// control commands, recording) arrive via [`GuiHooks`].
61/// Virtual suppression/forwarding counts - what the filter proxy
62/// would do to this frame under `mode`. The GUI has no proxy; the
63/// counters show ingest-filtered plus mode-suppressed contacts so the
64/// status line matches `--filter` semantics.
65fn virtual_suppression(
66    engine: &Engine,
67    mode: FilterMode,
68    touches: TouchSet<'_>,
69) -> (ContactCount, ContactCount) {
70    let total = touches.len() as u32;
71    let suppressed = match mode {
72        FilterMode::Off => 0,
73        FilterMode::Immediate | FilterMode::Gated | FilterMode::Guarded => {
74            let bound = engine.bound_slots();
75            let field = engine.field_suppressed_slots(touches);
76            touches
77                .slots_where(|t| bound.contains(&t.slot) || field.contains(&t.slot))
78                .len() as u32
79        }
80        FilterMode::TangiblesOnly => total,
81    };
82    (suppressed.into(), (total - suppressed).into())
83}

85pub fn evdev_thread(
86    device_path: String,
87    patterns: PatternBuf,
88    state: Arc<Mutex<SharedState>>,
89    config: DetectionConfig,
90    rotation: DisplayRotation,
91    device_profiles: Option<Arc<PathBuf>>,
92    hooks: GuiHooks,
93) {
94    let mut device = match TouchDevice::open(std::path::Path::new(&device_path)) {
95        Ok(d) => d,
96        Err(e) => {
97            if let Ok(mut s) = state.lock() {
98                s.device_error = Some(fl!(
99                    language_loader(),
100                    "device-error",
101                    error = e.to_string()
102                ));
103            }
104            return;
105        }
106    };
107    if config.import_slots {
108        device.queue_state_import();
109    }
110
111    let mut engine = Engine::new(patterns);
112    if let Some(mm) = config.edge {
113        engine.set_edge_margin(mm);
114    }
115    let (
116        dead_zones_visible,
117        spells_visible,
118        background_alpha,
119        edge_margin_visible,
120        fields_visible,
121        ruler_visible,
122        positions_visible,
123        filter_mode,
124        touch_passthrough,
125    ) = state
126        .lock()
127        .map(|s| {
128            (
129                s.dead_zones_visible,
130                s.spells_visible,
131                s.background_alpha.percent(),
132                s.edge_margin_visible,
133                s.fields_visible,
134                s.ruler_visible,
135                s.positions_visible,
136                s.filter_mode,
137                s.touch_passthrough,
138            )
139        })
140        .unwrap_or_default();
141    engine.set_fields(config.fields);
142    engine.set_dead_zones(config.dead_zones);
143    engine.set_completion(config.completion);
144
145    // Event hub for interface crates (GraphQL/D-Bus/MCP) - the GUI runs
146    // its own device loop, so it owns the runtime side like the
147    // headless modes do.
148    let profiles: Vec<ProfileInfo> = engine.patterns().iter().map(ProfileInfo::from).collect();
149    let (mut runtime, handle) = EngineRuntime::new(Vec::new(), profiles);
150    runtime.set_completion(config.completion);
151    runtime.set_display_rotation(rotation);
152    runtime.set_field_sets(engine.field_sets().to_vec());
153    runtime.set_dead_zones(engine.dead_zones().to_vec());
154    if let Ok(mut s) = state.lock() {
155        s.command_handle = Some(handle.clone());
156    }
157    runtime.set_dead_zones_visible(dead_zones_visible);
158    runtime.set_spells_visible(spells_visible);
159    runtime.set_background_alpha(background_alpha.into());
160    runtime.set_edge_margin_visible(edge_margin_visible);
161    runtime.set_fields_visible(fields_visible);
162    runtime.set_ruler_visible(ruler_visible);
163    runtime.set_positions_visible(positions_visible);
164    runtime.set_filter_mode(filter_mode);
165    runtime.set_touch_passthrough(touch_passthrough);
166    (hooks.spawn_interfaces)(&handle);
167
168    let mut touch_state = TouchState::new();
169    if let Some(cal) =
170        (hooks.effective_calibration)(&device, device_profiles.as_deref().map(PathBuf::as_path))
171    {
172        touch_state.set_calibration(cal);
173        engine.set_calibration(cal);
174        if let Ok(mut s) = state.lock() {
175            s.calibration = Some(cal);
176            // The effective calibration differs from the firmware
177            // values exactly when a device profile overrode them.
178            s.calibration_source = Some(if device.calibration() == Some(cal) {
179                CalibrationSource::Device
180            } else {
181                CalibrationSource::Profile
182            });
183        }
184    }
185    // Report the calibration the engine actually uses (profile override
186    // applied), not the firmware-reported one - clients size their field
187    // sets from these millimeter bounds.
188    let device_info = DeviceInfo::of(device.name(), engine.calibration().as_ref());
189    if let Ok(mut s) = state.lock() {
190        s.device_info = Some(device_info.clone());
191    }
192    runtime.set_device_info(device_info);
193
194    let mut stable_count: Option<usize> = None;
195    let mut stable_frames: u32 = 0;
196    let device_fd = device.as_raw_fd();
197    let session_start = std::time::Instant::now();
198    let mut frame_no = 0u64;
199    let mut recording: Option<RecordingSession> = None;
200
201    loop {
202        // Periodic wake for queued control commands even without input.
203        let mut pfd = libc::pollfd {
204            fd: device_fd,
205            events: libc::POLLIN,
206            revents: 0,
207        };
208        let ready = unsafe { libc::poll(&mut pfd, 1, 250) };
209        if ready < 0 {
210            if unsafe { *libc::__errno_location() } == libc::EINTR {
211                continue;
212            }
213            error!("poll error - evdev loop exits");
214            break;
215        }
216        if ready == 0 {
217            (hooks.drain_control_commands)(
218                &mut runtime,
219                &mut recording,
220                &mut engine,
221                device.name(),
222            );
223            runtime.set_recording_status(recording.as_ref().map(hooks.status_of_session));
224            if let Ok(mut s) = state.lock() {
225                s.message = runtime.message().map(str::to_string);
226                s.visible = runtime.gui_visible();
227                s.rotation = runtime.display_rotation();
228                s.dead_zones_visible = runtime.dead_zones_visible();
229                s.spells_visible = runtime.spells_visible();
230                s.dead_zones = engine.dead_zones().to_vec();
231                s.trajectories = engine
232                    .gesture_trajectories()
233                    .values()
234                    .map(|traj| traj.points.clone())
235                    .collect();
236                s.edge_margin = engine.edge_margin();
237                s.edge_margin_visible = runtime.edge_margin_visible();
238                s.fields_visible = runtime.fields_visible();
239                s.field_sets = engine.field_sets().to_vec();
240                s.field_set_rows = field_set_rows(&engine);
241                s.profiles = handle.profiles();
242                s.ruler_visible = runtime.ruler_visible();
243                s.grid_visible = runtime.grid_visible();
244                s.positions_visible = runtime.positions_visible();
245                s.touch_passthrough = runtime.touch_passthrough();
246                s.filter_mode = runtime.filter_mode();
247                s.completion = engine.completion();
248                s.legacy_mode = runtime.legacy_mode();
249                s.filter_stability = runtime
250                    .filter_stability()
251                    .map(|frames| frames.as_u64())
252                    .unwrap_or(0);
253                s.metrics = runtime.metrics().clone();
254                s.pattern_styles = engine
255                    .patterns()
256                    .iter()
257                    .map(|pattern| crate::pattern_style::PatternStyle {
258                        pattern_id: pattern.id,
259                        color: pattern.color,
260                        shape: pattern.shape.clone(),
261                    })
262                    .collect();
263                s.incoming_contacts = engine.incoming_contacts();
264                s.edge_filtered = engine.edge_filtered();
265                let (mode_suppressed, forwarded) =
266                    virtual_suppression(&engine, runtime.filter_mode(), (&*s.touches).into());
267                s.mode_suppressed = mode_suppressed;
268                s.forwarded = forwarded;
269                s.field_set_count = engine.field_set_count();
270                s.field_count = engine.field_count();
271                s.occupied_fields = engine.occupied_field_count();
272                if let Some(request) = s.background_alpha_request.take() {
273                    runtime.set_background_alpha(request.percent().into());
274                }
275                s.background_alpha =
276                    BackgroundAlpha::from_percent(runtime.background_alpha().percent());
277            }
278            runtime.publish_snapshot(engine.touches(), engine.tracked());
279            continue;
280        }
281
282        let events = match device.fetch_events() {
283            Ok(events) => events,
284            Err(e) => {
285                error!(error = %e, "device read failed - evdev loop exits");
286                break;
287            }
288        };
289        for raw in events {
290            if let Some(session) = &recording {
291                session.record_event(&raw, session_start.elapsed().as_secs_f64());
292            }
293            touch_state.process(&raw);
294
295            // The engine builds its own touch state from every event —
296            // feeding it only SYN_REPORTs leaves that state empty and no
297            // figure ever matches.
298            let frame = if mode_is_live(&state) {
299                if recording.is_some() {
300                    engine.process_with_diagnostics(&raw)
301                } else {
302                    engine.process(&raw)
303                }
304            } else {
305                None
306            };
307            if let Some(ref frame) = frame {
308                frame_no += 1;
309                runtime.publish(frame, engine.tracked());
310                (hooks.drain_control_commands)(
311                    &mut runtime,
312                    &mut recording,
313                    &mut engine,
314                    device.name(),
315                );
316                // A finished spell-training session queues its spell
317                // as a persistent registration through the normal
318                // command path (drain → `spells.d` + lifecycle event).
319                if let Some(spell) = engine.take_trained_spell() {
320                    info!(name = %spell.name, "spell training completed");
321                    let (command, _rx) = ControlCommand::register(
322                        Registration::Spell(spell),
323                        None,
324                        Lifetime::Persistent,
325                    );
326                    let _ = handle.try_send_command(command);
327                }
328                runtime.set_recording_status(recording.as_ref().map(hooks.status_of_session));
329                if let Ok(mut s) = state.lock() {
330                    s.message = runtime.message().map(str::to_string);
331                    s.visible = runtime.gui_visible();
332                    s.rotation = runtime.display_rotation();
333                    s.dead_zones_visible = runtime.dead_zones_visible();
334                    s.spells_visible = runtime.spells_visible();
335                    s.dead_zones = engine.dead_zones().to_vec();
336                    s.trajectories = engine
337                        .gesture_trajectories()
338                        .values()
339                        .map(|traj| traj.points.clone())
340                        .collect();
341                    if let Some(matched) = frame.spell_matches.first() {
342                        s.last_spell = Some((matched.clone(), std::time::Instant::now()));
343                    }
344                    s.dead_zone_filtered = frame.dead_zone_filtered;
345                    s.edge_margin = engine.edge_margin();
346                    s.edge_margin_visible = runtime.edge_margin_visible();
347                    s.fields_visible = runtime.fields_visible();
348                    s.field_sets = engine.field_sets().to_vec();
349                    s.field_set_rows = field_set_rows(&engine);
350                    s.profiles = handle.profiles();
351                    s.ruler_visible = runtime.ruler_visible();
352                    s.grid_visible = runtime.grid_visible();
353                    s.positions_visible = runtime.positions_visible();
354                    s.touch_passthrough = runtime.touch_passthrough();
355                    s.filter_mode = runtime.filter_mode();
356                    s.completion = engine.completion();
357                    s.legacy_mode = runtime.legacy_mode();
358                    s.filter_stability = runtime
359                        .filter_stability()
360                        .map(|frames| frames.as_u64())
361                        .unwrap_or(0);
362                    s.metrics = runtime.metrics().clone();
363                    s.pattern_styles = engine
364                        .patterns()
365                        .iter()
366                        .map(|pattern| crate::pattern_style::PatternStyle {
367                            pattern_id: pattern.id,
368                            color: pattern.color,
369                            shape: pattern.shape.clone(),
370                        })
371                        .collect();
372                    s.incoming_contacts = engine.incoming_contacts();
373                    s.edge_filtered = engine.edge_filtered();
374                    let (mode_suppressed, forwarded) = virtual_suppression(
375                        &engine,
376                        runtime.filter_mode(),
377                        (&frame.touches).into(),
378                    );
379                    s.mode_suppressed = mode_suppressed;
380                    s.forwarded = forwarded;
381                    s.field_set_count = engine.field_set_count();
382                    s.field_count = engine.field_count();
383                    s.occupied_fields = engine.occupied_field_count();
384                    if let Some(request) = s.background_alpha_request.take() {
385                        runtime.set_background_alpha(request.percent().into());
386                    }
387                    s.background_alpha =
388                        BackgroundAlpha::from_percent(runtime.background_alpha().percent());
389                }
390                if let Some(profile) = runtime.training_tick(&frame.touches) {
391                    info!(name = %profile.name, "training completed");
392                    runtime.set_message(None);
393                    (hooks.register_profile)(&mut runtime, &mut engine, profile);
394                }
395                if let Some(session) = &mut recording {
396                    session.record_frame(
397                        FrameNumber::new(frame_no),
398                        Milliseconds::from(session_start.elapsed()),
399                        frame,
400                        engine.tracked(),
401                    );
402                }
403            }
404
405            if raw.is_syn_report() {
406                // In live mode the frame's own touch list is authoritative:
407                // `touch_indices` in matches index into it (slot-sorted).
408                // The GUI's touch_state (HashMap order) would misalign them
409                // and draw figure lines across objects.
410                let touches = match &frame {
411                    Some(engine_frame) => engine_frame.touches.clone(),
412                    None => touch_state.active_touches(),
413                };
414                let count = touches.len();
415                let matches = frame.map(|f| f.matches).unwrap_or_default();
416
417                // Stability check (for training mode).
418                if stable_count == Some(count) && count >= 3 {
419                    stable_frames += 1;
420                } else {
421                    stable_count = Some(count);
422                    stable_frames = 0;
423                }
424
425                if let Ok(mut s) = state.lock() {
426                    s.touches = touches;
427                    s.tracked = engine.tracked().to_vec();
428                    s.matches = matches;
429                    s.stable_frames = stable_frames;
430                    s.stable = stable_frames >= STABLE_THRESHOLD && count >= 3;
431                }
432            }
433        }
434    }
