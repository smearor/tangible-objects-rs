33/// Consecutive same-size frames before a point set counts as stable
34/// for profile capture.
35pub(crate) const STABLE_THRESHOLD: u32 = 10;
36
37/// Device loop thread: opens the evdev device, runs the engine per
38/// `SYN_REPORT` frame, publishes snapshots/events through the runtime
39/// and writes the GUI's [`SharedState`]. Daemon semantics (interfaces,
40/// control commands, recording) arrive via [`GuiHooks`].
41/// Virtual suppression/forwarding counts - what the filter proxy
42/// would do to this frame under `mode`. The GUI has no proxy; the
43/// counters show ingest-filtered plus mode-suppressed contacts so the
44/// status line matches `--filter` semantics.
45fn virtual_suppression(
46    engine: &Engine,
47    mode: FilterMode,
48    touches: TouchSet<'_>,
49) -> (ContactCount, ContactCount) {
50    let total = touches.len() as u32;
51    let suppressed = match mode {
52        FilterMode::Off => 0,
53        FilterMode::Immediate | FilterMode::Gated | FilterMode::Guarded => {
54            let bound = engine.bound_slots();
55            let field = engine.field_suppressed_slots(touches);
56            touches
57                .slots_where(|t| bound.contains(&t.slot) || field.contains(&t.slot))
58                .len() as u32
59        }
60        FilterMode::TangiblesOnly => total,
61    };
62    (suppressed.into(), (total - suppressed).into())
63}

65pub fn evdev_thread(
66    device_path: String,
67    patterns: PatternBuf,
68    state: Arc<Mutex<SharedState>>,
69    config: DetectionConfig,
70    rotation: DisplayRotation,
71    device_profiles: Option<Arc<PathBuf>>,
72    hooks: GuiHooks,
73) {
74    let mut device = match TouchDevice::open(std::path::Path::new(&device_path)) {
75        Ok(d) => d,
76        Err(e) => {
77            if let Ok(mut s) = state.lock() {
78                s.device_error = Some(fl!(
79                    language_loader(),
80                    "device-error",
81                    error = e.to_string()
82                ));
83            }
84            return;
85        }
86    };
87    if config.import_slots {
88        device.queue_state_import();
89    }
90
91    let mut engine = Engine::new(patterns);
92    if let Some(mm) = config.edge {
93        engine.set_edge_margin(mm);
94    }
95    let (
96        dead_zones_visible,
97        background_alpha,
98        edge_margin_visible,
99        fields_visible,
100        ruler_visible,
101        positions_visible,
102        filter_mode,
103        touch_passthrough,
104    ) = state
105        .lock()
106        .map(|s| {
107            (
108                s.dead_zones_visible,
109                s.background_alpha.percent(),
110                s.edge_margin_visible,
111                s.fields_visible,
112                s.ruler_visible,
113                s.positions_visible,
114                s.filter_mode,
115                s.touch_passthrough,
116            )
117        })
118        .unwrap_or_default();
119    engine.set_fields(config.fields);
120    engine.set_dead_zones(config.dead_zones);
121    engine.set_completion(config.completion);
122
123    // Event hub for interface crates (GraphQL/D-Bus/MCP) - the GUI runs
124    // its own device loop, so it owns the runtime side like the
125    // headless modes do.
126    let profiles: Vec<ProfileInfo> = engine.patterns().iter().map(ProfileInfo::from).collect();
127    let (mut runtime, handle) = EngineRuntime::new(Vec::new(), profiles);
128    runtime.set_completion(config.completion);
129    runtime.set_display_rotation(rotation);
130    runtime.set_field_sets(engine.field_sets().to_vec());
131    runtime.set_dead_zones(engine.dead_zones().to_vec());
132    if let Ok(mut s) = state.lock() {
133        s.command_handle = Some(handle.clone());
134    }
135    runtime.set_dead_zones_visible(dead_zones_visible);
136    runtime.set_background_alpha(background_alpha.into());
137    runtime.set_edge_margin_visible(edge_margin_visible);
138    runtime.set_fields_visible(fields_visible);
139    runtime.set_ruler_visible(ruler_visible);
140    runtime.set_positions_visible(positions_visible);
141    runtime.set_filter_mode(filter_mode);
142    runtime.set_touch_passthrough(touch_passthrough);
143    (hooks.spawn_interfaces)(&handle);
144
145    let mut touch_state = TouchState::new();
146    if let Some(cal) =
147        (hooks.effective_calibration)(&device, device_profiles.as_deref().map(PathBuf::as_path))
148    {
149        touch_state.set_calibration(cal);
150        engine.set_calibration(cal);
151        if let Ok(mut s) = state.lock() {
152            s.calibration = Some(cal);
153        }
154    }
155    // Report the calibration the engine actually uses (profile override
156    // applied), not the firmware-reported one - clients size their field
157    // sets from these millimeter bounds.
158    runtime.set_device_info(DeviceInfo::of(device.name(), engine.calibration().as_ref()));
159
160    let mut stable_count: Option<usize> = None;
161    let mut stable_frames: u32 = 0;
162    let device_fd = device.as_raw_fd();
163    let session_start = std::time::Instant::now();
164    let mut frame_no = 0u64;
165    let mut recording: Option<RecordingSession> = None;
166
167    loop {
168        // Periodic wake for queued control commands even without input.
169        let mut pfd = libc::pollfd {
170            fd: device_fd,
171            events: libc::POLLIN,
172            revents: 0,
173        };
174        let ready = unsafe { libc::poll(&mut pfd, 1, 250) };
175        if ready < 0 {
176            if unsafe { *libc::__errno_location() } == libc::EINTR {
177                continue;
178            }
179            error!("poll error - evdev loop exits");
180            break;
181        }
182        if ready == 0 {
183            (hooks.drain_control_commands)(
184                &mut runtime,
185                &mut recording,
186                &mut engine,
187                device.name(),
188            );
189            runtime.set_recording_status(recording.as_ref().map(hooks.status_of_session));
190            if let Ok(mut s) = state.lock() {
191                s.message = runtime.message().map(str::to_string);
192                s.visible = runtime.gui_visible();
193                s.rotation = runtime.display_rotation();
194                s.dead_zones_visible = runtime.dead_zones_visible();
195                s.dead_zones = engine.dead_zones().to_vec();
196                s.edge_margin = engine.edge_margin();
197                s.edge_margin_visible = runtime.edge_margin_visible();
198                s.fields_visible = runtime.fields_visible();
199                s.field_sets = engine.field_sets().to_vec();
200                s.ruler_visible = runtime.ruler_visible();
201                s.positions_visible = runtime.positions_visible();
202                s.touch_passthrough = runtime.touch_passthrough();
203                s.filter_mode = runtime.filter_mode();
204                s.incoming_contacts = engine.incoming_contacts();
205                s.edge_filtered = engine.edge_filtered();
206                let (mode_suppressed, forwarded) =
207                    virtual_suppression(&engine, runtime.filter_mode(), (&*s.touches).into());
208                s.mode_suppressed = mode_suppressed;
209                s.forwarded = forwarded;
210                s.field_set_count = engine.field_set_count();
211                s.field_count = engine.field_count();
212                s.occupied_fields = engine.occupied_field_count();
213                if let Some(request) = s.background_alpha_request.take() {
214                    runtime.set_background_alpha(request.percent().into());
215                }
216                s.background_alpha =
217                    BackgroundAlpha::from_percent(runtime.background_alpha().percent());
218            }
219            runtime.publish_snapshot(engine.touches(), engine.tracked());
220            continue;
221        }
222
223        let events = match device.fetch_events() {
224            Ok(events) => events,
225            Err(e) => {
226                error!(error = %e, "device read failed - evdev loop exits");
227                break;
228            }
229        };
230        for raw in events {
231            if let Some(session) = &recording {
232                session.record_event(&raw, session_start.elapsed().as_secs_f64());
233            }
234            touch_state.process(&raw);
235
236            // The engine builds its own touch state from every event —
237            // feeding it only SYN_REPORTs leaves that state empty and no
238            // figure ever matches.
239            let frame = if mode_is_live(&state) {
240                if recording.is_some() {
241                    engine.process_with_diagnostics(&raw)
242                } else {
243                    engine.process(&raw)
244                }
245            } else {
246                None
247            };
248            if let Some(ref frame) = frame {
249                frame_no += 1;
250                runtime.publish(frame, engine.tracked());
251                (hooks.drain_control_commands)(
252                    &mut runtime,
253                    &mut recording,
254                    &mut engine,
255                    device.name(),
256                );
257                runtime.set_recording_status(recording.as_ref().map(hooks.status_of_session));
258                if let Ok(mut s) = state.lock() {
259                    s.message = runtime.message().map(str::to_string);
260                    s.visible = runtime.gui_visible();
261                    s.rotation = runtime.display_rotation();
262                    s.dead_zones_visible = runtime.dead_zones_visible();
263                    s.dead_zones = engine.dead_zones().to_vec();
264                    s.dead_zone_filtered = frame.dead_zone_filtered;
265                    s.edge_margin = engine.edge_margin();
266                    s.edge_margin_visible = runtime.edge_margin_visible();
267                    s.fields_visible = runtime.fields_visible();
268                    s.field_sets = engine.field_sets().to_vec();
269                    s.ruler_visible = runtime.ruler_visible();
270                    s.positions_visible = runtime.positions_visible();
271                    s.touch_passthrough = runtime.touch_passthrough();
272                    s.touch_passthrough = runtime.touch_passthrough();
273                    s.filter_mode = runtime.filter_mode();
274                    s.incoming_contacts = engine.incoming_contacts();
275                    s.edge_filtered = engine.edge_filtered();
276                    let (mode_suppressed, forwarded) = virtual_suppression(
277                        &engine,
278                        runtime.filter_mode(),
279                        (&frame.touches).into(),
280                    );
281                    s.mode_suppressed = mode_suppressed;
282                    s.forwarded = forwarded;
283                    s.field_set_count = engine.field_set_count();
284                    s.field_count = engine.field_count();
285                    s.occupied_fields = engine.occupied_field_count();
286                    if let Some(request) = s.background_alpha_request.take() {
287                        runtime.set_background_alpha(request.percent().into());
288                    }
289                    s.background_alpha =
290                        BackgroundAlpha::from_percent(runtime.background_alpha().percent());
291                }
292                if let Some(profile) = runtime.training_tick(&frame.touches) {
293                    info!(name = %profile.name, "training completed");
294                    runtime.set_message(None);
295                    (hooks.register_profile)(&mut runtime, &mut engine, profile);
296                }
297                if let Some(session) = &mut recording {
298                    session.record_frame(
299                        FrameNumber::new(frame_no),
300                        Milliseconds::from(session_start.elapsed()),
301                        frame,
302                        engine.tracked(),
303                    );
304                }
305            }
306
307            if raw.is_syn_report() {
308                // In live mode the frame's own touch list is authoritative:
309                // `touch_indices` in matches index into it (slot-sorted).
310                // The GUI's touch_state (HashMap order) would misalign them
311                // and draw figure lines across objects.
312                let touches = match &frame {
313                    Some(engine_frame) => engine_frame.touches.clone(),
314                    None => touch_state.active_touches(),
315                };
316                let count = touches.len();
317                let matches = frame.map(|f| f.matches).unwrap_or_default();
318
319                // Stability check (for training mode).
320                if stable_count == Some(count) && count >= 3 {
321                    stable_frames += 1;
322                } else {
323                    stable_count = Some(count);
324                    stable_frames = 0;
325                }
326
327                if let Ok(mut s) = state.lock() {
328                    s.touches = touches;
329                    s.tracked = engine.tracked().to_vec();
330                    s.matches = matches;
331                    s.stable_frames = stable_frames;
332                    s.stable = stable_frames >= STABLE_THRESHOLD && count >= 3;
333                }
334            }
335        }
336    }
