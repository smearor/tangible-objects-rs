34/// Consecutive same-size frames before a point set counts as stable
35/// for profile capture.
36pub(crate) const STABLE_THRESHOLD: u32 = 10;

38/// Field-set card rows for `SharedState` - set name, registration
39/// owner, enabled flag and field count per registered set.
40fn field_set_rows(engine: &Engine) -> Vec<FieldSetRow> {
41    engine
42        .field_sets()
43        .iter()
44        .map(|set| FieldSetRow {
45            name: set.name.clone(),
46            owner: engine.field_set_owner(&set.name).cloned(),
47            enabled: set.enabled,
48            field_count: set.fields.len(),
49        })
50        .collect()
51}

53/// Device loop thread: opens the evdev device, runs the engine per
54/// `SYN_REPORT` frame, publishes snapshots/events through the runtime
55/// and writes the GUI's [`SharedState`]. Daemon semantics (interfaces,
56/// control commands, recording) arrive via [`GuiHooks`].
57/// Virtual suppression/forwarding counts - what the filter proxy
58/// would do to this frame under `mode`. The GUI has no proxy; the
59/// counters show ingest-filtered plus mode-suppressed contacts so the
60/// status line matches `--filter` semantics.
61fn virtual_suppression(
62    engine: &Engine,
63    mode: FilterMode,
64    touches: TouchSet<'_>,
65) -> (ContactCount, ContactCount) {
66    let total = touches.len() as u32;
67    let suppressed = match mode {
68        FilterMode::Off => 0,
69        FilterMode::Immediate | FilterMode::Gated | FilterMode::Guarded => {
70            let bound = engine.bound_slots();
71            let field = engine.field_suppressed_slots(touches);
72            touches
73                .slots_where(|t| bound.contains(&t.slot) || field.contains(&t.slot))
74                .len() as u32
75        }
76        FilterMode::TangiblesOnly => total,
77    };
78    (suppressed.into(), (total - suppressed).into())
79}

81pub fn evdev_thread(
82    device_path: String,
83    patterns: PatternBuf,
84    state: Arc<Mutex<SharedState>>,
85    config: DetectionConfig,
86    rotation: DisplayRotation,
87    device_profiles: Option<Arc<PathBuf>>,
88    hooks: GuiHooks,
89) {
90    let mut device = match TouchDevice::open(std::path::Path::new(&device_path)) {
91        Ok(d) => d,
92        Err(e) => {
93            if let Ok(mut s) = state.lock() {
94                s.device_error = Some(fl!(
95                    language_loader(),
96                    "device-error",
97                    error = e.to_string()
98                ));
99            }
100            return;
101        }
102    };
103    if config.import_slots {
104        device.queue_state_import();
105    }
106
107    let mut engine = Engine::new(patterns);
108    if let Some(mm) = config.edge {
109        engine.set_edge_margin(mm);
110    }
111    let (
112        dead_zones_visible,
113        background_alpha,
114        edge_margin_visible,
115        fields_visible,
116        ruler_visible,
117        positions_visible,
118        filter_mode,
119        touch_passthrough,
120    ) = state
121        .lock()
122        .map(|s| {
123            (
124                s.dead_zones_visible,
125                s.background_alpha.percent(),
126                s.edge_margin_visible,
127                s.fields_visible,
128                s.ruler_visible,
129                s.positions_visible,
130                s.filter_mode,
131                s.touch_passthrough,
132            )
133        })
134        .unwrap_or_default();
135    engine.set_fields(config.fields);
136    engine.set_dead_zones(config.dead_zones);
137    engine.set_completion(config.completion);
138
139    // Event hub for interface crates (GraphQL/D-Bus/MCP) - the GUI runs
140    // its own device loop, so it owns the runtime side like the
141    // headless modes do.
142    let profiles: Vec<ProfileInfo> = engine.patterns().iter().map(ProfileInfo::from).collect();
143    let (mut runtime, handle) = EngineRuntime::new(Vec::new(), profiles);
144    runtime.set_completion(config.completion);
145    runtime.set_display_rotation(rotation);
146    runtime.set_field_sets(engine.field_sets().to_vec());
147    runtime.set_dead_zones(engine.dead_zones().to_vec());
148    if let Ok(mut s) = state.lock() {
149        s.command_handle = Some(handle.clone());
150    }
151    runtime.set_dead_zones_visible(dead_zones_visible);
152    runtime.set_background_alpha(background_alpha.into());
153    runtime.set_edge_margin_visible(edge_margin_visible);
154    runtime.set_fields_visible(fields_visible);
155    runtime.set_ruler_visible(ruler_visible);
156    runtime.set_positions_visible(positions_visible);
157    runtime.set_filter_mode(filter_mode);
158    runtime.set_touch_passthrough(touch_passthrough);
159    (hooks.spawn_interfaces)(&handle);
160
161    let mut touch_state = TouchState::new();
162    if let Some(cal) =
163        (hooks.effective_calibration)(&device, device_profiles.as_deref().map(PathBuf::as_path))
164    {
165        touch_state.set_calibration(cal);
166        engine.set_calibration(cal);
167        if let Ok(mut s) = state.lock() {
168            s.calibration = Some(cal);
169        }
170    }
171    // Report the calibration the engine actually uses (profile override
172    // applied), not the firmware-reported one - clients size their field
173    // sets from these millimeter bounds.
174    let device_info = DeviceInfo::of(device.name(), engine.calibration().as_ref());
175    if let Ok(mut s) = state.lock() {
176        s.device_info = Some(device_info.clone());
177    }
178    runtime.set_device_info(device_info);
179
180    let mut stable_count: Option<usize> = None;
181    let mut stable_frames: u32 = 0;
182    let device_fd = device.as_raw_fd();
183    let session_start = std::time::Instant::now();
184    let mut frame_no = 0u64;
185    let mut recording: Option<RecordingSession> = None;
186
187    loop {
188        // Periodic wake for queued control commands even without input.
189        let mut pfd = libc::pollfd {
190            fd: device_fd,
191            events: libc::POLLIN,
192            revents: 0,
193        };
194        let ready = unsafe { libc::poll(&mut pfd, 1, 250) };
195        if ready < 0 {
196            if unsafe { *libc::__errno_location() } == libc::EINTR {
197                continue;
198            }
199            error!("poll error - evdev loop exits");
200            break;
201        }
202        if ready == 0 {
203            (hooks.drain_control_commands)(
204                &mut runtime,
205                &mut recording,
206                &mut engine,
207                device.name(),
208            );
209            runtime.set_recording_status(recording.as_ref().map(hooks.status_of_session));
210            if let Ok(mut s) = state.lock() {
211                s.message = runtime.message().map(str::to_string);
212                s.visible = runtime.gui_visible();
213                s.rotation = runtime.display_rotation();
214                s.dead_zones_visible = runtime.dead_zones_visible();
215                s.dead_zones = engine.dead_zones().to_vec();
216                s.edge_margin = engine.edge_margin();
217                s.edge_margin_visible = runtime.edge_margin_visible();
218                s.fields_visible = runtime.fields_visible();
219                s.field_sets = engine.field_sets().to_vec();
220                s.field_set_rows = field_set_rows(&engine);
221                s.profiles = handle.profiles();
222                s.ruler_visible = runtime.ruler_visible();
223                s.grid_visible = runtime.grid_visible();
224                s.positions_visible = runtime.positions_visible();
225                s.touch_passthrough = runtime.touch_passthrough();
226                s.filter_mode = runtime.filter_mode();
227                s.incoming_contacts = engine.incoming_contacts();
228                s.edge_filtered = engine.edge_filtered();
229                let (mode_suppressed, forwarded) =
230                    virtual_suppression(&engine, runtime.filter_mode(), (&*s.touches).into());
231                s.mode_suppressed = mode_suppressed;
232                s.forwarded = forwarded;
233                s.field_set_count = engine.field_set_count();
234                s.field_count = engine.field_count();
235                s.occupied_fields = engine.occupied_field_count();
236                if let Some(request) = s.background_alpha_request.take() {
237                    runtime.set_background_alpha(request.percent().into());
238                }
239                s.background_alpha =
240                    BackgroundAlpha::from_percent(runtime.background_alpha().percent());
241            }
242            runtime.publish_snapshot(engine.touches(), engine.tracked());
243            continue;
244        }
245
246        let events = match device.fetch_events() {
247            Ok(events) => events,
248            Err(e) => {
249                error!(error = %e, "device read failed - evdev loop exits");
250                break;
251            }
252        };
253        for raw in events {
254            if let Some(session) = &recording {
255                session.record_event(&raw, session_start.elapsed().as_secs_f64());
256            }
257            touch_state.process(&raw);
258
259            // The engine builds its own touch state from every event —
260            // feeding it only SYN_REPORTs leaves that state empty and no
261            // figure ever matches.
262            let frame = if mode_is_live(&state) {
263                if recording.is_some() {
264                    engine.process_with_diagnostics(&raw)
265                } else {
266                    engine.process(&raw)
267                }
268            } else {
269                None
270            };
271            if let Some(ref frame) = frame {
272                frame_no += 1;
273                runtime.publish(frame, engine.tracked());
274                (hooks.drain_control_commands)(
275                    &mut runtime,
276                    &mut recording,
277                    &mut engine,
278                    device.name(),
279                );
280                runtime.set_recording_status(recording.as_ref().map(hooks.status_of_session));
281                if let Ok(mut s) = state.lock() {
282                    s.message = runtime.message().map(str::to_string);
283                    s.visible = runtime.gui_visible();
284                    s.rotation = runtime.display_rotation();
285                    s.dead_zones_visible = runtime.dead_zones_visible();
286                    s.dead_zones = engine.dead_zones().to_vec();
287                    s.dead_zone_filtered = frame.dead_zone_filtered;
288                    s.edge_margin = engine.edge_margin();
289                    s.edge_margin_visible = runtime.edge_margin_visible();
290                    s.fields_visible = runtime.fields_visible();
291                    s.field_sets = engine.field_sets().to_vec();
292                    s.field_set_rows = field_set_rows(&engine);
293                    s.profiles = handle.profiles();
294                    s.ruler_visible = runtime.ruler_visible();
295                    s.grid_visible = runtime.grid_visible();
296                    s.positions_visible = runtime.positions_visible();
297                    s.touch_passthrough = runtime.touch_passthrough();
298                    s.touch_passthrough = runtime.touch_passthrough();
299                    s.filter_mode = runtime.filter_mode();
300                    s.incoming_contacts = engine.incoming_contacts();
301                    s.edge_filtered = engine.edge_filtered();
302                    let (mode_suppressed, forwarded) = virtual_suppression(
303                        &engine,
304                        runtime.filter_mode(),
305                        (&frame.touches).into(),
306                    );
307                    s.mode_suppressed = mode_suppressed;
308                    s.forwarded = forwarded;
309                    s.field_set_count = engine.field_set_count();
310                    s.field_count = engine.field_count();
311                    s.occupied_fields = engine.occupied_field_count();
312                    if let Some(request) = s.background_alpha_request.take() {
313                        runtime.set_background_alpha(request.percent().into());
314                    }
315                    s.background_alpha =
316                        BackgroundAlpha::from_percent(runtime.background_alpha().percent());
317                }
318                if let Some(profile) = runtime.training_tick(&frame.touches) {
319                    info!(name = %profile.name, "training completed");
320                    runtime.set_message(None);
321                    (hooks.register_profile)(&mut runtime, &mut engine, profile);
322                }
323                if let Some(session) = &mut recording {
324                    session.record_frame(
325                        FrameNumber::new(frame_no),
326                        Milliseconds::from(session_start.elapsed()),
327                        frame,
328                        engine.tracked(),
329                    );
330                }
331            }
332
333            if raw.is_syn_report() {
334                // In live mode the frame's own touch list is authoritative:
335                // `touch_indices` in matches index into it (slot-sorted).
336                // The GUI's touch_state (HashMap order) would misalign them
337                // and draw figure lines across objects.
338                let touches = match &frame {
339                    Some(engine_frame) => engine_frame.touches.clone(),
340                    None => touch_state.active_touches(),
341                };
342                let count = touches.len();
343                let matches = frame.map(|f| f.matches).unwrap_or_default();
344
345                // Stability check (for training mode).
346                if stable_count == Some(count) && count >= 3 {
347                    stable_frames += 1;
348                } else {
349                    stable_count = Some(count);
350                    stable_frames = 0;
351                }
352
353                if let Ok(mut s) = state.lock() {
354                    s.touches = touches;
355                    s.tracked = engine.tracked().to_vec();
356                    s.matches = matches;
357                    s.stable_frames = stable_frames;
358                    s.stable = stable_frames >= STABLE_THRESHOLD && count >= 3;
359                }
360            }
361        }
362    }
