1//! The main-menu card - the always-open root card whose toggle list
2//! opens and closes the other cards on the board (GUI.md).
3
4use std::cell::RefCell;
5use std::rc::Rc;
6
7use floating_board::CardId;
8use floating_board::FloatingBoard;
9use gtk4::Box as GtkBox;
10use gtk4::Orientation;
11use gtk4::ToggleButton;
12use gtk4::prelude::*;
13
14/// The menu card's own id - it is skipped in its toggle list so it can
15/// never close itself.
16pub(crate) const MENU_CARD_ID: &str = "menu";
17
18/// Builds the menu-card content: one toggle per non-menu board card
19/// showing that card's header title. The list rebuilds on
20/// `card-added`/`card-removed` (the training card joins and leaves
21/// with the mode), toggle states re-sync on `card-visibility-changed`
22/// (a card closed via its X button unchecks its menu entry).
23pub(crate) fn create_menu_card(board: &FloatingBoard) -> GtkBox {
24    let body = GtkBox::new(Orientation::Vertical, 4);
25    let toggles: Rc<RefCell<Vec<(CardId, ToggleButton)>>> = Rc::new(RefCell::new(Vec::new()));
26
27    let rebuild = {
28        let board = board.clone();
29        let toggles = toggles.clone();
30        let body = body.clone();
31        Rc::new(move || {
32            while let Some(child) = body.first_child() {
33                body.remove(&child);
34            }
35            let mut toggles = toggles.borrow_mut();
36            toggles.clear();
37            for card in board.cards() {
38                let id = card.id();
39                if id == CardId::new(MENU_CARD_ID) {
40                    continue;
41                }
42                let button = ToggleButton::with_label(&card.title());
43                button.set_active(board.is_card_visible(&id).unwrap_or(true));
44                let board_click = board.clone();
45                let id_click = id.clone();
46                button.connect_toggled(move |button| {
47                    let _ = board_click.set_card_visible(&id_click, button.is_active());
48                });
49                body.append(&button);
50                toggles.push((id, button));
51            }
52        })
53    };
54
55    rebuild();
56
57    // A card added or removed later (training mode) rebuilds the list.
58    let rebuild_added = rebuild.clone();
59    board.connect_local("card-added", false, move |_| {
60        rebuild_added();
61        None
62    });
63    let rebuild_removed = rebuild.clone();
64    board.connect_local("card-removed", false, move |_| {
65        rebuild_removed();
66        None
67    });
68
69    // A card hidden or shown outside the menu (its X button) keeps the
70    // toggles truthful - the re-fired `connect_toggled` is a no-op
71    // because the board already holds the target visibility.
72    let toggles_sync = toggles;
73    let board_sync = board.clone();
74    board.connect_local("card-visibility-changed", false, move |args| {
75        if let Ok(card) = args[1].get::<floating_board::Card>() {
76            let id = card.id();
77            let visible = board_sync.is_card_visible(&id).unwrap_or(true);
78            if let Some((_, button)) = toggles_sync
79                .borrow()
80                .iter()
81                .find(|(toggle_id, _)| *toggle_id == id)
82                && button.is_active() != visible
83            {
84                button.set_active(visible);
85            }
86        }
87        None
88    });
89
90    body
91}
