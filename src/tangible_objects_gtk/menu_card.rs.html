1//! The main-menu card - the always-open root card whose toggle list
2//! opens and closes the other cards on the board (GUI.md). Its Arrange
3//! action tidies the visible cards via [`crate::card_layout::arrange_strategy`];
4//! folded, the card collapses to a round burger-menu FAB.
5
6use std::cell::RefCell;
7use std::rc::Rc;
8
9use floating_board::CardId;
10use floating_board::FloatingBoard;
11use gtk4::Box as GtkBox;
12use gtk4::Button;
13use gtk4::Image;
14use gtk4::Orientation;
15use gtk4::ToggleButton;
16use gtk4::prelude::*;
17use i18n_embed_fl::fl;
18use tracing::warn;
19
20use crate::card_layout::arrange_strategy;
21use crate::language_loader;
22
23/// The menu card's own id - it is skipped in its toggle list so it can
24/// never close itself.
25pub(crate) const MENU_CARD_ID: &str = "menu";
26
27/// Builds the menu-card content (toggle list plus Arrange action) and
28/// its folded-content FAB. The list rebuilds on
29/// `card-added`/`card-removed` (the training card joins and leaves
30/// with the mode), toggle states re-sync on `card-visibility-changed`
31/// (a card closed via its X button unchecks its menu entry). The FAB
32/// is display-only - the card's `unfold_on_click` unfolds it on tap
33/// and `WholeCardFolded` drags it (a real `Button` would claim the
34/// press and break both).
35pub(crate) fn create_menu_card(board: &FloatingBoard) -> (GtkBox, GtkBox) {
36    let outer = GtkBox::new(Orientation::Vertical, 6);
37    let body = GtkBox::new(Orientation::Vertical, 4);
38    let toggles: Rc<RefCell<Vec<(CardId, ToggleButton)>>> = Rc::new(RefCell::new(Vec::new()));
39
40    let rebuild = {
41        let board = board.clone();
42        let toggles = toggles.clone();
43        let body = body.clone();
44        Rc::new(move || {
45            while let Some(child) = body.first_child() {
46                body.remove(&child);
47            }
48            let mut toggles = toggles.borrow_mut();
49            toggles.clear();
50            for card in board.cards() {
51                let id = card.id();
52                if id == CardId::new(MENU_CARD_ID) {
53                    continue;
54                }
55                let button = ToggleButton::with_label(&card.title());
56                button.set_active(board.is_card_visible(&id).unwrap_or(true));
57                let board_click = board.clone();
58                let id_click = id.clone();
59                button.connect_toggled(move |button| {
60                    let _ = board_click.set_card_visible(&id_click, button.is_active());
61                });
62                body.append(&button);
63                toggles.push((id, button));
64            }
65        })
66    };
67
68    rebuild();
69
70    // A card added or removed later (training mode) rebuilds the list.
71    let rebuild_added = rebuild.clone();
72    board.connect_local("card-added", false, move |_| {
73        rebuild_added();
74        None
75    });
76    let rebuild_removed = rebuild.clone();
77    board.connect_local("card-removed", false, move |_| {
78        rebuild_removed();
79        None
80    });
81
82    // A card hidden or shown outside the menu (its X button) keeps the
83    // toggles truthful - the re-fired `connect_toggled` is a no-op
84    // because the board already holds the target visibility.
85    let toggles_sync = toggles;
86    let board_sync = board.clone();
87    board.connect_local("card-visibility-changed", false, move |args| {
88        if let Ok(card) = args[1].get::<floating_board::Card>() {
89            let id = card.id();
90            let visible = board_sync.is_card_visible(&id).unwrap_or(true);
91            if let Some((_, button)) = toggles_sync
92                .borrow()
93                .iter()
94                .find(|(toggle_id, _)| *toggle_id == id)
95                && button.is_active() != visible
96            {
97                button.set_active(visible);
98            }
99        }
100        None
101    });
102
103    // Arrange: flows the visible cards into tidy rows (animated) -
104    // `None` selects all visible, non-sticky cards; a `DoesNotFit`
105    // report leaves the board untouched.
106    let arrange = Button::with_label(&fl!(language_loader(), "menu-arrange"));
107    let board_arrange = board.clone();
108    arrange.connect_clicked(move |_| {
109        if let Err(error) = board_arrange.apply_layout(&arrange_strategy(), None, true) {
110            warn!(%error, "card arrangement does not fit");
111        }
112    });
113
114    outer.append(&body);
115    outer.append(&arrange);
116
117    // The folded content: a round burger-menu FAB. Display-only, so
118    // no child gesture competes with the card's unfold tap and drag.
119    let fab = GtkBox::new(Orientation::Vertical, 0);
120    fab.add_css_class("round-button");
121    fab.set_tooltip_text(Some(&fl!(language_loader(), "card-menu-title")));
122    let icon = Image::from_icon_name("open-menu-symbolic");
123    icon.set_hexpand(true);
124    icon.set_vexpand(true);
125    fab.append(&icon);
126
127    (outer, fab)
128}
