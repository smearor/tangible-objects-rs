21/// Builds the calibration-card content: a `label | value` grid
22/// refreshed from `SharedState` every 100 ms - the calibration lands
23/// once the evdev loop applied the effective calibration.
24pub fn create_calibration_card(state: &Arc<Mutex<SharedState>>) -> GtkBox {
25    let grid = Grid::new();
26    grid.set_column_spacing(12);
27    grid.set_row_spacing(4);
28
29    let source_value = add_row(&grid, 0, &fl!(language_loader(), "cal-source"));
30    let surface_value = add_row(&grid, 1, &fl!(language_loader(), "cal-surface"));
31    let x_range_value = add_row(&grid, 2, &fl!(language_loader(), "device-x-range"));
32    let y_range_value = add_row(&grid, 3, &fl!(language_loader(), "device-y-range"));
33    let res_value = add_row(&grid, 4, &fl!(language_loader(), "device-resolution"));
34
35    let state_tick = state.clone();
36    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
37        let s = match state_tick.lock() {
38            Ok(s) => s,
39            Err(_) => return ControlFlow::Continue,
40        };
41        source_value.set_label(&match s.calibration_source {
42            Some(CalibrationSource::Device) => {
43                fl!(language_loader(), "cal-source-device")
44            }
45            Some(CalibrationSource::Profile) => {
46                fl!(language_loader(), "cal-source-profile")
47            }
48            None => "–".to_string(),
49        });
50        match &s.calibration {
51            Some(cal) => {
52                surface_value.set_label(&format!(
53                    "{:.0} × {:.0} mm",
54                    cal.width_mm().as_f64(),
55                    cal.height_mm().as_f64()
56                ));
57                x_range_value.set_label(&format!(
58                    "{:.0} – {:.0}",
59                    cal.x_min.as_f64(),
60                    cal.x_max.as_f64()
61                ));
62                y_range_value.set_label(&format!(
63                    "{:.0} – {:.0}",
64                    cal.y_min.as_f64(),
65                    cal.y_max.as_f64()
66                ));
67                res_value.set_label(&format!(
68                    "{:.2} × {:.2}/mm",
69                    cal.x_res.as_f64(),
70                    cal.y_res.as_f64()
71                ));
72            }
73            None => {
74                for value in [&surface_value, &x_range_value, &y_range_value, &res_value] {
75                    value.set_label("–");
76                }
77            }
78        }
79        ControlFlow::Continue
80    });
81
82    let body = GtkBox::new(Orientation::Vertical, 0);
83    body.append(&grid);
84    body
85}

87/// Appends a `label | value` row to `grid` at `row`; returns the
88/// value label so the refresh tick can rewrite it.
89fn add_row(grid: &Grid, row: i32, name: &str) -> Label {
90    let name_label = Label::new(Some(name));
91    name_label.set_halign(gtk4::Align::Start);
92    name_label.set_opacity(0.6);
93    let value = Label::new(Some("–"));
94    value.set_halign(gtk4::Align::End);
95    grid.attach(&name_label, 0, row, 1, 1);
96    grid.attach(&value, 1, row, 1, 1);
97    value
98}
