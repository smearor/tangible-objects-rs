34/// Tolerance the saved profile takes (scalar mm).
35const DEFAULT_TOLERANCE: PatternTolerance = PatternTolerance::new(3.0);

37/// `SharedState` flag reader - the overlay toggles sync their
38/// checked state from it.
39type OverlayRead = fn(&SharedState) -> bool;

41/// Builds the control bar widget and wires its callbacks.
42pub fn create_control_bar(
43    state: &Arc<Mutex<SharedState>>,
44    profiles_dir: &Option<String>,
45    drawing_area: &DrawingArea,
46    bg_provider: &gtk4::CssProvider,
47) -> GtkBox {
48    let bar = GtkBox::new(Orientation::Horizontal, 8);
49    bar.set_valign(Align::End);
50    bar.set_halign(Align::Fill);
51    bar.set_margin_bottom(8);
52    bar.set_margin_start(12);
53    bar.set_margin_end(12);
54    bar.set_spacing(8);
55
56    // Mode toggle.
57    let mode_btn = Button::with_label(&fl!(language_loader(), "mode-live"));
58    mode_btn.add_css_class("suggested-action");
59
60    // Name entry.
61    let name_entry = Entry::new();
62    name_entry.set_placeholder_text(Some(&fl!(language_loader(), "figure-name-placeholder")));
63    name_entry.set_width_chars(20);
64    name_entry.set_sensitive(false);
65
66    // Save button.
67    let save_btn = Button::with_label(&fl!(language_loader(), "save-profile"));
68    save_btn.set_sensitive(false);
69
70    // Alpha toggle - cycles the background opacity presets; the
71    // touches/labels/chrome stay opaque, only the canvas fill thins.
72    let alpha_btn = Button::with_label(&fl!(
73        language_loader(),
74        "background-alpha",
75        percent = (state
76            .lock()
77            .map(|s| s.background_alpha.percent() as i64)
78            .unwrap_or(100))
79    ));
80    let state_alpha = state.clone();
81    let alpha_btn_clone = alpha_btn.clone();
82    let da_alpha = drawing_area.clone();
83    let provider_alpha = bg_provider.clone();
84    alpha_btn.connect_clicked(move |_| {
85        let Ok(mut s) = state_alpha.lock() else {
86            return;
87        };
88        s.background_alpha = s.background_alpha.next();
89        s.background_alpha_request = Some(s.background_alpha);
90        alpha_btn_clone.set_label(&fl!(
91            language_loader(),
92            "background-alpha",
93            percent = (s.background_alpha.percent() as i64)
94        ));
95        provider_alpha.load_from_data(&s.background_alpha.surface_bg_css());
96        da_alpha.queue_draw();
97    });
98
99    // Status label.
100    let status = Label::new(Some(""));
101    status.set_hexpand(true);
102    status.set_halign(Align::End);
103
104    // Mode toggle callback.
105    let state_toggle = state.clone();
106    let mode_btn_clone = mode_btn.clone();
107    let name_entry_clone = name_entry.clone();
108    let save_btn_clone = save_btn.clone();
109    mode_btn.connect_clicked(move |_| {
110        let is_training = match state_toggle.lock() {
111            Ok(mut s) => {
112                s.mode = if s.mode == GuiMode::Live {
113                    GuiMode::Training
114                } else {
115                    GuiMode::Live
116                };
117                s.stable_frames = 0;
118                s.stable = false;
119                s.mode == GuiMode::Training
120            }
121            Err(_) => return,
122        };
123
124        let label = if is_training {
125            fl!(language_loader(), "mode-training")
126        } else {
127            fl!(language_loader(), "mode-live")
128        };
129        mode_btn_clone.set_label(&label);
130        name_entry_clone.set_sensitive(is_training);
131        save_btn_clone.set_sensitive(false);
132    });
133
134    // Save callback.
135    let state_save = state.clone();
136    let name_entry_save = name_entry.clone();
137    let profiles_dir_save = profiles_dir.clone();
138    save_btn.connect_clicked(move |_| {
139        let name: String = name_entry_save.text().to_string();
140        if name.is_empty() {
141            return;
142        }
143
144        let points: Vec<Vec2<Millimeters>> = match state_save.lock() {
145            Ok(s) => {
146                if !s.stable {
147                    return;
148                }
149                s.touches.iter().map(|t| t.pos).collect()
150            }
151            Err(_) => return,
152        };
153
154        if points.len() < 3 {
155            return;
156        }
157
158        let profile = ProfileFile::from_points(&name, &points, DEFAULT_TOLERANCE);
159        let dir = profiles_dir_save.as_deref().unwrap_or("profiles");
160        std::fs::create_dir_all(dir).ok();
161        let filename = format!("{}/{}.toml", dir, name.to_lowercase().replace(' ', "-"));
162        let filepath = PathBuf::from(&filename);
163
164        match profile.save(&filepath) {
165            Ok(()) => info!(path = %filepath.display(), "profile saved"),
166            Err(e) => error!(error = %e, "profile save failed"),
167        }
168    });
169
170    bar.append(&mode_btn);
171    bar.append(&name_entry);
172    bar.append(&save_btn);
173    bar.append(&alpha_btn);
174    bar.append(&status);
175
176    // Second row: overlay toggles (send the same ControlCommands the
177    // interfaces do) + the FilterMode button set.
178    let toggles = GtkBox::new(Orientation::Horizontal, 8);
179    toggles.set_halign(Align::End);
180
181    let mut overlay_buttons: Vec<(ToggleButton, OverlayRead)> = Vec::new();
182    let mut add_overlay_toggle =
183        |label: &str, read: OverlayRead, command: fn(bool) -> ControlCommand| {
184            let btn = ToggleButton::with_label(label);
185            btn.set_active(state.lock().map(|s| read(&s)).unwrap_or_default());
186            let state_click = state.clone();
187            btn.connect_toggled(move |b| {
188                send_command(&state_click, command(b.is_active()));
189            });
190            toggles.append(&btn);
191            overlay_buttons.push((btn, read));
192        };
193    add_overlay_toggle(
194        &fl!(language_loader(), "toggle-edge-margin"),
195        |s| s.edge_margin_visible,
196        ControlCommand::SetEdgeMarginVisible,
197    );
198    add_overlay_toggle(
199        &fl!(language_loader(), "toggle-fields"),
200        |s| s.fields_visible,
201        ControlCommand::SetFieldsVisible,
202    );
203    add_overlay_toggle(
204        &fl!(language_loader(), "toggle-ruler"),
205        |s| s.ruler_visible,
206        ControlCommand::SetRulerVisible,
207    );
208    add_overlay_toggle(
209        &fl!(language_loader(), "toggle-positions"),
210        |s| s.positions_visible,
211        ControlCommand::SetPositionsVisible,
212    );
213    add_overlay_toggle(
214        &fl!(language_loader(), "toggle-passthrough"),
215        |s| s.touch_passthrough,
216        ControlCommand::SetTouchPassthrough,
217    );
218
219    // FilterMode button set - one checked at a time, commands flow
220    // through the control channel (honored by the proxy in --filter;
221    // display state otherwise).
222    let current_mode = state.lock().map(|s| s.filter_mode).unwrap_or_default();
223    let add_mode_button = |leader: &mut Option<ToggleButton>,
224                           buttons: &mut Vec<(ToggleButton, FilterMode)>,
225                           mode: FilterMode,
226                           label: &str| {
227        let btn = ToggleButton::with_label(label);
228        if let Some(group) = &*leader {
229            btn.set_group(Some(group));
230        } else {
231            *leader = Some(btn.clone());
232        }
233        btn.set_active(mode == current_mode);
234        let state_mode = state.clone();
235        btn.connect_toggled(move |b| {
236            if b.is_active() {
237                send_command(
238                    &state_mode,
239                    ControlCommand::SetFilterMode {
240                        mode,
241                        stability: None,
242                        stage: None,
243                    },
244                );
245            }
246        });
247        toggles.append(&btn);
248        buttons.push((btn, mode));
249    };
250    let mut mode_group: Option<ToggleButton> = None;
251    let mut mode_buttons: Vec<(ToggleButton, FilterMode)> = Vec::new();
252    add_mode_button(
253        &mut mode_group,
254        &mut mode_buttons,
255        FilterMode::Off,
256        &fl!(language_loader(), "filter-mode-off"),
257    );
258    add_mode_button(
259        &mut mode_group,
260        &mut mode_buttons,
261        FilterMode::Immediate,
262        &fl!(language_loader(), "filter-mode-immediate"),
263    );
264    add_mode_button(
265        &mut mode_group,
266        &mut mode_buttons,
267        FilterMode::Gated,
268        &fl!(language_loader(), "filter-mode-gated"),
269    );
270    add_mode_button(
271        &mut mode_group,
272        &mut mode_buttons,
273        FilterMode::Guarded,
274        &fl!(language_loader(), "filter-mode-guarded"),
275    );
276    add_mode_button(
277        &mut mode_group,
278        &mut mode_buttons,
279        FilterMode::TangiblesOnly,
280        &fl!(language_loader(), "filter-mode-tangibles-only"),
281    );
282
283    // Status update inside the redraw timer.
284    let state_status = state.clone();
285    let status_clone = status.clone();
286    let save_btn_status = save_btn.clone();
287    let alpha_btn_status = alpha_btn.clone();
288    let provider_status = bg_provider.clone();
289    let da = drawing_area.clone();
290    let mut applied_alpha = state_status
291        .lock()
292        .map(|s| s.background_alpha)
293        .unwrap_or_else(|_| crate::BackgroundAlpha::from_percent(100));
294    let mut applied_overlays = state_status
295        .lock()
296        .map(|s| {
297            overlay_buttons
298                .iter()
299                .map(|(_, read)| read(&s))
300                .collect::<Vec<_>>()
301        })
302        .unwrap_or_default();
303    let mut applied_mode = current_mode;
304    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
305        let s = match state_status.lock() {
306            Ok(s) => s,
307            Err(_) => return ControlFlow::Continue,
308        };
309        // Alpha changes from `SetBackgroundAlpha` commands land in the
310        // shared state via the device loop - the provider reload and
311        // button label follow here (the toggle applies both itself).
312        if s.background_alpha != applied_alpha {
313            applied_alpha = s.background_alpha;
314            provider_status.load_from_data(&applied_alpha.surface_bg_css());
315            alpha_btn_status.set_label(&fl!(
316                language_loader(),
317                "background-alpha",
318                percent = (applied_alpha.percent() as i64)
319            ));
320        }
321        // Overlay/mode changes from interface commands land in shared
322        // state via the device loop - sync the buttons (set_active
323        // re-fires toggled and re-sends an idempotent command).
324        for ((btn, read), applied) in overlay_buttons.iter().zip(applied_overlays.iter_mut()) {
325            let current = read(&s);
326            if current != *applied {
327                *applied = current;
328                btn.set_active(current);
329            }
330        }
331        if s.filter_mode != applied_mode {
332            applied_mode = s.filter_mode;
333            for (btn, mode) in &mode_buttons {
334                btn.set_active(*mode == applied_mode);
335            }
336        }
337        let count = s.touches.len();
338        match s.mode {
339            GuiMode::Live => {
340                let n = s.matches.len();
341                status_clone.set_label(&fl!(
342                    language_loader(),
343                    "live-status",
344                    figures = (n as i64),
345                    touches = (count as i64),
346                    occupied = (i64::from(s.occupied_fields.as_u32())),
347                    fields = (i64::from(s.field_count.as_u32())),
348                    sets = (i64::from(s.field_set_count.as_u32())),
349                    zones = (s.dead_zones.len() as i64),
350                    filtered = (i64::from(s.dead_zone_filtered.as_u32())),
351                    suppressed = (i64::from(
352                        s.edge_filtered.as_u32()
353                            + s.dead_zone_filtered.as_u32()
354                            + s.mode_suppressed.as_u32()
355                    )),
356                    forwarded = (i64::from(s.forwarded.as_u32()))
357                ));
358            }
359            GuiMode::Training => {
360                if let Some(err) = &s.device_error {
361                    status_clone.set_label(err);
362                } else if count < 3 {
363                    status_clone.set_label(&fl!(
364                        language_loader(),
365                        "wait-for-points",
366                        count = (count as i64)
367                    ));
368                } else if s.stable {
369                    status_clone.set_label(&fl!(language_loader(), "stable-hint"));
370                } else {
371                    status_clone.set_label(&fl!(
372                        language_loader(),
373                        "stabilizing",
374                        stable = (s.stable_frames as i64),
375                        threshold = (STABLE_THRESHOLD as i64)
376                    ));
377                }
378                save_btn_status.set_sensitive(s.stable);
379            }
380        }
381        drop(s);
382        da.queue_draw();
383        ControlFlow::Continue
384    });
385
386    let outer = GtkBox::new(Orientation::Vertical, 4);
387    outer.set_valign(Align::End);
388    outer.append(&toggles);
389    outer.append(&bar);
390
391    // Touch passthrough: the surface input region is reduced to the
392    // control-bar box so touches elsewhere reach the desktop beneath
393    // (Wayland `set_input_region`; a no-op elsewhere). Re-applied
394    // whenever the box's allocation changes.
395    let mut applied_passthrough: Option<(i32, i32, i32, i32)> = None;
396    let outer_sync = outer.clone();
397    let state_passthrough = state.clone();
398    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
399        // Read the flag under lock, drop it before the GTK surface
400        // calls - `set_input_region` may dispatch into event handlers
401        // that take `state` themselves.
402        let enabled = state_passthrough
403            .lock()
404            .map(|s| s.touch_passthrough)
405            .unwrap_or_default();
406        if enabled {
407            let rect = outer_surface_rect(&outer_sync);
408            if applied_passthrough != Some(rect)
409                && let Some(surface) = outer_sync.native().and_then(|n| n.surface())
410                && rect.2 > 0
411                && rect.3 > 0
412            {
413                applied_passthrough = Some(rect);
414                let region = gtk4::cairo::Region::create_rectangle(
415                    &gtk4::cairo::RectangleInt::new(rect.0, rect.1, rect.2, rect.3),
416                );
417                surface.set_input_region(Some(&region));
418            }
419        } else if applied_passthrough.is_some() {
420            applied_passthrough = None;
421            if let Some(surface) = outer_sync.native().and_then(|n| n.surface()) {
422                surface.set_input_region(None);
423            }
424        }
425        ControlFlow::Continue
426    });
427
428    outer
429}

431/// The widget's allocation rectangle in surface coordinates - the
432/// input-region rect of the control-bar box.
433fn outer_surface_rect(widget: &impl gtk4::prelude::WidgetExt) -> (i32, i32, i32, i32) {
434    let Some(native) = widget.native() else {
435        return (0, 0, widget.width(), widget.height());
436    };
437    let origin = widget
438        .compute_point(&native, &gtk4::graphene::Point::new(0.0, 0.0))
439        .map(|p| (p.x().round() as i32, p.y().round() as i32))
440        .unwrap_or_default();
441    (origin.0, origin.1, widget.width(), widget.height())
442}

444/// Sends a `ControlCommand` over the runtime's command channel - the
445/// same path the interface servers take. Uses `try_lock`: the
446/// status-sync timer calls `set_active` on the toggle buttons while
447/// holding `state`, which re-fires `connect_toggled` on this same
448/// thread - a blocking `lock()` would deadlock on the non-recursive
449/// mutex. Dropping the (idempotent) command on contention is safe.
450fn send_command(state: &Arc<Mutex<SharedState>>, command: ControlCommand) {
451    if let Ok(s) = state.try_lock()
452        && let Some(handle) = &s.command_handle
453    {
454        handle.try_send_command(command).ok();
455    }
