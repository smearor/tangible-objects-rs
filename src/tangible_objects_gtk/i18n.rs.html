1//! Localization loader - Fluent assets shared with `crates/tui` via
2//! the workspace-root `i18n/` directory (I18N.md).
3
4use i18n_embed::DesktopLanguageRequester;
5use i18n_embed::LanguageRequester;
6use i18n_embed::fluent::FluentLanguageLoader;
7use i18n_embed::fluent::fluent_language_loader;
8use rust_embed::RustEmbed;
9use std::sync::LazyLock;
10
11/// Compile-time embedded localization assets (workspace `i18n/`).
12#[derive(RustEmbed)]
13#[folder = "../../i18n/"]
14struct Localizations;
15
16static LANGUAGE_LOADER: LazyLock<FluentLanguageLoader> = LazyLock::new(|| {
17    let loader: FluentLanguageLoader = fluent_language_loader!();
18    let requested = DesktopLanguageRequester::new().requested_languages();
19    if i18n_embed::select(&loader, &Localizations, &requested).is_err() {
20        // fallback_language ("en") stays loaded - a missing locale is
21        // not a failure the GUI should surface.
22    }
23    loader
24});
25
26/// The shared fluent language loader - passed to `fl!`.
27pub fn language_loader() -> &'static FluentLanguageLoader {
28    &LANGUAGE_LOADER
29}
