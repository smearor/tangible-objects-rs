1//! `MarginLayout` - a [`LayoutStrategy`] decorator that insets the
2//! arrangement bounds by a uniform margin before delegating to an
3//! inner strategy, restoring the edge breathing room the former
4//! hand-rolled default arrangement kept.
5
6use smearor_commons_geometry::Pixels;
7use smearor_commons_geometry::Rect;
8use smearor_commons_layout::FlowLayout;
9use smearor_commons_layout::LayoutError;
10use smearor_commons_layout::LayoutItem;
11use smearor_commons_layout::LayoutStrategy;
12use smearor_commons_layout::Placement;
13
14/// Insets the arrangement bounds by `margin` on every side, then
15/// defers the packing to `inner`.
16#[derive(Debug, Clone, Copy)]
17pub(crate) struct MarginLayout {
18    /// The wrapped packing strategy.
19    inner: FlowLayout<Pixels>,
20    /// Uniform margin around the arrangement bounds.
21    margin: Pixels,
22}
23
24impl MarginLayout {
25    /// A margin-inset layout around `inner`.
26    pub(crate) fn new(inner: FlowLayout<Pixels>, margin: Pixels) -> Self {
27        Self { inner, margin }
28    }
29}
30
31impl LayoutStrategy<Pixels> for MarginLayout {
32    fn arrange(
33        &self,
34        bounds: Rect<Pixels>,
35        items: &[LayoutItem<Pixels>],
36    ) -> Result<Vec<Placement<Pixels>>, LayoutError> {
37        self.inner.arrange(bounds.inset(self.margin), items)
38    }
39}
