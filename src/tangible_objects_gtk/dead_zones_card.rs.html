1//! The dead-zones card - registered dead-zone points (name, position,
2//! radius, active hit, remove button) plus an `Add` button that
3//! registers a zone at the single active touch point.

32/// Suppression radius a GUI-created dead zone takes (mm).
33const DEAD_ZONE_RADIUS_MM: f64 = 20.0;
34
35/// Widgets of one dead-zone row - kept across ticks so only the
36/// volatile cells rewrite.
37struct DeadZoneRow {
38    /// The zone this row displays.
39    name: DeadZoneName,
40    /// Surface position cell.
41    position: Label,
42    /// Suppression radius cell.
43    radius: Label,
44    /// Hit indicator - a touch currently inside the zone.
45    hit: Label,
46}
47
48/// Builds the dead-zones card content: one row per registered zone
49/// (`Name | Position | Radius | Hit | Remove`) plus the `Add` button -
50/// armed only while exactly one touch is live, registering a
51/// `DEAD_ZONE_RADIUS_MM` zone at that contact's position. Adds and
52/// removals stay session-scoped; the Save button commits both to
53/// `dead_zones.d`.
54pub fn create_dead_zones_card(state: &Arc<Mutex<SharedState>>) -> GtkBox {
55    let grid = Grid::new();
56    grid.set_column_spacing(12);
57    grid.set_row_spacing(4);
58
59    let headers = [
60        fl!(language_loader(), "dead-zone-name"),
61        fl!(language_loader(), "dead-zone-position"),
62        fl!(language_loader(), "dead-zone-radius"),
63        fl!(language_loader(), "dead-zone-hit"),
64        fl!(language_loader(), "dead-zone-remove"),
65    ];
66    for (col, label) in headers.iter().enumerate() {
67        let header = Label::new(Some(label));
68        header.set_halign(Align::Start);
69        header.set_opacity(0.6);
70        grid.attach(&header, col as i32, 0, 1, 1);
71    }
72
73    // Adds and removals are session-scoped; the Save button commits
74    // the whole current set via `PersistDeadZones` (writes store files
75    // for live zones, deletes entries for removed ones).
76    let save_btn = Button::with_label(&fl!(language_loader(), "dead-zone-save"));
77    let state_save = state.clone();
78    save_btn.connect_clicked(move |_| {
79        send_command(&state_save, ControlCommand::PersistDeadZones);
80    });
81    let add_btn = Button::with_label(&fl!(language_loader(), "dead-zone-add"));
82    let state_add = state.clone();
83    add_btn.connect_clicked(move |_| {
84        let zone = {
85            let Ok(s) = state_add.lock() else {
86                return;
87            };
88            let Some(touch) = single_touch(&s) else {
89                return;
90            };
91            DeadZone::new(
92                DeadZoneName::new(next_zone_name(&s)),
93                touch.pos,
94                Millimeters::new(DEAD_ZONE_RADIUS_MM),
95            )
96        };
97        let (reply, _ignored) = oneshot::channel();
98        send_command(
99            &state_add,
100            ControlCommand::Register {
101                registration: Box::new(Registration::DeadZone(zone)),
102                owner: None,
103                lifetime: Lifetime::Session,
104                reply,
105            },
106        );
107    });
108
109    let rows: Rc<RefCell<Vec<DeadZoneRow>>> = Rc::new(RefCell::new(Vec::new()));
110    let state_tick = state.clone();
111    let rows_tick = rows.clone();
112    let grid_tick = grid.clone();
113    let add_btn_tick = add_btn.clone();
114    let empty_label = fl!(language_loader(), "dead-zones-empty");
115    let empty_shown = Rc::new(std::cell::Cell::new(false));
116    // Overlay auto-show: opening the card switches the dead-zone
117    // overlay on; closing restores the previous state. `auto_vis`:
118    // 0 = untouched, 1 = enable sent (pending sync), 2 = confirmed ours.
119    let was_open = std::cell::Cell::new(false);
120    let auto_vis = std::cell::Cell::new(0u8);
121    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
122        let s = match state_tick.lock() {
123            Ok(s) => s,
124            Err(_) => return ControlFlow::Continue,
125        };
126        let open = grid_tick.is_visible();
127        if open && !was_open.get() && !s.dead_zones_visible {
128            if let Some(handle) = &s.command_handle {
129                handle
130                    .try_send_command(ControlCommand::SetDeadZonesVisible(true))
131                    .ok();
132            }
133            auto_vis.set(1);
134        } else if !open && was_open.get() && auto_vis.get() > 0 {
135            if let Some(handle) = &s.command_handle {
136                handle
137                    .try_send_command(ControlCommand::SetDeadZonesVisible(false))
138                    .ok();
139            }
140            auto_vis.set(0);
141        }
142        if open && auto_vis.get() == 1 && s.dead_zones_visible {
143            auto_vis.set(2);
144        } else if auto_vis.get() == 2 && !s.dead_zones_visible {
145            // The user switched the overlay off while the card was
146            // open - we no longer own it, closing leaves it off anyway.
147            auto_vis.set(0);
148        }
149        was_open.set(open);
150        add_btn_tick.set_sensitive(single_touch(&s).is_some());
151        let mut rows = rows_tick.borrow_mut();
152        let names: Vec<&DeadZoneName> = s.dead_zones.iter().map(|zone| &zone.name).collect();
153        let stale = rows.len() != names.len()
154            || rows
155                .iter()
156                .zip(names.iter())
157                .any(|(row, name)| row.name != **name);
158        if stale {
159            let below_header = rows.len() + usize::from(empty_shown.get());
160            for _ in 0..below_header {
161                grid_tick.remove_row(1);
162            }
163            rows.clear();
164            empty_shown.set(false);
165            if names.is_empty() {
166                let label = Label::new(Some(&empty_label));
167                label.set_halign(Align::Start);
168                label.set_opacity(0.5);
169                grid_tick.attach(&label, 0, 1, 5, 1);
170                empty_shown.set(true);
171            } else {
172                for (row_index, name) in names.iter().enumerate() {
173                    let name_label = Label::new(Some(name.as_str()));
174                    name_label.set_halign(Align::Start);
175                    let position = Label::new(Some("–"));
176                    position.set_halign(Align::Start);
177                    let radius = Label::new(Some("–"));
178                    radius.set_halign(Align::Start);
179                    let hit = Label::new(Some(""));
180                    hit.set_halign(Align::Start);
181                    let remove = Button::from_icon_name("user-trash-symbolic");
182                    let zone_name = (*name).clone();
183                    let state_remove = state_tick.clone();
184                    remove.connect_clicked(move |_| {
185                        send_command(
186                            &state_remove,
187                            ControlCommand::UnregisterDeadZone {
188                                name: zone_name.clone(),
189                                // Session-scoped removal - the store
190                                // file survives until Save commits it.
191                                unpersist: false,
192                            },
193                        );
194                    });
195                    let grid_row = (row_index + 1) as i32;
196                    grid_tick.attach(&name_label, 0, grid_row, 1, 1);
197                    grid_tick.attach(&position, 1, grid_row, 1, 1);
198                    grid_tick.attach(&radius, 2, grid_row, 1, 1);
199                    grid_tick.attach(&hit, 3, grid_row, 1, 1);
200                    grid_tick.attach(&remove, 4, grid_row, 1, 1);
201                    rows.push(DeadZoneRow {
202                        name: (*name).clone(),
203                        position,
204                        radius,
205                        hit,
206                    });
207                }
208            }
209        }
210        for (row, zone) in rows.iter().zip(s.dead_zones.iter()) {
211            row.position
212                .set_label(&format!("{:.0},{:.0} mm", zone.center.x, zone.center.y));
213            row.radius
214                .set_label(&format!("{:.0} mm", zone.radius.as_f64()));
215            row.hit.set_label(
216                if s.unfiltered_touches.iter().any(|t| zone.contains(&t.pos)) {
217                    "●"
218                } else {
219                    ""
220                },
221            );
222        }
223        ControlFlow::Continue
224    });
225
226    let body = GtkBox::new(Orientation::Vertical, 6);
227    body.append(&grid);
228    let actions = GtkBox::new(Orientation::Horizontal, 6);
229    actions.append(&add_btn);
230    actions.append(&save_btn);
231    body.append(&actions);
232    body
233}

235/// The single live contact when exactly one is active (the `Add`
236/// button's precondition).
237fn single_touch(s: &SharedState) -> Option<&tangible_objects_model::TouchPoint> {
238    match s.touches.len() {
239        1 => s.touches.first(),
240        _ => None,
241    }
242}
243
244/// Next free `zone-N` name for a GUI-created dead zone.
245fn next_zone_name(s: &SharedState) -> String {
246    (1..)
247        .map(|index| format!("zone-{index}"))
248        .find(|name| !s.dead_zones.iter().any(|zone| zone.name.as_str() == name))
249        .unwrap_or_else(|| "zone-gui".to_string())
250}
