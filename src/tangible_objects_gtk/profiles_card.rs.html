36/// Widgets of one profile row - kept across ticks so only the
37/// volatile cells rewrite.
38struct ProfileRow {
39    /// The profile this row displays.
40    id: PatternId,
41    /// Registering application cell.
42    owner: Label,
43    /// Enable switch - sends `SetProfileEnabled`.
44    enabled: Switch,
45    /// `state-set` connection - the sync blocks it while writing
46    /// the switch so `set_active` cannot echo `SetProfileEnabled`
47    /// back to the device loop.
48    enabled_signal: SignalHandlerId,
49    /// Toggle the user requested (`SetProfileEnabled` was sent) but
50    /// `s.profiles` has not confirmed yet - the sync leaves the
51    /// switch alone while it pends so it cannot snap back.
52    enabled_pending: Rc<Cell<Option<(bool, Instant)>>>,
53    /// Color preview + picker - sends `SetProfileColor`.
54    color: gtk4::ColorDialogButton,
55    /// Current surface position cell.
56    position: Label,
57}
58
59/// Builds the profiles-card content: a `Name | Owner | Enabled |
60/// Position` grid refreshed from [`SharedState`] every 100 ms. Rows
61/// rebuild only when the registered set changes; the volatile cells
62/// (enabled flag, live position) update in place.
63pub fn create_profiles_card(
64    state: &Arc<Mutex<SharedState>>,
65    profiles_dir: Option<PathBuf>,
66) -> GtkBox {
67    let grid = Grid::new();
68    grid.set_column_spacing(10);
69    grid.set_row_spacing(4);
70    for (column, header) in [
71        fl!(language_loader(), "col-name"),
72        fl!(language_loader(), "col-owner"),
73        fl!(language_loader(), "col-enabled"),
74        fl!(language_loader(), "col-color"),
75        fl!(language_loader(), "col-position"),
76    ]
77    .iter()
78    .enumerate()
79    {
80        let label = Label::new(Some(header));
81        label.set_halign(gtk4::Align::Start);
82        label.add_css_class("dim-label");
83        grid.attach(&label, column as i32, 0, 1, 1);
84    }
85
86    let rows: Rc<RefCell<Vec<ProfileRow>>> = Rc::new(RefCell::new(Vec::new()));
87    let state_tick = state.clone();
88    let rows_tick = rows.clone();
89    let grid_tick = grid.clone();
90    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
91        let grid = &grid_tick;
92        let s = match state_tick.lock() {
93            Ok(s) => s,
94            Err(_) => return ControlFlow::Continue,
95        };
96        let mut rows = rows_tick.borrow_mut();
97        if rows
98            .iter()
99            .map(|row| row.id)
100            .ne(s.profiles.iter().map(|profile| profile.pattern_id))
101        {
102            // Registration set changed - drop every data row below the
103            // header and rebuild in `s.profiles` order.
104            let row_count = rows.len() as i32;
105            rows.clear();
106            for _ in 0..row_count {
107                grid.remove_row(1);
108            }
109            for (index, profile) in s.profiles.iter().enumerate() {
110                let name = Label::new(Some(profile.name.as_str()));
111                name.set_halign(Align::Start);
112                let owner = Label::new(None);
113                owner.set_halign(Align::Start);
114                let enabled = Switch::new();
115                enabled.set_halign(Align::Start);
116                let pattern_id = profile.pattern_id;
117                let state_enabled = state_tick.clone();
118                let enabled_pending = Rc::new(Cell::new(None));
119                let enabled_signal = {
120                    let enabled_pending = enabled_pending.clone();
121                    enabled.connect_state_set(move |_, active| {
122                        enabled_pending.set(Some((active, Instant::now())));
123                        send_command(
124                            &state_enabled,
125                            ControlCommand::SetProfileEnabled {
126                                id: pattern_id,
127                                enabled: active,
128                            },
129                        );
130                        gtk4::glib::Propagation::Proceed
131                    })
132                };
133                let color = ColorDialogButton::new(Some(ColorDialog::new()));
134                let accent = crate::draw::figure_color(profile.pattern_id);
135                color.set_rgba(&accent.into());
136                color.set_halign(Align::Start);
137                let pattern_id_color = profile.pattern_id;
138                let state_color = state_tick.clone();
139                color.connect_notify_local(Some("rgba"), move |button, _| {
140                    send_command(
141                        &state_color,
142                        ControlCommand::SetProfileColor {
143                            id: pattern_id_color,
144                            color: FigureColor::from(Color::<f64>::from(button.rgba())),
145                        },
146                    );
147                });
148                let position = Label::new(None);
149                position.set_halign(Align::Start);
150                let row = index as i32 + 1;
151                grid.attach(&name, 0, row, 1, 1);
152                grid.attach(&owner, 1, row, 1, 1);
153                grid.attach(&enabled, 2, row, 1, 1);
154                grid.attach(&color, 3, row, 1, 1);
155                grid.attach(&position, 4, row, 1, 1);
156                rows.push(ProfileRow {
157                    id: profile.pattern_id,
158                    owner,
159                    enabled,
160                    enabled_signal,
161                    enabled_pending,
162                    color,
163                    position,
164                });
165            }
166        }
167        for (profile, row) in s.profiles.iter().zip(rows.iter()) {
168            // Reflect the live color (declared `color=` or a
169            // `SetProfileColor` from elsewhere).
170            let expected = s
171                .pattern_styles
172                .iter()
173                .find(|style| style.pattern_id == profile.pattern_id)
174                .and_then(|style| style.color)
175                .map(|color| color.0)
176                .unwrap_or_else(|| crate::draw::figure_color(profile.pattern_id));
177            let current = row.color.rgba();
178            let epsilon = 1.0 / 255.0;
179            if (current.red() as f64 - expected.r).abs() > epsilon
180                || (current.green() as f64 - expected.g).abs() > epsilon
181                || (current.blue() as f64 - expected.b).abs() > epsilon
182            {
183                row.color.set_rgba(&expected.into());
184            }
185            row.owner.set_label(
186                profile
187                    .owner
188                    .as_ref()
189                    .map(|owner| owner.as_str())
190                    .unwrap_or("–"),
191            );
192            // `set_active` emits `state-set` too and cancels an
193            // in-flight toggle animation even on an unchanged value
194            // (`gtk_switch_end_toggle_animation` runs before the
195            // value check) - a write inside the user's ~100 ms toggle
196            // window would eat the `state-set` before the command
197            // ever sends. Skip writes while a user toggle is
198            // unconfirmed and block the handler around real syncs.
199            let confirmed = match row.enabled_pending.get() {
200                Some((requested, _)) if requested == profile.enabled => true,
201                Some((_, at)) if at.elapsed() < Duration::from_millis(500) => false,
202                _ => true,
203            };
204            if confirmed {
205                row.enabled_pending.set(None);
206                if row.enabled.is_active() != profile.enabled {
207                    row.enabled.block_signal(&row.enabled_signal);
208                    row.enabled.set_active(profile.enabled);
209                    row.enabled.unblock_signal(&row.enabled_signal);
210                }
211            }
212            let position = s
213                .tracked
214                .iter()
215                .find(|tangible| tangible.pattern_id == profile.pattern_id)
216                .map(|tangible| format!("{:.0},{:.0} mm", tangible.center.x, tangible.center.y))
217                .unwrap_or_else(|| "–".to_string());
218            row.position.set_label(&position);
219        }
220        ControlFlow::Continue
221    });
222
223    // Persists the runtime `enabled`/`color` overrides of every row
224    // back into the profile files (`SaveProfiles` drains through the
225    // device loop like the switches' commands do).
226    let save = Button::with_label(&fl!(language_loader(), "profiles-save"));
227    save.set_halign(Align::End);
228    let state_save = state.clone();
229    save.connect_clicked(move |_| {
230        send_command(
231            &state_save,
232            ControlCommand::SaveProfiles {
233                profiles_dir: profiles_dir.clone(),
234            },
235        );
236    });
237
238    let body = GtkBox::new(Orientation::Vertical, 6);
239    body.append(&grid);
240    body.append(&save);
241    body
242}
