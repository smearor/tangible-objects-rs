37/// Widgets of one profile row - kept across ticks so only the
38/// volatile cells rewrite.
39struct ProfileRow {
40    /// The profile this row displays.
41    id: PatternId,
42    /// Editable name cell - the Save button commits changed texts
43    /// as renames.
44    name: gtk4::Entry,
45    /// Pin count cell - the figure's contact count.
46    points: Label,
47    /// Registering application cell.
48    owner: Label,
49    /// Enable switch - sends `SetProfileEnabled`.
50    enabled: Switch,
51    /// `state-set` connection - the sync blocks it while writing
52    /// the switch so `set_active` cannot echo `SetProfileEnabled`
53    /// back to the device loop.
54    enabled_signal: SignalHandlerId,
55    /// Toggle the user requested (`SetProfileEnabled` was sent) but
56    /// `s.profiles` has not confirmed yet - the sync leaves the
57    /// switch alone while it pends so it cannot snap back.
58    enabled_pending: Rc<Cell<Option<(bool, Instant)>>>,
59    /// Color preview + picker - sends `SetProfileColor`.
60    color: gtk4::ColorDialogButton,
61    /// Current surface position cell.
62    position: Label,
63}
64
65/// Builds the profiles-card content: a `Name | Owner | Enabled |
66/// Position` grid refreshed from [`SharedState`] every 100 ms. Rows
67/// rebuild only when the registered set changes; the volatile cells
68/// (enabled flag, live position) update in place.
69pub fn create_profiles_card(
70    state: &Arc<Mutex<SharedState>>,
71    profiles_dir: Option<PathBuf>,
72) -> GtkBox {
73    let grid = Grid::new();
74    grid.set_column_spacing(10);
75    grid.set_row_spacing(4);
76    for (column, header) in [
77        fl!(language_loader(), "col-name"),
78        fl!(language_loader(), "col-points"),
79        fl!(language_loader(), "col-owner"),
80        fl!(language_loader(), "col-enabled"),
81        fl!(language_loader(), "col-color"),
82        fl!(language_loader(), "col-position"),
83    ]
84    .iter()
85    .enumerate()
86    {
87        let label = Label::new(Some(header));
88        label.set_halign(Align::Start);
89        label.add_css_class("dim-label");
90        grid.attach(&label, column as i32, 0, 1, 1);
91    }
92
93    let rows: Rc<RefCell<Vec<ProfileRow>>> = Rc::new(RefCell::new(Vec::new()));
94    let state_tick = state.clone();
95    let rows_tick = rows.clone();
96    let grid_tick = grid.clone();
97    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
98        let grid = &grid_tick;
99        let s = match state_tick.lock() {
100            Ok(s) => s,
101            Err(_) => return ControlFlow::Continue,
102        };
103        let mut rows = rows_tick.borrow_mut();
104        if rows
105            .iter()
106            .map(|row| row.id)
107            .ne(s.profiles.iter().map(|profile| profile.pattern_id))
108        {
109            // Registration set changed - drop every data row below the
110            // header and rebuild in `s.profiles` order.
111            let row_count = rows.len() as i32;
112            rows.clear();
113            for _ in 0..row_count {
114                grid.remove_row(1);
115            }
116            for (index, profile) in s.profiles.iter().enumerate() {
117                let name = gtk4::Entry::new();
118                name.set_text(profile.name.as_str());
119                name.set_width_chars(14);
120                let points = Label::new(None);
121                points.set_halign(Align::Start);
122                let owner = Label::new(None);
123                owner.set_halign(Align::Start);
124                let enabled = Switch::new();
125                enabled.set_halign(Align::Start);
126                let pattern_id = profile.pattern_id;
127                let state_enabled = state_tick.clone();
128                let enabled_pending = Rc::new(Cell::new(None));
129                let enabled_signal = {
130                    let enabled_pending = enabled_pending.clone();
131                    enabled.connect_state_set(move |_, active| {
132                        enabled_pending.set(Some((active, Instant::now())));
133                        send_command(
134                            &state_enabled,
135                            ControlCommand::SetProfileEnabled {
136                                id: pattern_id,
137                                enabled: active,
138                            },
139                        );
140                        gtk4::glib::Propagation::Proceed
141                    })
142                };
143                let color = ColorDialogButton::new(Some(ColorDialog::new()));
144                let accent = crate::draw::figure_color(profile.pattern_id);
145                color.set_rgba(&accent.into());
146                color.set_halign(Align::Start);
147                let pattern_id_color = profile.pattern_id;
148                let state_color = state_tick.clone();
149                color.connect_notify_local(Some("rgba"), move |button, _| {
150                    send_command(
151                        &state_color,
152                        ControlCommand::SetProfileColor {
153                            id: pattern_id_color,
154                            color: FigureColor::from(Color::<f64>::from(button.rgba())),
155                        },
156                    );
157                });
158                let position = Label::new(None);
159                position.set_halign(Align::Start);
160                let row = index as i32 + 1;
161                grid.attach(&name, 0, row, 1, 1);
162                grid.attach(&points, 1, row, 1, 1);
163                grid.attach(&owner, 2, row, 1, 1);
164                grid.attach(&enabled, 3, row, 1, 1);
165                grid.attach(&color, 4, row, 1, 1);
166                grid.attach(&position, 5, row, 1, 1);
167                rows.push(ProfileRow {
168                    id: profile.pattern_id,
169                    name,
170                    points,
171                    owner,
172                    enabled,
173                    enabled_signal,
174                    enabled_pending,
175                    color,
176                    position,
177                });
178            }
179        }
180        for (profile, row) in s.profiles.iter().zip(rows.iter()) {
181            // Sync the entry only while the user is not editing it -
182            // a rewrite under an active cursor would drop keystrokes.
183            if row.name.text() != profile.name.as_str() && !row.name.has_focus() {
184                row.name.set_text(profile.name.as_str());
185            }
186            row.points.set_label(&profile.num_points.to_string());
187            // Reflect the live color (declared `color=` or a
188            // `SetProfileColor` from elsewhere).
189            let expected = s
190                .pattern_styles
191                .iter()
192                .find(|style| style.pattern_id == profile.pattern_id)
193                .and_then(|style| style.color)
194                .map(|color| color.0)
195                .unwrap_or_else(|| crate::draw::figure_color(profile.pattern_id));
196            let current = row.color.rgba();
197            let epsilon = 1.0 / 255.0;
198            if (current.red() as f64 - expected.r).abs() > epsilon
199                || (current.green() as f64 - expected.g).abs() > epsilon
200                || (current.blue() as f64 - expected.b).abs() > epsilon
201            {
202                row.color.set_rgba(&expected.into());
203            }
204            row.owner.set_label(
205                profile
206                    .owner
207                    .as_ref()
208                    .map(|owner| owner.as_str())
209                    .unwrap_or("–"),
210            );
211            // `set_active` emits `state-set` too and cancels an
212            // in-flight toggle animation even on an unchanged value
213            // (`gtk_switch_end_toggle_animation` runs before the
214            // value check) - a write inside the user's ~100 ms toggle
215            // window would eat the `state-set` before the command
216            // ever sends. Skip writes while a user toggle is
217            // unconfirmed and block the handler around real syncs.
218            let confirmed = match row.enabled_pending.get() {
219                Some((requested, _)) if requested == profile.enabled => true,
220                Some((_, at)) if at.elapsed() < Duration::from_millis(500) => false,
221                _ => true,
222            };
223            if confirmed {
224                row.enabled_pending.set(None);
225                if row.enabled.is_active() != profile.enabled {
226                    row.enabled.block_signal(&row.enabled_signal);
227                    row.enabled.set_active(profile.enabled);
228                    row.enabled.unblock_signal(&row.enabled_signal);
229                }
230            }
231            let position = s
232                .tracked
233                .iter()
234                .find(|tangible| tangible.pattern_id == profile.pattern_id)
235                .map(|tangible| format!("{:.0},{:.0} mm", tangible.center.x, tangible.center.y))
236                .unwrap_or_else(|| "–".to_string());
237            row.position.set_label(&position);
238        }
239        ControlFlow::Continue
240    });
241
242    // Persists the runtime `enabled`/`color` overrides of every row
243    // back into the profile files (`SaveProfiles` drains through the
244    // device loop like the switches' commands do). Name entries that
245    // differ from the live name travel as renames - files move to
246    // `<new-name>.toml` and the pattern keeps its `PatternId`.
247    let save = Button::with_label(&fl!(language_loader(), "profiles-save"));
248    save.set_halign(Align::End);
249    let state_save = state.clone();
250    let rows_save = rows.clone();
251    save.connect_clicked(move |_| {
252        let renames = {
253            let Ok(s) = state_save.lock() else {
254                return;
255            };
256            rows_save
257                .borrow()
258                .iter()
259                .filter_map(|row| {
260                    s.profiles
261                        .iter()
262                        .find(|profile| profile.pattern_id == row.id)
263                        .filter(|profile| {
264                            let text = row.name.text();
265                            !text.is_empty() && text != profile.name.as_str()
266                        })
267                        .map(|profile| {
268                            (
269                                profile.pattern_id,
270                                PatternName::new(row.name.text().as_str()),
271                            )
272                        })
273                })
274                .collect()
275        };
276        send_command(
277            &state_save,
278            ControlCommand::SaveProfiles {
279                profiles_dir: profiles_dir.clone(),
280                renames,
281            },
282        );
283    });
284
285    let body = GtkBox::new(Orientation::Vertical, 6);
286    body.append(&grid);
287    body.append(&save);
288    body
289}
