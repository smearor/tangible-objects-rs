1//! Card layout persistence - `gui-cards.toml` under
2//! `$XDG_CONFIG_HOME/tangible-objects/` (GUI.md §3 Persistence):
3//! position, rotation and scale of every `FloatingBoard` card
4//! restore on the next GUI start.

21/// Layout file name inside the `tangible-objects` config dir.
22const LAYOUT_FILE: &str = "gui-cards.toml";
23/// Minimum distance of the default arrangement from every board edge.
24const EDGE_MARGIN: f64 = 100.0;
25/// Gap between neighboring cards in the default row arrangement.
26const CARD_GAP: f64 = 24.0;
27/// Debounce window for writing `layout_snapshot()` back - card
28/// gestures emit `geometry-changed` continuously while dragging.
29const SAVE_DEBOUNCE: Duration = Duration::from_millis(500);
30
31/// The layout path: `$XDG_CONFIG_HOME/tangible-objects/gui-cards.toml`,
32/// falling back to `~/.config/tangible-objects/gui-cards.toml` (same
33/// resolution the daemon's runtime store uses).
34fn layout_path() -> Option<PathBuf> {
35    let base = std::env::var_os("XDG_CONFIG_HOME")
36        .map(PathBuf::from)
37        .or_else(|| std::env::var_os("HOME").map(|home| PathBuf::from(home).join(".config")))?;
38    Some(base.join("tangible-objects").join(LAYOUT_FILE))
39}

41/// Loads the persisted layout; `None` when the file is missing,
42/// unreadable or holds no placements.
43pub(crate) fn load_layout() -> Option<BoardLayout> {
44    let path = layout_path()?;
45    let content = std::fs::read_to_string(&path).ok()?;
46    toml::from_str(&content)
47        .map_err(|error| warn!(path = %path.display(), %error, "card layout ignored"))
48        .ok()
49        .filter(|layout: &BoardLayout| !layout.is_empty())
50}
51
52/// Writes the layout snapshot into the config dir (best effort - a
53/// read-only home dir only loses persistence).
54fn save_layout(layout: &BoardLayout) {
55    let Some(path) = layout_path() else {
56        return;
57    };
58    let Some(dir) = path.parent() else {
59        return;
60    };
61    match toml::to_string_pretty(layout) {
62        Ok(serialized) => {
63            if let Err(error) =
64                std::fs::create_dir_all(dir).and_then(|_| std::fs::write(&path, serialized))
65            {
66                warn!(path = %path.display(), %error, "card layout save failed");
67            }
68        }
69        Err(error) => warn!(%error, "card layout serialization failed"),
70    }
71}

73/// Wires layout persistence for `board`:
74/// - restores `gui-cards.toml` once real bounds exist (clamping needs
75///   them - earlier restores would collapse every position),
76/// - arranges the default side-by-side layout when nothing was saved,
77/// - writes `layout_snapshot()` back [`SAVE_DEBOUNCE`] after the last
78///   `geometry-changed`.
79///
80/// `layout` is the result of [`load_layout`] - the caller loads it
81/// once so other consumers (mode-driven cards) can peek at saved
82/// placements too.
83pub(crate) fn wire_layout(board: &FloatingBoard, layout: Option<BoardLayout>) {
84    let board_apply = board.clone();
85    // Poll until bounds and card footprints exist - restoring or
86    // arranging earlier would clamp every position to the origin.
87    gtk4::glib::timeout_add_local(Duration::from_millis(50), move || {
88        let allocated = board_apply.width() > 0
89            && board_apply.height() > 0
90            && board_apply
91                .cards()
92                .iter()
93                .all(|card| card_extent(card).0 > 0.0);
94        if !allocated {
95            return ControlFlow::Continue;
96        }
97        match &layout {
98            Some(layout) => board_apply.restore_layout(layout),
99            None => arrange_default(&board_apply),
100        }
101        ControlFlow::Break
102    });
103
104    let dirty = Rc::new(Cell::new(false));
105    let dirty_flag = dirty.clone();
106    board.connect_local("geometry-changed", false, move |_| {
107        dirty_flag.set(true);
108        None
109    });
110    let board_save = board.clone();
111    gtk4::glib::timeout_add_local(SAVE_DEBOUNCE, move || {
112        if dirty.replace(false) {
113            save_layout(&board_save.layout_snapshot());
114        }
115        ControlFlow::Continue
116    });
117}

119/// The default layout for first runs: cards side by side starting at
120/// (`EDGE_MARGIN`, `EDGE_MARGIN`), wrapping to a new row whenever the
121/// next card would leave the board's right margin.
122fn arrange_default(board: &FloatingBoard) {
123    let board_w = f64::from(board.width());
124    let mut x = EDGE_MARGIN;
125    let mut y = EDGE_MARGIN;
126    let mut row_h = 0.0_f64;
127    for card in board.cards() {
128        let (w, h) = card_extent(&card);
129        if x > EDGE_MARGIN && x + w > board_w - EDGE_MARGIN {
130            x = EDGE_MARGIN;
131            y += row_h + CARD_GAP;
132            row_h = 0.0;
133        }
134        // `CardGeometry.position` is the card's center point.
135        card.set_position(CardPosition::new(Vec2::new(x + w / 2.0, y + h / 2.0)));
136        x += w + CARD_GAP;
137        row_h = row_h.max(h);
138    }
139}

141/// A card's occupied extent: the rotated footprint when allocated,
142/// else its measured minimum size as a floor estimate.
143fn card_extent(card: &Card) -> (f64, f64) {
144    let extents = gtk4::cairo::RectangleInt::new(0, 0, 0, 0);
145    card.footprint().extents(&extents);
146    if extents.width() > 0 && extents.height() > 0 {
147        (f64::from(extents.width()), f64::from(extents.height()))
148    } else {
149        let (w, _, _, _) = card.measure(Orientation::Horizontal, -1);
150        let (h, _, _, _) = card.measure(Orientation::Vertical, -1);
151        (f64::from(w).max(150.0), f64::from(h).max(80.0))
152    }
153}
