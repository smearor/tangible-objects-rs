1//! Card layout persistence - `gui-cards.toml` under
2//! `$XDG_CONFIG_HOME/tangible-objects/` (GUI.md §3 Persistence):
3//! position, rotation and scale of every `FloatingBoard` card
4//! restore on the next GUI start.

25/// Layout file name inside the `tangible-objects` config dir.
26const LAYOUT_FILE: &str = "gui-cards.toml";
27/// Minimum distance of the default arrangement from every board edge.
28const EDGE_MARGIN: f64 = 100.0;
29/// Gap between neighboring cards in the default row arrangement.
30const CARD_GAP: f64 = 24.0;
31
32/// The tidy-up strategy shared by the default arrangement and the
33/// menu's Arrange action: [`FlowLayout`] rows wrapped to fit, inside
34/// bounds inset by [`EDGE_MARGIN`].
35pub(crate) fn arrange_strategy() -> MarginLayout {
36    MarginLayout::new(
37        FlowLayout::<Pixels> {
38            gap: Distance::new(CARD_GAP),
39            ..Default::default()
40        },
41        Pixels::new(EDGE_MARGIN),
42    )
43}
44/// Debounce window for writing `layout_snapshot()` back - card
45/// gestures emit `geometry-changed` continuously while dragging.
46const SAVE_DEBOUNCE: Duration = Duration::from_millis(500);
47
48/// The layout path: `$XDG_CONFIG_HOME/tangible-objects/gui-cards.toml`,
49/// falling back to `~/.config/tangible-objects/gui-cards.toml` (same
50/// resolution the daemon's runtime store uses).
51fn layout_path() -> Option<PathBuf> {
52    let base = std::env::var_os("XDG_CONFIG_HOME")
53        .map(PathBuf::from)
54        .or_else(|| std::env::var_os("HOME").map(|home| PathBuf::from(home).join(".config")))?;
55    Some(base.join("tangible-objects").join(LAYOUT_FILE))
56}
57
58/// Loads the persisted layout; `None` when the file is missing,
59/// unreadable or holds no placements.
60pub(crate) fn load_layout() -> Option<BoardLayout> {
61    let path = layout_path()?;
62    let content = std::fs::read_to_string(&path).ok()?;
63    toml::from_str(&content)
64        .map_err(|error| warn!(path = %path.display(), %error, "card layout ignored"))
65        .ok()
66        .filter(|layout: &BoardLayout| !layout.is_empty())
67}
68
69/// Writes the layout snapshot into the config dir (best effort - a
70/// read-only home dir only loses persistence).
71fn save_layout(layout: &BoardLayout) {
72    let Some(path) = layout_path() else {
73        return;
74    };
75    let Some(dir) = path.parent() else {
76        return;
77    };
78    match toml::to_string_pretty(layout) {
79        Ok(serialized) => {
80            if let Err(error) =
81                std::fs::create_dir_all(dir).and_then(|_| std::fs::write(&path, serialized))
82            {
83                warn!(path = %path.display(), %error, "card layout save failed");
84            }
85        }
86        Err(error) => warn!(%error, "card layout serialization failed"),
87    }
88}

90/// Wires layout persistence for `board`:
91/// - restores `gui-cards.toml` once real bounds exist (clamping needs
92///   them - earlier restores would collapse every position),
93/// - arranges the default side-by-side layout when nothing was saved,
94/// - writes `layout_snapshot()` back [`SAVE_DEBOUNCE`] after the last
95///   `geometry-changed`.
96///
97/// `layout` is the result of [`load_layout`] - the caller loads it
98/// once so other consumers (mode-driven cards) can peek at saved
99/// placements too.
100pub(crate) fn wire_layout(board: &FloatingBoard, layout: Option<BoardLayout>) {
101    let board_apply = board.clone();
102    // Poll until bounds and card footprints exist - restoring or
103    // arranging earlier would clamp every position to the origin.
104    gtk4::glib::timeout_add_local(Duration::from_millis(50), move || {
105        let allocated = board_apply.width() > 0
106            && board_apply.height() > 0
107            && board_apply
108                .cards()
109                .iter()
110                .all(|card| card_extent(card).0 > 0.0);
111        if !allocated {
112            return ControlFlow::Continue;
113        }
114        match &layout {
115            Some(layout) => board_apply.restore_layout(layout),
116            // No saved layout: flow every card (hidden ones included -
117            // the menu opens them later at their arranged spot).
118            None => {
119                let ids: Vec<CardId> = board_apply.cards().iter().map(|card| card.id()).collect();
120                if let Err(error) = board_apply.apply_layout(&arrange_strategy(), Some(&ids), false)
121                {
122                    warn!(%error, "default card arrangement does not fit");
123                }
124            }
125        }
126        ControlFlow::Break
127    });
128
129    let dirty = Rc::new(Cell::new(false));
130    let dirty_flag = dirty.clone();
131    board.connect_local("geometry-changed", false, move |_| {
132        dirty_flag.set(true);
133        None
134    });
135    let board_save = board.clone();
136    gtk4::glib::timeout_add_local(SAVE_DEBOUNCE, move || {
137        if dirty.replace(false) {
138            save_layout(&board_save.layout_snapshot());
139        }
140        ControlFlow::Continue
141    });
142}

144/// A card's occupied extent: the rotated footprint when allocated,
145/// else its measured minimum size as a floor estimate.
146fn card_extent(card: &Card) -> (f64, f64) {
147    let extents = gtk4::cairo::RectangleInt::new(0, 0, 0, 0);
148    card.footprint().extents(&extents);
149    if extents.width() > 0 && extents.height() > 0 {
150        (f64::from(extents.width()), f64::from(extents.height()))
151    } else {
152        let (w, _, _, _) = card.measure(Orientation::Horizontal, -1);
153        let (h, _, _, _) = card.measure(Orientation::Vertical, -1);
154        (f64::from(w).max(150.0), f64::from(h).max(80.0))
155    }
156}
