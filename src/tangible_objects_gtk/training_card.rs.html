50/// Tolerance the saved profile takes (scalar mm).
51const DEFAULT_TOLERANCE: PatternTolerance = PatternTolerance::new(3.0);
52
53/// Builds the training-card content: name entry + save button + a
54/// stability status line. The entry activates in training mode, the
55/// save button once the contact set held stable
56/// (`STABLE_THRESHOLD` frames, ≥3 points) - the status line explains
57/// whichever gate is currently closed.
58pub fn create_training_card(
59    state: &Arc<Mutex<SharedState>>,
60    profiles_dir: &Option<String>,
61) -> GtkBox {
62    let name_entry = Entry::new();
63    name_entry.set_placeholder_text(Some(&fl!(language_loader(), "figure-name-placeholder")));
64    name_entry.set_width_chars(20);
65    name_entry.set_sensitive(false);
66
67    let save_btn = Button::with_label(&fl!(language_loader(), "save-profile"));
68    save_btn.set_sensitive(false);
69
70    let status = Label::new(Some(""));
71    status.set_halign(Align::Start);
72
73    // Save callback.
74    let state_save = state.clone();
75    let name_entry_save = name_entry.clone();
76    let profiles_dir_save = profiles_dir.clone();
77    save_btn.connect_clicked(move |_| {
78        let name: String = name_entry_save.text().to_string();
79        if name.is_empty() {
80            return;
81        }
82
83        let points: Vec<Vec2<Millimeters>> = match state_save.lock() {
84            Ok(s) => {
85                if !s.stable {
86                    return;
87                }
88                s.touches.iter().map(|t| t.pos).collect()
89            }
90            Err(_) => return,
91        };
92
93        if points.len() < 3 {
94            return;
95        }
96
97        let profile = ProfileFile::from_points(&name, &points, DEFAULT_TOLERANCE);
98        let dir = profiles_dir_save.as_deref().unwrap_or("profiles");
99        std::fs::create_dir_all(dir).ok();
100        let filename = format!("{}/{}.toml", dir, name.to_lowercase().replace(' ', "-"));
101        let filepath = PathBuf::from(&filename);
102
103        match profile.save(&filepath) {
104            Ok(()) => {
105                info!(path = %filepath.display(), "profile saved");
106                // The file itself is the persistence - register the
107                // pattern for this session so it matches and shows up
108                // on the profiles card without a restart.
109                let (reply, _ignored) = oneshot::channel();
110                send_command(
111                    &state_save,
112                    ControlCommand::Register {
113                        registration: Box::new(Registration::Pattern(profile)),
114                        owner: None,
115                        lifetime: Lifetime::Session,
116                        reply,
117                    },
118                );
119            }
120            Err(e) => error!(error = %e, "profile save failed"),
121        }
122    });
123
124    let body = GtkBox::new(Orientation::Vertical, 6);
125    body.append(&name_entry);
126    body.append(&save_btn);
127    body.append(&status);
128
129    // Sensitivity + status text follow the shared state every tick -
130    // the mode itself flips via the info card's mode button.
131    let state_status = state.clone();
132    let name_entry_status = name_entry.clone();
133    let save_btn_status = save_btn.clone();
134    let mut applied_mode = state_status.lock().map(|s| s.mode).unwrap_or(GuiMode::Live);
135    // The card is built while training mode is already active -
136    // `applied_mode` starts at Training then, so the entry's initial
137    // `set_sensitive` below must run unconditionally on the first
138    // tick, not only on a mode transition.
139    name_entry_status.set_sensitive(applied_mode == GuiMode::Training);
140    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
141        let s = match state_status.lock() {
142            Ok(s) => s,
143            Err(_) => return ControlFlow::Continue,
144        };
145        if s.mode != applied_mode {
146            applied_mode = s.mode;
147            name_entry_status.set_sensitive(applied_mode == GuiMode::Training);
148        }
149        let count = s.touches.len();
150        match s.mode {
151            GuiMode::Live | GuiMode::SpellTraining => {
152                status.set_label(&fl!(language_loader(), "train-hint"));
153            }
154            GuiMode::Training => {
155                if let Some(err) = &s.device_error {
156                    status.set_label(err);
157                } else if let Some(name) = &s.training_match {
158                    status.set_label(&fl!(
159                        language_loader(),
160                        "train-duplicate",
161                        name = name.as_str()
162                    ));
163                } else if count < 3 {
164                    status.set_label(&fl!(
165                        language_loader(),
166                        "wait-for-points",
167                        count = (count as i64)
168                    ));
169                } else if s.stable {
170                    status.set_label(&fl!(language_loader(), "stable-hint"));
171                } else {
172                    status.set_label(&fl!(
173                        language_loader(),
174                        "stabilizing",
175                        stable = (s.stable_frames as i64),
176                        threshold = (STABLE_THRESHOLD as i64)
177                    ));
178                }
179            }
180        }
181        save_btn_status.set_sensitive(s.stable);
182        ControlFlow::Continue
183    });
184
185    body
186}

188/// Board id of the training card - the mode watcher adds/removes it,
189/// so it stays out of the static card set.
190const TRAINING_CARD_ID: &str = "training";
191
192/// The training card's color scheme: a muted green header signals
193/// capture mode - consistent with the `TouchStable` scene color that
194/// marks the stable contact set the card waits for.
195fn training_palette() -> CardPalette {
196    let mut palette = CardPalette::new();
197    palette.insert(CardColor::Header, Color::<f64>::new(0.1, 0.3, 0.15));
198    palette
199}

201/// Poll cadence for mode changes (same 100 ms as the other card
202/// syncs).
203const MODE_POLL: Duration = Duration::from_millis(100);
204
205/// Wires the training card's mode lifecycle: entering training mode
206/// adds the card to the board (restoring its last geometry and fold
207/// state), leaving removes it again. `initial` is the placement the
208/// persisted layout held for the card - the first add restores it
209/// exactly; later cycles reuse the pose captured at removal.
210pub fn wire_training_card(
211    board: &FloatingBoard,
212    state: &Arc<Mutex<SharedState>>,
213    profiles_dir: &Option<String>,
214    initial: Option<CardPlacement>,
215) {
216    let last = Rc::new(RefCell::new(initial));
217    let board_tick = board.clone();
218    let state_tick = state.clone();
219    let profiles_dir_tick = profiles_dir.clone();
220    let card_id = CardId::new(TRAINING_CARD_ID);
221    let mut mode = state.lock().map(|s| s.mode).unwrap_or(GuiMode::Live);
222    if mode == GuiMode::Training {
223        add_training_card(&board_tick, &state_tick, &profiles_dir_tick, &last);
224    }
225    gtk4::glib::timeout_add_local(MODE_POLL, move || {
226        let Ok(s) = state_tick.lock() else {
227            return ControlFlow::Continue;
228        };
229        let current = s.mode;
230        drop(s);
231        if current == mode {
232            return ControlFlow::Continue;
233        }
234        mode = current;
235        match mode {
236            GuiMode::Training => {
237                add_training_card(&board_tick, &state_tick, &profiles_dir_tick, &last);
238            }
239            GuiMode::Live | GuiMode::SpellTraining => {
240                if let Ok(card) = board_tick.card(&card_id) {
241                    // Stash the pose so re-adding restores it; the
242                    // periodic layout save skips absent cards anyway.
243                    *last.borrow_mut() = Some(CardPlacement::new(
244                        card_id.clone(),
245                        card.geometry(),
246                        card.folded(),
247                    ));
248                    let _ = board_tick.remove_card(&card_id);
249                }
250            }
251        }
252        ControlFlow::Continue
253    });
254}

256/// Adds the training card to `board` and applies the stashed
257/// placement (geometry + fold) when one exists.
258fn add_training_card(
259    board: &FloatingBoard,
260    state: &Arc<Mutex<SharedState>>,
261    profiles_dir: &Option<String>,
262    last: &Rc<RefCell<Option<CardPlacement>>>,
263) {
264    let card_id = CardId::new(TRAINING_CARD_ID);
265    if board.card(&card_id).is_ok() {
266        return;
267    }
268    let body = create_training_card(state, profiles_dir);
269    let _ = board.add_card_with_id(
270        card_id.clone(),
271        &body,
272        CardConfig {
273            title: Some(fl!(language_loader(), "card-training-title")),
274            position: CardAnchor::BottomLeft,
275            anchor_margin: Pixels::new(100.0),
276            bounds_mode: BoundsMode::Contained,
277            content_padding: crate::CARD_CONTENT_PADDING,
278            foldable: true,
279            unfold_on_click: true,
280            resizable: true,
281            rotate_grip: true,
282            palette: Some(training_palette()),
283            chrome_style: ChromeStyle::Adwaita,
284            chrome: Some(crate::card_chrome(false)),
285            ..Default::default()
286        },
287    );
288    let Ok(card) = board.card(&card_id) else {
289        return;
290    };
291    if let Some(placement) = &*last.borrow() {
292        card.set_geometry(&placement.geometry);
293        card.set_folded(placement.folded);
294    }
