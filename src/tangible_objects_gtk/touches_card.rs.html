1//! The touches card - live raw contacts (slot, tracking id, surface
2//! position) as a movable card on the `FloatingBoard`.
3
4use std::cell::RefCell;
5use std::rc::Rc;
6use std::sync::Arc;
7use std::sync::Mutex;
8use std::time::Duration;
9
10use gtk4::Align;
11use gtk4::Box as GtkBox;
12use gtk4::Grid;
13use gtk4::Label;
14use gtk4::Orientation;
15use gtk4::glib::ControlFlow;
16use gtk4::prelude::*;
17use i18n_embed_fl::fl;
18
19use crate::language_loader;
20use crate::shared_state::SharedState;
21
22/// Rows are reserved in chunks of this size and never removed once
23/// created - the card always keeps room for the highest chunk it has
24/// seen, so contacts coming and going do not resize it.
25const TOUCH_ROW_CHUNK: usize = 5;
26
27/// Widgets of one contact row - rows persist (blanked while unused),
28/// so the grid keeps a stable size while contacts come and go.
29struct TouchRow {
30    /// Multitouch slot cell.
31    slot: Label,
32    /// Kernel tracking id cell.
33    tracking_id: Label,
34    /// Current surface position cell.
35    position: Label,
36}
37
38/// Builds the touches-card content: `slot | tracking id | x,y mm`
39/// rows on a persistent grid sized in `TOUCH_ROW_CHUNK` chunks (grown
40/// ahead, never shrunk) so the card does not resize as contacts come
41/// and go; cell contents updated every 100 ms.
42pub fn create_touches_card(state: &Arc<Mutex<SharedState>>) -> GtkBox {
43    let grid = Grid::new();
44    grid.set_column_spacing(12);
45    grid.set_row_spacing(4);
46
47    let headers = [
48        fl!(language_loader(), "touch-slot"),
49        fl!(language_loader(), "touch-id"),
50        fl!(language_loader(), "touch-pos"),
51    ];
52    for (col, label) in headers.iter().enumerate() {
53        let header = Label::new(Some(label));
54        header.set_halign(Align::Start);
55        header.set_opacity(0.6);
56        grid.attach(&header, col as i32, 0, 1, 1);
57    }
58
59    // Invisible sizing labels keep the columns from collapsing when
60    // few contacts (or none) are live - they measure like real
61    // content but never render.
62    for (col, sample) in ["99", "9999999", "9999,9999 mm"].iter().enumerate() {
63        let ghost = Label::new(Some(sample));
64        ghost.set_halign(Align::Start);
65        ghost.set_opacity(0.0);
66        grid.attach(&ghost, col as i32, 1, 1, 1);
67    }
68
69    let empty_label = Label::new(Some(&fl!(language_loader(), "touches-empty")));
70    empty_label.set_halign(Align::Start);
71    empty_label.set_opacity(0.5);
72    empty_label.set_visible(false);
73    grid.attach(&empty_label, 0, 1, 3, 1);
74
75    let rows: Rc<RefCell<Vec<TouchRow>>> = Rc::new(RefCell::new(Vec::new()));
76    let state_tick = state.clone();
77    let rows_tick = rows.clone();
78    let grid_tick = grid.clone();
79    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
80        let s = match state_tick.lock() {
81            Ok(s) => s,
82            Err(_) => return ControlFlow::Continue,
83        };
84        let mut rows = rows_tick.borrow_mut();
85        // One chunk is always reserved ahead: reaching the current
86        // capacity appends another chunk instead of resizing the card
87        // for every new contact.
88        let needed = TOUCH_ROW_CHUNK * (s.touches.len() / TOUCH_ROW_CHUNK + 1);
89        while rows.len() < needed {
90            let slot = Label::new(Some(" "));
91            slot.set_halign(Align::Start);
92            let tracking_id = Label::new(Some(" "));
93            tracking_id.set_halign(Align::Start);
94            let position = Label::new(Some(" "));
95            position.set_halign(Align::End);
96            let grid_row = (rows.len() + 1) as i32;
97            grid_tick.attach(&slot, 0, grid_row, 1, 1);
98            grid_tick.attach(&tracking_id, 1, grid_row, 1, 1);
99            grid_tick.attach(&position, 2, grid_row, 1, 1);
100            rows.push(TouchRow {
101                slot,
102                tracking_id,
103                position,
104            });
105        }
106        empty_label.set_visible(s.touches.is_empty());
107        for (index, row) in rows.iter().enumerate() {
108            match s.touches.get(index) {
109                Some(touch) => {
110                    row.slot.set_label(&touch.slot.as_u32().to_string());
111                    row.tracking_id
112                        .set_label(&touch.tracking_id.as_i32().to_string());
113                    row.position
114                        .set_label(&format!("{:.0},{:.0} mm", touch.pos.x, touch.pos.y));
115                }
116                None => {
117                    row.slot.set_label(" ");
118                    row.tracking_id.set_label(" ");
119                    row.position.set_label(" ");
120                }
121            }
122        }
123        ControlFlow::Continue
124    });
125
126    let body = GtkBox::new(Orientation::Vertical, 0);
127    body.append(&grid);
128    body
129}
