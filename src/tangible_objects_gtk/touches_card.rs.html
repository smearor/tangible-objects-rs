1//! The touches card - live raw contacts (slot, tracking id, surface
2//! position) as a movable card on the `FloatingBoard`.
3
4use std::cell::RefCell;
5use std::rc::Rc;
6use std::sync::Arc;
7use std::sync::Mutex;
8use std::time::Duration;
9
10use gtk4::Box as GtkBox;
11use gtk4::Grid;
12use gtk4::Label;
13use gtk4::Orientation;
14use gtk4::glib::ControlFlow;
15use gtk4::prelude::*;
16use i18n_embed_fl::fl;
17
18use crate::language_loader;
19use crate::shared_state::SharedState;
20
21/// Widgets of one contact row - the row exists while the contact
22/// (slot + tracking id) is live; only the position label rewrites.
23struct TouchRow {
24    /// Multitouch slot the contact occupies.
25    slot: u32,
26    /// Kernel tracking id of the contact lifetime.
27    tracking_id: i32,
28    /// Current surface position cell.
29    position: Label,
30}
31
32/// Builds the touches-card content: `slot | tracking id | x,y mm`
33/// rows rebuilt whenever the contact set changes (a slot that swaps
34/// tracking ids is a fresh contact), positions updated every 100 ms.
35pub fn create_touches_card(state: &Arc<Mutex<SharedState>>) -> GtkBox {
36    let grid = Grid::new();
37    grid.set_column_spacing(12);
38    grid.set_row_spacing(4);
39
40    let headers = [
41        fl!(language_loader(), "touch-slot"),
42        fl!(language_loader(), "touch-id"),
43        fl!(language_loader(), "touch-pos"),
44    ];
45    for (col, label) in headers.iter().enumerate() {
46        let header = Label::new(Some(label));
47        header.set_halign(gtk4::Align::Start);
48        header.set_opacity(0.6);
49        grid.attach(&header, col as i32, 0, 1, 1);
50    }
51
52    let rows: Rc<RefCell<Vec<TouchRow>>> = Rc::new(RefCell::new(Vec::new()));
53    let empty_shown = Rc::new(std::cell::Cell::new(false));
54    let state_tick = state.clone();
55    let rows_tick = rows.clone();
56    let grid_tick = grid.clone();
57    let empty_label = fl!(language_loader(), "touches-empty");
58    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
59        let s = match state_tick.lock() {
60            Ok(s) => s,
61            Err(_) => return ControlFlow::Continue,
62        };
63        let mut rows = rows_tick.borrow_mut();
64        let contacts: Vec<_> = s
65            .touches
66            .iter()
67            .map(|t| (t.slot.as_u32(), t.tracking_id.as_i32()))
68            .collect();
69        let stale = rows.len() != contacts.len()
70            || rows
71                .iter()
72                .zip(contacts.iter())
73                .any(|(row, (slot, id))| row.slot != *slot || row.tracking_id != *id);
74        if stale {
75            // Rebuild rows 1..: contact rows plus a possibly visible
76            // "no contacts" label (new, ended or re-tracked contact).
77            let below_header = rows.len() + usize::from(empty_shown.get());
78            for _ in 0..below_header {
79                grid_tick.remove_row(1);
80            }
81            rows.clear();
82            empty_shown.set(false);
83            if contacts.is_empty() {
84                let label = Label::new(Some(&empty_label));
85                label.set_halign(gtk4::Align::Start);
86                label.set_opacity(0.5);
87                grid_tick.attach(&label, 0, 1, 3, 1);
88                empty_shown.set(true);
89            } else {
90                for (row_index, (slot, tracking_id)) in contacts.iter().enumerate() {
91                    let slot_label = Label::new(Some(&slot.to_string()));
92                    slot_label.set_halign(gtk4::Align::Start);
93                    let id_label = Label::new(Some(&tracking_id.to_string()));
94                    id_label.set_halign(gtk4::Align::Start);
95                    let position = Label::new(Some("–"));
96                    position.set_halign(gtk4::Align::End);
97                    let grid_row = (row_index + 1) as i32;
98                    grid_tick.attach(&slot_label, 0, grid_row, 1, 1);
99                    grid_tick.attach(&id_label, 1, grid_row, 1, 1);
100                    grid_tick.attach(&position, 2, grid_row, 1, 1);
101                    rows.push(TouchRow {
102                        slot: *slot,
103                        tracking_id: *tracking_id,
104                        position,
105                    });
106                }
107            }
108        }
109        for (row, touch) in rows.iter().zip(s.touches.iter()) {
110            row.position
111                .set_label(&format!("{:.0},{:.0} mm", touch.pos.x, touch.pos.y));
112        }
113        ControlFlow::Continue
114    });
115
116    let body = GtkBox::new(Orientation::Vertical, 0);
117    body.append(&grid);
118    body
119}
