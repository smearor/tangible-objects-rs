21/// Builds the metrics-card content: a `label | value` grid refreshed
22/// from `SharedState` every 100 ms. The fps row derives from the
23/// frame counter delta between ticks - a rolling rate, not the
24/// cumulative average.
25pub fn create_metrics_card(state: &Arc<Mutex<SharedState>>) -> GtkBox {
26    let grid = Grid::new();
27    grid.set_column_spacing(12);
28    grid.set_row_spacing(4);
29
30    let frames_value = add_row(&grid, 0, &fl!(language_loader(), "metric-frames"));
31    let fps_value = add_row(&grid, 1, &fl!(language_loader(), "metric-fps"));
32    let contacts_value = add_row(&grid, 2, &fl!(language_loader(), "metric-contacts"));
33    let tracked_value = add_row(&grid, 3, &fl!(language_loader(), "metric-tracked"));
34    let active_value = add_row(&grid, 4, &fl!(language_loader(), "metric-active"));
35    let generic_value = add_row(&grid, 5, &fl!(language_loader(), "metric-generic"));
36    let candidates_value = add_row(&grid, 6, &fl!(language_loader(), "metric-candidates"));
37    let tangibles_value = add_row(&grid, 7, &fl!(language_loader(), "metric-tangibles"));
38
39    let state_tick = state.clone();
40    let mut last_sample: Option<(u64, Instant)> = None;
41    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
42        let s = match state_tick.lock() {
43            Ok(s) => s,
44            Err(_) => return ControlFlow::Continue,
45        };
46        let metrics = &s.metrics;
47        let frames = metrics.frames.as_u64();
48        frames_value.set_label(&frames.to_string());
49        let now = Instant::now();
50        fps_value.set_label(&match last_sample.replace((frames, now)) {
51            Some((prev_frames, prev_at)) => {
52                let elapsed = now.duration_since(prev_at).as_secs_f64();
53                format!("{:.1}", (frames - prev_frames) as f64 / elapsed.max(1e-6))
54            }
55            None => "–".to_string(),
56        });
57        contacts_value.set_label(&metrics.contacts.as_u32().to_string());
58        tracked_value.set_label(&format!(
59            "{} ({:.0}%)",
60            metrics.tracked_frames.as_u64(),
61            percent(metrics.tracked_frames.as_u64(), frames)
62        ));
63        active_value.set_label(&format!(
64            "{} ({:.0}%)",
65            metrics.active_frames.as_u64(),
66            percent(metrics.active_frames.as_u64(), frames)
67        ));
68        generic_value.set_label(&format!(
69            "{} ({:.0}%)",
70            metrics.generic_frames.as_u64(),
71            percent(metrics.generic_frames.as_u64(), frames)
72        ));
73        candidates_value.set_label(&metrics.candidates_tested.as_u64().to_string());
74        tangibles_value.set_label(&metrics.tangible_count.as_u64().to_string());
75        ControlFlow::Continue
76    });
77
78    let body = GtkBox::new(Orientation::Vertical, 0);
79    body.append(&grid);
80    body
81}

83/// `part` as a percentage of `total`, `0` for an empty session.
84fn percent(part: u64, total: u64) -> f64 {
85    if total == 0 {
86        0.0
87    } else {
88        part as f64 / total as f64 * 100.0
89    }
90}

92/// Appends a `label | value` row to `grid` at `row`; returns the
93/// value label so the refresh tick can rewrite it.
94fn add_row(grid: &Grid, row: i32, name: &str) -> Label {
95    let name_label = Label::new(Some(name));
96    name_label.set_halign(gtk4::Align::Start);
97    name_label.set_opacity(0.6);
98    let value = Label::new(Some("–"));
99    value.set_halign(gtk4::Align::End);
100    grid.attach(&name_label, 0, row, 1, 1);
101    grid.attach(&value, 1, row, 1, 1);
102    value
103}
