1//! `count_badge` - the compact folded-content widget for data cards:
2//! an icon plus a live `x / y` count. The badge is display-only - no
3//! child widget claims the press, so the card's own gestures handle
4//! it: `unfold_on_click` unfolds on tap and `WholeCardFolded` drags
5//! the folded card (a `Button` here would claim the sequence and
6//! break both).
7
8use gtk4::Box as GtkBox;
9use gtk4::Image;
10use gtk4::Label;
11use gtk4::Orientation;
12use gtk4::prelude::*;
13
14/// Builds a folded-content badge: `icon_name` centered over a smaller
15/// count label. Returns the badge widget (for
16/// `CardConfig::folded_content`) and the label the caller refreshes
17/// from its state ticks.
18pub(crate) fn count_badge(icon_name: &str, tooltip: &str) -> (GtkBox, Label) {
19    let badge = GtkBox::new(Orientation::Vertical, 0);
20    badge.set_tooltip_text(Some(tooltip));
21    let icon = Image::from_icon_name(icon_name);
22    icon.set_pixel_size(36);
23    icon.set_hexpand(true);
24    icon.set_vexpand(true);
25    badge.append(&icon);
26    let count = Label::new(None);
27    count.add_css_class("caption");
28    badge.append(&count);
29    (badge, count)
30}
