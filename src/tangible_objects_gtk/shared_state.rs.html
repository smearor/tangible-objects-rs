20/// Snapshot the draw functions consume - written by the device thread
21/// once per `SYN_REPORT` frame, read by the ~30 fps GTK timer.
22pub struct SharedState {
23    /// Active touch points.
24    pub touches: Vec<TouchPoint>,
25    /// Raw pattern matches of the last frame (live view).
26    pub matches: Vec<PatternMatch>,
27    /// Currently tracked tangibles (pending flags render dimmed).
28    pub tracked: Vec<TrackedTangible>,
29    /// Effective axis calibration (mm conversion for the canvas scale).
30    pub calibration: Option<Calibration>,
31    /// Consecutive frames the contact set held the same size.
32    pub stable_frames: u32,
33    /// Contact set stable long enough for profile capture.
34    pub stable: bool,
35    /// Live vs. training visualization.
36    pub mode: GuiMode,
37    /// Device-open failure shown instead of the surface.
38    pub device_error: Option<String>,
39    /// User-facing message pushed via an interface (`ShowMessage`).
40    pub message: Option<String>,
41    /// Whether the window should be shown (`SetGuiVisible`).
42    pub visible: bool,
43    /// Display rotation the mm→px mapping applies (`--display-rotation`)
44    /// - the evdev stream always speaks the panel's native frame.
45    pub rotation: DisplayRotation,
46    /// Canvas background opacity (`--background-alpha`, the alpha
47    /// toggle and `SetBackgroundAlpha` commands).
48    pub background_alpha: BackgroundAlpha,
49    /// Alpha change requested by the control-bar toggle - the device
50    /// thread drains it into the runtime (`SetBackgroundAlpha`
51    /// semantics) so the interface-visible state stays authoritative.
52    pub background_alpha_request: Option<BackgroundAlpha>,
53    /// Registered dead zones - the overlay data set.
54    pub dead_zones: Vec<DeadZone>,
55    /// Whether the dead-zone overlay is drawn
56    /// (`SetDeadZonesVisible`/`--dead-zones-visible`).
57    pub dead_zones_visible: bool,
58    /// Contacts dropped by dead-zone filtering on the last frame.
59    pub dead_zone_filtered: ContactCount,
60    /// Registered field sets (snapshot `field_sets` count).
61    pub field_set_count: FieldSetCount,
62    /// Occupancy fields across all field sets.
63    pub field_count: FieldCount,
64    /// Fields currently occupied.
65    pub occupied_fields: FieldCount,
66}
67
68impl SharedState {
69    /// Initial state - hidden-only devices start visible (the
70    /// `--gui` flag semantics), `SetGuiVisible` toggles afterwards.
71    pub fn new(
72        visible: bool,
73        rotation: DisplayRotation,
74        background_alpha: BackgroundAlpha,
75        dead_zones_visible: bool,
76    ) -> Arc<Mutex<Self>> {
77        Arc::new(Mutex::new(Self {
78            touches: Vec::new(),
79            matches: Vec::new(),
80            tracked: Vec::new(),
81            calibration: None,
82            stable_frames: 0,
83            stable: false,
84            mode: GuiMode::Live,
85            device_error: None,
86            message: None,
87            visible,
88            rotation,
89            background_alpha,
90            background_alpha_request: None,
91            dead_zones: Vec::new(),
92            dead_zones_visible,
93            dead_zone_filtered: ContactCount::default(),
94            field_set_count: FieldSetCount::default(),
95            field_count: FieldCount::default(),
96            occupied_fields: FieldCount::default(),
97        }))
98    }
99}
100
101/// Whether the shared state is in live mode.
102pub fn mode_is_live(state: &Arc<Mutex<SharedState>>) -> bool {
103    state
104        .lock()
105        .map(|s| s.mode == GuiMode::Live)
106        .unwrap_or(true)
