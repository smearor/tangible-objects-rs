1//! The render model shared between the device thread (writer) and the
2//! GTK main loop (reader) - the GUI's entire cross-thread contract.
3
4use std::sync::Arc;

7use tangible_objects_model::Calibration;

12
13use crate::BackgroundAlpha;

16/// Snapshot the draw functions consume - written by the device thread
17/// once per `SYN_REPORT` frame, read by the ~30 fps GTK timer.
18pub struct SharedState {
19    /// Active touch points.
20    pub touches: Vec<TouchPoint>,
21    /// Raw pattern matches of the last frame (live view).
22    pub matches: Vec<PatternMatch>,
23    /// Currently tracked tangibles (pending flags render dimmed).
24    pub tracked: Vec<TrackedTangible>,
25    /// Effective axis calibration (mm conversion for the canvas scale).
26    pub calibration: Option<Calibration>,
27    /// Consecutive frames the contact set held the same size.
28    pub stable_frames: u32,
29    /// Contact set stable long enough for profile capture.
30    pub stable: bool,
31    /// Live vs. training visualization.
32    pub mode: GuiMode,
33    /// Device-open failure shown instead of the surface.
34    pub device_error: Option<String>,
35    /// User-facing message pushed via an interface (`ShowMessage`).
36    pub message: Option<String>,
37    /// Whether the window should be shown (`SetGuiVisible`).
38    pub visible: bool,
39    /// Display rotation the mm→px mapping applies (`--display-rotation`)
40    /// - the evdev stream always speaks the panel's native frame.
41    pub rotation: DisplayRotation,
42    /// Canvas background opacity (`--background-alpha` and the alpha
43    /// toggle). GUI-local display state - no control command.
44    pub background_alpha: BackgroundAlpha,
45}
46
47impl SharedState {
48    /// Initial state - hidden-only devices start visible (the
49    /// `--gui` flag semantics), `SetGuiVisible` toggles afterwards.
50    pub fn new(
51        visible: bool,
52        rotation: DisplayRotation,
53        background_alpha: BackgroundAlpha,
54    ) -> Arc<Mutex<Self>> {
55        Arc::new(Mutex::new(Self {
56            touches: Vec::new(),
57            matches: Vec::new(),
58            tracked: Vec::new(),
59            calibration: None,
60            stable_frames: 0,
61            stable: false,
62            mode: GuiMode::Live,
63            device_error: None,
64            message: None,
65            visible,
66            rotation,
67            background_alpha,
68        }))
69    }
70}
71
72/// Whether the shared state is in live mode.
73pub fn mode_is_live(state: &Arc<Mutex<SharedState>>) -> bool {
74    state
75        .lock()
76        .map(|s| s.mode == GuiMode::Live)
77        .unwrap_or(true)
