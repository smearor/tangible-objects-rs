25/// Per-figure accent color - stable hash over the pattern ID.
26fn figure_color(id: PatternId) -> (f64, f64, f64) {
27    match id.as_u32() % 6 {
28        0 => (0.2, 0.6, 1.0),
29        1 => (0.2, 0.9, 0.3),
30        2 => (1.0, 0.5, 0.2),
31        3 => (1.0, 0.3, 0.5),
32        4 => (0.8, 0.3, 1.0),
33        _ => (1.0, 0.9, 0.2),
34    }
35}

37/// Draws the full frame: background, overlays, mode scene.
38pub fn draw_scene(cr: &cairo::Context, width: i32, height: i32, state: &Arc<Mutex<SharedState>>) {
39    let w = width as f64;
40    let h = height as f64;
41
42    let s = match state.lock() {
43        Ok(s) => s,
44        Err(_) => return,
45    };
46
47    // The canvas background is the drawing area's own CSS
48    // `background-color` (`.surface-bg`, alpha via `BackgroundAlpha`)
49    // - a cairo fill would sit *above* that layer as an opaque cover,
50    // so nothing is painted here; the widget stays transparent and the
51    // desktop shows through below 100% alpha.
52
53    // Show device error.
54    if let Some(err) = &s.device_error {
55        cr.set_source_rgb(1.0, 0.3, 0.3);
56        cr.set_font_size(24.0);
57        cr.move_to(20.0, 40.0);
58        let _ = cr.show_text(err);
59        return;
60    }
61
62    // User-facing message pushed via an interface (ShowMessage).
63    if let Some(msg) = &s.message {
64        cr.set_source_rgb(1.0, 0.9, 0.2);
65        cr.set_font_size(42.0);
66        if let Ok(ext) = cr.text_extents(msg) {
67            cr.move_to((w - ext.width()) / 2.0, 64.0);
68        } else {
69            cr.move_to(20.0, 64.0);
70        }
71        let _ = cr.show_text(msg);
72    }
73
74    let cal = match s.calibration {
75        Some(c) => c,
76        None => {
77            cr.set_source_rgb(0.8, 0.8, 0.8);
78            cr.set_font_size(20.0);
79            cr.move_to(20.0, 40.0);
80            let _ = cr.show_text(&fl!(language_loader(), "no-calibration"));
81            return;
82        }
83    };
84
85    let mode = s.mode;
86    let touches = &s.touches;
87    let matches = &s.matches;
88
89    // Map touch points from mm to screen pixels.
90    let rotation = s.rotation;
91    let to_screen = |pos: &Vec2<Millimeters>| -> Vec2<Pixels> {
92        cal.mm_to_screen(*pos, Vec2::new(w, h), rotation)
93    };
94
95    if s.dead_zones_visible {
96        draw_dead_zones(cr, &s.dead_zones, &to_screen);
97    }
98    if s.edge_margin_visible {
99        draw_edge_margin(cr, s.edge_margin, &cal, &to_screen);
100    }
101    if s.fields_visible {
102        draw_fields(cr, &s.field_sets, &to_screen);
103    }
104
105    if mode == GuiMode::Training {
106        draw_training(cr, w, h, touches.into(), &to_screen, &s);
107    } else {
108        draw_live(
109            cr,
110            touches.into(),
111            matches,
112            (&s.tracked).into(),
113            &to_screen,
114            &s,
115        );
116    }
117
118    if s.positions_visible {
119        draw_positions(
120            cr,
121            touches.into(),
122            (&s.tracked).into(),
123            &s.field_sets,
124            &to_screen,
125        );
126    }
127    if s.ruler_visible {
128        draw_ruler(cr, w, h, &cal, &to_screen);
129    }
130}

132/// Dead-zone overlay: a dashed circle and the zone's name on every
133/// suppressed spot, drawn beneath the figure/touch layer.
134fn draw_dead_zones(
135    cr: &cairo::Context,
136    zones: &[DeadZone],
137    to_screen: &impl Fn(&Vec2<Millimeters>) -> Vec2<Pixels>,
138) {
139    for zone in zones {
140        let center = to_screen(&zone.center);
141        // The mm→px scale may be anisotropic - sample the radius in
142        // both axes and take the mean so the circle stays circular.
143        let rx = to_screen(&(zone.center + Vec2::new(zone.radius.as_f64(), 0.0)))
144            .distance(&center)
145            .as_f64();
146        let ry = to_screen(&(zone.center + Vec2::new(0.0, zone.radius.as_f64())))
147            .distance(&center)
148            .as_f64();
149        let radius = (rx + ry) / 2.0;
150
151        cr.set_source_rgba(1.0, 0.3, 0.25, 0.10);
152        cr.arc(center.x, center.y, radius, 0.0, 2.0 * std::f64::consts::PI);
153        let _ = cr.fill();
154        cr.set_source_rgba(1.0, 0.3, 0.25, 0.85);
155        cr.set_dash(&[6.0, 4.0], 0.0);
156        cr.set_line_width(2.0);
157        cr.arc(center.x, center.y, radius, 0.0, 2.0 * std::f64::consts::PI);
158        let _ = cr.stroke();
159        cr.set_dash(&[], 0.0);
160
161        cr.set_font_size(16.0);
162        cr.set_source_rgba(1.0, 0.55, 0.5, 0.9);
163        let name = zone.name.as_str();
164        let label_y = center.y + radius + 18.0;
165        match cr.text_extents(name) {
166            Ok(ext) => cr.move_to(center.x - ext.width() / 2.0, label_y),
167            Err(_) => cr.move_to(center.x, label_y),
168        }
169        let _ = cr.show_text(name);
170    }
171}

173/// Edge-margin overlay: semi-transparent bands and a dashed inner
174/// border marking the strip where contacts are ingest-filtered.
175fn draw_edge_margin(
176    cr: &cairo::Context,
177    margin: Millimeters,
178    cal: &Calibration,
179    to_screen: &impl Fn(&Vec2<Millimeters>) -> Vec2<Pixels>,
180) {
181    let m = margin.as_f64();
182    if m <= 0.0 {
183        return;
184    }
185    let w_mm = cal.width_mm().as_f64();
186    let h_mm = cal.height_mm().as_f64();
187    let origin = to_screen(&Vec2::<Millimeters>::new(0.0, 0.0));
188    let extent = to_screen(&Vec2::<Millimeters>::new(w_mm, h_mm));
189    let inner_tl = to_screen(&Vec2::<Millimeters>::new(m, m));
190    let inner_br = to_screen(&Vec2::<Millimeters>::new(w_mm - m, h_mm - m));
191
192    cr.set_source_rgba(1.0, 0.35, 0.2, 0.10);
193    for rect in [
194        (origin.x, origin.y, extent.x, inner_tl.y),     // top band
195        (origin.x, inner_br.y, extent.x, extent.y),     // bottom band
196        (origin.x, inner_tl.y, inner_tl.x, inner_br.y), // left band
197        (inner_br.x, inner_tl.y, extent.x, inner_br.y), // right band
198    ] {
199        if rect.2 > rect.0 && rect.3 > rect.1 {
200            cr.rectangle(rect.0, rect.1, rect.2 - rect.0, rect.3 - rect.1);
201            let _ = cr.fill();
202        }
203    }
204
205    cr.set_dash(&[8.0, 5.0], 0.0);
206    cr.set_line_width(2.0);
207    cr.set_source_rgba(1.0, 0.35, 0.2, 0.65);
208    cr.rectangle(
209        inner_tl.x,
210        inner_tl.y,
211        inner_br.x - inner_tl.x,
212        inner_br.y - inner_tl.y,
213    );
214    let _ = cr.stroke();
215    cr.set_dash(&[], 0.0);
216}

218/// Field-set overlay: a dashed, semi-transparent region per field -
219/// the occupancy cells the engine counts (suppressed in filter mode).
220fn draw_fields(
221    cr: &cairo::Context,
222    sets: &[FieldSet],
223    to_screen: &impl Fn(&Vec2<Millimeters>) -> Vec2<Pixels>,
224) {
225    for set in sets {
226        for field in &set.fields {
227            let region = field_screen_region(set, field);
228            let tl = to_screen(&Vec2::<Millimeters>::new(
229                region.x_min.as_f64(),
230                region.y_min.as_f64(),
231            ));
232            let br = to_screen(&Vec2::<Millimeters>::new(
233                region.x_max.as_f64(),
234                region.y_max.as_f64(),
235            ));
236
237            cr.set_source_rgba(0.2, 0.8, 1.0, 0.08);
238            cr.rectangle(tl.x, tl.y, br.x - tl.x, br.y - tl.y);
239            let _ = cr.fill();
240            cr.set_dash(&[6.0, 4.0], 0.0);
241            cr.set_line_width(1.5);
242            cr.set_source_rgba(0.2, 0.8, 1.0, 0.6);
243            cr.rectangle(tl.x, tl.y, br.x - tl.x, br.y - tl.y);
244            let _ = cr.stroke();
245            cr.set_dash(&[], 0.0);
246
247            cr.set_font_size(11.0);
248            cr.set_source_rgba(0.55, 0.9, 1.0, 0.85);
249            cr.move_to(tl.x + 4.0, tl.y + 14.0);
250            let _ = cr.show_text(field.name.as_str());
251        }
252    }
253}

255/// A field's region resolved into screen-millimeter space - anchored
256/// sets carry local/normalized regions relative to their window.
257fn field_screen_region(set: &FieldSet, field: &tangible_objects_model::Field) -> Region {
258    let Some(anchor) = &set.window else {
259        return field.region;
260    };
261    let width = anchor.width.map(|w| w.as_f64()).unwrap_or(1.0);
262    let height = anchor.height.map(|h| h.as_f64()).unwrap_or(1.0);
263    let x = anchor.x_offset.as_f64();
264    let y = anchor.y_offset.as_f64();
265    Region {
266        x_min: Millimeters::new(x + field.region.x_min.as_f64() * width),
267        y_min: Millimeters::new(y + field.region.y_min.as_f64() * height),
268        x_max: Millimeters::new(x + field.region.x_max.as_f64() * width),
269        y_max: Millimeters::new(y + field.region.y_max.as_f64() * height),
270    }
271}

273/// Ruler overlay: semi-transparent strips with pixel ticks on the
274/// left edge and millimeter ticks on the bottom edge.
275fn draw_ruler(
276    cr: &cairo::Context,
277    w: f64,
278    h: f64,
279    cal: &Calibration,
280    to_screen: &impl Fn(&Vec2<Millimeters>) -> Vec2<Pixels>,
281) {
282    const RULER_W: f64 = 56.0;
283    const RULER_H: f64 = 36.0;
284
285    cr.set_source_rgba(0.04, 0.04, 0.06, 0.5);
286    cr.rectangle(0.0, 0.0, RULER_W, h);
287    let _ = cr.fill();
288    cr.rectangle(0.0, h - RULER_H, w, RULER_H);
289    let _ = cr.fill();
290
291    cr.set_font_size(10.0);
292    cr.set_source_rgba(0.9, 0.9, 0.9, 0.85);
293
294    // Left edge: pixel ticks every 100 px.
295    let mut y = 0.0;
296    while y <= h {
297        cr.move_to(RULER_W - 8.0, y);
298        cr.line_to(RULER_W, y);
299        let _ = cr.stroke();
300        cr.move_to(4.0, y + 3.0);
301        let _ = cr.show_text(&format!("{y} px"));
302        y += 100.0;
303    }
304
305    // Bottom edge: millimeter ticks every 50 mm.
306    let mut mm = 0.0;
307    while mm <= cal.width_mm().as_f64() {
308        let x = to_screen(&Vec2::<Millimeters>::new(mm, 0.0)).x;
309        if x <= w {
310            cr.move_to(x, h - 8.0);
311            cr.line_to(x, h);
312            let _ = cr.stroke();
313            cr.move_to(x + 3.0, h - 4.0);
314            let _ = cr.show_text(&format!("{mm} mm"));
315        }
316        mm += 50.0;
317    }
318}

320/// Positions overlay: mm + pixel coordinates at every touch point,
321/// tangible center and field-set field.
322fn draw_positions(
323    cr: &cairo::Context,
324    touches: TouchSet<'_>,
325    tracked: TrackedSet<'_>,
326    field_sets: &[FieldSet],
327    to_screen: &impl Fn(&Vec2<Millimeters>) -> Vec2<Pixels>,
328) {
329    cr.select_font_face("Sans", cairo::FontSlant::Normal, cairo::FontWeight::Normal);
330    cr.set_font_size(11.0);
331    cr.set_source_rgba(0.85, 0.85, 0.85, 0.85);
332
333    for touch in touches.iter() {
334        draw_position_label(cr, &touch.pos, to_screen, 10.0, -8.0);
335    }
336    for tangible in tracked.iter() {
337        draw_position_label(cr, &tangible.center, to_screen, 10.0, 18.0);
338    }
339    for set in field_sets {
340        for field in &set.fields {
341            let region = field_screen_region(set, field);
342            draw_position_label(cr, &region.center(), to_screen, 10.0, -8.0);
343        }
344    }
345}

347/// Writes the `x_mm,y_mm mm | x_px,y_px px` label at `pos` offset by
348/// `dx`/`dy` screen pixels.
349fn draw_position_label(
350    cr: &cairo::Context,
351    pos: &Vec2<Millimeters>,
352    to_screen: &impl Fn(&Vec2<Millimeters>) -> Vec2<Pixels>,
353    dx: f64,
354    dy: f64,
355) {
356    let px = to_screen(pos);
357    cr.move_to(px.x + dx, px.y + dy);
358    let _ = cr.show_text(&fl!(
359        language_loader(),
360        "position-label",
361        x_mm = (pos.x as i64),
362        y_mm = (pos.y as i64),
363        x_px = (px.x as i64),
364        y_px = (px.y as i64)
365    ));
366}

368/// Training view: complete-graph point set + stability status.
369fn draw_training(
370    cr: &cairo::Context,
371    _w: f64,
372    _h: f64,
373    touches: TouchSet<'_>,
374    to_screen: &impl Fn(&Vec2<Millimeters>) -> Vec2<Pixels>,
375    s: &SharedState,
376) {
377    let count = touches.len();
378    if count == 0 {
379        cr.set_source_rgb(0.6, 0.6, 0.6);
380        cr.set_font_size(22.0);
381        cr.move_to(20.0, 40.0);
382        let _ = cr.show_text(&fl!(language_loader(), "place-figure"));
383        return;
384    }
385
386    // Draw all pairwise connections (complete graph).
387    let line_alpha = if s.stable { 0.6 } else { 0.25 };
388    cr.set_line_width(2.0);
389    cr.set_source_rgba(0.4, 0.7, 1.0, line_alpha);
390    for i in 0..count {
391        for j in (i + 1)..count {
392            let p1 = to_screen(&touches[i].pos);
393            let p2 = to_screen(&touches[j].pos);
394            cr.move_to(p1.x, p1.y);
395            cr.line_to(p2.x, p2.y);
396            let _ = cr.stroke();
397        }
398    }
399
400    // Touch points as circles.
401    let point_color = if s.stable {
402        (0.3, 0.9, 0.4)
403    } else {
404        (0.5, 0.7, 1.0)
405    };
406    for tp in touches {
407        let p = to_screen(&tp.pos);
408        cr.set_source_rgb(point_color.0, point_color.1, point_color.2);
409        cr.arc(p.x, p.y, 12.0, 0.0, 2.0 * std::f64::consts::PI);
410        let _ = cr.fill();
411
412        // Ring.
413        cr.set_source_rgb(1.0, 1.0, 1.0);
414        cr.set_line_width(2.0);
415        cr.arc(p.x, p.y, 12.0, 0.0, 2.0 * std::f64::consts::PI);
416        let _ = cr.stroke();
417    }
418
419    // Status text top left.
420    cr.set_source_rgb(1.0, 1.0, 1.0);
421    cr.set_font_size(20.0);
422    cr.move_to(20.0, 30.0);
423    let status_text = if count < 3 {
424        fl!(language_loader(), "wait-for-points", count = (count as i64))
425    } else if s.stable {
426        fl!(language_loader(), "stable-status", count = (count as i64))
427    } else {
428        fl!(
429            language_loader(),
430            "stabilizing-status",
431            stable = (s.stable_frames as i64),
432            threshold = (STABLE_THRESHOLD as i64),
433            count = (count as i64)
434        )
435    };
436    let _ = cr.show_text(&status_text);
437
438    // Coordinates next to the points.
439    cr.set_font_size(12.0);
440    cr.set_source_rgb(0.7, 0.7, 0.7);
441    for (i, tp) in touches.iter().enumerate() {
442        let p = to_screen(&tp.pos);
443        cr.move_to(p.x + 16.0, p.y - 12.0);
444        let _ = cr.show_text(&fl!(
445            language_loader(),
446            "pin-label",
447            index = (i as i64),
448            x = (tp.pos.x.round() as i64),
449            y = (tp.pos.y.round() as i64)
450        ));
451    }
452}

454/// Live view: matched figures with names + unmatched points dimmed.
455fn draw_live(
456    cr: &cairo::Context,
457    touches: TouchSet<'_>,
458    matches: &[PatternMatch],
459    tracked: TrackedSet<'_>,
460    to_screen: &impl Fn(&Vec2<Millimeters>) -> Vec2<Pixels>,
461    s: &SharedState,
462) {
463    // Collect matched indices.
464    let mut matched_indices: std::collections::HashSet<usize> = std::collections::HashSet::new();
465    for m in matches {
466        for &idx in &m.touch_indices {
467            matched_indices.insert(idx);
468        }
469    }
470
471    // Draw figures (lines + names).
472    for m in matches {
473        let (r, g, b) = figure_color(m.pattern_id);
474        // Pending (unconfirmed partial spawn) - dim the whole figure.
475        let pending = tracked.iter().any(|t| {
476            t.pending
477                && t.pattern_id == m.pattern_id
478                && t.center.distance(&m.center).as_f64() < 50.0
479        });
480        let alpha = if pending { 0.25 } else { 1.0 };
481        let n = m.touch_indices.len();
482        if n < 2 {
483            continue;
484        }
485
486        // Complete graph.
487        cr.set_source_rgba(r, g, b, 0.5 * alpha);
488        cr.set_line_width(2.5);
489        for i in 0..n {
490            for j in (i + 1)..n {
491                let p1 = to_screen(&touches[m.touch_indices[i]].pos);
492                let p2 = to_screen(&touches[m.touch_indices[j]].pos);
493                cr.move_to(p1.x, p1.y);
494                cr.line_to(p2.x, p2.y);
495                let _ = cr.stroke();
496            }
497        }
498
499        // Touch points.
500        for &idx in &m.touch_indices {
501            let p = to_screen(&touches[idx].pos);
502            cr.set_source_rgba(r, g, b, alpha);
503            cr.arc(p.x, p.y, 14.0, 0.0, 2.0 * std::f64::consts::PI);
504            let _ = cr.fill();
505            cr.set_source_rgb(1.0, 1.0, 1.0);
506            cr.set_line_width(2.0);
507            cr.arc(p.x, p.y, 14.0, 0.0, 2.0 * std::f64::consts::PI);
508            let _ = cr.stroke();
509        }
510
511        // Name above the figure (outside the object).
512        let ys: Vec<f64> = m
513            .touch_indices
514            .iter()
515            .map(|&idx| to_screen(&touches[idx].pos).y)
516            .collect();
517        let top_y = ys.iter().cloned().fold(f64::INFINITY, f64::min);
518        let cx = to_screen(&m.center).x;
519        cr.set_source_rgba(1.0, 1.0, 1.0, alpha);
520        cr.select_font_face("Sans", cairo::FontSlant::Normal, cairo::FontWeight::Bold);
521        cr.set_font_size(22.0);
522        if let Ok(extents) = cr.text_extents(m.pattern_name.as_str()) {
523            let tx = cx - extents.width() / 2.0;
524            let ty = (top_y - 28.0).max(extents.height() + 4.0);
525            cr.move_to(tx, ty);
526            let _ = cr.show_text(m.pattern_name.as_str());
527        }
528    }
529
530    // Unmatched touch points (gray).
531    for (i, tp) in touches.iter().enumerate() {
532        if matched_indices.contains(&i) {
533            continue;
534        }
535        let p = to_screen(&tp.pos);
536        cr.set_source_rgb(0.4, 0.4, 0.4);
537        cr.arc(p.x, p.y, 8.0, 0.0, 2.0 * std::f64::consts::PI);
538        let _ = cr.fill();
539    }
540
541    // Info top left.
542    cr.set_source_rgb(0.7, 0.7, 0.7);
543    cr.select_font_face("Sans", cairo::FontSlant::Normal, cairo::FontWeight::Normal);
544    cr.set_font_size(16.0);
545    cr.move_to(20.0, 30.0);
546    let _ = cr.show_text(&fl!(
547        language_loader(),
548        "live-info",
549        figures = (matches.len() as i64),
550        touches = (touches.len() as i64),
551        occupied = (i64::from(s.occupied_fields.as_u32())),
552        fields = (i64::from(s.field_count.as_u32())),
553        zones = (s.dead_zones.len() as i64),
554        filtered = (i64::from(s.dead_zone_filtered.as_u32())),
555        suppressed = (i64::from(
556            s.edge_filtered.as_u32() + s.dead_zone_filtered.as_u32() + s.mode_suppressed.as_u32()
557        )),
558        forwarded = (i64::from(s.forwarded.as_u32()))
559    ));
560}
