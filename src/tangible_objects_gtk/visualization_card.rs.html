1//! The visualization card - the canvas overlay toggles (edge margin,
2//! fields, ruler, grid, touch passthrough) as a movable card on the
3//! `FloatingBoard` (concepts/planned/GUI.md §3).
4
5use std::sync::Arc;
6use std::sync::Mutex;
7use std::time::Duration;
8
9use gtk4::Box as GtkBox;
10use gtk4::Button;
11use gtk4::DrawingArea;
12use gtk4::Orientation;
13use gtk4::Separator;
14use gtk4::ToggleButton;
15use gtk4::glib::ControlFlow;
16use gtk4::prelude::*;
17use i18n_embed_fl::fl;
18use tangible_objects_model::TangibleRenderMode;
19use tangible_objects_runtime::ControlCommand;
20
21use crate::language_loader;
22use crate::shared_state::SharedState;
23use crate::shared_state::send_command;
24
25/// `SharedState` flag reader - the overlay toggles sync their
26/// checked state from it.
27type OverlayRead = fn(&SharedState) -> bool;
28
29/// Builds the visualization-card content: one toggle per canvas
30/// overlay plus the background-alpha cycler, synced back from
31/// [`SharedState`] every 100 ms - interface mutations
32/// (GraphQL/D-Bus/MCP) flip the buttons, button clicks send
33/// `ControlCommand`s over the same channel.
34pub fn create_visualization_card(
35    state: &Arc<Mutex<SharedState>>,
36    drawing_area: &DrawingArea,
37    bg_provider: &gtk4::CssProvider,
38) -> GtkBox {
39    let toggles = GtkBox::new(Orientation::Vertical, 4);
40
41    let mut overlay_buttons: Vec<(ToggleButton, OverlayRead)> = Vec::new();
42    let mut add_overlay_toggle =
43        |label: &str, read: OverlayRead, command: fn(bool) -> ControlCommand| {
44            let btn = ToggleButton::with_label(label);
45            btn.set_active(state.lock().map(|s| read(&s)).unwrap_or_default());
46            let state_click = state.clone();
47            btn.connect_toggled(move |b| {
48                send_command(&state_click, command(b.is_active()));
49            });
50            toggles.append(&btn);
51            overlay_buttons.push((btn, read));
52        };
53    add_overlay_toggle(
54        &fl!(language_loader(), "toggle-edge-margin"),
55        |s| s.edge_margin_visible,
56        ControlCommand::SetEdgeMarginVisible,
57    );
58    add_overlay_toggle(
59        &fl!(language_loader(), "toggle-fields"),
60        |s| s.fields_visible,
61        ControlCommand::SetFieldsVisible,
62    );
63    add_overlay_toggle(
64        &fl!(language_loader(), "toggle-dead-zones"),
65        |s| s.dead_zones_visible,
66        ControlCommand::SetDeadZonesVisible,
67    );
68    add_overlay_toggle(
69        &fl!(language_loader(), "toggle-spells"),
70        |s| s.spells_visible,
71        ControlCommand::SetSpellsVisible,
72    );
73    add_overlay_toggle(
74        &fl!(language_loader(), "toggle-ruler"),
75        |s| s.ruler_visible,
76        ControlCommand::SetRulerVisible,
77    );
78    add_overlay_toggle(
79        &fl!(language_loader(), "toggle-grid"),
80        |s| s.grid_visible,
81        ControlCommand::SetGridVisible,
82    );
83    add_overlay_toggle(
84        &fl!(language_loader(), "toggle-positions"),
85        |s| s.positions_visible,
86        ControlCommand::SetPositionsVisible,
87    );
88    add_overlay_toggle(
89        &fl!(language_loader(), "toggle-passthrough"),
90        |s| s.touch_passthrough,
91        ControlCommand::SetTouchPassthrough,
92    );
93
94    // Render-mode cycler - GUI-local: touch points, physical shape,
95    // or both (GUI.md tangible render modes).
96    let render_btn = Button::with_label(&fl!(language_loader(), "render-mode-both"));
97    let state_render = state.clone();
98    render_btn.connect_clicked(move |_| {
99        let Ok(mut s) = state_render.lock() else {
100            return;
101        };
102        s.render_mode = match s.render_mode {
103            TangibleRenderMode::TouchPoints => TangibleRenderMode::PhysicalGeometry,
104            TangibleRenderMode::PhysicalGeometry => {
105                TangibleRenderMode::TouchPointsAndPhysicalGeometry
106            }
107            TangibleRenderMode::TouchPointsAndPhysicalGeometry => TangibleRenderMode::TouchPoints,
108        };
109    });
110    toggles.append(&render_btn);
111
112    toggles.append(&Separator::new(Orientation::Horizontal));
113
114    // Alpha cycler - rotates the background opacity presets; touches,
115    // labels and chrome stay opaque, only the canvas fill thins.
116    let alpha_btn = Button::with_label(&fl!(
117        language_loader(),
118        "background-alpha",
119        percent = (state
120            .lock()
121            .map(|s| s.background_alpha.percent() as i64)
122            .unwrap_or(100))
123    ));
124    let state_alpha = state.clone();
125    let alpha_btn_click = alpha_btn.clone();
126    let provider_alpha = bg_provider.clone();
127    let da_alpha = drawing_area.clone();
128    alpha_btn.connect_clicked(move |_| {
129        let Ok(mut s) = state_alpha.lock() else {
130            return;
131        };
132        s.background_alpha = s.background_alpha.next();
133        s.background_alpha_request = Some(s.background_alpha);
134        alpha_btn_click.set_label(&fl!(
135            language_loader(),
136            "background-alpha",
137            percent = (s.background_alpha.percent() as i64)
138        ));
139        provider_alpha.load_from_string(&s.background_alpha.surface_bg_css());
140        da_alpha.queue_draw();
141    });
142    toggles.append(&alpha_btn);
143
144    let render_btn_sync = render_btn.clone();
145    let state_sync = state.clone();
146    let mut applied = state_sync
147        .lock()
148        .map(|s| {
149            overlay_buttons
150                .iter()
151                .map(|(_, read)| read(&s))
152                .collect::<Vec<_>>()
153        })
154        .unwrap_or_default();
155    let provider_sync = bg_provider.clone();
156    let alpha_btn_sync = alpha_btn.clone();
157    let mut applied_alpha = state_sync
158        .lock()
159        .map(|s| s.background_alpha)
160        .unwrap_or_else(|_| crate::BackgroundAlpha::from_percent(100));
161    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
162        let s = match state_sync.lock() {
163            Ok(s) => s,
164            Err(_) => return ControlFlow::Continue,
165        };
166        // Alpha changes from `SetBackgroundAlpha` commands land in the
167        // shared state via the device loop - the provider reload and
168        // button label follow here (the click applies both itself).
169        if s.background_alpha != applied_alpha {
170            applied_alpha = s.background_alpha;
171            provider_sync.load_from_string(&applied_alpha.surface_bg_css());
172            alpha_btn_sync.set_label(&fl!(
173                language_loader(),
174                "background-alpha",
175                percent = (applied_alpha.percent() as i64)
176            ));
177        }
178        // Overlay changes from interface commands land in shared
179        // state via the device loop - sync the buttons (set_active
180        // re-fires toggled and re-sends an idempotent command).
181        for ((btn, read), applied) in overlay_buttons.iter().zip(applied.iter_mut()) {
182            let current = read(&s);
183            if current != *applied {
184                *applied = current;
185                btn.set_active(current);
186            }
187        }
188        render_btn_sync.set_label(&match s.render_mode {
189            TangibleRenderMode::TouchPoints => fl!(language_loader(), "render-mode-points"),
190            TangibleRenderMode::PhysicalGeometry => {
191                fl!(language_loader(), "render-mode-geometry")
192            }
193            TangibleRenderMode::TouchPointsAndPhysicalGeometry => {
194                fl!(language_loader(), "render-mode-both")
195            }
196        });
197        ControlFlow::Continue
198    });
199
200    let body = GtkBox::new(Orientation::Vertical, 0);
201    body.append(&toggles);
202    body
203}
