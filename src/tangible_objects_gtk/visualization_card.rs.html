1//! The visualization card - the canvas overlay toggles (edge margin,
2//! fields, ruler, grid, touch passthrough) as a movable card on the
3//! `FloatingBoard` (concepts/planned/GUI.md §3).
4
5use std::sync::Arc;
6use std::sync::Mutex;
7use std::time::Duration;
8
9use gtk4::Box as GtkBox;
10use gtk4::Button;
11use gtk4::DrawingArea;
12use gtk4::Orientation;
13use gtk4::Separator;
14use gtk4::ToggleButton;
15use gtk4::glib::ControlFlow;
16use gtk4::prelude::*;
17use i18n_embed_fl::fl;
18use tangible_objects_runtime::ControlCommand;
19
20use crate::language_loader;
21use crate::shared_state::SharedState;
22use crate::shared_state::send_command;
23
24/// `SharedState` flag reader - the overlay toggles sync their
25/// checked state from it.
26type OverlayRead = fn(&SharedState) -> bool;
27
28/// Builds the visualization-card content: one toggle per canvas
29/// overlay plus the background-alpha cycler, synced back from
30/// [`SharedState`] every 100 ms - interface mutations
31/// (GraphQL/D-Bus/MCP) flip the buttons, button clicks send
32/// `ControlCommand`s over the same channel.
33pub fn create_visualization_card(
34    state: &Arc<Mutex<SharedState>>,
35    drawing_area: &DrawingArea,
36    bg_provider: &gtk4::CssProvider,
37) -> GtkBox {
38    let toggles = GtkBox::new(Orientation::Vertical, 4);
39
40    let mut overlay_buttons: Vec<(ToggleButton, OverlayRead)> = Vec::new();
41    let mut add_overlay_toggle =
42        |label: &str, read: OverlayRead, command: fn(bool) -> ControlCommand| {
43            let btn = ToggleButton::with_label(label);
44            btn.set_active(state.lock().map(|s| read(&s)).unwrap_or_default());
45            let state_click = state.clone();
46            btn.connect_toggled(move |b| {
47                send_command(&state_click, command(b.is_active()));
48            });
49            toggles.append(&btn);
50            overlay_buttons.push((btn, read));
51        };
52    add_overlay_toggle(
53        &fl!(language_loader(), "toggle-edge-margin"),
54        |s| s.edge_margin_visible,
55        ControlCommand::SetEdgeMarginVisible,
56    );
57    add_overlay_toggle(
58        &fl!(language_loader(), "toggle-fields"),
59        |s| s.fields_visible,
60        ControlCommand::SetFieldsVisible,
61    );
62    add_overlay_toggle(
63        &fl!(language_loader(), "toggle-ruler"),
64        |s| s.ruler_visible,
65        ControlCommand::SetRulerVisible,
66    );
67    add_overlay_toggle(
68        &fl!(language_loader(), "toggle-grid"),
69        |s| s.grid_visible,
70        ControlCommand::SetGridVisible,
71    );
72    add_overlay_toggle(
73        &fl!(language_loader(), "toggle-positions"),
74        |s| s.positions_visible,
75        ControlCommand::SetPositionsVisible,
76    );
77    add_overlay_toggle(
78        &fl!(language_loader(), "toggle-passthrough"),
79        |s| s.touch_passthrough,
80        ControlCommand::SetTouchPassthrough,
81    );
82
83    toggles.append(&Separator::new(Orientation::Horizontal));
84
85    // Alpha cycler - rotates the background opacity presets; touches,
86    // labels and chrome stay opaque, only the canvas fill thins.
87    let alpha_btn = Button::with_label(&fl!(
88        language_loader(),
89        "background-alpha",
90        percent = (state
91            .lock()
92            .map(|s| s.background_alpha.percent() as i64)
93            .unwrap_or(100))
94    ));
95    let state_alpha = state.clone();
96    let alpha_btn_click = alpha_btn.clone();
97    let provider_alpha = bg_provider.clone();
98    let da_alpha = drawing_area.clone();
99    alpha_btn.connect_clicked(move |_| {
100        let Ok(mut s) = state_alpha.lock() else {
101            return;
102        };
103        s.background_alpha = s.background_alpha.next();
104        s.background_alpha_request = Some(s.background_alpha);
105        alpha_btn_click.set_label(&fl!(
106            language_loader(),
107            "background-alpha",
108            percent = (s.background_alpha.percent() as i64)
109        ));
110        provider_alpha.load_from_data(&s.background_alpha.surface_bg_css());
111        da_alpha.queue_draw();
112    });
113    toggles.append(&alpha_btn);
114
115    let state_sync = state.clone();
116    let mut applied = state_sync
117        .lock()
118        .map(|s| {
119            overlay_buttons
120                .iter()
121                .map(|(_, read)| read(&s))
122                .collect::<Vec<_>>()
123        })
124        .unwrap_or_default();
125    let provider_sync = bg_provider.clone();
126    let alpha_btn_sync = alpha_btn.clone();
127    let mut applied_alpha = state_sync
128        .lock()
129        .map(|s| s.background_alpha)
130        .unwrap_or_else(|_| crate::BackgroundAlpha::from_percent(100));
131    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
132        let s = match state_sync.lock() {
133            Ok(s) => s,
134            Err(_) => return ControlFlow::Continue,
135        };
136        // Alpha changes from `SetBackgroundAlpha` commands land in the
137        // shared state via the device loop - the provider reload and
138        // button label follow here (the click applies both itself).
139        if s.background_alpha != applied_alpha {
140            applied_alpha = s.background_alpha;
141            provider_sync.load_from_data(&applied_alpha.surface_bg_css());
142            alpha_btn_sync.set_label(&fl!(
143                language_loader(),
144                "background-alpha",
145                percent = (applied_alpha.percent() as i64)
146            ));
147        }
148        // Overlay changes from interface commands land in shared
149        // state via the device loop - sync the buttons (set_active
150        // re-fires toggled and re-sends an idempotent command).
151        for ((btn, read), applied) in overlay_buttons.iter().zip(applied.iter_mut()) {
152            let current = read(&s);
153            if current != *applied {
154                *applied = current;
155                btn.set_active(current);
156            }
157        }
158        ControlFlow::Continue
159    });
160
161    let body = GtkBox::new(Orientation::Vertical, 0);
162    body.append(&toggles);
163    body
164}
