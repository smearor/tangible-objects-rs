23/// Builds the info-card content widget: a `label | value` grid the
24/// update timer refreshes from [`SharedState`] every 100 ms - the same
25/// cadence the control bar's status label uses.
26pub fn create_info_card(state: &Arc<Mutex<SharedState>>) -> GtkBox {
27    let grid = Grid::new();
28    grid.set_column_spacing(12);
29    grid.set_row_spacing(4);
30
31    // Row 0: the Live/Training mode button - it fills the row in
32    // place of a `Mode | value` label pair (moved from the control
33    // bar with the card migration).
34    let mode_btn = Button::with_label(&fl!(language_loader(), "mode-live"));
35    mode_btn.add_css_class("suggested-action");
36    let state_mode = state.clone();
37    mode_btn.connect_clicked(move |b| {
38        let label = match state_mode.lock() {
39            Ok(mut s) => {
40                s.mode = if s.mode == GuiMode::Live {
41                    GuiMode::Training
42                } else {
43                    GuiMode::Live
44                };
45                s.stable_frames = 0;
46                s.stable = false;
47                match s.mode {
48                    GuiMode::Live => fl!(language_loader(), "mode-live"),
49                    GuiMode::Training => fl!(language_loader(), "mode-training"),
50                }
51            }
52            Err(_) => return,
53        };
54        b.set_label(&label);
55    });
56    grid.attach(&mode_btn, 0, 0, 2, 1);
57
58    let touches_value = add_row(&grid, 1, &fl!(language_loader(), "info-touches"));
59    let figures_value = add_row(&grid, 2, &fl!(language_loader(), "info-figures"));
60    let tracked_value = add_row(&grid, 3, &fl!(language_loader(), "info-tracked"));
61    let fields_value = add_row(&grid, 4, &fl!(language_loader(), "info-fields"));
62    let zones_value = add_row(&grid, 5, &fl!(language_loader(), "info-dead-zones"));
63    let filtered_value = add_row(&grid, 6, &fl!(language_loader(), "info-filtered"));
64    let stability_value = add_row(&grid, 7, &fl!(language_loader(), "info-stability"));
65
66    let state_tick = state.clone();
67    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
68        let s = match state_tick.lock() {
69            Ok(s) => s,
70            Err(_) => return ControlFlow::Continue,
71        };
72        mode_btn.set_label(&match s.mode {
73            GuiMode::Live => fl!(language_loader(), "mode-live"),
74            GuiMode::Training => fl!(language_loader(), "mode-training"),
75        });
76        touches_value.set_label(&s.touches.len().to_string());
77        figures_value.set_label(&s.matches.len().to_string());
78        tracked_value.set_label(&s.tracked.len().to_string());
79        fields_value.set_label(&fl!(
80            language_loader(),
81            "info-fields-value",
82            occupied = (i64::from(s.occupied_fields.as_u32())),
83            total = (i64::from(s.field_count.as_u32())),
84            sets = (i64::from(s.field_set_count.as_u32()))
85        ));
86        zones_value.set_label(&fl!(
87            language_loader(),
88            "info-zones-value",
89            zones = (s.dead_zones.len() as i64),
90            filtered = (i64::from(s.dead_zone_filtered.as_u32()))
91        ));
92        filtered_value.set_label(&fl!(
93            language_loader(),
94            "info-filtered-value",
95            suppressed = (i64::from(
96                s.edge_filtered.as_u32()
97                    + s.dead_zone_filtered.as_u32()
98                    + s.mode_suppressed.as_u32()
99            )),
100            forwarded = (i64::from(s.forwarded.as_u32()))
101        ));
102        stability_value.set_label(&if s.stable {
103            fl!(language_loader(), "info-stable")
104        } else {
105            fl!(
106                language_loader(),
107                "info-stabilizing",
108                stable = (s.stable_frames as i64),
109                threshold = (STABLE_THRESHOLD as i64)
110            )
111        });
112        ControlFlow::Continue
113    });
114
115    let body = GtkBox::new(Orientation::Vertical, 0);
116    body.append(&grid);
117    body
118}

120/// Appends a `label | value` row to `grid` and returns the value label -
121/// the row name is rendered dimmed in column 0, the live value in
122/// column 1.
123fn add_row(grid: &Grid, row: i32, label: &str) -> Label {
124    let name = Label::new(Some(label));
125    name.set_halign(gtk4::Align::Start);
126    name.add_css_class("dim-label");
127    grid.attach(&name, 0, row, 1, 1);
128
129    let value = Label::new(Some("–"));
130    value.set_halign(gtk4::Align::Start);
131    grid.attach(&value, 1, row, 1, 1);
132    value
133}
