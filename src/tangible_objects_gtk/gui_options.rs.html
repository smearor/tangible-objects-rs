6/// Display-side options for [`crate::run_gui`] - window presentation,
7/// not detection behavior (`DetectionConfig` stays on the engine side).
8#[derive(Debug, Copy, Clone)]
9pub struct GuiOptions {
10    /// Whether the window starts visible (`SetGuiVisible` toggles
11    /// afterwards).
12    pub visible: bool,
13    /// Quarter turns applied when mapping surface millimeters onto
14    /// screen pixels - the panel's native landscape frame stays in the
15    /// evdev stream even when the desktop shows the screen rotated
16    /// (e.g. GNOME portrait mode).
17    pub rotation: DisplayRotation,
18    /// Canvas background opacity. Below opaque the window shows the
19    /// desktop beneath the touch visualization - the touches, figures
20    /// and UI chrome stay opaque; only the black fill translucents
21    /// (`cairo` alpha, not window opacity).
22    pub background_alpha: BackgroundAlpha,
23    /// Whether the dead-zone overlay starts drawn
24    /// (`SetDeadZonesVisible` commands toggle afterwards).
25    pub dead_zones_visible: bool,
26    /// Whether the edge-margin overlay starts drawn
27    /// (`SetEdgeMarginVisible` commands toggle afterwards).
28    pub edge_margin_visible: bool,
29    /// Whether the field-set overlay starts drawn
30    /// (`SetFieldsVisible` commands toggle afterwards).
31    pub fields_visible: bool,
32    /// Whether the ruler overlay starts drawn
33    /// (`SetRulerVisible` commands toggle afterwards).
34    pub ruler_visible: bool,
35    /// Whether the grid overlay starts drawn
36    /// (`SetGridVisible` commands toggle afterwards).
37    pub grid_visible: bool,
38    /// Whether the positions overlay starts drawn
39    /// (`SetPositionsVisible` commands toggle afterwards).
40    pub positions_visible: bool,
41    /// Whether touches pass through the window to the desktop except
42    /// on GUI elements (`SetTouchPassthrough` commands toggle
43    /// afterwards; Wayland only).
44    pub touch_passthrough: bool,
45    /// Initial filter mode shown in the GUI (`--filter-mode`) -
46    /// display state, honored by the proxy in `--filter` mode.
47    pub filter_mode: FilterMode,
48}
