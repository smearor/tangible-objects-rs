1//! Daemon-embedded GTK visualization (`concepts/planned/GUI.md` +
2//! WORKSPACE.md Phase 3).
3//!
4//! Embedded library, not a remote client: the embedding daemon calls
5//! [`run_gui`], which runs the engine/device loop on a background
6//! thread (in-process, zero-latency event flow) and renders the
7//! [`SharedState`] in the GTK main loop. Daemon-mode semantics
8//! (interfaces, control commands, recording, calibration) arrive via
9//! [`GuiHooks`] so this crate never links the daemon crate.

86/// Content padding shared by every card - the wider chrome frame
87/// makes the resize/rotate grips and the quad-finger gestures easier
88/// to hit on a touchscreen.
89pub(crate) const CARD_CONTENT_PADDING: Pixels = Pixels::new(20.0);

91/// Fullscreen GTK visualization with control bar and evdev-driven
92/// drawing. `patterns` is pre-loaded by the daemon (`load_patterns`);
93/// `visible` carries the initial window state (`--gui` shows by
94/// default; `SetGuiVisible` commands toggle afterwards).
95pub fn run_gui(
96    device_path: String,
97    profiles_dir: Option<String>,
98    patterns: PatternBuf,
99    device_profiles: Option<PathBuf>,
100    config: DetectionConfig,
101    options: GuiOptions,
102    hooks: GuiHooks,
103) {
104    let state = SharedState::new(&options);
105
106    // Spawn evdev thread - the in-process engine loop.
107    let state_clone = state.clone();
108    let device_profiles_arc = device_profiles.map(Arc::new);
109    let device_path_thread = device_path.clone();
110    std::thread::spawn(move || {
111        evdev_thread::evdev_thread(
112            device_path_thread,
113            patterns,
114            state_clone,
115            config,
116            options.rotation,
117            device_profiles_arc,
118            hooks,
119        );
120    });
121
122    let app = Application::builder()
123        .application_id("dev.aschaeffer.tangible-objects")
124        .build();
125
126    let state_app = state.clone();
127    let profiles_dir_app = profiles_dir.clone();
128    let device_path_app = device_path.clone();
129    let visible_app = options.visible;
130    app.connect_activate(move |app| {
131        let drawing_area = DrawingArea::new();
132        drawing_area.set_hexpand(true);
133        drawing_area.set_vexpand(true);
134
135        // The canvas darkening lives in the widget's CSS background
136        // layer (`.surface-bg`) - its alpha blends against the
137        // transparent window, so the toggle sees the desktop beneath.
138        // The cairo draw stays content-only on top of it.
139        let bg_provider = CssProvider::new();
140        bg_provider.load_from_string(&options.background_alpha.surface_bg_css());
141        drawing_area.add_css_class("surface-bg");
142        if let Some(display) = Display::default() {
143            gtk4::style_context_add_provider_for_display(
144                &display,
145                &bg_provider,
146                gtk4::STYLE_PROVIDER_PRIORITY_APPLICATION,
147            );
148        }
149
150        let state_draw = state_app.clone();
151        drawing_area.set_draw_func(move |_, cr, w, h| {
152            draw::draw_scene(cr, Size::new(w, h), &state_draw);
153        });
154
155        // Layout: the drawing area fills the window; all controls live
156        // on floating cards.
157        let overlay = Overlay::new();
158        overlay.set_child(Some(&drawing_area));
159
160        // Canvas redraw cadence - the evdev thread writes
161        // `SharedState` per frame; this tick repaints at ~10 Hz.
162        let da_tick = drawing_area.clone();
163        gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
164            da_tick.queue_draw();
165            ControlFlow::Continue
166        });
167
168        // Free-floating cards on top of the surface view
169        // (concepts/planned/GUI.md §3): the board stacks a transparent
170        // card layer above the overlay; gestures landing outside a
171        // card fall through to the canvas.
172        let board = FloatingBoard::new(&overlay);
173        let info_body = info_card::create_info_card(&state_app);
174        // The profiles card's Save button needs the `--profiles`
175        // source dir - built before the uniform card loop.
176        let profiles_body = profiles_card::create_profiles_card(
177            &state_app,
178            profiles_dir_app.clone().map(PathBuf::from),
179        );
180        let _ = board.add_card_with_id(
181            "info",
182            &info_body,
183            CardConfig {
184                title: Some(fl!(language_loader(), "info-card-title")),
185                position: CardAnchor::TopRight,
186                anchor_margin: Pixels::new(100.0),
187                bounds_mode: BoundsMode::Contained,
188                content_padding: CARD_CONTENT_PADDING,
189                foldable: true,
190                resizable: true,
191                rotate_grip: true,
192                closable: true,
193                close_action: CardCloseAction::Hide,
194                ..Default::default()
195            },
196        );
197        for (id, body, title, anchor) in [
198            (
199                "visualization",
200                visualization_card::create_visualization_card(
201                    &state_app,
202                    &drawing_area,
203                    &bg_provider,
204                ),
205                fl!(language_loader(), "card-visualization-title"),
206                CardAnchor::TopLeft,
207            ),
208            (
209                "filter",
210                filter_card::create_filter_card(&state_app),
211                fl!(language_loader(), "card-filter-title"),
212                CardAnchor::CenterLeft,
213            ),
214            (
215                "device",
216                device_card::create_device_card(&state_app, &device_path_app),
217                fl!(language_loader(), "card-device-title"),
218                CardAnchor::BottomCenter,
219            ),
220            (
221                "parameters",
222                parameters_card::create_parameters_card(&state_app),
223                fl!(language_loader(), "card-parameters-title"),
224                CardAnchor::BottomRight,
225            ),
226            (
227                "touches",
228                touches_card::create_touches_card(&state_app),
229                fl!(language_loader(), "card-touches-title"),
230                CardAnchor::BottomCenter,
231            ),
232            (
233                "calibration",
234                calibration_card::create_calibration_card(&state_app),
235                fl!(language_loader(), "card-calibration-title"),
236                CardAnchor::Center,
237            ),
238            (
239                "metrics",
240                metrics_card::create_metrics_card(&state_app),
241                fl!(language_loader(), "card-metrics-title"),
242                CardAnchor::CenterRight,
243            ),
244            (
245                "profiles",
246                profiles_body,
247                fl!(language_loader(), "card-profiles-title"),
248                CardAnchor::CenterRight,
249            ),
250            (
251                "field-sets",
252                field_sets_card::create_field_sets_card(&state_app),
253                fl!(language_loader(), "card-field-sets-title"),
254                CardAnchor::TopCenter,
255            ),
256            (
257                "dead-zones",
258                dead_zones_card::create_dead_zones_card(&state_app),
259                fl!(language_loader(), "card-dead-zones-title"),
260                CardAnchor::BottomLeft,
261            ),
262            (
263                "seats",
264                seats_card::create_seats_card(&state_app),
265                fl!(language_loader(), "card-seats-title"),
266                CardAnchor::CenterRight,
267            ),
268        ] {
269            let _ = board.add_card_with_id(
270                id,
271                &body,
272                CardConfig {
273                    title: Some(title),
274                    position: anchor,
275                    anchor_margin: Pixels::new(100.0),
276                    bounds_mode: BoundsMode::Contained,
277                    content_padding: CARD_CONTENT_PADDING,
278                    foldable: true,
279                    resizable: true,
280                    rotate_grip: true,
281                    // X hides the card (not remove) - the menu re-opens it.
282                    closable: true,
283                    close_action: CardCloseAction::Hide,
284                    ..Default::default()
285                },
286            );
287        }
288
289        // Main-menu card: always-open root listing every other card -
290        // the only card that cannot be closed (`closable: false`).
291        let menu_body = menu_card::create_menu_card(&board);
292        let _ = board.add_card_with_id(
293            menu_card::MENU_CARD_ID,
294            &menu_body,
295            CardConfig {
296                title: Some(fl!(language_loader(), "card-menu-title")),
297                position: CardAnchor::TopLeft,
298                anchor_margin: Pixels::new(100.0),
299                bounds_mode: BoundsMode::Contained,
300                content_padding: CARD_CONTENT_PADDING,
301                foldable: true,
302                resizable: true,
303                rotate_grip: true,
304                ..Default::default()
305            },
306        );
307
308        // Card layout: restore `gui-cards.toml` (or arrange the
309        // default side-by-side), persist on geometry changes.
310        let layout = card_layout::load_layout();
311        // The training card exists only in training mode - its saved
312        // placement (if any) seeds the add/remove watcher.
313        let training_placement = layout.as_ref().and_then(|layout| {
314            layout
315                .cards
316                .iter()
317                .find(|card| card.id == CardId::new("training"))
318                .cloned()
319        });
320        let spell_training_placement = layout.as_ref().and_then(|layout| {
321            layout
322                .cards
323                .iter()
324                .find(|card| card.id == CardId::new("spell-training"))
325                .cloned()
326        });
327        card_layout::wire_layout(&board, layout);
328        training_card::wire_training_card(
329            &board,
330            &state_app,
331            &profiles_dir_app,
332            training_placement,
333        );
334        spell_training_card::wire_spell_training_card(&board, &state_app, spell_training_placement);
335
336        // Initially only the menu is open (GUI.md): every other card
337        // starts hidden and is opened via the menu's toggle list.
338        for card in board.cards() {
339            if card.id() != CardId::new(menu_card::MENU_CARD_ID) {
340                let _ = board.set_card_visible(&card.id(), false);
341            }
342        }
343
344        // Touch passthrough: input reaches the cards; everything else
345        // falls through to the desktop.
346        passthrough::wire_passthrough(&board, &state_app);
347
348        let window = ApplicationWindow::builder()
349            .application(app)
350            .title(fl!(language_loader(), "window-title"))
351            .default_width(1600)
352            .default_height(1200)
353            .child(&board)
354            .build();
355
356        // Input-region probe: a capture-phase legacy controller counts
357        // every touch down the surface receives. The diagnostics merge
358        // the counter so a dead card splits into "event never arrived"
359        // (applied region wrong) vs "arrived but eaten" (gesture claim).
360        let state_probe = state_app.clone();
361        let window_probe = window.downgrade();
362        let probe = EventControllerLegacy::new();
363        probe.set_propagation_phase(PropagationPhase::Capture);
364        probe.connect_event(move |_, event| {
365            let event_type = event.event_type();
366            if let Ok(mut s) = state_probe.lock() {
367                *s.event_counts.entry(format!("{event_type:?}")).or_insert(0) += 1;
368                if matches!(event_type, EventType::TouchBegin | EventType::ButtonPress) {
369                    let (x, y) = event.position().unwrap_or_default();
370                    s.last_event_position = Some(Vec2::<Pixels>::new(x, y));
371                }
372            }
373            if matches!(event_type, EventType::TouchBegin | EventType::ButtonPress)
374                && let Some(window) = window_probe.upgrade()
375            {
376                let (x, y) = event.position().unwrap_or_default();
377                let picked = window.pick(
378                    x,
379                    y,
380                    gtk4::PickFlags::DEFAULT | gtk4::PickFlags::INSENSITIVE,
381                );
382                let mut ancestry = Vec::new();
383                let mut current = picked.clone();
384                while let Some(widget) = current {
385                    ancestry.push(format!(
386                        "{}(mapped:{},sensitive:{},can_target:{})",
387                        widget.type_().name(),
388                        widget.is_mapped(),
389                        widget.is_sensitive(),
390                        widget.can_target()
391                    ));
392                    current = widget.parent();
393                }
394                debug!(
395                    event = ?event_type,
396                    x,
397                    y,
398                    picked = ?picked.map(|w| w.type_().name().to_string()),
399                    ancestry = %ancestry.join(" <- "),
400                    "pick diagnostic"
401                );
402            }
403            Propagation::Proceed
404        });
405        window.add_controller(probe);
406
407        // `--touch-passthrough` puts the window into the layer-shell
408        // Overlay layer: a toplevel loses stacking when the
409        // through-click focuses the window beneath (GNOME), and GTK4
410        // has no `keep_above`. Must run before `present`; runtime
411        // toggles only adjust the input region. `init_layer_shell`
412        // degrades to a regular toplevel when the protocol is missing
413        // - logging makes the fallback visible on remote machines.
414        if options.touch_passthrough {
415            if gtk4_layer_shell::is_supported() {
416                info!("layer-shell overlay enabled (touch passthrough)");
417            } else {
418                warn!("layer-shell not supported - window stays a regular toplevel");
419            }
420
421            window.init_layer_shell();
422            window.set_layer(Layer::Overlay);
423            window.set_anchor(Edge::Top, true);
424            window.set_anchor(Edge::Bottom, true);
425            window.set_anchor(Edge::Left, true);
426            window.set_anchor(Edge::Right, true);
427            // -1 = ignore other layer surfaces' exclusive zones - the
428            // surface spans edge-to-edge over bars/docks reserved by
429            // other apps (e.g. waybar on the same Overlay layer).
430            window.set_exclusive_zone(-1);
431            // Stable namespace so compositor rules can match the
432            // surface (e.g. Hyprland `layerrule`).
433            window.set_namespace(Some("tangible-objects"));
434            window.set_keyboard_mode(KeyboardMode::None);
435        }
436
437        // Window background transparent so the canvas fill's alpha
438        // (`background_alpha` toggle) actually shows the desktop
439        // beneath - the theme paints the window background opaque by
440        // default, covering the translucent cairo fill. Only the
441        // window *background* is cleared; contents stay opaque.
442        let provider = gtk4::CssProvider::new();
443        provider.load_from_string("window { background-color: transparent; }");
444        if let Some(display) = Display::default() {
445            gtk4::style_context_add_provider_for_display(
446                &display,
447                &provider,
448                gtk4::STYLE_PROVIDER_PRIORITY_APPLICATION,
449            );
450        }
451
452        window.fullscreen();
453        window.set_visible(visible_app);
454
455        // Redraw timer ~30fps + visibility sync (SetGuiVisible).
456        let state_tick = state_app.clone();
457        let da_tick = drawing_area.clone();
458        let win_tick = window.clone();
459        gtk4::glib::timeout_add_local(Duration::from_millis(33), move || {
460            if let Ok(s) = state_tick.lock()
461                && win_tick.is_visible() != s.visible
462            {
463                win_tick.set_visible(s.visible);
464            }
465            da_tick.queue_draw();
466            ControlFlow::Continue
467        });
468
469        // Escape exits fullscreen.
470        let key_controller = EventControllerKey::new();
471        key_controller.connect_key_pressed(|_, keyval, _, _| {
472            if keyval == Key::Escape {
473                std::process::exit(0);
474            }
475            Propagation::Proceed
476        });
477        window.add_controller(key_controller);
478    });
479
480    // `run()` would hand `std::env::args` to `g_application_run` - GTK
481    // would reject daemon flags like `--gui`/`--device` as unknown
482    // options ("Unbekannte Option"), so only argv[0] is passed.
483    app.run_with_args(&["tangible-objects"]);
484}
