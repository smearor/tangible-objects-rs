1//! Daemon-embedded GTK visualization (`concepts/planned/GUI.md` +
2//! WORKSPACE.md Phase 3).
3//!
4//! Embedded library, not a remote client: the embedding daemon calls
5//! [`run_gui`], which runs the engine/device loop on a background
6//! thread (in-process, zero-latency event flow) and renders the
7//! [`SharedState`] in the GTK main loop. Daemon-mode semantics
8//! (interfaces, control commands, recording, calibration) arrive via
9//! [`GuiHooks`] so this crate never links the daemon crate.

58/// Fullscreen GTK visualization with control bar and evdev-driven
59/// drawing. `patterns` is pre-loaded by the daemon (`load_patterns`);
60/// `visible` carries the initial window state (`--gui` shows by
61/// default; `SetGuiVisible` commands toggle afterwards).
62pub fn run_gui(
63    device_path: String,
64    profiles_dir: Option<String>,
65    patterns: PatternBuf,
66    device_profiles: Option<PathBuf>,
67    config: DetectionConfig,
68    options: GuiOptions,
69    hooks: GuiHooks,
70) {
71    let state = SharedState::new(&options);
72
73    // Spawn evdev thread - the in-process engine loop.
74    let state_clone = state.clone();
75    let device_profiles_arc = device_profiles.map(Arc::new);
76    let device_path_thread = device_path.clone();
77    std::thread::spawn(move || {
78        evdev_thread::evdev_thread(
79            device_path_thread,
80            patterns,
81            state_clone,
82            config,
83            options.rotation,
84            device_profiles_arc,
85            hooks,
86        );
87    });
88
89    let app = Application::builder()
90        .application_id("dev.aschaeffer.tangible-objects")
91        .build();
92
93    let state_app = state.clone();
94    let profiles_dir_app = profiles_dir.clone();
95    let device_path_app = device_path.clone();
96    let visible_app = options.visible;
97    app.connect_activate(move |app| {
98        let drawing_area = DrawingArea::new();
99        drawing_area.set_hexpand(true);
100        drawing_area.set_vexpand(true);
101
102        // The canvas darkening lives in the widget's CSS background
103        // layer (`.surface-bg`) - its alpha blends against the
104        // transparent window, so the toggle sees the desktop beneath.
105        // The cairo draw stays content-only on top of it.
106        let bg_provider = gtk4::CssProvider::new();
107        bg_provider.load_from_data(&options.background_alpha.surface_bg_css());
108        drawing_area.add_css_class("surface-bg");
109        drawing_area
110            .style_context()
111            .add_provider(&bg_provider, gtk4::STYLE_PROVIDER_PRIORITY_APPLICATION);
112
113        let state_draw = state_app.clone();
114        drawing_area.set_draw_func(move |_, cr, w, h| {
115            draw::draw_scene(cr, w, h, &state_draw);
116        });
117
118        // Layout: the drawing area fills the window; all controls live
119        // on floating cards.
120        let overlay = gtk4::Overlay::new();
121        overlay.set_child(Some(&drawing_area));
122
123        // Canvas redraw cadence - the evdev thread writes
124        // `SharedState` per frame; this tick repaints at ~10 Hz.
125        let da_tick = drawing_area.clone();
126        gtk4::glib::timeout_add_local(std::time::Duration::from_millis(100), move || {
127            da_tick.queue_draw();
128            gtk4::glib::ControlFlow::Continue
129        });
130
131        // Free-floating cards on top of the surface view
132        // (concepts/planned/GUI.md §3): the board stacks a transparent
133        // card layer above the overlay; gestures landing outside a
134        // card fall through to the canvas.
135        let board = FloatingBoard::new(&overlay);
136        let info_body = info_card::create_info_card(&state_app);
137        let _ = board.add_card_with_id(
138            "info",
139            &info_body,
140            CardConfig {
141                title: Some(fl!(language_loader(), "info-card-title")),
142                position: CardAnchor::TopRight,
143                anchor_margin: 100.0,
144                bounds_mode: BoundsMode::Contained,
145                foldable: true,
146                ..Default::default()
147            },
148        );
149        for (id, body, title, anchor) in [
150            (
151                "visualization",
152                visualization_card::create_visualization_card(
153                    &state_app,
154                    &drawing_area,
155                    &bg_provider,
156                ),
157                fl!(language_loader(), "card-visualization-title"),
158                CardAnchor::TopLeft,
159            ),
160            (
161                "filter",
162                filter_card::create_filter_card(&state_app),
163                fl!(language_loader(), "card-filter-title"),
164                CardAnchor::CenterLeft,
165            ),
166            (
167                "device",
168                device_card::create_device_card(&state_app, &device_path_app),
169                fl!(language_loader(), "card-device-title"),
170                CardAnchor::BottomCenter,
171            ),
172            (
173                "profiles",
174                profiles_card::create_profiles_card(&state_app),
175                fl!(language_loader(), "card-profiles-title"),
176                CardAnchor::CenterRight,
177            ),
178            (
179                "field-sets",
180                field_sets_card::create_field_sets_card(&state_app),
181                fl!(language_loader(), "card-field-sets-title"),
182                CardAnchor::TopCenter,
183            ),
184        ] {
185            let _ = board.add_card_with_id(
186                id,
187                &body,
188                CardConfig {
189                    title: Some(title),
190                    position: anchor,
191                    anchor_margin: 100.0,
192                    bounds_mode: BoundsMode::Contained,
193                    foldable: true,
194                    ..Default::default()
195                },
196            );
197        }
198
199        // Card layout: restore `gui-cards.toml` (or arrange the
200        // default side-by-side), persist on geometry changes.
201        let layout = card_layout::load_layout();
202        // The training card exists only in training mode - its saved
203        // placement (if any) seeds the add/remove watcher.
204        let training_placement = layout.as_ref().and_then(|layout| {
205            layout
206                .cards
207                .iter()
208                .find(|card| card.id == CardId::new("training"))
209                .cloned()
210        });
211        card_layout::wire_layout(&board, layout);
212        training_card::wire_training_card(
213            &board,
214            &state_app,
215            &profiles_dir_app,
216            training_placement,
217        );
218
219        // Touch passthrough: input reaches the cards; everything else
220        // falls through to the desktop.
221        passthrough::wire_passthrough(&board, &state_app);
222
223        let window = ApplicationWindow::builder()
224            .application(app)
225            .title(fl!(language_loader(), "window-title"))
226            .default_width(1600)
227            .default_height(1200)
228            .child(&board)
229            .build();
230
231        // `--touch-passthrough` puts the window into the layer-shell
232        // Overlay layer: a toplevel loses stacking when the
233        // through-click focuses the window beneath (GNOME), and GTK4
234        // has no `keep_above`. Must run before `present`; runtime
235        // toggles only adjust the input region. `init_layer_shell`
236        // degrades to a regular toplevel when the protocol is missing
237        // - logging makes the fallback visible on remote machines.
238        if options.touch_passthrough {
239            use gtk4_layer_shell::Edge;
240            use gtk4_layer_shell::KeyboardMode;
241            use gtk4_layer_shell::Layer;
242            use gtk4_layer_shell::LayerShell;
243
244            if gtk4_layer_shell::is_supported() {
245                info!("layer-shell overlay enabled (touch passthrough)");
246            } else {
247                warn!("layer-shell not supported - window stays a regular toplevel");
248            }
249
250            window.init_layer_shell();
251            window.set_layer(Layer::Overlay);
252            window.set_anchor(Edge::Top, true);
253            window.set_anchor(Edge::Bottom, true);
254            window.set_anchor(Edge::Left, true);
255            window.set_anchor(Edge::Right, true);
256            // -1 = ignore other layer surfaces' exclusive zones - the
257            // surface spans edge-to-edge over bars/docks reserved by
258            // other apps (e.g. waybar on the same Overlay layer).
259            window.set_exclusive_zone(-1);
260            // Stable namespace so compositor rules can match the
261            // surface (e.g. Hyprland `layerrule`).
262            window.set_namespace(Some("tangible-objects"));
263            window.set_keyboard_mode(KeyboardMode::None);
264        }
265
266        // Window background transparent so the canvas fill's alpha
267        // (`background_alpha` toggle) actually shows the desktop
268        // beneath - the theme paints the window background opaque by
269        // default, covering the translucent cairo fill. Only the
270        // window *background* is cleared; contents stay opaque.
271        let provider = gtk4::CssProvider::new();
272        provider.load_from_data("window { background-color: transparent; }");
273        if let Some(display) = gtk4::gdk::Display::default() {
274            gtk4::style_context_add_provider_for_display(
275                &display,
276                &provider,
277                gtk4::STYLE_PROVIDER_PRIORITY_APPLICATION,
278            );
279        }
280
281        window.fullscreen();
282        window.set_visible(visible_app);
283
284        // Redraw timer ~30fps + visibility sync (SetGuiVisible).
285        let state_tick = state_app.clone();
286        let da_tick = drawing_area.clone();
287        let win_tick = window.clone();
288        gtk4::glib::timeout_add_local(Duration::from_millis(33), move || {
289            if let Ok(s) = state_tick.lock()
290                && win_tick.is_visible() != s.visible
291            {
292                win_tick.set_visible(s.visible);
293            }
294            da_tick.queue_draw();
295            ControlFlow::Continue
296        });
297
298        // Escape exits fullscreen.
299        let key_controller = gtk4::EventControllerKey::new();
300        key_controller.connect_key_pressed(|_, keyval, _, _| {
301            if keyval == gtk4::gdk::Key::Escape {
302                std::process::exit(0);
303            }
304            Propagation::Proceed
305        });
306        window.add_controller(key_controller);
307    });
308
309    // `run()` would hand `std::env::args` to `g_application_run` - GTK
310    // would reject daemon flags like `--gui`/`--device` as unknown
311    // options ("Unbekannte Option"), so only argv[0] is passed.
312    app.run_with_args(&["tangible-objects"]);
313}
