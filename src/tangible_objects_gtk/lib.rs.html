1//! Daemon-embedded GTK visualization (`concepts/planned/GUI.md` +
2//! WORKSPACE.md Phase 3).
3//!
4//! Embedded library, not a remote client: the embedding daemon calls
5//! [`run_gui`], which runs the engine/device loop on a background
6//! thread (in-process, zero-latency event flow) and renders the
7//! [`SharedState`] in the GTK main loop. Daemon-mode semantics
8//! (interfaces, control commands, recording, calibration) arrive via
9//! [`GuiHooks`] so this crate never links the daemon crate.

42/// Fullscreen GTK visualization with control bar and evdev-driven
43/// drawing. `patterns` is pre-loaded by the daemon (`load_patterns`);
44/// `visible` carries the initial window state (`--gui` shows by
45/// default; `SetGuiVisible` commands toggle afterwards).
46pub fn run_gui(
47    device_path: String,
48    profiles_dir: Option<String>,
49    patterns: PatternBuf,
50    device_profiles: Option<PathBuf>,
51    config: DetectionConfig,
52    options: GuiOptions,
53    hooks: GuiHooks,
54) {
55    let state = SharedState::new(
56        options.visible,
57        options.rotation,
58        options.background_alpha,
59        options.dead_zones_visible,
60    );
61
62    // Spawn evdev thread - the in-process engine loop.
63    let state_clone = state.clone();
64    let device_profiles_arc = device_profiles.map(Arc::new);
65    std::thread::spawn(move || {
66        evdev_thread::evdev_thread(
67            device_path,
68            patterns,
69            state_clone,
70            config,
71            options.rotation,
72            device_profiles_arc,
73            hooks,
74        );
75    });
76
77    let app = Application::builder()
78        .application_id("dev.aschaeffer.tangible-objects")
79        .build();
80
81    let state_app = state.clone();
82    let profiles_dir_app = profiles_dir.clone();
83    let visible_app = options.visible;
84    app.connect_activate(move |app| {
85        let drawing_area = DrawingArea::new();
86        drawing_area.set_hexpand(true);
87        drawing_area.set_vexpand(true);
88
89        // The canvas darkening lives in the widget's CSS background
90        // layer (`.surface-bg`) - its alpha blends against the
91        // transparent window, so the toggle sees the desktop beneath.
92        // The cairo draw stays content-only on top of it.
93        let bg_provider = gtk4::CssProvider::new();
94        bg_provider.load_from_data(&options.background_alpha.surface_bg_css());
95        drawing_area.add_css_class("surface-bg");
96        drawing_area
97            .style_context()
98            .add_provider(&bg_provider, gtk4::STYLE_PROVIDER_PRIORITY_APPLICATION);
99
100        let state_draw = state_app.clone();
101        drawing_area.set_draw_func(move |_, cr, w, h| {
102            draw::draw_scene(cr, w, h, &state_draw);
103        });
104
105        // Control bar.
106        let control_bar = control_bar::create_control_bar(
107            &state_app,
108            &profiles_dir_app,
109            &drawing_area,
110            &bg_provider,
111        );
112
113        // Layout: drawing area + control bar overlay.
114        let overlay = gtk4::Overlay::new();
115        overlay.set_child(Some(&drawing_area));
116        overlay.add_overlay(&control_bar);
117
118        let window = ApplicationWindow::builder()
119            .application(app)
120            .title(fl!(language_loader(), "window-title"))
121            .default_width(1600)
122            .default_height(1200)
123            .child(&overlay)
124            .build();
125
126        // Window background transparent so the canvas fill's alpha
127        // (`background_alpha` toggle) actually shows the desktop
128        // beneath - the theme paints the window background opaque by
129        // default, covering the translucent cairo fill. Only the
130        // window *background* is cleared; contents stay opaque.
131        let provider = gtk4::CssProvider::new();
132        provider.load_from_data("window { background-color: transparent; }");
133        if let Some(display) = gtk4::gdk::Display::default() {
134            gtk4::style_context_add_provider_for_display(
135                &display,
136                &provider,
137                gtk4::STYLE_PROVIDER_PRIORITY_APPLICATION,
138            );
139        }
140
141        window.fullscreen();
142        window.set_visible(visible_app);
143
144        // Redraw timer ~30fps + visibility sync (SetGuiVisible).
145        let state_tick = state_app.clone();
146        let da_tick = drawing_area.clone();
147        let win_tick = window.clone();
148        gtk4::glib::timeout_add_local(Duration::from_millis(33), move || {
149            if let Ok(s) = state_tick.lock()
150                && win_tick.is_visible() != s.visible
151            {
152                win_tick.set_visible(s.visible);
153            }
154            da_tick.queue_draw();
155            ControlFlow::Continue
156        });
157
158        // Escape exits fullscreen.
159        let key_controller = gtk4::EventControllerKey::new();
160        key_controller.connect_key_pressed(|_, keyval, _, _| {
161            if keyval == gtk4::gdk::Key::Escape {
162                std::process::exit(0);
163            }
164            Propagation::Proceed
165        });
166        window.add_controller(key_controller);
167    });
168
169    // `run()` would hand `std::env::args` to `g_application_run` - GTK
170    // would reject daemon flags like `--gui`/`--device` as unknown
171    // options ("Unbekannte Option"), so only argv[0] is passed.
172    app.run_with_args(&["tangible-objects"]);
173}
