1//! Daemon-embedded GTK visualization (`concepts/planned/GUI.md` +
2//! WORKSPACE.md Phase 3).
3//!
4//! Embedded library, not a remote client: the embedding daemon calls
5//! [`run_gui`], which runs the engine/device loop on a background
6//! thread (in-process, zero-latency event flow) and renders the
7//! [`SharedState`] in the GTK main loop. Daemon-mode semantics
8//! (interfaces, control commands, recording, calibration) arrive via
9//! [`GuiHooks`] so this crate never links the daemon crate.

92/// Content padding shared by every card - the wider chrome frame
93/// makes the resize/rotate grips and the quad-finger gestures easier
94/// to hit on a touchscreen.
95pub(crate) const CARD_CONTENT_PADDING: Pixels = Pixels::new(20.0);
96
97/// The chrome slot list every card shares, modeled on the
98/// floating-board demo: the rotate steppers flank the title bar ends
99/// (counter-clockwise leading, clockwise trailing), title and window
100/// buttons sit inline between them, resize and rotate grips stay
101/// corner affordances. `closable` controls the close button - the
102/// menu card and the mode-bound training card omit it.
103pub(crate) fn card_chrome(closable: bool) -> Vec<ChromeSlot> {
104    let mut slots = vec![
105        ChromeSlot::inline_start(ChromeElement::RotateCounterClockwise),
106        ChromeSlot::inline(ChromeElement::Title),
107        ChromeSlot::inline(ChromeElement::Minimize),
108    ];
109    if closable {
110        slots.push(ChromeSlot::inline(ChromeElement::Close));
111    }
112    slots.extend([
113        ChromeSlot::inline_end(ChromeElement::RotateClockwise),
114        ChromeSlot::inline(ChromeElement::ResizeGrip),
115        ChromeSlot::inline(ChromeElement::RotateGrip),
116    ]);
117    slots
118}

120/// Fullscreen GTK visualization with control bar and evdev-driven
121/// drawing. `patterns` is pre-loaded by the daemon (`load_patterns`);
122/// `visible` carries the initial window state (`--gui` shows by
123/// default; `SetGuiVisible` commands toggle afterwards).
124pub fn run_gui(
125    device_path: String,
126    profiles_dir: Option<String>,
127    patterns: PatternBuf,
128    device_profiles: Option<PathBuf>,
129    config: DetectionConfig,
130    options: GuiOptions,
131    hooks: GuiHooks,
132) {
133    // libadwaita must initialize before the first adw widget is built -
134    // every card renders `ChromeStyle::Adwaita`.
135    if let Err(error) = floating_board::adw::init() {
136        warn!(%error, "libadwaita init failed - card chrome renders unstyled");
137    }
138
139    let state = SharedState::new(&options);
140
141    // Spawn evdev thread - the in-process engine loop.
142    let state_clone = state.clone();
143    let device_profiles_arc = device_profiles.map(Arc::new);
144    let device_path_thread = device_path.clone();
145    std::thread::spawn(move || {
146        evdev_thread::evdev_thread(
147            device_path_thread,
148            patterns,
149            state_clone,
150            config,
151            options.rotation,
152            device_profiles_arc,
153            hooks,
154        );
155    });
156
157    let app = Application::builder()
158        .application_id("dev.aschaeffer.tangible-objects")
159        .build();
160
161    let state_app = state.clone();
162    let profiles_dir_app = profiles_dir.clone();
163    let device_path_app = device_path.clone();
164    let visible_app = options.visible;
165    app.connect_activate(move |app| {
166        let drawing_area = DrawingArea::new();
167        drawing_area.set_hexpand(true);
168        drawing_area.set_vexpand(true);
169
170        // The canvas darkening lives in the widget's CSS background
171        // layer (`.surface-bg`) - its alpha blends against the
172        // transparent window, so the toggle sees the desktop beneath.
173        // The cairo draw stays content-only on top of it.
174        let bg_provider = CssProvider::new();
175        bg_provider.load_from_string(&options.background_alpha.surface_bg_css());
176        drawing_area.add_css_class("surface-bg");
177        if let Some(display) = Display::default() {
178            gtk4::style_context_add_provider_for_display(
179                &display,
180                &bg_provider,
181                gtk4::STYLE_PROVIDER_PRIORITY_APPLICATION,
182            );
183        }
184
185        let state_draw = state_app.clone();
186        drawing_area.set_draw_func(move |_, cr, w, h| {
187            draw::draw_scene(cr, Size::new(w, h), &state_draw);
188        });
189
190        // Layout: the drawing area fills the window; all controls live
191        // on floating cards.
192        let overlay = Overlay::new();
193        overlay.set_child(Some(&drawing_area));
194
195        // Canvas redraw cadence - the evdev thread writes
196        // `SharedState` per frame; this tick repaints at ~10 Hz.
197        let da_tick = drawing_area.clone();
198        gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
199            da_tick.queue_draw();
200            ControlFlow::Continue
201        });
202
203        // Free-floating cards on top of the surface view
204        // (concepts/planned/GUI.md §3): the board stacks a transparent
205        // card layer above the overlay; gestures landing outside a
206        // card fall through to the canvas.
207        let board = FloatingBoard::new(&overlay);
208        let info_body = info_card::create_info_card(&state_app);
209        // The profiles card's Save button needs the `--profiles`
210        // source dir - built before the uniform card loop. The data
211        // cards below also build their folded-content count badges.
212        let (profiles_body, profiles_badge) = profiles_card::create_profiles_card(
213            &state_app,
214            profiles_dir_app.clone().map(PathBuf::from),
215        );
216        let (field_sets_body, field_sets_badge) =
217            field_sets_card::create_field_sets_card(&state_app);
218        let (seats_body, seats_badge) = seats_card::create_seats_card(&state_app);
219        let _ = board.add_card_with_id(
220            "info",
221            &info_body,
222            CardConfig {
223                title: Some(fl!(language_loader(), "info-card-title")),
224                position: CardAnchor::TopRight,
225                anchor_margin: Pixels::new(100.0),
226                bounds_mode: BoundsMode::Contained,
227                content_padding: CARD_CONTENT_PADDING,
228                foldable: true,
229                resizable: true,
230                rotate_grip: true,
231                closable: true,
232                close_action: CardCloseAction::Hide,
233                unfold_on_click: true,
234                drag_handle: DragHandle::WholeCardFolded,
235                chrome_style: ChromeStyle::Adwaita,
236                chrome: Some(card_chrome(true)),
237                ..Default::default()
238            },
239        );
240        for (id, body, title, anchor, folded) in [
241            (
242                "visualization",
243                visualization_card::create_visualization_card(
244                    &state_app,
245                    &drawing_area,
246                    &bg_provider,
247                ),
248                fl!(language_loader(), "card-visualization-title"),
249                CardAnchor::TopLeft,
250                None,
251            ),
252            (
253                "filter",
254                filter_card::create_filter_card(&state_app),
255                fl!(language_loader(), "card-filter-title"),
256                CardAnchor::CenterLeft,
257                None,
258            ),
259            (
260                "device",
261                device_card::create_device_card(&state_app, &device_path_app),
262                fl!(language_loader(), "card-device-title"),
263                CardAnchor::BottomCenter,
264                None,
265            ),
266            (
267                "parameters",
268                parameters_card::create_parameters_card(&state_app),
269                fl!(language_loader(), "card-parameters-title"),
270                CardAnchor::BottomRight,
271                None,
272            ),
273            (
274                "touches",
275                touches_card::create_touches_card(&state_app),
276                fl!(language_loader(), "card-touches-title"),
277                CardAnchor::BottomCenter,
278                None,
279            ),
280            (
281                "calibration",
282                calibration_card::create_calibration_card(&state_app),
283                fl!(language_loader(), "card-calibration-title"),
284                CardAnchor::Center,
285                None,
286            ),
287            (
288                "metrics",
289                metrics_card::create_metrics_card(&state_app),
290                fl!(language_loader(), "card-metrics-title"),
291                CardAnchor::CenterRight,
292                None,
293            ),
294            (
295                "profiles",
296                profiles_body,
297                fl!(language_loader(), "card-profiles-title"),
298                CardAnchor::CenterRight,
299                Some(profiles_badge),
300            ),
301            (
302                "field-sets",
303                field_sets_body,
304                fl!(language_loader(), "card-field-sets-title"),
305                CardAnchor::TopCenter,
306                Some(field_sets_badge),
307            ),
308            (
309                "dead-zones",
310                dead_zones_card::create_dead_zones_card(&state_app),
311                fl!(language_loader(), "card-dead-zones-title"),
312                CardAnchor::BottomLeft,
313                None,
314            ),
315        ] {
316            let _ = board.add_card_with_id(
317                id,
318                &body,
319                CardConfig {
320                    title: Some(title),
321                    position: anchor,
322                    anchor_margin: Pixels::new(100.0),
323                    bounds_mode: BoundsMode::Contained,
324                    content_padding: CARD_CONTENT_PADDING,
325                    foldable: true,
326                    resizable: true,
327                    rotate_grip: true,
328                    // X hides the card (not remove) - the menu re-opens it.
329                    closable: true,
330                    close_action: CardCloseAction::Hide,
331                    folded_content: folded.as_ref().map(|badge| badge.clone().upcast()),
332                    unfold_on_click: true,
333                    drag_handle: DragHandle::WholeCardFolded,
334                    chrome_style: ChromeStyle::Adwaita,
335                    chrome: Some(card_chrome(true)),
336                    ..Default::default()
337                },
338            );
339        }
340
341        // The seats card lives outside the uniform loop: its folded
342        // badge doubles as a draggable FAB (`WholeCardFolded` - the
343        // whole folded surface drags, the unfolded card keeps its
344        // header drag; `unfold_on_click` taps it open again). Its
345        // content is display-only, nothing inside competes for
346        // gestures.
347        let _ = board.add_card_with_id(
348            "seats",
349            &seats_body,
350            CardConfig {
351                title: Some(fl!(language_loader(), "card-seats-title")),
352                position: CardAnchor::CenterRight,
353                anchor_margin: Pixels::new(100.0),
354                bounds_mode: BoundsMode::Contained,
355                content_padding: CARD_CONTENT_PADDING,
356                foldable: true,
357                resizable: true,
358                rotate_grip: true,
359                closable: true,
360                close_action: CardCloseAction::Hide,
361                folded_content: Some(seats_badge.clone().upcast()),
362                unfold_on_click: true,
363                drag_handle: DragHandle::WholeCardFolded,
364                chrome_style: ChromeStyle::Adwaita,
365                chrome: Some(card_chrome(true)),
366                ..Default::default()
367            },
368        );
369
370        // Main-menu card: always-open root listing every other card -
371        // the only card that cannot be closed (`closable: false`).
372        // Folded, it collapses to a round burger-menu FAB.
373        let (menu_body, menu_fab) = menu_card::create_menu_card(&board);
374        let _ = board.add_card_with_id(
375            menu_card::MENU_CARD_ID,
376            &menu_body,
377            CardConfig {
378                title: Some(fl!(language_loader(), "card-menu-title")),
379                position: CardAnchor::TopLeft,
380                anchor_margin: Pixels::new(100.0),
381                bounds_mode: BoundsMode::Contained,
382                content_padding: CARD_CONTENT_PADDING,
383                foldable: true,
384                folded_content: Some(menu_fab.clone().upcast()),
385                unfold_on_click: true,
386                drag_handle: DragHandle::WholeCardFolded,
387                resizable: true,
388                rotate_grip: true,
389                chrome_style: ChromeStyle::Adwaita,
390                chrome: Some(card_chrome(false)),
391                ..Default::default()
392            },
393        );
394
395        // Card layout: restore `gui-cards.toml` (or arrange the
396        // default side-by-side), persist on geometry changes.
397        let layout = card_layout::load_layout();
398        // The training card exists only in training mode - its saved
399        // placement (if any) seeds the add/remove watcher.
400        let training_placement = layout.as_ref().and_then(|layout| {
401            layout
402                .cards
403                .iter()
404                .find(|card| card.id == CardId::new("training"))
405                .cloned()
406        });
407        let spell_training_placement = layout.as_ref().and_then(|layout| {
408            layout
409                .cards
410                .iter()
411                .find(|card| card.id == CardId::new("spell-training"))
412                .cloned()
413        });
414        card_layout::wire_layout(&board, layout);
415        training_card::wire_training_card(
416            &board,
417            &state_app,
418            &profiles_dir_app,
419            training_placement,
420        );
421        spell_training_card::wire_spell_training_card(&board, &state_app, spell_training_placement);
422
423        // Initially only the menu is open (GUI.md): every other card
424        // starts hidden and is opened via the menu's toggle list.
425        for card in board.cards() {
426            if card.id() != CardId::new(menu_card::MENU_CARD_ID) {
427                let _ = board.set_card_visible(&card.id(), false);
428            }
429        }
430
431        // Touch passthrough: input reaches the cards; everything else
432        // falls through to the desktop.
433        passthrough::wire_passthrough(&board, &state_app);
434
435        let window = ApplicationWindow::builder()
436            .application(app)
437            .title(fl!(language_loader(), "window-title"))
438            .default_width(1600)
439            .default_height(1200)
440            .child(&board)
441            .build();
442
443        // Input-region probe: a capture-phase legacy controller counts
444        // every touch down the surface receives. The diagnostics merge
445        // the counter so a dead card splits into "event never arrived"
446        // (applied region wrong) vs "arrived but eaten" (gesture claim).
447        let state_probe = state_app.clone();
448        let window_probe = window.downgrade();
449        let probe = EventControllerLegacy::new();
450        probe.set_propagation_phase(PropagationPhase::Capture);
451        probe.connect_event(move |_, event| {
452            let event_type = event.event_type();
453            if let Ok(mut s) = state_probe.lock() {
454                *s.event_counts.entry(format!("{event_type:?}")).or_insert(0) += 1;
455                if matches!(event_type, EventType::TouchBegin | EventType::ButtonPress) {
456                    let (x, y) = event.position().unwrap_or_default();
457                    s.last_event_position = Some(Vec2::<Pixels>::new(x, y));
458                }
459            }
460            if matches!(event_type, EventType::TouchBegin | EventType::ButtonPress)
461                && let Some(window) = window_probe.upgrade()
462            {
463                let (x, y) = event.position().unwrap_or_default();
464                let picked = window.pick(
465                    x,
466                    y,
467                    gtk4::PickFlags::DEFAULT | gtk4::PickFlags::INSENSITIVE,
468                );
469                let mut ancestry = Vec::new();
470                let mut current = picked.clone();
471                while let Some(widget) = current {
472                    ancestry.push(format!(
473                        "{}(mapped:{},sensitive:{},can_target:{})",
474                        widget.type_().name(),
475                        widget.is_mapped(),
476                        widget.is_sensitive(),
477                        widget.can_target()
478                    ));
479                    current = widget.parent();
480                }
481                debug!(
482                    event = ?event_type,
483                    x,
484                    y,
485                    picked = ?picked.map(|w| w.type_().name().to_string()),
486                    ancestry = %ancestry.join(" <- "),
487                    "pick diagnostic"
488                );
489            }
490            Propagation::Proceed
491        });
492        window.add_controller(probe);
493
494        // `--touch-passthrough` puts the window into the layer-shell
495        // Overlay layer: a toplevel loses stacking when the
496        // through-click focuses the window beneath (GNOME), and GTK4
497        // has no `keep_above`. Must run before `present`; runtime
498        // toggles only adjust the input region. `init_layer_shell`
499        // degrades to a regular toplevel when the protocol is missing
500        // - logging makes the fallback visible on remote machines.
501        if options.touch_passthrough {
502            if gtk4_layer_shell::is_supported() {
503                info!("layer-shell overlay enabled (touch passthrough)");
504            } else {
505                warn!("layer-shell not supported - window stays a regular toplevel");
506            }
507
508            window.init_layer_shell();
509            window.set_layer(Layer::Overlay);
510            window.set_anchor(Edge::Top, true);
511            window.set_anchor(Edge::Bottom, true);
512            window.set_anchor(Edge::Left, true);
513            window.set_anchor(Edge::Right, true);
514            // -1 = ignore other layer surfaces' exclusive zones - the
515            // surface spans edge-to-edge over bars/docks reserved by
516            // other apps (e.g. waybar on the same Overlay layer).
517            window.set_exclusive_zone(-1);
518            // Stable namespace so compositor rules can match the
519            // surface (e.g. Hyprland `layerrule`).
520            window.set_namespace(Some("tangible-objects"));
521            window.set_keyboard_mode(KeyboardMode::None);
522        }
523
524        // Window background transparent so the canvas fill's alpha
525        // (`background_alpha` toggle) actually shows the desktop
526        // beneath - the theme paints the window background opaque by
527        // default, covering the translucent cairo fill. Only the
528        // window *background* is cleared; contents stay opaque.
529        let provider = gtk4::CssProvider::new();
530        // `.round-button` turns the menu card's folded-content FAB into
531        // a circle (same recipe as the floating-board demo). `.badge-fab`
532        // is the thumb-sized variant for count badges: translucent
533        // enough to keep the table scene readable underneath.
534        provider.load_from_string(
535            "window { background-color: transparent; } \
536             .round-button { border-radius: 50%; min-width: 46px; min-height: 46px; padding: 0; border: 1px solid alpha(@theme_fg_color, 0.8); background-color: alpha(@theme_bg_color, 0.5); } \
537             .badge-fab { border-radius: 50%; min-width: 92px; min-height: 92px; padding: 0; \
538             border: 2px solid alpha(@theme_fg_color, 0.8); background-color: alpha(@theme_bg_color, 0.5); }",
539        );
540        if let Some(display) = Display::default() {
541            gtk4::style_context_add_provider_for_display(
542                &display,
543                &provider,
544                gtk4::STYLE_PROVIDER_PRIORITY_APPLICATION,
545            );
546        }
547
548        window.fullscreen();
549        window.set_visible(visible_app);
550
551        // Redraw timer ~30fps + visibility sync (SetGuiVisible).
552        let state_tick = state_app.clone();
553        let da_tick = drawing_area.clone();
554        let win_tick = window.clone();
555        gtk4::glib::timeout_add_local(Duration::from_millis(33), move || {
556            if let Ok(s) = state_tick.lock()
557                && win_tick.is_visible() != s.visible
558            {
559                win_tick.set_visible(s.visible);
560            }
561            da_tick.queue_draw();
562            ControlFlow::Continue
563        });
564
565        // Escape exits fullscreen.
566        let key_controller = EventControllerKey::new();
567        key_controller.connect_key_pressed(|_, keyval, _, _| {
568            if keyval == Key::Escape {
569                std::process::exit(0);
570            }
571            Propagation::Proceed
572        });
573        window.add_controller(key_controller);
574    });
575
576    // `run()` would hand `std::env::args` to `g_application_run` - GTK
577    // would reject daemon flags like `--gui`/`--device` as unknown
578    // options ("Unbekannte Option"), so only argv[0] is passed.
579    app.run_with_args(&["tangible-objects"]);
580}
