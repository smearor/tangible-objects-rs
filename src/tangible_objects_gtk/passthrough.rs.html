1//! Touch passthrough - the Wayland `set_input_region` sync that lets
2//! touches fall through to the desktop beneath except where a card is
3//! hit (`SetTouchPassthrough` / `--touch-passthrough`).
4
5use floating_board::FloatingBoard;
6use gtk4::glib::ControlFlow;
7use gtk4::prelude::*;
8use std::sync::Arc;
9use std::sync::Mutex;
10use std::time::Duration;
11use tracing::debug;
12
13use crate::shared_state::SharedState;

15/// Wires the touch-passthrough input-region sync.
16///
17/// While enabled, the surface accepts input only on the board's
18/// cards; every other touch falls through to the desktop beneath
19/// (Wayland `set_input_region`; a no-op elsewhere). The region is
20/// rebuilt on every tick: card footprints move, rotate and scale with
21/// the gestures. Called once from `run_gui` after the `FloatingBoard`
22/// exists.
23pub fn wire_passthrough(board: &FloatingBoard, state: &Arc<Mutex<SharedState>>) {
24    let board_sync = board.clone();
25    let state_passthrough = state.clone();
26    let mut applied = false;
27    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
28        // Read the flag under lock, drop it before the GTK surface
29        // calls - `set_input_region` may dispatch into event handlers
30        // that take `state` themselves.
31        let enabled = state_passthrough
32            .lock()
33            .map(|s| s.touch_passthrough)
34            .unwrap_or_default();
35        if enabled {
36            // The board must not reset the region on its next
37            // allocation - the consumer owns it while passthrough runs.
38            board_sync.set_input_region_managed(true);
39            if let Some(region) = passthrough_region(&board_sync)
40                && let Some(surface) = board_sync.native().and_then(|n| n.surface())
41            {
42                surface.set_input_region(Some(&region));
43                applied = true;
44            }
45            // Debug: dump each card's footprint vs its visible geometry -
46            // a card missing from the union is completely input-dead.
47            for card in board_sync.cards() {
48                let geometry = card.geometry();
49                let footprint = card.footprint();
50                let extents = gtk4::cairo::RectangleInt::new(0, 0, 0, 0);
51                footprint.extents(&extents);
52                debug!(
53                    card = %card.id(),
54                    position = ?geometry.position,
55                    rotation = ?geometry.rotation,
56                    scale = ?geometry.scale,
57                    footprint = ?(extents.x(), extents.y(), extents.width(), extents.height()),
58                    visible = %card.is_visible(),
59                    "passthrough card footprint"
60                );
61            }
62        } else if applied {
63            applied = false;
64            board_sync.set_input_region_managed(false);
65            if let Some(surface) = board_sync.native().and_then(|n| n.surface()) {
66                surface.set_input_region(None);
67            }
68        }
69        ControlFlow::Continue
70    });
71}

73/// The passthrough input region in surface coordinates: the rotated
74/// footprints of every card on the board.
75fn passthrough_region(board: &FloatingBoard) -> Option<gtk4::cairo::Region> {
76    let native = board.native()?;
77    let cards = board.cards_region();
78    if cards.is_empty() {
79        return Some(cards);
80    }
81    let origin = board.compute_point(&native, &gtk4::graphene::Point::new(0.0, 0.0))?;
82    cards.translate(origin.x().round() as i32, origin.y().round() as i32);
83    Some(cards)
84}
