33/// Wires the touch-passthrough input-region sync.
34///
35/// While enabled, the surface accepts input on the board's cards and
36/// inside registered dead zones; every other touch falls through to
37/// the desktop beneath (Wayland `set_input_region`; a no-op
38/// elsewhere). Dead zones must stay inside the region so their
39/// contacts are swallowed by the overlay - the engine drops them at
40/// ingest, so the application below must not receive them either.
41/// The region is rebuilt on every tick: card footprints move, rotate
42/// and scale with the gestures. Called once from `run_gui` after the
43/// `FloatingBoard` exists.
44pub fn wire_passthrough(board: &FloatingBoard, state: &Arc<Mutex<SharedState>>) {
45    let board_sync = board.clone();
46    let state_passthrough = state.clone();
47    let mut applied = false;
48    let mut last_region_hash = 0u64;
49    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
50        // Read the flag and the zone inputs under lock, drop it before
51        // the GTK surface calls - `set_input_region` may dispatch into
52        // event handlers that take `state` themselves.
53        let (enabled, dead_zones, calibration, rotation, event_counts, last_event_position) =
54            state_passthrough
55                .lock()
56                .map(|s| {
57                    (
58                        s.touch_passthrough,
59                        s.dead_zones.clone(),
60                        s.calibration,
61                        s.rotation,
62                        s.event_counts.clone(),
63                        s.last_event_position,
64                    )
65                })
66                .unwrap_or_default();
67        // Card hit areas are collected every tick regardless of the
68        // flag - the device loop filters contacts that land on a card
69        // out of the touch set (a card tap must not disturb training
70        // stability or draw a surface dot).
71        let footprints = card_footprints(&board_sync);
72        if enabled {
73            // The board must not reset the region on its next
74            // allocation - the consumer owns it while passthrough runs.
75            board_sync.set_input_region_managed(true);
76            let mut diagnostics = InputRegionDiagnostics {
77                event_counts,
78                last_event_position,
79                ..Default::default()
80            };
81            if let Some((region, origin)) =
82                passthrough_region(&board_sync, &dead_zones, calibration, rotation)
83            {
84                diagnostics.origin = origin;
85                diagnostics.rectangles = region.num_rectangles() as usize;
86                diagnostics.extents = region_extents(&region);
87                diagnostics.status = region.status().err().map(|e| e.to_string());
88                diagnostics.first_rectangle = first_rectangle(&region);
89                log_region_change(&region, &footprints, &mut last_region_hash);
90                if let Some(surface) = board_sync.native().and_then(|n| n.surface()) {
91                    surface.set_input_region(Some(&region));
92                    diagnostics.applied = true;
93                    applied = true;
94                }
95            }
96            diagnostics.cards = footprints.clone();
97            if let Ok(mut s) = state_passthrough.lock() {
98                s.input_region_diagnostics = diagnostics;
99            }
100        } else {
101            if applied {
102                applied = false;
103                board_sync.set_input_region_managed(false);
104                if let Some(surface) = board_sync.native().and_then(|n| n.surface()) {
105                    surface.set_input_region(None);
106                }
107            }
108            if let Ok(mut s) = state_passthrough.lock() {
109                s.input_region_diagnostics = InputRegionDiagnostics::default();
110            }
111        }
112        if let Ok(mut s) = state_passthrough.lock() {
113            s.card_footprints = footprints;
114        }
115        ControlFlow::Continue
116    });
117}

119/// Per-card hit areas in board-layer pixels - every card on the
120/// board with its position, rotation, scale and the rasterized
121/// footprint bounds. Consumed by the input-region diagnostics and by
122/// the device loop's card-touch filter, so it is collected on every
123/// tick whether passthrough is enabled or not.
124fn card_footprints(board: &FloatingBoard) -> Vec<CardFootprint> {
125    board
126        .cards()
127        .iter()
128        .map(|card| {
129            let geometry = card.geometry();
130            let footprint = card.footprint();
131            let extents = region_extents(&footprint);
132            let rendered = rendered_bounds(board, card);
133            let placed = card.placed_position();
134            let mut chain = Vec::new();
135            if let Some(bounds) = rendered {
136                let center = Vec2::new(bounds.min.x + bounds.size.x / 2.0, bounds.min.y + bounds.size.y / 2.0);
137                let drift = center - Vec2::from(geometry.position);
138                if card.is_visible() && (drift.x.abs() > 1.0 || drift.y.abs() > 1.0) {
139                    chain = ancestor_chain(board, card);
140                    debug!(
141                        card = %card.id(),
142                        position = ?Vec2::<Pixels>::from(geometry.position),
143                        placed = ?placed.map(|p| (p.x, p.y)),
144                        rendered = ?(bounds.min.x, bounds.min.y, bounds.size.x, bounds.size.y),
145                        drift = ?(drift.x, drift.y),
146                        chain = ?chain,
147                        "rendered card drifted from its model position - input region cannot cover it"
148                    );
149                }
150            }
151            debug!(
152                card = %card.id(),
153                position = ?geometry.position,
154                rotation = ?geometry.rotation,
155                scale = ?geometry.scale,
156                footprint = ?extents.map(|e| (e.min.x, e.min.y, e.size.x, e.size.y)),
157                visible = %card.is_visible(),
158                "passthrough card footprint"
159            );
160            let (_, natural) = card.measure_size();
161            CardFootprint {
162                card_id: CardId::new(card.id().to_string()),
163                position: geometry.position.into(),
164                rotation: geometry.rotation,
165                scale: ScaleFactor::new(geometry.scale.into()),
166                visible: card.is_visible(),
167                footprint: extents,
168                natural: natural.with_unit::<Pixels>(),
169                allocated: card.allocated_size().with_unit::<Pixels>(),
170                rendered,
171                placed,
172                chain,
173                footprint_status: footprint.status().err().map(|e| e.to_string()),
174                footprint_rectangles: footprint.num_rectangles() as usize,
175                first_rectangle: first_rectangle(&footprint),
176            }
177        })
178        .collect()
179}

181/// The card's rendered bounds in board coordinates - the axis-aligned
182/// box `compute_bounds` reports for the actual widget with every
183/// rotation and scale of its transform stack applied. This is the
184/// ground truth of where the card draws; the model footprint
185/// (`entry_footprint`) only matches it when the layout kept the card
186/// centered on its stored position.
187fn rendered_bounds(board: &FloatingBoard, card: &floating_board::Card) -> Option<Rect<Pixels>> {
188    let bounds = card.compute_bounds(board)?;
189    Some(Rect::new(
190        Vec2::new(f64::from(bounds.x()), f64::from(bounds.y())),
191        Vec2::new(f64::from(bounds.width()), f64::from(bounds.height())),
192    ))
193}

195/// Every ancestor of the card up to the board with its widget type,
196/// its `(0,0)` origin mapped to board coordinates and its GTK
197/// allocation - the layer-by-layer view that pinpoints which level
198/// of the card → rotation → scaling → layer chain introduces a
199/// render-vs-placement offset.
200fn ancestor_chain(board: &FloatingBoard, card: &floating_board::Card) -> Vec<String> {
201    let mut chain = Vec::new();
202    let mut widget: Option<gtk4::Widget> = Some(card.clone().upcast());
203    while let Some(current) = widget {
204        let origin = current.compute_point(board, &gtk4::graphene::Point::new(0.0, 0.0));
205        let bounds = current.compute_bounds(&current);
206        chain.push(format!(
207            "{}: origin={:?} alloc={:?}",
208            current.type_().name(),
209            origin,
210            bounds
211        ));
212        widget = current.parent();
213    }
214    chain
215}

218/// The passthrough input region in surface coordinates: the rotated
219/// footprints of every card on the board, unioned with the rasterized
220/// screen circles of all registered dead zones. Returns the region
221/// plus the board-to-surface origin offset it was translated by.
222fn passthrough_region(
223    board: &FloatingBoard,
224    dead_zones: &[DeadZone],
225    calibration: Option<Calibration>,
226    rotation: DisplayRotation,
227) -> Option<(Region, Vec2<Pixels>)> {
228    let native = board.native()?;
229    let region = board.cards_region();
230    union_dead_zones(&region, board, dead_zones, calibration, rotation);
231    if region.is_empty() {
232        return Some((region, Vec2::ZERO));
233    }
234    let origin = board.compute_point(&native, &gtk4::graphene::Point::new(0.0, 0.0))?;
235    region.translate(origin.x().round() as i32, origin.y().round() as i32);
236    Some((
237        region,
238        Vec2::<Pixels>::new(f64::from(origin.x()), f64::from(origin.y())),
239    ))
240}

242/// Integer extents of a cairo region - `None` when the region is
243/// empty. Shared by the unioned input region and every card
244/// footprint in the diagnostics payload.
245fn region_extents(region: &Region) -> Option<Rect<Pixels, i32>> {
246    if region.is_empty() {
247        return None;
248    }
249    let extents = RectangleInt::new(0, 0, 0, 0);
250    region.extents(&extents);
251    Some(Rect::new(
252        Vec2::new(extents.x(), extents.y()),
253        Vec2::new(extents.width(), extents.height()),
254    ))
255}

257/// The region's first stored rectangle - a direct coordinate probe
258/// that bypasses `extents` entirely.
259fn first_rectangle(region: &Region) -> Option<Rect<Pixels, i32>> {
260    if region.is_empty() || region.num_rectangles() <= 0 {
261        return None;
262    }
263    let rect = region.rectangle(0);
264    Some(Rect::new(
265        Vec2::new(rect.x(), rect.y()),
266        Vec2::new(rect.width(), rect.height()),
267    ))
268}

270/// Logs the full rectangle list whenever the unioned region changes -
271/// the MCP diagnostics only expose count/first-rect, which is not
272/// enough to tell which rectangles `set_input_region` actually got.
273/// Also probes `contains_point` at every visible card's footprint
274/// center so a corrupted union (rectangles stored but not covering
275/// their area) shows up immediately.
276fn log_region_change(region: &Region, footprints: &[CardFootprint], last_hash: &mut u64) {
277    let mut hasher = std::collections::hash_map::DefaultHasher::new();
278    for index in 0..region.num_rectangles() {
279        let rect = region.rectangle(index);
280        rect.x().hash(&mut hasher);
281        rect.y().hash(&mut hasher);
282        rect.width().hash(&mut hasher);
283        rect.height().hash(&mut hasher);
284    }
285    let hash = hasher.finish();
286    if hash == *last_hash {
287        return;
288    }
289    *last_hash = hash;
290    let rectangles = (0..region.num_rectangles())
291        .map(|index| {
292            let rect = region.rectangle(index);
293            format!(
294                "{}+{} {}x{}",
295                rect.x(),
296                rect.y(),
297                rect.width(),
298                rect.height()
299            )
300        })
301        .collect::<Vec<_>>()
302        .join(" ");
303    let extents = RectangleInt::new(0, 0, 0, 0);
304    region.extents(&extents);
305    let coverage = footprints
306        .iter()
307        .filter(|footprint| footprint.visible)
308        .filter_map(|footprint| {
309            footprint.footprint.as_ref().map(|rect| {
310                let center_x = rect.min.x + rect.size.x / 2;
311                let center_y = rect.min.y + rect.size.y / 2;
312                let model = region.contains_point(center_x, center_y);
313                let rendered = footprint
314                    .rendered
315                    .as_ref()
316                    .map(|bounds| {
317                        region.contains_point(
318                            (bounds.min.x + bounds.size.x / 2.0).round() as i32,
319                            (bounds.min.y + bounds.size.y / 2.0).round() as i32,
320                        )
321                    })
322                    .map(|covered| if covered { 't' } else { 'F' })
323                    .unwrap_or('?');
324                format!(
325                    "{}={}/{}",
326                    footprint.card_id,
327                    if model { 't' } else { 'F' },
328                    rendered
329                )
330            })
331        })
332        .collect::<Vec<_>>()
333        .join(" ");
334    debug!(
335        status = ?region.status(),
336        extents = ?(extents.x(), extents.y(), extents.width(), extents.height()),
337        %coverage,
338        %rectangles,
339        "passthrough input region"
340    );
341}

343/// Unions every dead zone's screen circle into `region` (board
344/// coordinates - the card layer spans the same canvas the mm→px
345/// mapping draws on, so the existing board→surface translation
346/// applies to both).
347fn union_dead_zones(
348    region: &Region,
349    board: &FloatingBoard,
350    dead_zones: &[DeadZone],
351    calibration: Option<Calibration>,
352    rotation: DisplayRotation,
353) {
354    let Some(calibration) = calibration else {
355        return;
356    };
357    let canvas = Vec2::<Pixels>::new(board.width() as f64, board.height() as f64);
358    if canvas.x <= 0.0 || canvas.y <= 0.0 {
359        return;
360    }
361    for zone in dead_zones {
362        let center = calibration.mm_to_screen(zone.center, canvas, rotation);
363        let radius = dead_zone_radius_px(zone, &calibration, canvas, rotation);
364        if radius <= 0.0 {
365            continue;
366        }
367        for rect in Circle::new(center, radius).region_rects() {
368            if let Err(error) = region.union_rectangle(&RectangleInt::from(rect)) {
369                warn!(%error, zone = %zone.name, "failed to union dead-zone strip");
370            }
371        }
372    }
373}

375/// The zone's suppression radius in screen pixels - mean of both
376/// axes, matching the overlay draw (`draw_dead_zones`) so the input
377/// region covers the drawn circle.
378fn dead_zone_radius_px(
379    zone: &DeadZone,
380    calibration: &Calibration,
381    canvas: Vec2<Pixels>,
382    rotation: DisplayRotation,
383) -> f64 {
384    let center = calibration.mm_to_screen(zone.center, canvas, rotation);
385    let radius = zone.radius.as_f64();
386    let rx = calibration
387        .mm_to_screen(zone.center + Vec2::new(radius, 0.0), canvas, rotation)
388        .distance(&center)
389        .as_f64();
390    let ry = calibration
391        .mm_to_screen(zone.center + Vec2::new(0.0, radius), canvas, rotation)
392        .distance(&center)
393        .as_f64();
394    (rx + ry) / 2.0
395}

398mod tests {
399    use cairo::RectangleInt;
400    use smearor_commons_geometry::Circle;

404    /// Every disk point lies inside a generated strip; each strip is a
405    /// 1-px row inside the circle's bounding box.
406    #[test]
407    fn circle_rects_cover_the_disk() {
408        let center = Vec2::<Pixels>::new(100.4, 80.6);
409        let radius = 30.0;
410        let strips: Vec<RectangleInt> = Circle::new(center, radius)
411            .region_rects()
412            .into_iter()
413            .map(RectangleInt::from)
414            .collect();
415        assert!(!strips.is_empty());
416        assert!(strips.iter().all(|strip| strip.height() == 1));
417        assert!(strips.iter().all(|strip| {
418            strip.y() >= (center.y - radius).floor() as i32
419                && strip.y() < (center.y + radius).ceil() as i32
420        }));
421
422        let mut y = center.y - radius;
423        while y <= center.y + radius {
424            let mut x = center.x - radius;
425            while x <= center.x + radius {
426                if (x - center.x).powi(2) + (y - center.y).powi(2) <= radius * radius {
427                    let covered = strips.iter().any(|strip| {
428                        x >= f64::from(strip.x())
429                            && x < f64::from(strip.x() + strip.width())
430                            && y >= f64::from(strip.y())
431                            && y < f64::from(strip.y() + 1)
432                    });
433                    assert!(covered, "disk point ({x}, {y}) not covered");
434                }
435                x += 0.25;
436            }
437            y += 0.25;
438        }
439    }

441    /// cairo `Region::extents` must write the union bounds back into
442    /// the passed `RectangleInt` - the diagnostics rely on it.
443    #[test]
444    fn cairo_extents_write_back() {
445        let region = gtk4::cairo::Region::create();
446        region
447            .union_rectangle(&gtk4::cairo::RectangleInt::new(10, 20, 30, 40))

451        assert_eq!(
452            (extents.x(), extents.y(), extents.width(), extents.height()),
