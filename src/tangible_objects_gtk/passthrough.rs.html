32/// Wires the touch-passthrough input-region sync.
33///
34/// While enabled, the surface accepts input on the board's cards and
35/// inside registered dead zones; every other touch falls through to
36/// the desktop beneath (Wayland `set_input_region`; a no-op
37/// elsewhere). Dead zones must stay inside the region so their
38/// contacts are swallowed by the overlay - the engine drops them at
39/// ingest, so the application below must not receive them either.
40/// The region is rebuilt on every tick: card footprints move, rotate
41/// and scale with the gestures. Called once from `run_gui` after the
42/// `FloatingBoard` exists.
43pub fn wire_passthrough(board: &FloatingBoard, state: &Arc<Mutex<SharedState>>) {
44    let board_sync = board.clone();
45    let state_passthrough = state.clone();
46    let mut applied = false;
47    let mut last_region_hash = 0u64;
48    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
49        // Read the flag and the zone inputs under lock, drop it before
50        // the GTK surface calls - `set_input_region` may dispatch into
51        // event handlers that take `state` themselves.
52        let (enabled, dead_zones, calibration, rotation, event_counts, last_event_position) =
53            state_passthrough
54                .lock()
55                .map(|s| {
56                    (
57                        s.touch_passthrough,
58                        s.dead_zones.clone(),
59                        s.calibration,
60                        s.rotation,
61                        s.event_counts.clone(),
62                        s.last_event_position,
63                    )
64                })
65                .unwrap_or_default();
66        // Card hit areas are collected every tick regardless of the
67        // flag - the device loop filters contacts that land on a card
68        // out of the touch set (a card tap must not disturb training
69        // stability or draw a surface dot).
70        let footprints = card_footprints(&board_sync);
71        if enabled {
72            // The board must not reset the region on its next
73            // allocation - the consumer owns it while passthrough runs.
74            board_sync.set_input_region_managed(true);
75            let mut diagnostics = InputRegionDiagnostics {
76                event_counts,
77                last_event_position,
78                ..Default::default()
79            };
80            if let Some((region, origin)) =
81                passthrough_region(&board_sync, &dead_zones, calibration, rotation)
82            {
83                diagnostics.origin = origin;
84                diagnostics.rectangles = region.num_rectangles() as usize;
85                diagnostics.extents = region_extents(&region);
86                diagnostics.status = region.status().err().map(|e| e.to_string());
87                diagnostics.first_rectangle = first_rectangle(&region);
88                log_region_change(&region, &footprints, &mut last_region_hash);
89                if let Some(surface) = board_sync.native().and_then(|n| n.surface()) {
90                    surface.set_input_region(Some(&region));
91                    diagnostics.applied = true;
92                    applied = true;
93                }
94            }
95            diagnostics.cards = footprints.clone();
96            if let Ok(mut s) = state_passthrough.lock() {
97                s.input_region_diagnostics = diagnostics;
98            }
99        } else {
100            if applied {
101                applied = false;
102                board_sync.set_input_region_managed(false);
103                if let Some(surface) = board_sync.native().and_then(|n| n.surface()) {
104                    surface.set_input_region(None);
105                }
106            }
107            if let Ok(mut s) = state_passthrough.lock() {
108                s.input_region_diagnostics = InputRegionDiagnostics::default();
109            }
110        }
111        if let Ok(mut s) = state_passthrough.lock() {
112            s.card_footprints = footprints;
113        }
114        ControlFlow::Continue
115    });
116}

118/// Per-card hit areas in board-layer pixels - every card on the
119/// board with its position, rotation, scale and the rasterized
120/// footprint bounds. Consumed by the input-region diagnostics and by
121/// the device loop's card-touch filter, so it is collected on every
122/// tick whether passthrough is enabled or not.
123fn card_footprints(board: &FloatingBoard) -> Vec<CardFootprint> {
124    board
125        .cards()
126        .iter()
127        .map(|card| {
128            let geometry = card.geometry();
129            let footprint = card.footprint();
130            let extents = region_extents(&footprint);
131            let rendered = rendered_bounds(board, &card);
132            let placed = placed_position(&card);
133            let mut chain = Vec::new();
134            if let Some(bounds) = rendered {
135                let center = Vec2::new(bounds.min.x + bounds.size.x / 2.0, bounds.min.y + bounds.size.y / 2.0);
136                let drift = center - Vec2::from(geometry.position);
137                if card.is_visible() && (drift.x.abs() > 1.0 || drift.y.abs() > 1.0) {
138                    chain = ancestor_chain(board, &card);
139                    debug!(
140                        card = %card.id(),
141                        position = ?Vec2::<Pixels>::from(geometry.position),
142                        placed = ?placed.map(|p| (p.x, p.y)),
143                        rendered = ?(bounds.min.x, bounds.min.y, bounds.size.x, bounds.size.y),
144                        drift = ?(drift.x, drift.y),
145                        chain = ?chain,
146                        "rendered card drifted from its model position - input region cannot cover it"
147                    );
148                }
149            }
150            debug!(
151                card = %card.id(),
152                position = ?geometry.position,
153                rotation = ?geometry.rotation,
154                scale = ?geometry.scale,
155                footprint = ?extents.map(|e| (e.min.x, e.min.y, e.size.x, e.size.y)),
156                visible = %card.is_visible(),
157                "passthrough card footprint"
158            );
159            let (_, natural) = card.measure_size();
160            CardFootprint {
161                card_id: CardId::new(card.id().to_string()),
162                position: geometry.position.into(),
163                rotation: geometry.rotation,
164                scale: ScaleFactor::new(geometry.scale.into()),
165                visible: card.is_visible(),
166                footprint: extents,
167                natural: natural.with_unit::<Pixels>(),
168                allocated: card.allocated_size().with_unit::<Pixels>(),
169                rendered,
170                placed,
171                chain,
172                footprint_status: footprint.status().err().map(|e| e.to_string()),
173                footprint_rectangles: footprint.num_rectangles() as usize,
174                first_rectangle: first_rectangle(&footprint),
175            }
176        })
177        .collect()
178}

180/// The card's rendered bounds in board coordinates - the axis-aligned
181/// box `compute_bounds` reports for the actual widget with every
182/// rotation and scale of its transform stack applied. This is the
183/// ground truth of where the card draws; the model footprint
184/// (`entry_footprint`) only matches it when the layout kept the card
185/// centered on its stored position.
186fn rendered_bounds(board: &FloatingBoard, card: &floating_board::Card) -> Option<Rect<Pixels>> {
187    let bounds = card.compute_bounds(board)?;
188    Some(Rect::new(
189        Vec2::new(f64::from(bounds.x()), f64::from(bounds.y())),
190        Vec2::new(f64::from(bounds.width()), f64::from(bounds.height())),
191    ))
192}

194/// Every ancestor of the card up to the board with its widget type,
195/// its `(0,0)` origin mapped to board coordinates and its GTK
196/// allocation - the layer-by-layer view that pinpoints which level
197/// of the card → rotation → scaling → layer chain introduces a
198/// render-vs-placement offset.
199fn ancestor_chain(board: &FloatingBoard, card: &floating_board::Card) -> Vec<String> {
200    let mut chain = Vec::new();
201    let mut widget: Option<gtk4::Widget> = Some(card.clone().upcast());
202    while let Some(current) = widget {
203        let origin = current
204            .compute_point(board, &gtk4::graphene::Point::new(0.0, 0.0))
205            .map(|point| (point.x(), point.y()));
206        let bounds = current.compute_bounds(&current).map(|rect| (rect.width(), rect.height()));
207        chain.push(format!("{}: origin={:?} alloc={:?}", current.type_().name(), origin, bounds));
208        widget = current.parent();
209    }
210    chain
211}

213/// The transform stack's stored position on the card layer - the
214/// `(x, y)` `gtk4::Fixed` was last asked to place it at. The widget
215/// chain is card → rotation → scaling → layer, so the scaling
216/// widget's parent is the layer and `child_position` reports the
217/// stored `put` spot. `None` when the widget tree is not attached
218/// yet. Comparing `placed` with `rendered.min` shows whether the
219/// layer actually moved the stack where `apply_position` put it;
220/// comparing `placed + size/2` with `position` shows whether the
221/// board's model and its placement agree.
222fn placed_position(card: &floating_board::Card) -> Option<Vec2<Pixels>> {
223    let scaling = card.parent()?.parent()?;
224    let layer = scaling.parent()?;
225    let fixed = layer.downcast::<gtk4::Fixed>().ok()?;
226    let (x, y) = fixed.child_position(&scaling);
227    Some(Vec2::new(f64::from(x), f64::from(y)))
228}

230/// The passthrough input region in surface coordinates: the rotated
231/// footprints of every card on the board, unioned with the rasterized
232/// screen circles of all registered dead zones. Returns the region
233/// plus the board-to-surface origin offset it was translated by.
234fn passthrough_region(
235    board: &FloatingBoard,
236    dead_zones: &[DeadZone],
237    calibration: Option<Calibration>,
238    rotation: DisplayRotation,
239) -> Option<(Region, Vec2<Pixels>)> {
240    let native = board.native()?;
241    let region = board.cards_region();
242    union_dead_zones(&region, board, dead_zones, calibration, rotation);
243    if region.is_empty() {
244        return Some((region, Vec2::ZERO));
245    }
246    let origin = board.compute_point(&native, &gtk4::graphene::Point::new(0.0, 0.0))?;
247    region.translate(origin.x().round() as i32, origin.y().round() as i32);
248    Some((
249        region,
250        Vec2::<Pixels>::new(f64::from(origin.x()), f64::from(origin.y())),
251    ))
252}

254/// Integer extents of a cairo region - `None` when the region is
255/// empty. Shared by the unioned input region and every card
256/// footprint in the diagnostics payload.
257fn region_extents(region: &Region) -> Option<Rect<Pixels, i32>> {
258    if region.is_empty() {
259        return None;
260    }
261    let extents = RectangleInt::new(0, 0, 0, 0);
262    region.extents(&extents);
263    Some(Rect::new(
264        Vec2::new(extents.x(), extents.y()),
265        Vec2::new(extents.width(), extents.height()),
266    ))
267}

269/// The region's first stored rectangle - a direct coordinate probe
270/// that bypasses `extents` entirely.
271fn first_rectangle(region: &Region) -> Option<Rect<Pixels, i32>> {
272    if region.is_empty() || region.num_rectangles() <= 0 {
273        return None;
274    }
275    let rect = region.rectangle(0);
276    Some(Rect::new(
277        Vec2::new(rect.x(), rect.y()),
278        Vec2::new(rect.width(), rect.height()),
279    ))
280}

282/// Logs the full rectangle list whenever the unioned region changes -
283/// the MCP diagnostics only expose count/first-rect, which is not
284/// enough to tell which rectangles `set_input_region` actually got.
285/// Also probes `contains_point` at every visible card's footprint
286/// center so a corrupted union (rectangles stored but not covering
287/// their area) shows up immediately.
288fn log_region_change(region: &Region, footprints: &[CardFootprint], last_hash: &mut u64) {
289    let mut hasher = std::collections::hash_map::DefaultHasher::new();
290    for index in 0..region.num_rectangles() {
291        let rect = region.rectangle(index);
292        rect.x().hash(&mut hasher);
293        rect.y().hash(&mut hasher);
294        rect.width().hash(&mut hasher);
295        rect.height().hash(&mut hasher);
296    }
297    let hash = hasher.finish();
298    if hash == *last_hash {
299        return;
300    }
301    *last_hash = hash;
302    let rectangles = (0..region.num_rectangles())
303        .map(|index| {
304            let rect = region.rectangle(index);
305            format!(
306                "{}+{} {}x{}",
307                rect.x(),
308                rect.y(),
309                rect.width(),
310                rect.height()
311            )
312        })
313        .collect::<Vec<_>>()
314        .join(" ");
315    let extents = RectangleInt::new(0, 0, 0, 0);
316    region.extents(&extents);
317    let coverage = footprints
318        .iter()
319        .filter(|footprint| footprint.visible)
320        .filter_map(|footprint| {
321            footprint.footprint.as_ref().map(|rect| {
322                let center_x = rect.min.x + rect.size.x / 2;
323                let center_y = rect.min.y + rect.size.y / 2;
324                let model = region.contains_point(center_x, center_y);
325                let rendered = footprint
326                    .rendered
327                    .as_ref()
328                    .map(|bounds| {
329                        region.contains_point(
330                            (bounds.min.x + bounds.size.x / 2.0).round() as i32,
331                            (bounds.min.y + bounds.size.y / 2.0).round() as i32,
332                        )
333                    })
334                    .map(|covered| if covered { 't' } else { 'F' })
335                    .unwrap_or('?');
336                format!("{}={}/{}", footprint.card_id, if model { 't' } else { 'F' }, rendered)
337            })
338        })
339        .collect::<Vec<_>>()
340        .join(" ");
341    debug!(
342        status = ?region.status(),
343        extents = ?(extents.x(), extents.y(), extents.width(), extents.height()),
344        %coverage,
345        %rectangles,
346        "passthrough input region"
347    );
348}

350/// Unions every dead zone's screen circle into `region` (board
351/// coordinates - the card layer spans the same canvas the mm→px
352/// mapping draws on, so the existing board→surface translation
353/// applies to both).
354fn union_dead_zones(
355    region: &Region,
356    board: &FloatingBoard,
357    dead_zones: &[DeadZone],
358    calibration: Option<Calibration>,
359    rotation: DisplayRotation,
360) {
361    let Some(calibration) = calibration else {
362        return;
363    };
364    let canvas = Vec2::<Pixels>::new(board.width() as f64, board.height() as f64);
365    if canvas.x <= 0.0 || canvas.y <= 0.0 {
366        return;
367    }
368    for zone in dead_zones {
369        let center = calibration.mm_to_screen(zone.center, canvas, rotation);
370        let radius = dead_zone_radius_px(zone, &calibration, canvas, rotation);
371        if radius <= 0.0 {
372            continue;
373        }
374        for strip in circle_rects(center, radius) {
375            if let Err(error) = region.union_rectangle(&strip) {
376                warn!(%error, zone = %zone.name, "failed to union dead-zone strip");
377            }
378        }
379    }
380}

382/// The zone's suppression radius in screen pixels - mean of both
383/// axes, matching the overlay draw (`draw_dead_zones`) so the input
384/// region covers the drawn circle.
385fn dead_zone_radius_px(
386    zone: &DeadZone,
387    calibration: &Calibration,
388    canvas: Vec2<Pixels>,
389    rotation: DisplayRotation,
390) -> f64 {
391    let center = calibration.mm_to_screen(zone.center, canvas, rotation);
392    let radius = zone.radius.as_f64();
393    let rx = calibration
394        .mm_to_screen(zone.center + Vec2::new(radius, 0.0), canvas, rotation)
395        .distance(&center)
396        .as_f64();
397    let ry = calibration
398        .mm_to_screen(zone.center + Vec2::new(0.0, radius), canvas, rotation)
399        .distance(&center)
400        .as_f64();
401    (rx + ry) / 2.0
402}

404/// Rasterizes a screen circle into 1-px-high [`gtk4::cairo::RectangleInt`]
405/// strips - cairo regions are rectangle unions, so the disk becomes one
406/// row per scanline. Each row uses the chord at its center-nearest edge
407/// (the widest the circle reaches inside that row), so the union covers
408/// every point the zone's `contains` would hit.
409fn circle_rects(center: Vec2<Pixels>, radius: f64) -> Vec<RectangleInt> {
410    let top = (center.y - radius).floor() as i32;
411    let bottom = (center.y + radius).ceil() as i32;
412    (top..bottom)
413        .filter_map(|y| {
414            // Distance from the center to the row's nearest edge - zero
415            // when the row straddles the center line.
416            let nearest = (f64::from(y) - center.y)
417                .max(center.y - f64::from(y + 1))
418                .max(0.0);
419            if nearest > radius {
420                return None;
421            }
422            let half = (radius * radius - nearest * nearest).sqrt().max(0.5);
423            let left = (center.x - half).floor() as i32;
424            let right = (center.x + half).ceil() as i32;
425            Some(RectangleInt::new(left, y, right - left, 1))
426        })
427        .collect()
428}

436    /// Every disk point lies inside a generated strip; each strip is a
437    /// 1-px row inside the circle's bounding box.
438    #[test]
439    fn circle_rects_cover_the_disk() {
440        let center = Vec2::<Pixels>::new(100.4, 80.6);
441        let radius = 30.0;
442        let strips = circle_rects(center, radius);
443        assert!(!strips.is_empty());
444        assert!(strips.iter().all(|strip| strip.height() == 1));
445        assert!(strips.iter().all(|strip| {
446            strip.y() >= (center.y - radius).floor() as i32
447                && strip.y() < (center.y + radius).ceil() as i32
448        }));
449
450        let mut y = center.y - radius;
451        while y <= center.y + radius {
452            let mut x = center.x - radius;
453            while x <= center.x + radius {
454                if (x - center.x).powi(2) + (y - center.y).powi(2) <= radius * radius {
455                    let covered = strips.iter().any(|strip| {
456                        x >= f64::from(strip.x())
457                            && x < f64::from(strip.x() + strip.width())
458                            && y >= f64::from(strip.y())
459                            && y < f64::from(strip.y() + 1)
460                    });
461                    assert!(covered, "disk point ({x}, {y}) not covered");
462                }
463                x += 0.25;
464            }
465            y += 0.25;
466        }
467    }
