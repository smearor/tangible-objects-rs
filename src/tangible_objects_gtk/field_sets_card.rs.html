1//! The field-sets card - the registered field sets (name, owner,
2//! enabled state, field count) as a movable card on the
3//! `FloatingBoard` (concepts/planned/GUI.md §3).
4
5use std::cell::RefCell;
6use std::rc::Rc;
7use std::sync::Arc;
8use std::sync::Mutex;
9use std::time::Duration;
10
11use gtk4::Box as GtkBox;
12use gtk4::Grid;
13use gtk4::Label;
14use gtk4::Orientation;
15use gtk4::Switch;
16use gtk4::glib::ControlFlow;
17use gtk4::prelude::*;
18use i18n_embed_fl::fl;
19use tangible_objects_model::FieldSetName;
20use tangible_objects_runtime::ControlCommand;
21
22use crate::language_loader;
23use crate::shared_state::SharedState;
24use crate::shared_state::send_command;

26/// Widgets of one field-set row - kept across ticks so only the
27/// volatile cells rewrite.
28struct FieldSetRowWidgets {
29    /// The set this row displays.
30    name: FieldSetName,
31    /// Registering application cell.
32    owner: Label,
33    /// Enable switch - sends `SetFieldSetEnabled`.
34    enabled: Switch,
35    /// Occupancy field count cell.
36    field_count: Label,
37}
38
39/// Builds the field-sets-card content: a `Name | Owner | Enabled |
40/// Fields` grid refreshed from [`SharedState`] every 100 ms. Rows
41/// rebuild only when the registered sets change; the volatile cells
42/// (enabled flag) update in place.
43pub fn create_field_sets_card(state: &Arc<Mutex<SharedState>>) -> GtkBox {
44    let grid = Grid::new();
45    grid.set_column_spacing(10);
46    grid.set_row_spacing(4);
47    for (column, header) in [
48        fl!(language_loader(), "col-name"),
49        fl!(language_loader(), "col-owner"),
50        fl!(language_loader(), "col-enabled"),
51        fl!(language_loader(), "col-fields"),
52    ]
53    .iter()
54    .enumerate()
55    {
56        let label = Label::new(Some(header));
57        label.set_halign(gtk4::Align::Start);
58        label.add_css_class("dim-label");
59        grid.attach(&label, column as i32, 0, 1, 1);
60    }
61
62    let rows: Rc<RefCell<Vec<FieldSetRowWidgets>>> = Rc::new(RefCell::new(Vec::new()));
63    let state_tick = state.clone();
64    let rows_tick = rows.clone();
65    let grid_tick = grid.clone();
66    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
67        let grid = &grid_tick;
68        let s = match state_tick.lock() {
69            Ok(s) => s,
70            Err(_) => return ControlFlow::Continue,
71        };
72        let mut rows = rows_tick.borrow_mut();
73        if rows
74            .iter()
75            .map(|row| row.name.as_str())
76            .ne(s.field_set_rows.iter().map(|row| row.name.as_str()))
77        {
78            // Registration set changed - drop every data row below the
79            // header and rebuild in `s.field_set_rows` order.
80            let row_count = rows.len() as i32;
81            rows.clear();
82            for _ in 0..row_count {
83                grid.remove_row(1);
84            }
85            for (index, set) in s.field_set_rows.iter().enumerate() {
86                let name = Label::new(Some(set.name.as_str()));
87                name.set_halign(gtk4::Align::Start);
88                let owner = Label::new(None);
89                owner.set_halign(gtk4::Align::Start);
90                let enabled = Switch::new();
91                enabled.set_halign(gtk4::Align::Start);
92                let set_name = set.name.clone();
93                let state_enabled = state_tick.clone();
94                enabled.connect_state_set(move |_, active| {
95                    send_command(
96                        &state_enabled,
97                        ControlCommand::SetFieldSetEnabled {
98                            name: set_name.clone(),
99                            enabled: active,
100                        },
101                    );
102                    gtk4::glib::Propagation::Proceed
103                });
104                let field_count = Label::new(None);
105                field_count.set_halign(gtk4::Align::Start);
106                let row = index as i32 + 1;
107                grid.attach(&name, 0, row, 1, 1);
108                grid.attach(&owner, 1, row, 1, 1);
109                grid.attach(&enabled, 2, row, 1, 1);
110                grid.attach(&field_count, 3, row, 1, 1);
111                rows.push(FieldSetRowWidgets {
112                    name: set.name.clone(),
113                    owner,
114                    enabled,
115                    field_count,
116                });
117            }
118        }
119        for (set, row) in s.field_set_rows.iter().zip(rows.iter()) {
120            row.owner.set_label(
121                set.owner
122                    .as_ref()
123                    .map(|owner| owner.as_str())
124                    .unwrap_or("–"),
125            );
126            // set_active emits no state-set signal - the sync does
127            // not echo commands back to the device loop.
128            row.enabled.set_active(set.enabled);
129            row.field_count.set_label(&set.field_count.to_string());
130        }
131        ControlFlow::Continue
132    });
133
134    let body = GtkBox::new(Orientation::Vertical, 0);
135    body.append(&grid);
136    body
137}
