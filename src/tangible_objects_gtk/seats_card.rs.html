2//! owner, marker profile, table side, edge fraction, radial
3//! orientation) as a movable card on the `FloatingBoard`

27/// Widgets of one seat row - kept across ticks so only the volatile
28/// cells (pose, marker state) rewrite.
29struct SeatRow {
30    /// The seat this row displays - the rebuild key.
31    id: SeatId,
32    /// Seat name cell - dimmed while the marker is absent.
33    name: Label,
34    /// Session owner cell.
35    owner: Label,
36    /// Marker profile name cell - the localized `seat-virtual`
37    /// placeholder for virtual seats.
38    profile: Label,
39    /// Organizational edge cell (`Bottom`/`Left`/`Top`/`Right`).
40    side: Label,
41    /// Fraction along the edge cell - `"–"` for unpinned offsets.
42    fraction: Label,
43    /// Radial orientation cell, degrees.
44    rotation: Label,
45}
46
47/// Builds the seats-card content: a `Seat name | Owner | Profile name |
48/// Table side | Fraction | Rotation` grid refreshed from
49/// [`SharedState`] every 100 ms. Rows rebuild only when the seat set
50/// changes; the pose cells update in place.
51pub fn create_seats_card(state: &Arc<Mutex<SharedState>>) -> GtkBox {
52    let grid = Grid::new();
53    grid.set_column_spacing(10);
54    grid.set_row_spacing(4);
55    for (column, header) in [
56        fl!(language_loader(), "col-seat-name"),
57        fl!(language_loader(), "col-owner"),
58        fl!(language_loader(), "col-profile-name"),
59        fl!(language_loader(), "col-side"),
60        fl!(language_loader(), "col-fraction"),
61        fl!(language_loader(), "col-rotation"),
62    ]
63    .iter()
64    .enumerate()
65    {
66        let label = Label::new(Some(header));
67        label.set_halign(Align::Start);
68        label.add_css_class("dim-label");
69        grid.attach(&label, column as i32, 0, 1, 1);
70    }
71
72    let rows: Rc<RefCell<Vec<SeatRow>>> = Rc::new(RefCell::new(Vec::new()));
73    let state_tick = state.clone();
74    let rows_tick = rows.clone();
75    let grid_tick = grid.clone();
76    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
77        let grid = &grid_tick;
78        let s = match state_tick.lock() {
79            Ok(s) => s,
80            Err(_) => return ControlFlow::Continue,
81        };
82        // Session order is stable; each session's roster keeps its
83        // declaration order - the flattened list is the card's row set.
84        let seats = s
85            .seat_sessions
86            .iter()
87            .flat_map(|session| session.seats.iter())
88            .collect::<Vec<_>>();
89        let mut rows = rows_tick.borrow_mut();
90        if rows
91            .iter()
92            .map(|row| row.id)
93            .ne(seats.iter().map(|seat| seat.id))
94        {
95            // Seat set changed - drop every data row below the header
96            // and rebuild in roster order.
97            let row_count = rows.len() as i32;
98            rows.clear();
99            for _ in 0..row_count {
100                grid.remove_row(1);
101            }
102            for (index, seat) in seats.iter().enumerate() {
103                let name = Label::new(None);
104                let owner = Label::new(None);
105                let profile = Label::new(None);
106                let side = Label::new(None);
107                let fraction = Label::new(None);
108                let rotation = Label::new(None);
109                let row = index as i32 + 1;
110                for (column, label) in [&name, &owner, &profile, &side, &fraction, &rotation]
111                    .iter()
112                    .enumerate()
113                {
114                    label.set_halign(Align::Start);
115                    grid.attach(*label, column as i32, row, 1, 1);
116                }
117                rows.push(SeatRow {
118                    id: seat.id,
119                    name,
120                    owner,
121                    profile,
122                    side,
123                    fraction,
124                    rotation,
125                });
126            }
127        }
128        for (seat, row) in seats.iter().zip(rows.iter()) {
129            row.name.set_label(seat.name.as_str());
130            // Absent markers dim the whole row - mirrors the seat
131            // overlay's alpha reduction while the marker is lifted.
132            let absent = seat.state == SeatMarkerState::Absent;
133            for label in [
134                &row.name,
135                &row.owner,
136                &row.profile,
137                &row.side,
138                &row.fraction,
139                &row.rotation,
140            ] {
141                if absent {
142                    label.add_css_class("dim-label");
143                } else {
144                    label.remove_css_class("dim-label");
145                }
146            }
147            row.owner.set_label(seat.owner.as_str());
148            let profile = seat
149                .marker
150                .map(|marker| {
151                    s.profiles
152                        .iter()
153                        .find(|profile| profile.pattern_id == marker)
154                        .map(|profile| profile.name.to_string())
155                        .unwrap_or_else(|| marker.to_string())
156                })
157                .unwrap_or_else(|| fl!(language_loader(), "seat-virtual"));
158            row.profile.set_label(&profile);
159            row.side.set_label(&match seat.pose.seat.side {
160                TableSide::Bottom => fl!(language_loader(), "side-bottom"),
161                TableSide::Left => fl!(language_loader(), "side-left"),
162                TableSide::Top => fl!(language_loader(), "side-top"),
163                TableSide::Right => fl!(language_loader(), "side-right"),
164            });
165            row.fraction.set_label(
166                &seat
167                    .pose
168                    .seat
169                    .offset
170                    .map(|offset| format!("{:.2}", offset.as_f64()))
171                    .unwrap_or_else(|| "–".to_string()),
172            );
173            row.rotation
174                .set_label(&format!("{:.0}°", seat.pose.orientation.to_degrees()));
175        }
176        ControlFlow::Continue
177    });
178
179    let body = GtkBox::new(Orientation::Vertical, 6);
180    body.append(&grid);
181    body
182}
