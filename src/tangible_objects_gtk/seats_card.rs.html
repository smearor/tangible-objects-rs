28/// Widgets of one seat row - kept across ticks so only the volatile
29/// cells (pose, marker state) rewrite.
30struct SeatRow {
31    /// The seat this row displays - the rebuild key.
32    id: SeatId,
33    /// Seat name cell - dimmed while the marker is absent.
34    name: Label,
35    /// Session owner cell.
36    owner: Label,
37    /// Marker profile name cell - the localized `seat-virtual`
38    /// placeholder for virtual seats.
39    profile: Label,
40    /// Organizational edge cell (`Bottom`/`Left`/`Top`/`Right`).
41    side: Label,
42    /// Fraction along the edge cell - `"–"` for unpinned offsets.
43    fraction: Label,
44    /// Radial orientation cell, degrees.
45    rotation: Label,
46}
47
48/// Builds the seats-card content: a `Seat name | Owner | Profile name |
49/// Table side | Fraction | Rotation` grid refreshed from
50/// [`SharedState`] every 100 ms. Rows rebuild only when the seat set
51/// changes; the pose cells update in place. The folded-content badge
52/// shows `anchored/total` seats alongside the card body. The card
53/// uses `DragHandle::WholeCardFolded` (configured in `lib.rs`):
54/// folded, the whole badge drags; unfolded, the header strip drags
55/// and a tap unfolds via `unfold_on_click`.
56pub fn create_seats_card(state: &Arc<Mutex<SharedState>>) -> (GtkBox, GtkBox) {
57    let grid = Grid::new();
58    grid.set_column_spacing(10);
59    grid.set_row_spacing(4);
60    for (column, header) in [
61        fl!(language_loader(), "col-seat-name"),
62        fl!(language_loader(), "col-owner"),
63        fl!(language_loader(), "col-profile-name"),
64        fl!(language_loader(), "col-side"),
65        fl!(language_loader(), "col-fraction"),
66        fl!(language_loader(), "col-rotation"),
67    ]
68    .iter()
69    .enumerate()
70    {
71        let label = Label::new(Some(header));
72        label.set_halign(Align::Start);
73        label.add_css_class("dim-label");
74        grid.attach(&label, column as i32, 0, 1, 1);
75    }
76
77    let (badge, badge_count) = count_badge(
78        "system-users-symbolic",
79        &fl!(language_loader(), "card-seats-title"),
80    );
81    // Thumb-sized translucent FAB (`.badge-fab` in `lib.rs`'s
82    // stylesheet) instead of a plain folded button.
83    badge.add_css_class("badge-fab");
84
85    let rows: Rc<RefCell<Vec<SeatRow>>> = Rc::new(RefCell::new(Vec::new()));
86    let state_tick = state.clone();
87    let rows_tick = rows.clone();
88    let grid_tick = grid.clone();
89    gtk4::glib::timeout_add_local(Duration::from_millis(100), move || {
90        let grid = &grid_tick;
91        let s = match state_tick.lock() {
92            Ok(s) => s,
93            Err(_) => return ControlFlow::Continue,
94        };
95        // Session order is stable; each session's roster keeps its
96        // declaration order - the flattened list is the card's row set.
97        let seats = s
98            .seat_sessions
99            .iter()
100            .flat_map(|session| session.seats.iter())
101            .collect::<Vec<_>>();
102        // The badge counts seats whose marker anchors the pose -
103        // `Absent` seats keep a last pose but nobody sits there.
104        let anchored = seats
105            .iter()
106            .filter(|seat| seat.state != SeatMarkerState::Absent)
107            .count();
108        badge_count.set_label(&format!("{anchored} / {}", seats.len()));
109        let mut rows = rows_tick.borrow_mut();
110        if rows
111            .iter()
112            .map(|row| row.id)
113            .ne(seats.iter().map(|seat| seat.id))
114        {
115            // Seat set changed - drop every data row below the header
116            // and rebuild in roster order.
117            let row_count = rows.len() as i32;
118            rows.clear();
119            for _ in 0..row_count {
120                grid.remove_row(1);
121            }
122            for (index, seat) in seats.iter().enumerate() {
123                let name = Label::new(None);
124                let owner = Label::new(None);
125                let profile = Label::new(None);
126                let side = Label::new(None);
127                let fraction = Label::new(None);
128                let rotation = Label::new(None);
129                let row = index as i32 + 1;
130                for (column, label) in [&name, &owner, &profile, &side, &fraction, &rotation]
131                    .iter()
132                    .enumerate()
133                {
134                    label.set_halign(Align::Start);
135                    grid.attach(*label, column as i32, row, 1, 1);
136                }
137                rows.push(SeatRow {
138                    id: seat.id,
139                    name,
140                    owner,
141                    profile,
142                    side,
143                    fraction,
144                    rotation,
145                });
146            }
147        }
148        for (seat, row) in seats.iter().zip(rows.iter()) {
149            row.name.set_label(seat.name.as_str());
150            // Absent markers dim the whole row - mirrors the seat
151            // overlay's alpha reduction while the marker is lifted.
152            let absent = seat.state == SeatMarkerState::Absent;
153            for label in [
154                &row.name,
155                &row.owner,
156                &row.profile,
157                &row.side,
158                &row.fraction,
159                &row.rotation,
160            ] {
161                if absent {
162                    label.add_css_class("dim-label");
163                } else {
164                    label.remove_css_class("dim-label");
165                }
166            }
167            row.owner.set_label(seat.owner.as_str());
168            let profile = seat
169                .marker
170                .map(|marker| {
171                    s.profiles
172                        .iter()
173                        .find(|profile| profile.pattern_id == marker)
174                        .map(|profile| profile.name.to_string())
175                        .unwrap_or_else(|| marker.to_string())
176                })
177                .unwrap_or_else(|| fl!(language_loader(), "seat-virtual"));
178            row.profile.set_label(&profile);
179            row.side.set_label(&match seat.pose.seat.side {
180                TableSide::Bottom => fl!(language_loader(), "side-bottom"),
181                TableSide::Left => fl!(language_loader(), "side-left"),
182                TableSide::Top => fl!(language_loader(), "side-top"),
183                TableSide::Right => fl!(language_loader(), "side-right"),
184            });
185            row.fraction.set_label(
186                &seat
187                    .pose
188                    .seat
189                    .offset
190                    .map(|offset| format!("{:.2}", offset.as_f64()))
191                    .unwrap_or_else(|| "–".to_string()),
192            );
193            row.rotation
194                .set_label(&format!("{:.0}°", seat.pose.orientation.to_degrees()));
195        }
196        ControlFlow::Continue
197    });
198
199    let body = GtkBox::new(Orientation::Vertical, 6);
200    body.append(&grid);
201    (body, badge)
202}
