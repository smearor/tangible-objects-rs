1//! `FieldSetRow` - one row of the field-sets card.
2
3use tangible_objects_model::FieldSetName;
4use tangible_objects_model::Owner;
5
6/// Display-relevant slice of a registered
7/// [`tangible_objects_model::FieldSet`] plus its registration owner -
8/// built per frame in `evdev_thread` from `engine.field_sets()` and
9/// `engine.field_set_owner()`.
10#[derive(Debug, Clone)]
11pub struct FieldSetRow {
12    /// The set's name.
13    pub name: FieldSetName,
14    /// Registering application; `None` for statically loaded sets.
15    pub owner: Option<Owner>,
16    /// Whether the set currently tracks occupancy
17    /// (`SetFieldSetEnabled` command).
18    pub enabled: bool,
19    /// Number of occupancy fields in the set.
20    pub field_count: usize,
21}
